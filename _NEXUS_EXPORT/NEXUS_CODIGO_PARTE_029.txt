from __future__ import annotations

import re
from typing import Any

from planner.intent import Intent
from planner.reference_resolver import ReferenceResolver
from planner.episode_query_resolver import EpisodeQueryResolver


class IntentRecognizer:
    """
    Convierte texto libre en un Intent estructurado.

    Soporta:
    - intenciones simples
    - intenciones contextuales
    - referencias semánticas
    - intenciones compuestas
    """

    def __init__(
        self,
        reference_resolver: ReferenceResolver | None = None,
    ) -> None:
        self._reference_resolver = (
            reference_resolver
            or ReferenceResolver()
        )
        self._episode_query_resolver = EpisodeQueryResolver()

    def recognize(
        self,
        text: str,
        context: dict[str, Any] | None = None,
    ) -> Intent:
        normalized = text.lower().strip()

        #
        # Compound Intent
        #
        # Se evalúa antes de las intenciones simples para evitar
        # que una acción contenida dentro de una orden compuesta
        # capture toda la solicitud.
        #

        # Conservamos el texto original para no perder mayúsculas
        # en valores que el usuario pide guardar.
        compound_parts = re.split(
            r"\s+(?:(?:y|e)\s+)?(?:luego|después|despues|además|ademas)\s+|\s+(?:y|e)\s+",
            text.strip(),
            maxsplit=1,
            flags=re.IGNORECASE,
        )

        if len(compound_parts) == 2:
            first_original = compound_parts[0].strip()
            second_original = compound_parts[1].strip()
            first_text = first_original.lower()
            second_text = second_original.lower()

            first_intent = self._recognize_single(
                first_text,
                context,
                original_text=first_original,
            )

            second_intent = self._recognize_single(
                second_text,
                context,
                original_text=second_original,
                previous_intent=first_intent,
            )

            if (
                first_intent.action != "system.unknown"
                and second_intent.action != "system.unknown"
            ):
                return Intent(
                    action=first_intent.action,
                    confidence=min(
                        first_intent.confidence,
                        second_intent.confidence,
                    ),
                    parameters=first_intent.parameters.copy(),
                    original_text=text,
                    sub_intents=[
                        Intent(
                            action=first_intent.action,
                            confidence=first_intent.confidence,
                            parameters=first_intent.parameters.copy(),
                            original_text=first_original,
                        ),
                        Intent(
                            action=second_intent.action,
                            confidence=second_intent.confidence,
                            parameters=second_intent.parameters.copy(),
                            original_text=second_original,
                        ),
                    ],
                )

        return self._recognize_single(
            normalized,
            context,
            original_text=text,
        )

    def _recognize_single(
        self,
        normalized: str,
        context: dict[str, Any] | None = None,
        original_text: str | None = None,
        previous_intent: Intent | None = None,
    ) -> Intent:
        text = original_text or normalized

                #
        # Semantic memory lookup by topic
        # Supports questions such as:
        # "¿Qué recordás sobre mi proyecto?"
        # "¿Qué tenés registrado sobre mi proyecto?"
        #

        
        # Semantic memory lookup by topic.
        # Supports both explicit memory questions and direct factual questions.
        broad_memory_match = re.search(
            r"^¿?\s*(?:qué|que)\s+"
            r"(?:recordás|recordas|recuerdas|tenés registrado|"
            r"tenes registrado|tienes registrado)\s+"
            r"(?:sobre|acerca de)\s+"
            r"(?:mí|mi persona|mi información|mi informacion)"
            r"[?!.]*$",
            normalized,
            re.IGNORECASE,
        )

        if broad_memory_match:
            return Intent(
                action="system.recall_fact",
                original_text=text,
                parameters={
                    "subject": "user",
                    "predicate": "__all__",
                },
            )

        direct_lookup_match = re.search(
            r"^¿?\s*(?:cuál|cual)\s+es\s+"
            r"(?:mi|mis)\s+(.+?)[?!.]*$",
            normalized,
            re.IGNORECASE,
        )

        if direct_lookup_match:
            predicate = direct_lookup_match.group(1).strip()
            if predicate:
                return Intent(
                    action="system.recall_fact",
                    original_text=text,
                    parameters={
                        "subject": "user",
                        "predicate": predicate,
                    },
                )

        memory_lookup_match = re.search(
            r"^¿?\s*(?:qué|que)\s+"
            r"(?:recordás|recordas|recuerdas|tenés registrado|"
            r"tenes registrado|tienes registrado)\s+"
            r"(?:sobre|acerca de|de)\s+(.+?)[?!.]*$",
            normalized,
            re.IGNORECASE,
        )

        if memory_lookup_match:
            predicate = memory_lookup_match.group(1).strip()

            if predicate:
                return Intent(
                    action="system.recall_fact",
                    original_text=text,
                    parameters={
                        "subject": "user",
                        "predicate": predicate,
                    },
                )


        #
        # Semantic reference resolution
        #
        # El resolver identifica el objeto al que hace referencia
        # el usuario. No decide por sí mismo qué acción ejecutar.
        #

        reference = self._reference_resolver.resolve(
            normalized,
            context,
        )

        #
        # Semantic Episode Query
        #
        # Interpreta preguntas sobre episodios anteriores.
        # El resolver solamente determina qué acción y qué campo
        # deben consultarse; la recuperación real queda a cargo
        # de QueryEpisodeAction.
        #

        episode_query = self._episode_query_resolver.resolve(
            normalized
        )

        if episode_query is not None:
            return Intent(
                action="system.query_episode",
                original_text=text,
                parameters={
                    "action": episode_query.action,
                    "field": episode_query.field,
                },
            )

        #
        # Explicit semantic memory
        #
        # V1: explicit requests to remember a fact and retrieve it later.
        #
        remember_match = re.match(
            r"^(?:record[aá]|recordame|acordate|acord[aá]te|"
            r"guard[aá]|guarda|aprend[eé]|memoriz[aá])\s+"
            r"(?:que\s+)?(.+?)\s+(?:se llama|es|son|está|esta)\s+"
            r"(.+?)[.!?]*$",
            text.strip(),
            re.IGNORECASE,
        )

        if remember_match:
            predicate = re.sub(
                r"^(?:mi|mis|el|la|los|las|un|una)\s+",
                "",
                remember_match.group(1).strip(),
                flags=re.IGNORECASE,
            )
            value = remember_match.group(2).strip().rstrip(".!?")
            if predicate and value:
                return Intent(
                    action="system.remember_fact",
                    original_text=text,
                    parameters={
                        "subject": "user",
                        "predicate": predicate.lower(),
                        "value": value,
                    },
                )

        if re.search(
            r"\b(?:qué|que)\s+(?:nombre|dato|información|informacion|color|valor|hecho|preferencia)\s+"
            r"(?:te\s+)?(?:ped[ií]\s+)?(?:recordar|guardar|memorizar)\b",
            normalized,
        ):
            return Intent(
                action="system.recall_last_fact",
                original_text=text,
                parameters={"subject": "user"},
            )

        recall_match = re.search(
            r"\b(?:cómo|como)\s+se\s+llama\s+"
            r"(?:(?:mi|mis|el|la|los|las|un|una)\s+)?(.+?)[?!.]*$",
            normalized,
        )
        if recall_match:
            predicate = recall_match.group(1).strip()
            if predicate:
                return Intent(
                    action="system.recall_fact",
                    original_text=text,
                    parameters={
                        "subject": "user",
                        "predicate": predicate,
                    },
                )

        #
        # Anaphoric references inside compound intents
        #
        # Permite que la segunda acción haga referencia al resultado
        # inmediato de la primera acción, por ejemplo:
        #
        #   "capturá la pantalla e inspeccionala"
        #   "capturá la pantalla y analizala"
        #   "sacá una captura y revisala"
        #
        # La referencia solo se resuelve cuando existe una intención
        # anterior conocida y el objeto referenciado es compatible.
        #

        if (
            previous_intent is not None
            and previous_intent.action == "system.screen_capture"
        ):
            capture_reference_patterns = (
                r"^(inspeccion[aá]la|inspeccionala)$",
                r"^(analiz[aá]la|analizala)$",
                r"^(revis[aá]la|revisala)$",
                r"^(mir[aá]la|mirala)$",
            )

            if any(
                re.fullmatch(
                    pattern,
                    normalized,
                )
                for pattern in capture_reference_patterns
            ):
                return Intent(
                    action="system.inspect_capture",
                    original_text=text,
                    parameters={},
                )

        #
        # Contextual Volume References
        #
        # Estas expresiones dependen de una interacción anterior.
        #
        # Ejemplos:
        #
        #   "ahora ponelo al 50%"
        #   "dejalo en 30%"
        #   "subilo al 70%"
        #
        # La presencia de una referencia contextual no crea una
        # capacidad. Solamente permite completar el significado
        # cuando la acción resultante ya es conocida.
        #

        contextual_volume = self._resolve_contextual_volume(
            normalized,
            context,
        )

        if contextual_volume is not None:
            return Intent(
                action="system.set_volume",
                original_text=text,
                parameters={
                    "level": contextual_volume,
                },
            )

        #
        # Contextual application reference
        #
        # La referencia semántica es el objeto ("esa aplicación"),
        # mientras que la frase completa contiene además la acción
        # ("abrí", "ejecutá", "iniciá").
        #
        # Por eso primero identificamos la estructura lingüística
        # y después entregamos solamente el referente al resolver.
        #

        contextual_open_patterns = (
            r"^(abr[ií]|abrir|ejecut[aá]|ejecutar|"
            r"inici[aá]|iniciar)\s+(esa\s+aplicaci[oó]n)$",
            r"^(abr[ií]|abrir|ejecut[aá]|ejecutar|"
            r"inici[aá]|iniciar)\s+(esa\s+app)$",
        )

        contextual_open_match = None

        for pattern in contextual_open_patterns:
            contextual_open_match = re.fullmatch(
                pattern,
                normalized,
            )

            if contextual_open_match:
                break

        if contextual_open_match:
            reference_text = (
                contextual_open_match.group(2)
            )

            reference = self._reference_resolver.resolve(
                reference_text,
                context,
            )

            if (
                reference is not None
                and reference.kind == "application"
            ):
                application = str(
                    reference.value
                ).strip()

                if application:
                    return Intent(
                        action="system.open_app",
                        original_text=text,
                        parameters={
                            "application": application,
                        },
                    )

        #
        # Open Application
        #
        match = re.match(
            r"^(abr[ií]|abrir|ejecut[aá]|ejecutar|"
            r"inici[aá]|iniciar)\s+(.+)$",
            normalized,
        )

        if match:
            application = match.group(2).strip()

            #
            # Evitamos arrastrar artículos innecesarios hacia
            # el launcher cuando el usuario dice:
            #
            #   "abrí la calculadora"
            #   "abrí el notepad"
            #

            application = re.sub(
                r"^(la|el|un|una)\s+",
                "",
                application,
            ).strip()

            return Intent(
                action="system.open_app",
                original_text=text,
                parameters={
                    "application": application,
                },
            )

        #
        # Screen Capture
        #

        screen_capture_patterns = (
            r"\bcaptur[aá]\s+la\s+pantalla\b",
            r"\bhacé\s+una\s+captura\s+de\s+pantalla\b",
            r"\bhace\s+una\s+captura\s+de\s+pantalla\b",
            r"\bsacá\s+una\s+captura\s+de\s+pantalla\b",
            r"\bsaca\s+una\s+captura\s+de\s+pantalla\b",
        )

        if any(
            re.search(
                pattern,
                normalized,
            )
            for pattern in screen_capture_patterns
        ):
            return Intent(
                action="system.screen_capture",
                original_text=text,
                parameters={},
            )

        #
        # Inspect Capture
        #
        # Además de las formas explícitas, una referencia semántica
        # a una captura previamente disponible permite expresiones
        # naturales como:
        #
        #   "inspeccioná esa captura"
        #   "analizá esa pantalla"
        #

        inspect_capture_patterns = (
            r"\binspeccion[aá]\s+(la\s+)?captura\b",
            r"\binspeccion[aá]\s+(la\s+)?pantalla\b",
            r"\banaliz[aá]\s+(la\s+)?captura\b",
            r"\banaliz[aá]\s+(la\s+)?pantalla\b",
            r"\brevis[aá]\s+(la\s+)?captura\b",
            r"\brevis[aá]\s+(la\s+)?pantalla\b",
            r"\bm[aí]r[aá]\s+(la\s+)?captura\b",
            r"\bm[aí]r[aá]\s+(la\s+)?pantalla\b",
            r"\binspeccion[aá]\s+esa\s+captura\b",
            r"\binspeccion[aá]\s+esa\s+pantalla\b",
            r"\banaliz[aá]\s+esa\s+captura\b",
            r"\banaliz[aá]\s+esa\s+pantalla\b",
            r"\brevis[aá]\s+esa\s+captura\b",
            r"\brevis[aá]\s+esa\s+pantalla\b",
            r"\bm[aí]r[aá]\s+esa\s+captura\b",
            r"\bm[aí]r[aá]\s+esa\s+pantalla\b",
        )

        if any(
            re.search(
                pattern,
                normalized,
            )
            for pattern in inspect_capture_patterns
        ):
            if (
                "esa captura" in normalized
                or "esa pantalla" in normalized
            ):
                if (
                    reference is not None
                    and reference.kind == "capture"
                ):
                    return Intent(
                        action="system.inspect_capture",
                        original_text=text,
                        parameters={},
                    )

                return Intent(
                    action="system.unknown",
                    original_text=text,
                    parameters={},
                )

            return Intent(
                action="system.inspect_capture",
                original_text=text,
                parameters={},
            )

        #
        # Contextual Close Application
        #
        # Permite referencias como:
        #
        #   "Abrí Chrome"
        #   "Cerralo"
        #   "Cerrala"
        #   "Cerrá esa aplicación"
        #   "Cerrá esa app"
        #
        # La aplicación solo se resuelve si existe una
        # referencia contextual explícita y válida.
        #

        contextual_close_patterns = (
            r"^(cerr[aá]lo|cerralo)$",
            r"^(cerr[aá]la|cerrala)$",
            r"^cerr[aá]\s+esa\s+aplicación$",
            r"^cerra\s+esa\s+aplicación$",
            r"^cerr[aá]\s+esa\s+app$",
            r"^cerra\s+esa\s+app$",
        )

        if context and any(
            re.fullmatch(
                pattern,
                normalized,
            )
            for pattern in contextual_close_patterns
        ):
            application = self._resolve_contextual_application(
                normalized,
                context,
            )

            if application:
                return Intent(
                    action="system.close_app",
                    original_text=text,
                    parameters={
                        "application": application,
                    },
                )

        #
        # Close Application
        #
        # Ejemplos:
        #
        #   "Cerrá notepad"
        #   "Cierra notepad"
        #   "Cerrar notepad"
        #   "Cerrá la calculadora"
        #   "Cierra Spotify"
        #

        match = re.match(
            r"^(cerr[aá]|cierra|cerrar)\s+(.+)$",
            normalized,
        )

        if match:
            application = match.group(2).strip()

            application = re.sub(
                r"^(la|el|un|una)\s+",
                "",
                application,
            ).strip()

            if application:
                return Intent(
                    action="system.close_app",
                    original_text=text,
                    parameters={
                        "application": application,
                    },
                )

        #
        # Mute
        #

        mute_patterns = (
            r"\bsilenci[aá]\b",
            r"\bsilencia\b",
            r"\bpon[eé]\s+mute\b",
            r"\bponelo\s+en\s+silencio\b",
            r"\bponelo\s+en\s+mute\b",
            r"\bactivar\s+silencio\b",
        )

        if any(
            re.search(
                pattern,
                normalized,
            )
            for pattern in mute_patterns
        ):
            return Intent(
                action="system.mute",
                original_text=text,
                parameters={},
            )

        #
        # Unmute
        #

        unmute_patterns = (
            r"\bactiv[aá]\s+el\s+sonido\b",
            r"\bactiv[aá]\s+nuevamente\s+el\s+sonido\b",
            r"\bquit[aá]\s+el\s+silencio\b",
            r"\bquit[aá]le\s+el\s+mute\b",
            r"\bsac[aá]le\s+el\s+mute\b",
            r"\bdesactiv[aá]\s+el\s+silencio\b",
            r"\bdesactiv[aá]le\s+el\s+mute\b",
            r"\bsac[aá]\s+el\s+silencio\b",
        )

        if any(
            re.search(
                pattern,
                normalized,
            )
            for pattern in unmute_patterns
        ):
            return Intent(
                action="system.unmute",
                original_text=text,
                parameters={},
            )

        #
        # Set Volume
        #

        match = re.search(
            r"(?:"
            r"(?:volumen|volume)\s+"
            r"(?:al|a|en)\s*"
            r"|"
            r"(?:subilo|subirlo|sube)\s+"
            r"(?:al|a|en)\s*"
            r"|"
            r"(?:bajalo|bajarlo|baja)\s+"
            r"(?:al|a|en)\s*"
            r"|"
            r"(?:ponelo|ponerlo|ponélo|ponelo)\s+"
            r"(?:al|a|en)\s*"
            r"|"
            r"(?:dejalo|dejarlo|déjalo)\s+"
            r"(?:al|a|en)\s*"
            r")"
            r"(\d+(?:[.,]\d+)?)\s*%?",
            normalized,
        )

        if match:
            level = float(
                match.group(1).replace(",", ".")
            )

            return Intent(
                action="system.set_volume",
                original_text=text,
                parameters={
                    "level": level,
                },
            )

        #
        # Get Current Volume
        #

        current_volume_patterns = (
            r"\ba cuánto está\b",
            r"\ba cuanto esta\b",
            r"\ba cuánto quedó\b",
            r"\ba cuanto quedo\b",
            r"\bvolumen actual\b",
            r"\bqué volumen tengo\b",
            r"\bque volumen tengo\b",
            r"\bqué volumen hay\b",
            r"\bque volumen hay\b",
        )

        if any(
            re.search(
                pattern,
                normalized,
            )
            for pattern in current_volume_patterns
        ):
            return Intent(
                action="system.get_volume",
                original_text=text,
                parameters={},
            )

        #
        # Get Current Mute State
        #

        mute_status_patterns = (
            r"\bestá silenciado\b",
            r"\besta silenciado\b",
            r"\bestá en mute\b",
            r"\besta en mute\b",
            r"\bsistema está silenciado\b",
            r"\bsistema esta silenciado\b",
            r"\bestá activado el mute\b",
            r"\besta activado el mute\b",
            r"\bhay sonido\b",
        )

        if any(
            re.search(
                pattern,
                normalized,
            )
            for pattern in mute_status_patterns
        ):
            return Intent(
                action="system.get_mute",
                original_text=text,
                parameters={},
            )

        #
        # Restore previous mute
        #

        if (
            re.search(
                r"(volv[eé]|restaur[aá]|dej[aá]|pon[eé]).*"
                r"(mute|silencio|silenciado|sonido)",
                normalized,
            )
            or re.search(
                r"(mute|silencio|silenciado|sonido).*"
                r"(anterior|previo|como estaba|como antes)",
                normalized,
            )
        ):
            return Intent(
                action="system.restore_previous_mute",
                original_text=text,
                parameters={},
            )

        #
        # Restore previous volume
        #

        if (
            re.search(
                r"(volv[eé]|restaur[aá]|dej[aá]|pon[eé]).*"
                r"(como estaba|como antes|anterior|previo)",
                normalized,
            )
            or re.search(
                r"restaur[aá]\s+(el\s+)?volumen",
                normalized,
            )
        ):
            return Intent(
                action="system.restore_previous_volume",
                original_text=text,
                parameters={},
            )

        #
        # Historical Volume - Previous
        #

        if (
            re.search(
                r"\b(antes|anterior|previamente)\b",
                normalized,
            )
            and re.search(
                r"\b(volumen|volume)\b",
                normalized,
            )
        ):
            return Intent(
                action="system.get_volume_history",
                original_text=text,
                parameters={
                    "mode": "previous",
                },
            )

        #
        # Historical Volume - Specific Value
        #

        percentage_match = re.search(
            r"(\d+(?:[.,]\d+)?)\s*%",
            normalized,
        )

        if percentage_match:
            requested_level = float(
                percentage_match.group(1).replace(",", ".")
            )

            historical_phrases = (
                "qué dejé",
                "que deje",
                "qué puse",
                "que puse",
                "qué tenía",
                "que tenia",
                "estuvo",
                "había estado",
                "habia estado",
                "dejé",
                "deje",
                "puse",
            )

            if any(
                phrase in normalized
                for phrase in historical_phrases
            ):
                return Intent(
                    action="system.get_volume_history",
                    original_text=text,
                    parameters={
                        "mode": "find",
                        "value": requested_level,
                    },
                )

        #
        # Contextual Volume History Compatibility
        #
        # Mantiene la capacidad anterior de consultar un valor
        # histórico específico cuando ese valor aparece en una
        # expresión contextual.
        #

        if context:
            percentage_match = re.search(
                r"(\d+(?:[.,]\d+)?)\s*%",
                normalized,
            )

            if percentage_match:
                requested_level = float(
                    percentage_match.group(1).replace(",", ".")
                )

                recent_interactions = context.get(
                    "recent_interactions",
                    [],
                )

                for interaction in reversed(
                    recent_interactions
                ):
                    if interaction.get("type") != "intent":
                        continue

                    data = interaction.get(
                        "data",
                        {},
                    )

                    if (
                        data.get("action")
                        != "system.set_volume"
                    ):
                        continue

                    parameters = data.get(
                        "parameters",
                        {},
                    )

                    level = parameters.get(
                        "level"
                    )

                    if level is None:
                        continue

                    try:
                        level = float(level)
                    except (
                        TypeError,
                        ValueError,
                    ):
                        continue

                    if level == requested_level:
                        return Intent(
                            action="system.get_volume_history",
                            original_text=text,
                            parameters={
                                "mode": "find",
                                "value": requested_level,
                            },
                        )

        #
        # Last Episode

        references = context.get(

            {},

        if isinstance(
            references,

                "last_application"

        recent_interactions = context.get(

                {},
            )

            if not isinstance(
                parameters,
                dict,
            ):
                continue

            application = parameters.get(
                "application"
            )

            if (
                isinstance(application, str)

            ):

                r"\bque cambie antes\b",
                r"\bqué hizo nexus antes\b",
                r"\bque hizo nexus antes\b",
            )
            return any(re.search(pattern, normalized) for pattern in episode_patterns)

        # Las demás acciones conservan la política de validación existente.
        return True

    def _validate_parameters(
        self,
        capability: dict[str, Any],
        parameters: dict[str, Any],
    ) -> bool:
        """Valida parámetros contra el esquema de la capacidad seleccionada."""
        schema = capability.get("parameters", {})
        if not isinstance(schema, dict):
            return True

        # Parámetros obligatorios.
        for name, definition in schema.items():
            if not isinstance(definition, dict):
                continue
            if definition.get("required", False) and name not in parameters:
                return False

        # No aceptar parámetros que no estén definidos en el esquema.
        for name in parameters:
            if name not in schema:
                return False

        # Comprobar los tipos de los parámetros recibidos.
        for name, value in parameters.items():
            definition = schema.get(name, {})
            parameter_type = definition.get("type")

            if parameter_type == "string":
                if not isinstance(value, str):
                    return False
            elif parameter_type == "number":
                if isinstance(value, bool) or not isinstance(value, (int, float)):
                    return False
            elif parameter_type == "integer":
                if isinstance(value, bool) or not isinstance(value, int):
                    return False

        return True

    def _unknown(
        self,
        text: str,
        confidence: float = 0.0,
    ) -> Intent:
        return Intent(
            action="system.unknown",
            confidence=confidence,
            parameters={},
            original_text=text,
        )

def build_engine(
    assistant_context: AssistantContext | None = None,
    assistant_session: AssistantSession | None = None,
) -> Engine:
    """
    Construye y configura completamente NEXUS.

    El AssistantContext es opcional para mantener compatibilidad
    con el arranque actual de NEXUS.
    """

    security_context = (
        assistant_session.context
        if assistant_session is not None
        else assistant_context
    )

    security_user = getattr(
        security_context,
        "user",
        None,
    )

    security_actor_id = getattr(
        security_user,
        "id",
        None,
    )

    local_console_actor = Actor.system(
        "local-console",
        name="NEXUS Local Console",
    )

    if (
        isinstance(security_actor_id, str)
        and security_actor_id.strip()
    ):
        security_actor = Actor.user(
            security_actor_id.strip()
        )
    else:
        security_actor = local_console_actor

    #
    # Working Memory
    #

    # Una única instancia compartida por todos
    # los componentes que necesiten memoria.

    memory = WorkingMemory()

    episode_memory = EpisodeMemory(
        memory
    )

    #
    # State History
    #

    state_history = StateHistory(
        memory
    )

    #
    # Kernel
    #

    kernel = Kernel()

    #
    # Runtime
    #

    runtime = Runtime(
        kernel=kernel
    )

    #
    # Provider Manager
    #

    provider_manager = ProviderManager()

    provider_manager.register(
        LocalProvider()
    )

    #
    # Action Registry
    #

    registry = ActionRegistry()

    #
    # Applications
    #

    registry.register(
        OpenAppAction(
            provider_manager
        )
    )

    registry.register(
        CloseAppAction()
    )

    registry.register(
        ProcessListAction(
            provider_manager
        )
    )

    registry.register(
        ScreenCaptureAction(
            provider_manager
        )
    )

    registry.register(
        InspectScreenAction()
    )

    registry.register(
        InspectCaptureAction()
    )

    registry.register(GetTimeAction())

    #
    # Episodic Memory
    #

    registry.register(
        QueryEpisodeAction(
            episode_memory
        )
    )

    #
    # Volume
    #

    registry.register(
        SetVolumeAction(
            provider_manager
        )
    )

    registry.register(
        GetVolumeAction(
            provider_manager
        )
    )

    registry.register(
        GetVolumeHistoryAction(
            state_history
        )
    )

    registry.register(
        GetRecentEpisodesAction(
            episode_memory
        )
    )

    registry.register(
        GetPreviousEpisodeAction(
            episode_memory
        )
    )

    registry.register(
        GetLastEpisodeAction(
            episode_memory
        )
    )

    registry.register(
        RestorePreviousVolumeAction(
            provider_manager,
            state_history,
        )
    )

    #
    # Mute / Unmute
    #

    registry.register(
        MuteAction(
            provider_manager
        )
    )

    registry.register(
        UnmuteAction(
            provider_manager
        )
    )

    registry.register(
        GetMuteAction(
            provider_manager
        )
    )

    registry.register(
        RestorePreviousMuteAction(
            provider_manager,
            state_history,
        )
    )

    #
    # Action Manager
    #

        #
    # Security: permisos de ejecución.
    #

    action_names = registry.list()

    permissions = frozenset(
        Permission(
            resource=action_name,
            action=PermissionAction.EXECUTE,
        )
        for action_name in action_names
    )


    local_role = Role(
        id="local_operator",
        name="Local Operator",
        permissions=permissions,
    )

    security_db_path = os.environ.get(
        "NEXUS_SECURITY_DB_PATH",
        str(Path.home() / ".nexus" / "security.db"),
    )

    security_database = SQLiteSecurityStore(security_db_path)
    role_store = SQLiteRoleStore(security_database)
    assignment_store = SQLiteAssignmentStore(security_database)

    stored_role = role_store.get(local_role.id)

    if stored_role is None:
        role_store.add(local_role)
    elif stored_role != local_role:
        role_store.update(local_role)

    persisted_local_role = role_store.get(local_role.id)

    if persisted_local_role is None:
        raise RuntimeError(
            "No se pudo recuperar el rol local desde SQLite."
        )

    assignment_store.add(
        ActorRoleAssignment(
            actor=local_console_actor,
            role=persisted_local_role,
        )
    )


    policy_store = ActionPolicyStore()

    for action_name in action_names:
        policy_store.register(
            ActionPolicy(
                action_name=action_name,
                permission=Permission(
                    resource=action_name,
                    action=PermissionAction.EXECUTE,
                ),
            )
        )

    authorization_service = AuthorizationService(
        assignment_store
    )

    action_authorization = ActionAuthorization(
        policy_store=policy_store,
        authorization=authorization_service,
    )

    manager = ActionManager(
        registry,
        action_authorization=action_authorization,
    )

    runtime.configure_actions(
        registry,
        manager
    )

    #
    # Tool Registry
    #

    tool_registry = ToolRegistry()

    tool_registry.register_many(
        provider_manager.tools()
    )

    tool_registry.register_many(
        registry.describe()
    )

    #
    # Exponer Tool Registry al Runtime
    #

    runtime.tool_registry = tool_registry

    #
    # Intent Providers
    #

    rule_provider = RuleIntentProvider(
        IntentRecognizer()
    )

    ollama_provider = OllamaIntentProvider(
        model="llama3:latest"
    )

    intent_provider = HybridIntentProvider(
        rule_provider=rule_provider,
        ollama_provider=ollama_provider,
    )

    #
    # Executor
    #

    executor = Executor(
        runtime
    )

    runtime.configure_executor(
        executor
    )

    session_manager = AssistantSessionManager()

    if assistant_session is not None:
        active_session = assistant_session

    elif assistant_context is not None:
        active_session = session_manager.start(
            assistant_context
        )

    else:
        active_session = None

    if active_session is not None:
        assistant_context = active_session.context

    #
    # Documents Capability
    #

    document_services = build_document_services()

    document_capability = DocumentCapability(
        context_access=document_services.context_access,
    )

    document_cognitive_access = DocumentCognitiveAccess(
        document_capability
    )

    #
    # Processes Capability
    #

    process_store = ProcessStore()

    process_resolver = ProcessResolver(
        process_store,
    )

    process_capability = ProcessCapability(
        store=process_store,
        resolver=process_resolver,
    )

    process_cognitive_access = ProcessCognitiveAccess(
        process_capability,
    )

    #
    # Rules Capability
    #

    rule_store = RuleStore()

    rule_resolver = RuleResolver(
        rule_store
    )

    rule_capability = RuleCapability(
        store=rule_store,
        resolver=rule_resolver,
    )

    rule_cognitive_access = RuleCognitiveAccess(
        rule_capability
    )

    #
    # Cognitive Capability Router
    #

    cognitive_capability_router = CognitiveCapabilityRouter(
        document_access=document_cognitive_access,
        process_access=process_cognitive_access,
        rule_access=rule_cognitive_access,
    )

    #
    # Engine
    #

    semantic_memory_db_path = os.environ.get(
        "NEXUS_SEMANTIC_MEMORY_DB_PATH",
        str(Path.home() / ".nexus" / "semantic_memory.db"),
    )
    semantic_memory = SemanticMemory(db_path=semantic_memory_db_path)

    return Engine(
        runtime,
        intent_provider=intent_provider,
        semantic_memory=semantic_memory,
        memory=memory,
        assistant_context=assistant_context,
        document_capability=document_capability,
        document_cognitive_access=document_cognitive_access,
        cognitive_capability_router=cognitive_capability_router,
        process_capability=process_capability,
        process_cognitive_access=process_cognitive_access,
        rule_capability=rule_capability,
        rule_cognitive_access=rule_cognitive_access,
        default_actor_id=security_actor.id,
        default_actor_type=security_actor.actor_type.value,
    )


def build_assistant_runtime(
    assistant_context: AssistantContext | None = None,
    assistant_session: AssistantSession | None = None,
) -> tuple[
    Engine,
    AssistantServices,
    AssistantServicesLifecycle,
]:
    """
    Construye el Engine cognitivo, los servicios del Assistant Layer
    y su lifecycle.

    El Engine, los servicios y su ciclo de vida mantienen
    responsabilidades separadas.
    """

    engine = build_engine(
        assistant_context=assistant_context,
        assistant_session=assistant_session,
    )

    services = build_assistant_services()

    lifecycle = AssistantServicesLifecycle(
        services
    )

    return engine, services, lifecycle


def _print_result(result) -> None:
    """
    Presenta el resultado de una ejecución
    de NEXUS de forma legible.
    """


    if result.success:

        print("NEXUS >", end=" ")

        response = (
            result.metadata.get("response")
            if isinstance(result.metadata, dict)
            else None
        )

        if response:
            print(response)
        elif result.message and result.message != "Execution completed successfully.":
            print(result.message)
        elif result.output:
            print(result.output)
        else:
            print("Listo.")

        return

    print(
        f"NEXUS > {result.message}"
    )



async def _interactive_loop(
    engine: Engine,
) -> None:
    """
    Loop interactivo persistente de NEXUS.

    El Engine y el event loop permanecen vivos
    durante toda la sesión.
    """

    print()

    print("Escribí una instrucción.")

    print("Escribí 'salir' para cerrar NEXUS.")

    print()

    while True:

        try:

            text = input(
                "Usuario > "
            ).strip()

        except (
            EOFError,
            KeyboardInterrupt,
        ):

            print()

            break

        if not text:
            continue

        if text.lower() in {
            "salir",
            "exit",
            "quit",
        }:

            break

        try:

            result = await engine.execute(
                text
            )

            _print_result(
                result
            )

        except Exception as exc:

            print(
                f"NEXUS > Error interno: {exc}"
            )

        print()



async def _main_async() -> None:
    """
    Inicializa NEXUS, mantiene una sesión interactiva
    persistente y gestiona el ciclo de vida de sus servicios.
    """

    print()
    print("=" * 50)
    print("          NEXUS Cognitive OS")
    print("=" * 50)
    print()

    engine, services, lifecycle = build_assistant_runtime()

    print("✓ Kernel")
    print("✓ Runtime")
    print("✓ Providers")
    print("✓ Actions")
    print("✓ Intent Providers")
    print("✓ Executor")
    print("✓ Engine")
    print("✓ Reminder Store")
    print("✓ Reminder Scheduler")
    print()

    try:
        await lifecycle.start()

        print("=" * 50)
        print("NEXUS READY")
        print("=" * 50)

        await _interactive_loop(engine)

    finally:
        await lifecycle.stop()

        close_semantic_memory = getattr(
            engine.semantic_memory,
            "close",
            None,
        )

        if callable(close_semantic_memory):
            close_semantic_memory()

        print()
        print("NEXUS detenido.")



def main() -> None:
    """
    Punto de entrada de NEXUS.
    """

    import asyncio

    asyncio.run(
        _main_async()
    )
