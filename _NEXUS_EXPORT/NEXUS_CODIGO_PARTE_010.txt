SYSTEM_PROMPT = """
Sos Nexus, un asistente de sistema local inteligente.

REGLAS ESTRICTAS:

SIEMPRE devolver SOLO uno de estos formatos:

Acción simple:
{ "action": "open_app", "value": "chrome" }

Texto:
{ "response": "texto" }

NUNCA usar:
- type
- data
- launch_app
- formatos inventados

Sé breve, directo y práctico.
"""


# =========================================================
# 🔥 Parser tolerante (corrige formatos raros del LLM)
# =========================================================
def normalize_result(result):

    if not isinstance(result, dict):
        return {"response": str(result)}

    # formato correcto
    if "action" in result or "response" in result:
        return result

    # formato inventado tipo:
    # { type: "LAUNCH_APP", data: { app_name: "Chrome" } }
    if "type" in result:
        t = str(result["type"]).lower()

        if "launch" in t or "open" in t:
            app = result.get("data", {}).get("app_name", "")
            return {"action": "open_app", "value": app}

    return {"response": str(result)}

# 🧠 Cerebro principal
# =========================================================
def ask_nexus(message: str, user=None):

    if user is None:
        user = {"id": "default"}

    user_id = user.get("id", "default")

    lower = message.lower()

    # =====================================================
    # 1️⃣ Tareas complejas → agente autónomo
    # =====================================================
    if any(x in lower for x in [
        "prepará", "analizá", "organiza todo",
        "hacé todo", "completo", "automáticamente"
    ]):
        result = run_autonomous(message, user)
        return {"response": result}

    # =====================================================
    # 2️⃣ Contextos inteligentes
    # =====================================================
    memory_context = search_memory(message)
    system_ctx = get_context()
    personal_memories = recall(message)
    semantic_mem = recall_semantic(user_id)

    # =====================================================
    # 3️⃣ Historial conversacional
    # =====================================================
    history = load_history()

    messages = [
        {"role": "system", "content": SYSTEM_PROMPT},
        {"role": "system", "content": f"Contexto del sistema:\n{system_ctx}"}
    ]

    if memory_context:
        messages.append({
            "role": "system",
            "content": f"Contexto de archivos:\n{memory_context}"
        })

    if personal_memories:
        messages.append({
            "role": "system",
            "content": f"Memoria personal relevante:\n{personal_memories}"
        })

    if semantic_mem:
        messages.append({
            "role": "system",
            "content": f"Memoria semántica del usuario:\n{semantic_mem}"
        })

    # últimos intercambios
    for h in history[-6:]:
        messages.append({"role": "user", "content": h["user"]})
        messages.append({"role": "assistant", "content": h["assistant"]})

    messages.append({"role": "user", "content": message})

    similar = search_similar(user.get("id","default"), message)

    if similar:
       memory_context = "\n".join([x[1] for x in similar])
       messages.append({
        "role": "system",
        "content": f"Memoria relevante:\n{memory_context}"
    })

    # =====================================================
    # 4️⃣ LLM
    # =====================================================
    try:
        res = ollama.chat(
            model="llama3",
            messages=messages
        )

        content = res["message"]["content"]

    except Exception as e:
        return {"response": f"Error LLM: {str(e)}"}
    
    save_vector_memory(user.get("id","default"), message)

    # =====================================================
    # 5️⃣ Parseo robusto
    # =====================================================
    try:
        parsed = json.loads(content)
        result = normalize_result(parsed)
    except Exception:
        result = {"response": content}

    # =====================================================
    # 6️⃣ Guardar historial
    # =====================================================
    history.append({
        "user": message,
        "assistant": result.get("response") or str(result)
    })

    save_history(history)

    # =====================================================
    # 7️⃣ Memoria personal
    # =====================================================
    if should_remember(message):
        save_memory(message)

    # =====================================================
    # 8️⃣ Aprendizaje semántico automático
    # =====================================================
    learn_from_interaction(
        user_id,
        message,
        result.get("response", "")
    )

    return result

import json
import ollama
from actions.system_actions import execute_action


SYSTEM = """
Sos Nexus Planner.

Convertí la orden del usuario en un PLAN JSON.

Formato:
{
  "steps": [
    {"action": "create_folder", "value": "PDF"},
    {"action": "move_files", "value": {"ext": ".pdf", "to": "PDF"}}
  ]
}

Acciones disponibles:
- open_app
- open_file
- create_folder
- move_files
- run_cmd
- close_active_app
- list_processes

Solo devolver JSON.
"""


def run_plan(user_message: str):

    res = ollama.chat(
        model="llama3",
        messages=[
            {"role": "system", "content": SYSTEM},
            {"role": "user", "content": user_message}
        ]
    )

    content = res["message"]["content"]

    try:
        plan = json.loads(content)
    except:
        return "No pude generar un plan."

    outputs = []

    for step in plan.get("steps", []):
        out = execute_action(step["action"], step["value"])
        outputs.append(out)

    return "\n".join(outputs)

        if len(history) < 5:
            return False

        # ejemplo simple: muchos errores seguidos
        errors = [h for h in history[-5:] if h.get("status") == "error"]

from brain.autonomous.agent import next_step
from actions.system_actions import execute_action
import time


MAX_STEPS = 10


def run_autonomous(goal, user=None):

    history = []
    step_count = 0

    while step_count < MAX_STEPS:

        plan = next_step(goal, history)

        if plan.get("done"):
            break

        steps = plan.get("steps", [])

        for s in steps:
            out = execute_action(s["action"], s["value"], user)
            history.append(out)

        step_count += 1
        time.sleep(1)

    return "\n".join(history)

def parse_local_command(text: str):

    # hora
    if "hora" in t:
        return {"action": "get_time"}

    # abrir app
    if t.startswith("abrir ") or t.startswith("abrí ") or t.startswith("open "):

        return {"action": "open_app", "value": app}

    return None

from collections import Counter
from core.database import get_connection


def analyze_patterns(limit=200):

    conn = get_connection()
    cur = conn.cursor()

    cur.execute("""
    SELECT step_name FROM step_executions
    ORDER BY started_at DESC
    LIMIT %s
    """, (limit,))

    rows = cur.fetchall()
    conn.close()

    steps = [r["step_name"] for r in rows]

    counter = Counter(steps)

    suggestions = []

    for step, count in counter.items():

        if count >= 5:
            suggestions.append({
                "type": "repeated_step",
                "step": step,
                "count": count,
                "message": f"Detecté que ejecutás '{step}' {count} veces. ¿Querés automatizarlo?"
            })

    return suggestions

import json
import math

from core.database import get_connection
from brain.semantic.embedder import embed


# ==========================================================
# COSINE SIMILARITY
# ==========================================================

def cosine(a, b):
    dot = sum(x * y for x, y in zip(a, b))
    mag_a = math.sqrt(sum(x * x for x in a))
    mag_b = math.sqrt(sum(x * x for x in b))
    return dot / (mag_a * mag_b + 1e-8)


# ==========================================================
# SEARCH SIMILAR
# ==========================================================

def search_similar(user_id, query, limit=5):

    q_vec = embed(query)

    conn = get_connection()
    cur = conn.cursor()

    cur.execute("""
        SELECT id, content, embedding, confidence
        FROM semantic_vectors
        WHERE user_id = %s
    """, (user_id,))

    rows = cur.fetchall()
    conn.close()

    scored = []

    for row in rows:
        emb = json.loads(row[2])  # embedding guardado como JSON
        confidence = row[3] if row[3] else 1.0

        score = cosine(q_vec, emb) * confidence

        scored.append({
            "id": row[0],
            "content": row[1],
            "score": score
        })

    # ordenar por score
    scored.sort(key=lambda x: x["score"], reverse=True)

    return scored[:limit]

from datetime import datetime, timezone

from core.database import get_connection


def save_semantic(user_id, key, value, confidence=0.6):

    conn = get_connection()
    cur = conn.cursor()

    # 🔍 Buscar existente
    cur.execute("""
        SELECT id, confidence
        FROM semantic_memory
        WHERE user_id = %s AND key = %s
    """, (user_id, key))

    row = cur.fetchone()

    now = datetime.now(timezone.utc)

    if row:
        semantic_id = row[0]
        current_conf = row[1]

        new_conf = min(1.0, current_conf + 0.1)

        cur.execute("""
            UPDATE semantic_memory
            SET value = %s,
                confidence = %s,
                updated_at = %s
            WHERE id = %s
        """, (value, new_conf, now, semantic_id))

    else:
        cur.execute("""
            INSERT INTO semantic_memory
            (user_id, key, value, confidence, created_at, updated_at)
            VALUES (%s, %s, %s, %s, %s, %s)
        """, (user_id, key, value, confidence, now, now))

    conn.commit()
    conn.close()

    now = datetime.now(timezone.utc)

    cur.execute("""
        INSERT INTO semantic_vectors

    """, (
        user_id,
        content,
        json.dumps(vec),
        0.6,
        now,
        now
    ))

    conn.commit()
    conn.close()

from core.database import get_connection


def analyze_workflows():

    conn = get_connection()
    cur = conn.cursor()

    cur.execute("""
        SELECT step, status, COUNT(*) as total
        FROM job_steps
        GROUP BY step, status
    """)

    rows = cur.fetchall()
    conn.close()

    problems = []

    for r in rows:

        step = r["step"]
        status = r["status"]
        total = r["total"]

        if status == "error" and total > 2:

            problems.append({
                "step": step,
                "issue": "step falla frecuentemente",
                "count": total
            })

    return problems

def running_processes(limit=10):

    procs = []

    for p in psutil.process_iter(['name', 'cpu_percent']):
        try:
            procs.append((p.info['name'], p.info['cpu_percent']))
        except Exception:
            pass

    procs.sort(key=lambda x: x[1], reverse=True)

    return [p[0] for p in procs[:limit]]

# ==========================================================
# PARSE SCHEDULE FROM NATURAL LANGUAGE
# ==========================================================

def parse_schedule(prompt):

    prompt = prompt.lower()

    # cada X minutos
    m = re.search(r"cada (\d+) minutos?", prompt)
    if m:
        return int(m.group(1)) * 60

    # cada X horas
    m = re.search(r"cada (\d+) horas?", prompt)
    if m:
        return int(m.group(1)) * 3600

    # cada hora
    if "cada hora" in prompt:
        return 3600

    # cada día
    if "cada día" in prompt or "cada dia" in prompt:
        return 86400

    return None


# ==========================================================
# COPILOT ENGINE
# ==========================================================

class CopilotEngine:

    def interpret(self, prompt):

        interval = parse_schedule(prompt)

        full_prompt = SYSTEM_PROMPT + "\n\n" + prompt

        r = requests.post(
            OLLAMA_URL,
            json={
                "model": "llama3",
                "prompt": full_prompt,
                "stream": False
            }
        )

        data = r.json()

        content = data.get("response", "")

        try:

            workflow = json.loads(content)

            # si detectamos intervalo temporal
            if interval:
                workflow["trigger"] = "schedule"
                workflow["interval"] = interval

            return workflow

        except Exception:
            return None

from core.database import get_connection


class CopilotDebugger:

    def analyze_last_failure(self):

        conn = get_connection()
        cur = conn.cursor()

        # último job fallido
        cur.execute("""
            SELECT id, workflow, result, updated_at
            FROM jobs
            WHERE status='error'
            ORDER BY updated_at DESC
            LIMIT 1
        """)

        job = cur.fetchone()

        if not job:
            conn.close()
            return {
                "message": "No encontré workflows fallidos recientemente."
            }

        job_id = job["id"]

        # pasos del job
        cur.execute("""
            SELECT step, status, result
            FROM job_steps
            WHERE job_id=%s
        """,(job_id,))

        steps = cur.fetchall()

        conn.close()

        failed_step = None

        for s in steps:
            if s["status"] == "error":
                failed_step = s
                break

        if failed_step:

            return {
                "workflow": job["workflow"],
                "message": f"El workflow falló en el paso '{failed_step['step']}'.",
                "error": failed_step["result"]
            }

        return {
            "workflow": job["workflow"],
            "message": "El workflow falló pero no encontré el paso exacto."
        }

from core.database import get_connection


class CopilotFixer:

    def propose_fix(self):

        conn = get_connection()
        cur = conn.cursor()

        # último job fallido
        cur.execute("""
            SELECT id, workflow, result
            FROM jobs
            WHERE status='error'
            ORDER BY updated_at DESC
            LIMIT 1
        """)

        job = cur.fetchone()

        if not job:
            conn.close()
            return {
                "message": "No encontré errores recientes."
            }

        error_text = job["result"] or ""

        suggestion = None

        # =========================
        # REGLAS DE CORRECCIÓN
        # =========================

        if "File not found" in error_text:
            suggestion = {
                "action": "add_step",
                "step": {
                    "type": "action",
                    "action": "create_folder",
                    "value": "missing_folder"
                },
                "reason": "El workflow intenta leer un archivo que no existe."
            }

        elif "Ruta no permitida" in error_text:
            suggestion = {
                "action": "change_path",
                "reason": "La ruta está bloqueada por sandbox."
            }

        elif "Workflow no encontrado" in error_text:
            suggestion = {
                "action": "verify_workflow_name",
                "reason": "El workflow solicitado no existe en la base de datos."
            }

        else:
            suggestion = {
                "action": "manual_review",
                "reason": "No pude determinar automáticamente la causa."
            }

        conn.close()

        return {
            "workflow": job["workflow"],
            "error": error_text,
            "suggestion": suggestion
        }

from pywinauto import Desktop



def list_ui_elements():

    windows = Desktop(
        backend="uia"
    ).windows()

    result = []

    for w in windows:

        try:

            title = w.window_text()

            rect = w.rectangle()

            result.append({

                "title": title,

                "x": rect.left,

                "y": rect.top,

                "w": rect.width(),

                "h": rect.height()
            })

        except:
            pass

    return result

def click_ui_element(
    title
):

    windows = Desktop(
        backend="uia"
    ).windows()

    for w in windows:

        try:

            if title.lower() in (
                w.window_text().lower()
            ):

                w.set_focus()

                return True

        except:
            pass

    return False

    try:

        }

    except:

        return None

import uuid
import json

from core.database import get_connection


def send_message(

    sender,
    receiver,
    message,
    metadata=None
):

    conn = get_connection()
    cur = conn.cursor()

    cur.execute("""
        INSERT INTO agent_messages
        (
            id,
            sender,
            receiver,
            message,
            metadata
        )
        VALUES (%s,%s,%s,%s,%s)
    """, (
        str(uuid.uuid4()),
        sender,
        receiver,
        message,
        json.dumps(metadata)
        if metadata
        else None
    ))

    conn.commit()
    conn.close()


def get_messages(
    receiver,
    limit=20
):

    conn = get_connection()
    cur = conn.cursor()

    cur.execute("""
        SELECT *
        FROM agent_messages
        WHERE receiver = %s
        ORDER BY created_at DESC
        LIMIT %s
    """, (
        receiver,
        limit
    ))

    rows = cur.fetchall()

    conn.close()

    return [dict(r) for r in rows]
