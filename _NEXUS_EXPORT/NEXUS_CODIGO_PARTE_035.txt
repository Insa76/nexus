from pathlib import Path
import zipfile

ROOT = Path(__file__).resolve().parent
OUTPUT = ROOT.parent / "NEXUS_APPS_PARTES"
OUTPUT.mkdir(exist_ok=True)

EXCLUDED = {
    "node_modules", "dist", "build", ".git",
    ".next", ".venv", "venv", "__pycache__",
    ".pytest_cache", "coverage", ".turbo"
}

EXCLUDED_EXT = {
    ".pyc", ".zip", ".7z", ".rar", ".gz"
}

MAX_BYTES = 35 * 1024 * 1024

APP_DIRS = [
    "apps", "backend", "dashboard", "desktop",
    "electron", "landing", "nexus-mobile"
]

files = []

for folder in APP_DIRS:
    base = ROOT / folder
    if not base.exists():
        continue

    for p in base.rglob("*"):
        if not p.is_file():
            continue

        rel = p.relative_to(ROOT)

        if any(part in EXCLUDED for part in rel.parts):
            continue

        if p.suffix.lower() in EXCLUDED_EXT:
            continue

        if p.name == ".env":
            continue

        if p.name.startswith(".env.") and p.name != ".env.example":
            continue

        files.append((p, rel, p.stat().st_size))

parts = []
current = []
current_size = 0

for item in files:
    path, rel, size = item

    if current and current_size + size > MAX_BYTES:
        parts.append(current)
        current = []
        current_size = 0

    current.append(item)
    current_size += size

if current:
    parts.append(current)

for index, group in enumerate(parts, 1):
    destination = OUTPUT / f"NEXUS_APPS_PARTE_{index:02}.zip"

    with zipfile.ZipFile(
        destination, "w",
        compression=zipfile.ZIP_DEFLATED,
        compresslevel=6
    ) as z:
        for path, rel, size in group:
            z.write(path, rel)

    print(
        f"{destination.name} | "
        f"{destination.stat().st_size / 1024 / 1024:.1f} MB | "
        f"{len(group)} archivos"
    )

print(f"\nTotal de partes: {len(parts)}")
print(f"Carpeta: {OUTPUT}")

from pathlib import Path
import os

ROOT = Path(__file__).resolve().parent
OUTPUT = ROOT / "NEXUS_APPS_TEXTO"

FOLDERS = [
    "apps",
    "backend",
    "dashboard",
    "desktop",
    "electron",
    "landing",
    "nexus-mobile",
]

EXCLUDED_DIRS = {
    ".git", "node_modules", "dist", "build", ".next",
    ".venv", "venv", "__pycache__", ".pytest_cache",
    "coverage", ".turbo", ".cache", "vendor",
}

EXCLUDED_FILES = {
    ".env", ".env.local", ".env.production",
    ".env.development", ".env.test",
}

EXTENSIONS = {
    ".py", ".js", ".jsx", ".ts", ".tsx",
    ".html", ".css", ".scss", ".sass",
    ".json", ".jsonc", ".md", ".txt",
    ".yml", ".yaml", ".toml", ".ini",
    ".cfg", ".conf", ".sql", ".sh",
    ".bat", ".ps1", ".xml", ".svg",
    ".prisma", ".graphql", ".gql",
    ".vue", ".svelte", ".mjs", ".cjs",
    ".properties", ".gitignore", ".editorconfig",
    ".dockerignore", ".example",
}

SPECIAL_FILES = {
    "Dockerfile", "Makefile", "Procfile",
    "LICENSE", "README", "CHANGELOG",
    "requirements", "Pipfile", "Justfile",
}

MAX_CHARS = 350_000
PART = 0
CURRENT = []
CURRENT_SIZE = 0
TOTAL_FILES = 0
SKIPPED = 0


def write_part():
    global PART, CURRENT, CURRENT_SIZE

    if not CURRENT:
        return

    PART += 1
    path = OUTPUT / f"NEXUS_APPS_TEXTO_{PART:03}.txt"

    content = "\n".join(CURRENT)
    path.write_text(content, encoding="utf-8")

    print(f"Creado: {path.name} ({len(content):,} caracteres)")

    CURRENT = []
    CURRENT_SIZE = 0


def add_text(text):
    global CURRENT_SIZE

    while text:
        available = MAX_CHARS - CURRENT_SIZE

        if available <= 0:
            write_part()
            available = MAX_CHARS

        piece = text[:available]
        CURRENT.append(piece)
        CURRENT_SIZE += len(piece)
        text = text[available:]

        if CURRENT_SIZE >= MAX_CHARS:
            write_part()


def read_file(path):
    data = path.read_bytes()

    if b"\x00" in data:
        return None

    for encoding in ("utf-8-sig", "utf-8", "cp1252"):
        try:
            return data.decode(encoding)
        except UnicodeDecodeError:
            continue

    return None


def main():
    global TOTAL_FILES, SKIPPED

    OUTPUT.mkdir(exist_ok=True)

    add_text(
        "NEXUS - EXPORTACION DE CODIGO FUENTE\n"
        "Formato: texto UTF-8\n"
        "Cada archivo conserva su ruta original.\n\n"
    )

    for folder in FOLDERS:
        base = ROOT / folder

        if not base.exists():
            print(f"No encontrada: {folder}")
            continue

        print(f"Procesando: {folder}")

        for current_root, dirs, files in os.walk(base):
            dirs[:] = sorted(
                d for d in dirs
                if d not in EXCLUDED_DIRS
            )

            for filename in sorted(files):
                path = Path(current_root) / filename

                if filename in EXCLUDED_FILES:
                    SKIPPED += 1
                    continue

                if filename.startswith(".env."):
                    SKIPPED += 1
                    continue

                if (
                    path.suffix.lower() not in EXTENSIONS
                    and filename not in SPECIAL_FILES
                ):
                    SKIPPED += 1
                    continue

                try:
                    content = read_file(path)

                    if content is None:
                        SKIPPED += 1
                        continue

                    relative = path.relative_to(ROOT).as_posix()

                    header = (
                        "\n\n"
                        + "=" * 75 + "\n"
                        + f"ARCHIVO: {relative}\n"
                        + "=" * 75 + "\n\n"
                    )

                    add_text(header)
                    add_text(content)
                    add_text(
                        f"\n\n"
                        f"--- FIN DEL ARCHIVO: {relative} ---\n"
                    )

                    TOTAL_FILES += 1

                except Exception as error:
                    print(f"Error leyendo {path}: {error}")
                    SKIPPED += 1

    write_part()

    summary = (
        f"Archivos exportados: {TOTAL_FILES}\n"
        f"Archivos excluidos o no legibles: {SKIPPED}\n"
        f"Partes generadas: {PART}\n"
    )

    (OUTPUT / "RESUMEN.txt").write_text(
        summary, encoding="utf-8"
    )

    print("\nEXPORTACION FINALIZADA")
    print(summary)
    print(f"Resultado: {OUTPUT}")


if __name__ == "__main__":
    main()

from pathlib import Path
import zipfile
import subprocess

ROOT = Path(__file__).resolve().parent
OUTPUT = ROOT.parent / "NEXUS_ENTREGA"

OUTPUT.mkdir(exist_ok=True)

EXCLUDED_DIRS = {
    ".git", ".venv", "venv", "node_modules",
    "__pycache__", ".pytest_cache", ".mypy_cache",
    ".ruff_cache", ".next", ".turbo", ".cache",
    "coverage", "htmlcov"
}

EXCLUDED_SUFFIXES = {
    ".pyc", ".pyo", ".zip", ".7z", ".rar",
    ".tar", ".gz"
}

GROUPS = {
    "01_CORE": ["packages", "tests", "docs"],
    "02_APPS": [
        "apps", "backend", "dashboard",
        "desktop", "electron", "landing", "nexus-mobile"
    ],
}

def include_file(path):
    if any(part in EXCLUDED_DIRS for part in path.parts):
        return False

    if path.suffix.lower() in EXCLUDED_SUFFIXES:
        return False

    if path.name == ".env":
        return False

    if path.name.startswith(".env.") and path.name != ".env.example":
        return False

    return True

def add_path(archive, path):
    if not path.exists():
        return

    if path.is_file():
        if include_file(path.relative_to(ROOT)):
            archive.write(path, path.relative_to(ROOT))
        return

    for item in path.rglob("*"):
        if item.is_file() and include_file(item.relative_to(ROOT)):
            archive.write(item, item.relative_to(ROOT))

def create_archive(name, paths):
    destination = OUTPUT / f"NEXUS_{name}.zip"

    with zipfile.ZipFile(
        destination, "w", compression=zipfile.ZIP_DEFLATED,
        compresslevel=6
    ) as archive:
        for relative in paths:
            add_path(archive, ROOT / relative)

    print(f"CREADO: {destination} ({destination.stat().st_size / 1024 / 1024:.1f} MB)")

# Paquete 1: motor, arquitectura y pruebas
create_archive("01_CORE", GROUPS["01_CORE"])

# Paquete 2: aplicaciones e interfaces
create_archive("02_APPS", GROUPS["02_APPS"])

# Paquete 3: todo el resto del proyecto y estado Git
excluded = set(GROUPS["01_CORE"] + GROUPS["02_APPS"])

remaining = [
    item.name for item in ROOT.iterdir()
    if item.name not in excluded
    and item.name != "exportar_nexus.py"
]

create_archive("03_RESTO", remaining)

# Estado Git para preservar los cambios locales
try:
    commands = [
        ["git", "status", "--short", "--branch"],
        ["git", "log", "-5", "--oneline"],
        ["git", "diff", "--stat"]
    ]

    report = []

    for command in commands:
        result = subprocess.run(
            command, cwd=ROOT, capture_output=True,
            text=True, encoding="utf-8", errors="replace"
        )
        report.append("$ " + " ".join(command))
        report.append(result.stdout)
        report.append(result.stderr)

    (OUTPUT / "NEXUS_GIT_STATE.txt").write_text(
        "\n".join(report), encoding="utf-8"
    )

except Exception as error:
    print("No se pudo obtener el estado Git:", error)

print("\nEXPORTACIÓN TERMINADA.")
print("Carpeta:", OUTPUT)

        $Block = @"

================================================================================
ARCHIVO: $Relative

$Content

================================================================================
FIN DEL ARCHIVO
================================================================================

"@

function escapeHtml(value) {

    return String(value)
        .replaceAll("&", "&amp;")
        .replaceAll("<", "&lt;")
        .replaceAll(">", "&gt;")
        .replaceAll('"', "&quot;")
        .replaceAll("'", "&#039;");
}


function pretty(value) {

    if (
        value === null ||
        value === undefined
    ) {
        return "null";
    }

    if (
        typeof value === "string"
    ) {
        return value;
    }

    return JSON.stringify(
        value,
        null,
        2
    );
}


function addMessage(
    type,
    content
) {

    const empty =
        conversation.querySelector(
            ".empty"
        );

    if (empty) {
        empty.remove();
    }

    const message =
        document.createElement("div");

    message.className =
        `message ${type}`;

    const label =
        type === "user"
            ? "Usuario"
            : "NEXUS";

    message.innerHTML = `
        <div class="message-label">
            ${label}
        </div>

        <div class="message-content">
            ${escapeHtml(content)}
        </div>
    `;

    conversation.appendChild(
        message
    );

    conversation.scrollTop =
        conversation.scrollHeight;
}


function renderTrace(trace) {

    traceContainer.innerHTML = "";

    const steps = [
        {
            key: "request",
            label: "Request",
        },
        {
        key: "context",
        label: "Cognitive Context",
        },
        {
            key: "intent",
            label: "Intent",
        },
        {
            key: "objective",
            label: "Objective",
        },
        {
            key: "plan",
            label: "Plan",
        },
        {
            key: "plan_final",
            label: "Final Plan",
        },
        {
            key: "result",
            label: "Runtime / Result",
        },
        {
            key: "error",
            label: "Error",
        },
    ];


    let rendered = 0;


    for (const step of steps) {

        const data =
            trace?.[step.key];

        if (
            data === null ||
            data === undefined
        ) {
            continue;
        }

        rendered++;

        const isError =
            step.key === "error";

        const element =
            document.createElement("div");

        element.className =
            `trace-step ${
                isError
                    ? "error"
                    : "success"
            }`;

        element.innerHTML = `
            <div class="trace-step-header">

                <div class="trace-name">

                    <span
                        class="${
                            isError
                                ? "trace-error"
                                : "trace-icon"
                        }"
                    >
                        ${
                            isError
                                ? "✕"
                                : "✓"
                        }
                    </span>

                    ${step.label}

                </div>

            </div>

            <pre class="trace-data">${escapeHtml(
                pretty(data)
            )}</pre>
        `;

        traceContainer.appendChild(
            element
        );
    }


    if (!rendered) {

        traceContainer.innerHTML = `
            <div class="empty">
                No hay datos de ejecución.
            </div>
        `;

        traceStatus.textContent =
            "Sin datos";

        return;
    }


    traceStatus.textContent =
        `${rendered} etapas`;
}


async function execute() {

    const text =
        commandInput.value.trim();

    if (!text) {
        return;
    }


    addMessage(
        "user",
        text
    );


    commandInput.value = "";

    executeButton.disabled = true;

    executeButton.textContent =
        "Ejecutando...";

    traceStatus.textContent =
        "Ejecutando";


    try {

        const response =
            await fetch(
                "/api/execute",
                {
                    method: "POST",

                    headers: {
                        "Content-Type":
                            "application/json",
                    },

                    body: JSON.stringify({
                        text,
                    }),
                }
            );


        const data =
            await response.json();


        if (
            data.success
        ) {

            addMessage(
                "nexus",
                pretty(data.output)
            );

        } else {

            addMessage(
                "nexus",
                data.message ||
                    "NEXUS devolvió un error."
            );
        }


        if (data.trace) {

            renderTrace(
                data.trace
            );
        }


        const memory =
            data.memory_count ?? 0;

        memoryHeader.textContent =
            `Memory: ${memory}`;

        memoryValue.textContent =
            `${memory} entries`;


    } catch (error) {

        addMessage(
            "nexus",
            `Error de conexión: ${error.message}`
        );

        traceStatus.textContent =
            "Error";


    } finally {

        executeButton.disabled = false;

        executeButton.textContent =
            "Ejecutar";
    }
}


async function loadStatus() {

    try {

        const response =
            await fetch(
                "/api/status"
            );

        const data =
            await response.json();


        if (
            data.status === "online"
        ) {

            statusText.textContent =
                "NEXUS online";

        }

        const memory =
            data.memory ?? 0;

        memoryHeader.textContent =
            `Memory: ${memory}`;

        memoryValue.textContent =
            `${memory} entries`;


    } catch {

        statusText.textContent =
            "NEXUS offline";
    }
}


executeButton.addEventListener(
    "click",
    execute
);


clearButton.addEventListener(

        conversation.innerHTML = `

                Todavía no hay ejecuciones.
            </div>

        traceContainer.innerHTML = `

                Ejecutá una instrucción para
                visualizar el recorrido interno
                de NEXUS.
            </div>
        `;

        traceStatus.textContent =
            "Esperando ejecución";
    }
);


commandInput.addEventListener(
    "keydown",
    (event) => {

        if (
            event.key === "Enter" &&
            (event.ctrlKey || event.metaKey)
        ) {

            execute();
        }
    }
);


loadStatus();
