from fastapi import Header, HTTPException, Depends
from jose import JWTError

from auth.security import decode_token


def get_current_user(authorization: str = Header(None)):

    if not authorization:
        raise HTTPException(status_code=401, detail="Token requerido")

    if not authorization.startswith("Bearer "):
        raise HTTPException(status_code=401, detail="Formato de token inválido")

    token = authorization.split(" ")[1]

    try:
        payload = decode_token(token)
        return payload

    except JWTError:
        raise HTTPException(status_code=401, detail="Token inválido")
    
def require_admin(user=Depends(get_current_user)):

    if user["role"] != "admin":
        raise HTTPException(status_code=403, detail="Admin only")

    return user

# ======================================================

SECRET = "nexus-super-secret"  # 🔥 MISMO QUE ANTES
ALGORITHM = "HS256"
TOKEN_EXPIRE_HOURS = 8

pwd = CryptContext(
    schemes=["argon2"],
    deprecated="auto"
)

# ======================================================
# PASSWORDS
# ======================================================

def hash_password(password: str) -> str:
    return pwd.hash(password)


def verify_password(password: str, hashed: str) -> bool:
    return pwd.verify(password, hashed)


# ======================================================
# JWT
# ======================================================

def create_token(data: dict) -> str:

    payload = data.copy()
    payload["exp"] = datetime.utcnow() + timedelta(hours=TOKEN_EXPIRE_HOURS)

    return jwt.encode(payload, SECRET, algorithm=ALGORITHM)


def decode_token(token: str) -> dict:

    try:
        return jwt.decode(token, SECRET, algorithms=[ALGORITHM])
    except JWTError:
        return None

def normalize_result(result):

    if not isinstance(result, dict):
        return {"response": str(result)}

    # formato correcto
    if "action" in result or "response" in result:
        return result

    # formato inventado tipo:
    # { type: "LAUNCH_APP", data: { app_name: "Chrome" } }
    if "type" in result:
        t = str(result["type"]).lower()

        if "launch" in t or "open" in t:
            app = result.get("data", {}).get("app_name", "")
            return {"action": "open_app", "value": app}

    return {"response": str(result)}

def ask_nexus(message: str, user=None):

    if user is None:
        user = {"id": "default"}

    lower = message.lower()

    if any(x in lower for x in [
        "prepará", "analizá", "organiza todo",
        "hacé todo", "completo", "automáticamente"
    ]):
        result = run_autonomous(message, user)
        return {"response": result}

    # =====================================================

    # =====================================================
    memory_context = search_memory(message)

    personal_memories = recall(message)
    semantic_mem = recall_semantic(user_id)

    # =====================================================
    # 3️⃣ Historial conversacional
    # =====================================================
    history = load_history()

    messages = [
        {"role": "system", "content": SYSTEM_PROMPT},
        {"role": "system", "content": f"Contexto del sistema:\n{system_ctx}"}
    ]

    if memory_context:
        messages.append({
            "role": "system",
            "content": f"Contexto de archivos:\n{memory_context}"
        })

    if personal_memories:
        messages.append({
            "role": "system",
            "content": f"Memoria personal relevante:\n{personal_memories}"
        })

    if semantic_mem:
        messages.append({
            "role": "system",
            "content": f"Memoria semántica del usuario:\n{semantic_mem}"
        })

    # últimos intercambios
    for h in history[-6:]:
        messages.append({"role": "user", "content": h["user"]})
        messages.append({"role": "assistant", "content": h["assistant"]})

    messages.append({"role": "user", "content": message})

    similar = search_similar(user.get("id","default"), message)

    if similar:
       memory_context = "\n".join([x[1] for x in similar])
       messages.append({
        "role": "system",
        "content": f"Memoria relevante:\n{memory_context}"
    })

    # =====================================================
    # 4️⃣ LLM
    # =====================================================
    try:
        res = ollama.chat(
            model="llama3",

        )

        content = res["message"]["content"]

    except Exception as e:
        return {"response": f"Error LLM: {str(e)}"}
    
    save_vector_memory(user.get("id","default"), message)

    # =====================================================
    # 5️⃣ Parseo robusto
    # =====================================================
    try:
        parsed = json.loads(content)
        result = normalize_result(parsed)

        result = {"response": content}

    # =====================================================
    # 6️⃣ Guardar historial
    # =====================================================
    history.append({
        "user": message,
        "assistant": result.get("response") or str(result)
    })

    if should_remember(message):
        save_memory(message)

    # =====================================================
    learn_from_interaction(
        user_id,
        message,
        result.get("response", "")
    )

    return result

        # ejemplo simple: muchos errores seguidos
        errors = [h for h in history[-5:] if h.get("status") == "error"]

        if len(errors) >= 3:

        return False

import json
import math

from core.database import get_connection
from brain.semantic.embedder import embed


# ==========================================================
# COSINE SIMILARITY
# ==========================================================

def cosine(a, b):
    dot = sum(x * y for x, y in zip(a, b))
    mag_a = math.sqrt(sum(x * x for x in a))
    mag_b = math.sqrt(sum(x * x for x in b))
    return dot / (mag_a * mag_b + 1e-8)


# ==========================================================
# SEARCH SIMILAR
# ==========================================================

def search_similar(user_id, query, limit=5):

    q_vec = embed(query)

    conn = get_connection()
    cur = conn.cursor()

    cur.execute("""
        SELECT id, content, embedding, confidence
        FROM semantic_vectors
        WHERE user_id = %s
    """, (user_id,))

    rows = cur.fetchall()
    conn.close()

    scored = []

    for row in rows:
        emb = json.loads(row[2])  # embedding guardado como JSON
        confidence = row[3] if row[3] else 1.0

        score = cosine(q_vec, emb) * confidence

        scored.append({
            "id": row[0],
            "content": row[1],
            "score": score
        })

    # ordenar por score
    scored.sort(key=lambda x: x["score"], reverse=True)

    return scored[:limit]

# ==========================================================
# PARSE SCHEDULE FROM NATURAL LANGUAGE
# ==========================================================

def parse_schedule(prompt):

    prompt = prompt.lower()

    # cada X minutos
    m = re.search(r"cada (\d+) minutos?", prompt)
    if m:
        return int(m.group(1)) * 60

    # cada X horas
    m = re.search(r"cada (\d+) horas?", prompt)
    if m:
        return int(m.group(1)) * 3600

    # cada hora
    if "cada hora" in prompt:
        return 3600

    # cada día
    if "cada día" in prompt or "cada dia" in prompt:
        return 86400

    return None


# ==========================================================
# COPILOT ENGINE
# ==========================================================

class CopilotEngine:

    def interpret(self, prompt):

        interval = parse_schedule(prompt)

        full_prompt = SYSTEM_PROMPT + "\n\n" + prompt

        r = requests.post(
            OLLAMA_URL,
            json={
                "model": "llama3",
                "prompt": full_prompt,
                "stream": False
            }
        )

        data = r.json()

        content = data.get("response", "")

        try:

            workflow = json.loads(content)

            # si detectamos intervalo temporal
            if interval:

            WHERE job_id=%s

                "workflow": job["workflow"],
                "message": f"El workflow falló en el paso '{failed_step['step']}'.",
                "error": failed_step["result"]
            }

            "workflow": job["workflow"],

class AnomalyDetector:

    def detect(self, history):

        if len(history) < 5:
            return None

        last = history[-1]
        prev = history[-5]

        # error spike
        if last["error_rate"] > prev["error_rate"] * 2 and last["error_rate"] > 0.2:
            return {
                "type": "error_spike",
                "message": "Estoy observando un aumento anormal de errores."
            }

        # queue growth
        if last["queue_size"] > prev["queue_size"] + 10:
            return {
                "type": "queue_growth",
                "message": "La cola de trabajos está creciendo rápidamente."
            }

        if len(last["open_circuits"]) >= 2:
            return {
                "type": "circuit_alert",
                "message": "Varios circuit breakers están abiertos."
            }

        return None

import math


def fake_embedding(text):

    text = text.lower()

    vector = []

    for c in text[:128]:

        vector.append(
            ord(c) / 255
        )

    while len(vector) < 128:
        vector.append(0)

    return vector


def cosine_similarity(a, b):

    dot = sum(x*y for x,y in zip(a,b))

    norm_a = math.sqrt(
        sum(x*x for x in a)
    )

    norm_b = math.sqrt(
        sum(x*x for x in b)
    )

    if norm_a == 0 or norm_b == 0:
        return 0

    return dot / (
        norm_a * norm_b
    )

class JsonFormatter(logging.Formatter):
    def format(self, record):

        log_record = {
            "timestamp": datetime.utcnow().isoformat(),
            "level": record.levelname,
            "message": record.getMessage(),
            "module": record.module,
        }

        if hasattr(record, "extra_data"):
            log_record["data"] = record.extra_data

        return json.dumps(log_record)

import time
from datetime import datetime


class Metrics:

    def __init__(self):
        self.start_time = time.time()

        # contadores
        self.counters = {}

        # histogramas
        self.histograms = {}

        # gauges (estado actual)
        self.gauges = {}

        # métricas globales
        self.total_workflows = 0
        self.total_steps = 0
        self.total_jobs = 0
        self.total_errors = 0
        self.active_jobs = 0

    # =========================
    # COUNTERS
    # =========================

    def inc(self, name):
        self.counters[name] = self.counters.get(name, 0) + 1

    def inc_job(self):
        self.inc("jobs_created")
        self.total_jobs += 1

    def inc_error(self):
        self.inc("jobs_failed")
        self.total_errors += 1

    # =========================
    # HISTOGRAMS
    # =========================

    def observe(self, name, value):
        if name not in self.histograms:
            self.histograms[name] = []

        self.histograms[name].append(value)

    # =========================
    # GAUGES
    # =========================

    def set_active_jobs(self, value):
        self.active_jobs = value
        self.gauges["active_jobs"] = value

    def set_gauge(self, name, value):
        self.gauges[name] = value

    # =========================
    # WORKFLOW METRICS
    # =========================

    def inc_workflow(self):
        self.total_workflows += 1

    def inc_step(self):
        self.total_steps += 1

    # =========================
    # UPTIME
    # =========================

    def uptime(self):
        return round(time.time() - self.start_time)

    # =========================
    # SNAPSHOT
    # =========================

    def snapshot(self):

        return {
            "uptime_seconds": self.uptime(),
            "total_workflows_executed": self.total_workflows,
            "total_steps_executed": self.total_steps,
            "total_jobs_created": self.total_jobs,
            "total_errors": self.total_errors,
            "active_jobs": self.active_jobs,
            "timestamp": datetime.utcnow().isoformat()
        }

    # =========================
    # CONCURRENCY INFO
    # =========================

    def concurrency_info(self, queue_size, workers):

        return {
            "queue_size": queue_size,
            "max_workers": workers
        }


metrics = Metrics()

    cur = conn.cursor()

    cur.execute("""
        INSERT INTO missions
        (
            id,
            goal,
            status
        )
        VALUES (%s,%s,%s)
    """, (
        mission_id,
        goal,
        "planning"
    ))

    conn.commit()

    # =========================
    # GENERATE DAG
    # =========================
    workflow = generate_workflow(
        goal,
        model
    )

    # =========================
    # EXECUTE
    # =========================
    job_id = await manager.create_job(
        workflow,
        "system"
    )

    # =========================
    # UPDATE
    # =========================
    cur.execute("""
        UPDATE missions
        SET status = %s,
            result = %s
        WHERE id = %s
    """, (
        "running",
        str({
            "job_id": job_id
        }),
        mission_id
    ))

    conn.commit()
    conn.close()

    return {
        "mission_id": mission_id,
        "job_id": job_id

model = YOLO("yolov8n.pt")


def detect_objects(
    image_path
):

    results = model(image_path)

    detections = []

    for r in results:

        boxes = r.boxes

        for box in boxes:

            cls = int(box.cls[0])

            conf = float(box.conf[0])

            coords = box.xyxy[0].tolist()

            label = model.names[cls]

            detections.append({

                "label": label,

                "confidence": conf,

                "x1": coords[0],

                "y1": coords[1],

                "x2": coords[2],

                "y2": coords[3]
            })

    return detections

BLOCKED_PATHS = [

    "C:/Windows",
    "C:/Program Files",
    "C:/Users"
]

BLOCKED_DOMAINS = [

    "malware",
    "phishing",
    "tor"
]

MAX_AUTONOMOUS_STEPS = 20


def validate_action(
    action,
    params
):

    # =========================
    # BLOCK DELETE
    # =========================
    if action == "delete_files":

        return False, "delete_files blocked"

    # =========================
    # VALIDATE PATHS
    # =========================
    path = (
        params.get("path")
        or params.get("to")
        or ""
    )

    for blocked in BLOCKED_PATHS:

        if blocked.lower() in path.lower():

            return False, (
                f"Blocked path: {blocked}"
            )

    # =========================
    # VALIDATE URLS
    # =========================
    url = params.get(
        "url",
        ""
    )

    for domain in BLOCKED_DOMAINS:

        if domain in url.lower():

            return False, (
                f"Blocked domain: {domain}"
            )

    return True, "safe"

from core.metrics import metrics
from resilience.circuit import circuit
from core.database import get_connection
from datetime import datetime
import json


class SelfModel:

    def __init__(self):
        self.last_evaluation = None
        self.history = []

    # =========================
    # SYSTEM SNAPSHOT
    # =========================

    def snapshot(self):

        errors = metrics.counters.get("jobs_failed", 0)
        done = metrics.counters.get("jobs_done", 0)

        total = errors + done

        error_rate = errors / total if total > 0 else 0

        open_circuits = [
            k for k, v in circuit.state.items()
            if v == "OPEN"
        ]

        queue_size = metrics.gauges.get("active_jobs", 0)

        snapshot = {
            "timestamp": datetime.utcnow().isoformat(),
            "error_rate": error_rate,
            "open_circuits": open_circuits,
            "queue_size": queue_size,
            "total_jobs": total
        }

        # guardar historial
        self.history.append(snapshot)

        # mantener historial corto (~5 minutos si loop es 10s)
        if len(self.history) > 30:
            self.history.pop(0)

        return snapshot

    # =========================
    # RISK EVALUATION
    # =========================

    def evaluate_risk(self, snapshot):

        if snapshot["open_circuits"]:
            return "HIGH"

        if snapshot["error_rate"] > 0.3:
            return "MEDIUM"

        return "LOW"

    # =========================
    # HEALTH SCORE
    # =========================

    def health_score(self, snapshot):

        score = 100

        # penalizar error rate
        score -= snapshot["error_rate"] * 50

        # penalizar circuit breakers abiertos
        score -= len(snapshot["open_circuits"]) * 20

        # penalizar cola grande
        if snapshot["queue_size"] > 10:
            score -= 10

        if score < 0:
            score = 0

        return round(score, 2)

    # =========================
    # ANOMALY DETECTION
    # =========================

    def detect_anomaly(self):

        if len(self.history) < 5:
            return None

        last = self.history[-1]
        prev = self.history[-5]

        # spike de errores
        if last["error_rate"] > prev["error_rate"] * 2 and last["error_rate"] > 0.2:
            return {
                "type": "error_spike",

            "memory": dict(row)
        })

        key=lambda x: x["score"],

import asyncio
import inspect
from collections import defaultdict


class EventBus:

    def __init__(self):
        self.listeners = defaultdict(list)

    def subscribe(self, event_type: str, handler):
        self.listeners[event_type].append(handler)

    async def publish(self, event_type: str, payload: dict):

        if event_type not in self.listeners:
            return

        for handler in self.listeners[event_type]:

            asyncio.create_task(
                self._safe_execute(handler, event_type, payload)
            )

    async def _safe_execute(self, handler, event_type, payload):

        try:
            sig = inspect.signature(handler)

            # Handler(event_type, payload)
            if len(sig.parameters) == 2:
                await handler(event_type, payload)

            # Handler(payload)
            else:
                await handler(payload)

        except Exception as e:
            print(f"⚠ Event handler error [{event_type}]:", e)


bus = EventBus()

import asyncio
from collections import defaultdict

from jobs.manager import manager

class RetryManager:

    def __init__(self):
        self.retry_count = defaultdict(int)

    async def handle_step_error(self, payload):

        workflow = payload["workflow"]
        job_id = payload.get("job_id")

        wf = get_workflow_by_name(workflow)

        if not wf:
            return

        retry_conf = wf.get("retry")
        if not retry_conf:
            return

        max_r = retry_conf.get("max_retries", 0)
        delay = retry_conf.get("delay_seconds", 0)
        backoff = retry_conf.get("backoff", "fixed")

        self.retry_count[job_id] += 1
        attempt = self.retry_count[job_id]

        if attempt > max_r:

            # 🔥 AQUÍ VA EL PASO 4
            await bus.publish("job.failed.final", {
                "workflow": workflow,
                "job_id": job_id,
                "tenant_id": wf.get("tenant_id"),
                "workspace_id": wf.get("workspace_id"),
                "error": "Max retries exceeded"
            })

            return

        # delay inteligente
        if backoff == "exponential":
            wait = delay * (2 ** (attempt - 1))
        else:
            wait = delay

        await asyncio.sleep(wait)

        await manager.create_job(
            workflow,
            wf.get("tenant_id"),
            wf.get("workspace_id")
        )


retry_manager = RetryManager()

MAX_WORKERS = 4
MAX_QUEUE = 50


# ==========================================================
# JOB MANAGER
# ==========================================================
class JobManager:

    def __init__(self):

        self.queue = asyncio.Queue(
            maxsize=MAX_QUEUE
        )

        self.max_workers = MAX_WORKERS

        self.semaphore = asyncio.Semaphore(
            self.max_workers
        )

        self.started = False

        self.workers = []

        self.join_tracker = {}

        self.paused_jobs = {}

        self.runtime_logs = []

        self.paused = False

    def add_runtime_log(
        self,
        message
    ):

        self.runtime_logs.append(message)

        if len(self.runtime_logs) > 100:
          self.runtime_logs.pop(0)

    # ======================================================
    # START
    # ======================================================
    async def start(self):

        if self.started:
            return

        logger.info("JobManager started")

        self.started = True

        self.recover_jobs()

        for i in range(self.max_workers):

            worker = asyncio.create_task(
                self.worker()
            )

            self.workers.append(worker)

        # 🔥 RUNTIME SUPERVISOR
        asyncio.create_task(
            self.runtime_supervisor_loop()
        )        

    # ======================================================
    # RECOVER
    # ======================================================
    def recover_jobs(self):

        conn = get_connection()
        cur = conn.cursor()

        cur.execute("""
            SELECT id, workflow, tenant_id
            FROM jobs
            WHERE status IN ('queued','running')
        """)

        rows = cur.fetchall()

        conn.close()

        for row in rows:

            try:

                self.queue.put_nowait({

                    "job_id": row["id"],

                    "workflow": row["workflow"],

                    "tenant_id": row["tenant_id"]
                })

            except:
                pass

    # ======================================================
    # WORKER
    # ======================================================
    async def worker(self):

        logger.info("Worker initialized")

        while True:

            if getattr(self, "paused", False):

                print("🛑 SYSTEM PAUSED")

                await asyncio.sleep(5)

                continue

            job = await self.queue.get()

            job_id = job["job_id"]

            workflow_input = job["workflow"]

            tenant_id = job["tenant_id"]

            print(f"\n🚀 NUEVO JOB: {job_id}")

            print(f"📦 INPUT: {workflow_input}")

            try:

                # ==========================================
                # NORMALIZAR
                # ==========================================
                if isinstance(workflow_input, str):

    async def execute_node(

        self.save_execution(

            "running"
        )

        print(f"⚙️ EXEC NODE: {action}")

        # ==================================================
        # JOIN NODE
        # ==================================================
        if action == "join":

            join_id = node["id"]

            incoming = len([
                e for e in edges
                if e["target"] == join_id
            ])

            if join_id not in self.join_tracker:

                self.join_tracker[join_id] = 0

            self.join_tracker[join_id] += 1

            print(
                f"🔗 JOIN {join_id}: "
                f"{self.join_tracker[join_id]}/{incoming}"
            )

            # esperar ramas
            if self.join_tracker[join_id] < incoming:
                return

            # reset
            self.join_tracker[join_id] = 0

        # ==================================================
        # WS RUNNING
        # ==================================================
        await ws.broadcast({

            "type": "job_log",

            "job_id": job_id,

            "node_id": node["id"],

            "message":
                f"⚙️ Ejecutando {action}",

            "action": action,

            "status": "running"
        })

        # ==================================================
        # RETRIES
        # ==================================================
        retry_count = node["data"].get(
            "retry",
            0
        )

        timeout = node["data"].get(
            "timeout",
            30
        )

        attempt = 0

        while True:

            try:

                result = await asyncio.wait_for(

                    execute_action(
                        action,
                        value
                    ),

                    timeout=timeout

                        await self.execute_node(

                            generated["nodes"],

                            generated["edges"],

                            job_id
                        )

                        return

                # ==========================================
                # APPROVAL
                # ==========================================
                if isinstance(result, dict):

                # ==========================================

                # ==========================================

                    action,

                    "error",

                ),
                None
            )

                n,
                nodes,
                edges,
                job_id
            )

            for n in next_nodes
        ])

        self.update_status(
            job_id,
            "running"
        )

        del self.paused_jobs[job_id]

import json
from pathlib import Path

FILE = Path("nexus_learning.json")


def detect_habits():

    if not FILE.exists():
        return []

    try:
        raw = FILE.read_text().strip()

        if not raw:
            return []

        data = json.loads(raw)

        if not isinstance(data, dict):
            return []

    except Exception:
        return []

    habits = []

    for action, times in data.items():
        if times >= 5:
            habits.append({
                "action": action,
                "count": times
            })

    # ordenar por frecuencia
    habits.sort(key=lambda x: x["count"], reverse=True)

    return habits

from core.database import get_connection

class LearningEngine:

    def analyze_recent_failures(self):

        conn = get_connection()
        cur = conn.cursor()

        cur.execute("""
            SELECT workflow, result
            FROM jobs
            WHERE status='error'
            ORDER BY updated_at DESC
            LIMIT 20
        """)

        rows = cur.fetchall()
        conn.close()

        if not rows:
            return None

        workflows = {}

        for r in rows:

            wf = r["workflow"]

            workflows[wf] = workflows.get(wf, 0) + 1

        for wf, count in workflows.items():

            if count >= 3:

                return {
                    "type": "repeated_failure",
                    "workflow": wf,
                    "message": f"El workflow {wf} está fallando repetidamente."
                }

        return None

        """, (
            datetime.utcnow(),
            learning["type"],
            learning.get("workflow"),

import os

from sentence_transformers import SentenceTransformer
from pypdf import PdfReader
import docx

model = SentenceTransformer("all-MiniLM-L6-v2")

client = chromadb.Client()
collection = client.get_or_create_collection("nexus_memory")


def read_file(path):
    ext = path.lower().split(".")[-1]

    if ext == "txt":
        return open(path, encoding="utf-8", errors="ignore").read()

    if ext == "pdf":
        reader = PdfReader(path)
        return "\n".join(p.extract_text() or "" for p in reader.pages)

    if ext == "docx":
        d = docx.Document(path)
        return "\n".join(p.text for p in d.paragraphs)

    return ""


def chunk_text(text, size=500):
    return [text[i:i+size] for i in range(0, len(text), size)]


def index_folder(folder):
    for root, _, files in os.walk(folder):
        for f in files:
            if f.endswith((".pdf", ".txt", ".docx")):
                path = os.path.join(root, f)
                print("Indexando:", path)

                content = read_file(path)
                chunks = chunk_text(content)

                for i, chunk in enumerate(chunks):
                    emb = model.encode(chunk).tolist()

                    collection.add(
                        documents=[chunk],
                        embeddings=[emb],
                        ids=[f"{path}_{i}"],
                        metadatas=[{"path": path}]
                    )

        if ws in self.clients:
            self.clients.remove(ws)

    async def broadcast(self, data):

        print(f"📡 Broadcasting: {data}")

        disconnected = []

        for c in self.clients:
            try:
                await c.send_json(data)
            except Exception as e:
                print("❌ Error WS:", e)
                disconnected.append(c)

        # limpiar conexiones muertas
        for d in disconnected:
            self.disconnect(d)

import time
from collections import defaultdict

from events.bus import bus
from brain.nexus_voice import speak


class CircuitBreaker:

    def __init__(self):
        self.failures = defaultdict(int)
        self.state = {}
        self.opened_at = {}

        self.threshold = 5
        self.cooldown_seconds = 30

    # --------------------------------------------------
    # INTERNAL KEY BUILDER
    # --------------------------------------------------

    def _key(self, workflow, tenant, workspace):
        return f"{workflow}:{tenant}:{workspace}"

    # --------------------------------------------------
    # CHECK STATE
    # --------------------------------------------------

    def is_open(self, workflow, tenant, workspace):

        key = self._key(workflow, tenant, workspace)
        state = self.state.get(key, "CLOSED")

        if state == "OPEN":

            opened = self.opened_at.get(key, 0)

            # cooldown para probar recuperación
            if time.time() - opened > self.cooldown_seconds:

                print(f"🟡 Circuit HALF_OPEN para {key}")

                self.state[key] = "HALF_OPEN"
                return False

            return True

        return False

    # --------------------------------------------------
    # FAILURE
    # --------------------------------------------------

    async def handle_failure(self, payload):

        workflow = payload["workflow"]
        tenant = payload.get("tenant_id", "default")
        workspace = payload.get("workspace_id", "main")

        key = self._key(workflow, tenant, workspace)

        self.failures[key] += 1

        if self.failures[key] >= self.threshold:

            self.state[key] = "OPEN"
            self.opened_at[key] = time.time()

            print(f"🔥 Circuit OPEN para {key}")

            await speak(
                f"He bloqueado temporalmente el workflow {workflow} por fallos repetidos.",
                "warning"
            )

            await bus.publish("circuit.opened", {
                "workflow": workflow,
                "tenant_id": tenant,
                "workspace_id": workspace
            })

    # --------------------------------------------------
    # SUCCESS
    # --------------------------------------------------

    async def handle_success(self, payload):

        workflow = payload["workflow"]
        tenant = payload.get("tenant_id", "default")
        workspace = payload.get("workspace_id", "main")

        key = self._key(workflow, tenant, workspace)

        if self.state.get(key) == "HALF_OPEN":

            self.state[key] = "CLOSED"
            self.failures[key] = 0

            print(f"🟢 Circuit CLOSED para {key}")

            await speak(
                f"He restaurado el funcionamiento normal del workflow {workflow}.",
                "info"
            )

            await bus.publish("circuit.closed", {
                "workflow": workflow,
                "tenant_id": tenant,
                "workspace_id": workspace
            })


circuit = CircuitBreaker()


def register_circuit():
    bus.subscribe("job.failed", circuit.handle_failure)
    bus.subscribe("job.done", circuit.handle_success)

            await self.run_schedule(row)
            return

        last = self.parse_datetime(row["last_run"])

        if not last:
            await self.run_schedule(row)
            return

        if (now - last).total_seconds() >= interval:
            await self.run_schedule(row)

    # =========================
    # CRON
    # =========================

    async def check_cron(self, row, now):

        expr = row["expression"]

        base = self.parse_datetime(row["last_run"]) or now

        itr = croniter(expr, base)
        next_run = itr.get_next(datetime)

        if now >= next_run:
            await self.run_schedule(row)

    # =========================
    # RUN
    # =========================

    async def run_schedule(self, row):

        try:

            await manager.create_job(row["workflow"])

            conn = get_connection()
            cur = conn.cursor()

            cur.execute("""
                UPDATE schedules
                SET last_run = %s
                WHERE id = %s
            """, (
                datetime.utcnow(),
                row["id"]
            ))

            conn.commit()
            conn.close()

        except Exception as e:
            print("🔥 Scheduler run error:", e)

    # =========================
    # HELPERS
    # =========================

    def parse_datetime(self, value):

        try:
            if isinstance(value, datetime):
                return value
            return datetime.fromisoformat(str(value))
        except Exception:
            return None

async def execute_step(
    step,
    user,
    context,
    workflow_name=None,
    tenant_id=None,
    workspace_id=None,
    job_id=None
):

    history = []
    step_type = step.get("type", "action")

    try:

        # ==================================================
        # 1️⃣ ACCIÓN NORMAL
        # ==================================================

        if step_type == "action":

            value = resolve_variables(step.get("value"), context)

            if step.get("input_type"):
                if not validate_type(value, step.get("input_type")):
                    raise Exception(
                        f"Input inválido para {step['action']}"
                    )

            await bus.publish("step.start", {
                "workflow": workflow_name,
                "step": step["action"],
                "job_id": job_id
            })

            out = execute_action(
                step["action"],
                value,
                user
            )

            # 🔥 FIX IMPORTANTE (si la acción es async)
            if asyncio.iscoroutine(out):
                out = await out

            if step.get("output_type"):
                if not validate_type(out, step.get("output_type")):
                    raise Exception(
                        f"Output inválido en {step['action']}"
                    )

            if step.get("assign"):
                context[step["assign"]] = out

            history.append({
                "type": "action",
                "action": step["action"],
                "result": out
            })

            # 🔥 LOG DEL STEP
            log_step(job_id, step["action"], "success", out)

            await bus.publish("step.success", {
                "workflow": workflow_name,
                "step": step["action"],
                "job_id": job_id
            })

            await manager.broadcast({
                "type": "workflow_step",
                "action": step["action"]
            })

            await manager.broadcast({
                "type": "workflow_result",
                "action": step["action"],
                "result": out
            })

            for child in step.get("next", []):
                child_result = await execute_step(
                    child,
                    user,
                    context,
                    workflow_name,
                    tenant_id,
                    workspace_id,
                    job_id
                )
                history.extend(child_result)

            return history


        # ==================================================
        # 2️⃣ CONDICIÓN
        # ==================================================

        if step_type == "condition":

            test_value = resolve_variables(
                step.get("value"),
                context
            )

            test_result = execute_action(
                step["if_action"],
                test_value,
                user
            )

            if asyncio.iscoroutine(test_result):
                test_result = await test_result

            condition_met = evaluate_condition(
                test_result,
                step.get("operator"),
                step.get("compare")
            )

            history.append({
                "type": "condition",
                "result": condition_met
            })

            branch = step.get("then") if condition_met else step.get("else_")

            if branch:
                for s in branch:
                    branch_result = await execute_step(
                        s,
                        user,
                        context,
                        workflow_name,
                        tenant_id,
                        workspace_id,
                        job_id
                    )
                    history.extend(branch_result)

            return history


        # ==================================================
        # 3️⃣ SUBWORKFLOW
        # ==================================================

        if step_type == "subworkflow":

            sub_name = step.get("name")

            await manager.broadcast({
                "type": "workflow_sub_start",
                "workflow": sub_name
            })

            sub_history = await run_workflow(
                sub_name,

            })

    except Exception as e:

# ==========================================================
# VARIABLE RESOLUTION
# ==========================================================

def resolve_variables(value, context):

    if value is None:
        return value

    if isinstance(value, str):
        for key, val in context.items():
            placeholder = "{{" + key + "}}"
            if placeholder in value:
                value = value.replace(
                    placeholder,
                    str(val)
                )
        return value

    if isinstance(value, dict):
        return {
            k: resolve_variables(v, context)
            for k, v in value.items()
        }

    return value


# ==========================================================
# CONDITION EVALUATOR
# ==========================================================

def evaluate_condition(result, operator, compare):

    if operator == "contains":
        return str(compare) in str(result)

    if operator == "equals":
        return str(result) == str(compare)

    if operator == "not_equals":
        return str(result) != str(compare)

    if operator == "gt":
        try:
            return float(result) > float(compare)
        except:
            return False

    if operator == "lt":
        try:
            return float(result) < float(compare)
        except:
            return False

    return False

def log_step(job_id, step, status, result):

        INSERT INTO job_steps
        (job_id, step, status, result, timestamp)
        VALUES (%s, %s, %s, %s, %s)
    """, (
        job_id,
        step,
        status,
        json.dumps(result) if result else "",
        datetime.utcnow()
    ))

    conn.commit()
    conn.close()

def save_workflow_version(

    trigger,
    steps,
    tenant_id="default",
    workspace_id="main"
):

    conn = get_connection()
    cur = conn.cursor()

    # 1️⃣ Obtener versión actual
    cur.execute("""
        SELECT MAX(version)
        FROM workflows
        WHERE name = %s
        AND tenant_id = %s
        AND workspace_id = %s
    """, (name, tenant_id, workspace_id))

    row = cur.fetchone()
    current_version = row[0] if row and row[0] else 0
    new_version = current_version + 1

    # 2️⃣ Desactivar versiones anteriores
    cur.execute("""
        UPDATE workflows
        SET active = FALSE
        WHERE name = %s
        AND tenant_id = %s
        AND workspace_id = %s
    """, (name, tenant_id, workspace_id))

    # 3️⃣ Crear definición
    definition = {
        "name": name,
        "trigger": trigger,
        "steps": steps
    }

    now = datetime.now(timezone.utc)

    # 4️⃣ Insertar nueva versión activa
    cur.execute("""
        INSERT INTO workflows
        (name, version, definition, created_at, tenant_id, workspace_id, active)
        VALUES (%s, %s, %s, %s, %s, %s, TRUE)
    """, (
        name,
        new_version,
        json.dumps(definition),
        now,
        tenant_id,
        workspace_id
    ))

    conn.commit()
    conn.close()

    return new_version

def get_workflow_by_name(name, tenant_id="default", workspace_id="main"):

    cur.execute("""
        SELECT definition
        FROM workflows
        WHERE name = %s
        AND tenant_id = %s
        AND workspace_id = %s
        AND active = 1
        ORDER BY version DESC
        LIMIT 1
    """, (name, tenant_id, workspace_id))

    conn.close()

    if not row:
        return None

    definition = row[0]

    if not definition:

    return json.loads(definition)

VALID_ACTIONS = {
    "create_folder",
    "move_files",
    "list_dir",
    "read_file",
    "write_file",
    "delete_file",
    "run_cmd",
    "open_url"
}

# acciones que requieren confirmación humana
DANGEROUS_ACTIONS = {
    "delete_file",
    "run_cmd"
}

MAX_STEPS = 25


def validate_workflow(workflow):

    if not workflow:
        raise Exception("Workflow vacío")

    if "steps" not in workflow:
        raise Exception("Workflow sin steps")

    steps = workflow["steps"]

    if not isinstance(steps, list):
        raise Exception("Steps debe ser una lista")

    if len(steps) > MAX_STEPS:
        raise Exception("Workflow demasiado grande")

    for step in steps:
        validate_step(step)

    return workflow


def validate_step(step):

    if not isinstance(step, dict):
        raise Exception("Step inválido")

    if "action" not in step and step.get("type") != "condition":
        raise Exception("Step sin acción")

    # =========================
    # ACTION STEPS
    # =========================

    if "action" in step:

        action = step["action"]

        if action not in VALID_ACTIONS:
            raise Exception(f"Acción no válida: {action}")

        # advertencia acciones peligrosas
        if action in DANGEROUS_ACTIONS:
            print(f"⚠ Acción peligrosa detectada: {action}")

        value = step.get("value")

        # ---------------------
        # autocorrecciones
        # ---------------------

        if action == "move_files":

            if isinstance(value, str):

                step["value"] = {
                    "ext": ".pdf",
                    "to": "PDF"
                }

        if action == "list_dir":

            if not value:
                step["value"] = "."

    # =========================
    # SUBSTEPS (next)
    # =========================

    if "next" in step:

        if not isinstance(step["next"], list):
            raise Exception("next debe ser lista")

        for s in step["next"]:
            validate_step(s)

    # =========================
    # CONDITIONS
    # =========================

    if step.get("type") == "condition":

        for branch in ["then", "else_"]:

            if branch in step:

                if not isinstance(step[branch], list):
                    raise Exception(f"{branch} debe ser lista")

                for s in step[branch]:
                    validate_step(s)
