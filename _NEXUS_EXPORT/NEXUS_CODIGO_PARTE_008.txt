import { mockPresence } from "./mockPresence";
import type {
  NexusPresenceEvent,
  NexusPresenceState,
  NexusPresenceStateModel,
} from "./types";

export interface PresenceAdapter {
  getCurrentState(): NexusPresenceStateModel;

  subscribe(
    listener: (state: NexusPresenceStateModel) => void,
  ): () => void;
}

const PRESENCE_API_URL =
  "http://127.0.0.1:8765/api/presence";

const PRESENCE_EVENTS_URL =
  "http://127.0.0.1:8765/api/presence/events";

const POLL_INTERVAL_MS = 1000;

class HttpPresenceAdapter implements PresenceAdapter {
  private state: NexusPresenceStateModel = mockPresence;

  private listeners = new Set<
    (state: NexusPresenceStateModel) => void
  >();

  private pollTimer: ReturnType<typeof setInterval> | null =
    null;

  private eventSource: EventSource | null = null;

  private refreshing = false;

  private sseConnected = false;

  /*
   * Timestamp of the first presence event of the current
   * cognitive cycle.
   */
  private currentCycleStartedAt: string | null = null;

  /*
   * Presence events belonging exclusively to the current
   * cognitive cycle.
   */
  private currentCycleEvents: NexusPresenceEvent[] = [];

  getCurrentState(): NexusPresenceStateModel {
    return this.state;
  }

  subscribe(
    listener: (state: NexusPresenceStateModel) => void,
  ): () => void {
    this.listeners.add(listener);

    listener(this.state);

    void this.refresh();

    this.startSSE();

    this.startPolling();

    return () => {
      this.listeners.delete(listener);

      if (this.listeners.size === 0) {
        this.stopSSE();
        this.stopPolling();
      }
    };
  }

  private startSSE(): void {
    if (this.eventSource !== null) {
      return;
    }

    if (typeof EventSource === "undefined") {
      return;
    }

    try {
      const eventSource = new EventSource(
        PRESENCE_EVENTS_URL,
      );

      this.eventSource = eventSource;

      eventSource.onopen = () => {
        this.sseConnected = true;
      };

      eventSource.addEventListener(
        "presence",
        (event) => {
          this.handlePresenceEvent(event);
        },
      );

      eventSource.onerror = () => {
        this.sseConnected = false;

        /*
         * EventSource automatically attempts to reconnect.
         *
         * HTTP snapshot polling remains available as fallback.
         */
      };
    } catch {
      this.sseConnected = false;
      this.eventSource = null;
    }
  }

  private stopSSE(): void {
    if (this.eventSource === null) {
      return;
    }

    this.eventSource.close();

    this.eventSource = null;
    this.sseConnected = false;
  }

  private handlePresenceEvent(
    event: MessageEvent<string>,
  ): void {
    try {
      const payload = JSON.parse(
        event.data,
      ) as {
        state?: string;
        timestamp?: string;
        message?: string | null;
        data?: unknown;
      };

      if (!payload.state) {
        return;
      }

      const state = this.normalizeState(
        payload.state,
      );

      if (state === null) {
        return;
      }

      const timestamp =
        payload.timestamp ??
        new Date().toISOString();

      /*
       * LISTENING marks the beginning of a new cognitive
       * cycle.
       *
       * Everything that was previously visible in the
       * timeline belongs to an older cycle.
       */
      if (state === "listening") {
        this.startNewCycle(timestamp);
      }

      const presenceEvent: NexusPresenceEvent = {
        id: `sse-${Date.now()}-${Math.random()
          .toString(36)
          .slice(2, 8)}`,
        type: state,
        timestamp,
        message:
          payload.message ??
          undefined,
        data: payload.data,
      };

      this.addCurrentCycleEvent(
        presenceEvent,
      );

      this.state = {
        ...this.state,
        state,
        events: this.currentCycleEvents,
      };

      this.emit();

      /*
       * Terminal states trigger a complete snapshot refresh.
       *
       * The snapshot supplies objective, plan, execution,
       * verification, memory and learning details while the
       * current-cycle SSE events remain the timeline source.
       */
      if (
        state === "completed" ||
        state === "failed"
      ) {
        void this.refresh();
      }
    } catch {
      /*
       * Invalid SSE payloads must never break the adapter.
       */
    }
  }

  private startNewCycle(
    timestamp: string,
  ): void {
    this.currentCycleStartedAt = timestamp;
    this.currentCycleEvents = [];
  }

  private addCurrentCycleEvent(
    event: NexusPresenceEvent,
  ): void {
    const exists =
      this.currentCycleEvents.some(
        (existing) =>
          existing.timestamp ===
            event.timestamp &&
          existing.type === event.type &&
          existing.message ===
            event.message,
      );

    if (exists) {
      return;
    }

    this.currentCycleEvents = [
      ...this.currentCycleEvents,
      event,
    ];
  }

  private normalizeState(
    value: string,
  ): NexusPresenceState | null {
    const states: NexusPresenceState[] = [
      "idle",
      "listening",
      "understanding",
      "thinking",
      "deciding",
      "executing",
      "verifying",
      "completed",
      "needs_confirmation",
      "failed",
      "recovering",
    ];

    if (
      states.includes(
        value as NexusPresenceState,
      )
    ) {
      return value as NexusPresenceState;
    }

    return null;
  }

  private startPolling(): void {
    if (this.pollTimer !== null) {
      return;
    }

    this.pollTimer = setInterval(() => {
      void this.refresh();
    }, POLL_INTERVAL_MS);
  }

  private stopPolling(): void {
    if (this.pollTimer === null) {
      return;
    }

    clearInterval(this.pollTimer);

    this.pollTimer = null;
  }

  private async refresh(): Promise<void> {
    if (this.refreshing) {
      return;
    }

    this.refreshing = true;

    try {
      const response = await fetch(
        PRESENCE_API_URL,
        {
          method: "GET",
          headers: {
            Accept: "application/json",
          },
        },
      );

      if (!response.ok) {
        throw new Error(
          `Presence API returned HTTP ${response.status}`,
        );
      }

      const nextState =
        (await response.json()) as NexusPresenceStateModel;

      if (
        !this.shouldAcceptSnapshot(
          nextState,
        )
      ) {
        return;
      }

      /*
       * Preserve the SSE event timeline whenever a current
       * cycle exists.
       *
       * The snapshot is still responsible for all the richer
       * cognitive data.
       */
      const events =
        this.resolveSnapshotEvents(
          nextState,
        );

      this.state = {
        ...nextState,
        events,
      };

      this.emit();
    } catch {
      /*
       * Preserve the last known state.
       */
    } finally {
      this.refreshing = false;
    }
  }

  private resolveSnapshotEvents(
    nextState: NexusPresenceStateModel,
  ): NexusPresenceEvent[] {
    /*
     * Once an SSE cycle has started, Presence events are the
     * authoritative source for the live/current timeline.
     */
    if (
      this.currentCycleStartedAt !== null
    ) {
      return this.currentCycleEvents;
    }

    /*
     * Before the first live cycle exists, use the snapshot.
     *
     * The snapshot may contain multiple historical cognitive
     * cycles. Keep only the latest one so the console timeline
     * represents the current cognitive cycle rather than
     * historical runtime activity.
     */
    return this.filterToLatestSnapshotCycle(
      nextState,
    );
  }

  private filterToLatestSnapshotCycle(
  presence: NexusPresenceStateModel,
): NexusPresenceEvent[] {
  if (presence.events.length === 0) {
    return [];
  }

  /*
   * Preferred boundary:
   *
   * The runtime stores the user request timestamp.
   * When available, it represents the beginning of the
   * latest request/cognitive cycle.
   */
  const requestTimestamp =
    presence.request?.timestamp;

  if (requestTimestamp) {
    const requestTime =
      Date.parse(requestTimestamp);

    if (!Number.isNaN(requestTime)) {
      return presence.events.filter(
        (event) => {
          const eventTime =
            Date.parse(event.timestamp);

          if (Number.isNaN(eventTime)) {
            return true;
          }

          return eventTime >= requestTime;
        },
      );
    }
  }

  /*
   * Snapshot fallback:
   *
   * Historical runtime snapshots use semantic event types
   * such as:
   *
   * request → context → intent → decision → objective
   * → plan → execution → verification → memory → learning
   *
   * Therefore REQUEST is the cycle boundary for snapshot
   * events.
   *
   * Find the LAST request event so only the newest cognitive
   * cycle remains visible in the console.
   */
  let latestRequestIndex = -1;

  for (
    let index = 0;
    index < presence.events.length;
    index += 1
  ) {
    const event = presence.events[index];

    if (event.type === "request") {
      latestRequestIndex = index;
    }
  }

  if (latestRequestIndex >= 0) {
    return presence.events.slice(
      latestRequestIndex,
    );
  }

  /*
   * If the snapshot contains no recognizable request
   * boundary, preserve the snapshot rather than guessing.
   */
  return presence.events;
}

  private shouldAcceptSnapshot(
    nextState: NexusPresenceStateModel,
  ): boolean {
    if (!this.sseConnected) {
      return true;
    }

    /*
     * Terminal snapshots are always accepted because they
     * contain the complete final execution information.
     */
    if (
      nextState.state === "completed" ||
      nextState.state === "failed"
    ) {
      return true;
    }

    return this.isSnapshotNewer(
      nextState,
      this.state,
    );
  }

  private isSnapshotNewer(
    nextState: NexusPresenceStateModel,
    currentState: NexusPresenceStateModel,
  ): boolean {
    const nextTimestamp =
      this.getLatestTimestamp(nextState);

    const currentTimestamp =
      this.getLatestTimestamp(currentState);

    if (
      nextTimestamp === null ||
      currentTimestamp === null
    ) {
      return true;
    }

    return (
      nextTimestamp >= currentTimestamp
    );
  }

  private getLatestTimestamp(
    state: NexusPresenceStateModel,
  ): number | null {
    const timestamps: number[] = [];

    if (state.request?.timestamp) {
      const value = Date.parse(
        state.request.timestamp,
      );

      if (!Number.isNaN(value)) {
        timestamps.push(value);
      }
    }

    for (const event of state.events) {
      const value = Date.parse(
        event.timestamp,
      );

      if (!Number.isNaN(value)) {
        timestamps.push(value);
      }
    }

    if (timestamps.length === 0) {
      return null;
    }

    return Math.max(...timestamps);
  }

  private emit(): void {
    for (const listener of this.listeners) {
      listener(this.state);
    }
  }
}

export const presenceAdapter =
  new HttpPresenceAdapter();

import type {
  NexusPresenceStateModel,
} from "./types";

export type CognitiveCycleStatus =
  | "active"
  | "completed"
  | "failed";

export type CognitiveCycleEvent = {
  id: string;
  type: string;
  timestamp: string;
  message?: string;
  data?: unknown;
};

export type CognitiveCycle = {
  id: string;
  startedAt: string;
  endedAt?: string;
  request?: CognitiveCycleEvent;
  events: CognitiveCycleEvent[];
  status: CognitiveCycleStatus;
  eventCount: number;
};

function getCycleStatus(
  events: CognitiveCycleEvent[],
): CognitiveCycleStatus {
  const lastEvent = events[events.length - 1];

  if (!lastEvent) {
    return "active";
  }

  if (
    lastEvent.type === "verification_failure" ||
    lastEvent.type === "failed"
  ) {
    return "failed";
  }

  if (
    lastEvent.type === "episode" ||
    lastEvent.type === "completed"
  ) {
    return "completed";
  }

  return "active";
}

function createCycle(
  events: CognitiveCycleEvent[],
): CognitiveCycle | null {
  if (events.length === 0) {
    return null;
  }

  const request = events.find(
    (event) => event.type === "user_request",
  );

  const startedAt =
    request?.timestamp ??
    events[0].timestamp;

  const lastEvent =
    events[events.length - 1];

  const status = getCycleStatus(events);

  return {
    id: request?.id ?? events[0].id,
    startedAt,
    endedAt:
      status === "active"
        ? undefined
        : lastEvent.timestamp,
    request,
    events,
    status,
    eventCount: events.length,
  };
}

export function buildCycleHistory(
  presence: NexusPresenceStateModel,
): CognitiveCycle[] {
  const sourceEvents = presence.events;

  if (sourceEvents.length === 0) {
    return [];
  }

  const cycles: CognitiveCycle[] = [];
  let currentEvents: CognitiveCycleEvent[] = [];

  for (const event of sourceEvents) {
    const normalizedEvent: CognitiveCycleEvent = {
      id: event.id,
      type: event.type,
      timestamp: event.timestamp,
      message: event.message,
      data: event.data,
    };

    if (
      normalizedEvent.type === "user_request" &&
      currentEvents.length > 0
    ) {
      const previousCycle = createCycle(
        currentEvents,
      );

      if (previousCycle) {
        cycles.push(previousCycle);
      }

      currentEvents = [];
    }

    currentEvents.push(normalizedEvent);
  }

  const lastCycle = createCycle(
    currentEvents,
  );

  if (lastCycle) {
    cycles.push(lastCycle);
  }

  return cycles;
}

export function getCurrentCycle(
  presence: NexusPresenceStateModel,
): CognitiveCycle | null {
  const cycles = buildCycleHistory(presence);

  return cycles.length > 0
    ? cycles[cycles.length - 1]
    : null;
}

export function getHistoricalCycles(
  presence: NexusPresenceStateModel,
): CognitiveCycle[] {
  const cycles = buildCycleHistory(presence);

  if (cycles.length <= 1) {
    return [];
  }

  return cycles.slice(0, -1);
}

export type StoredLearningCandidate = {
  id: string;
  insightId: string;
  title: string;
  knowledge: string;
  sourceCycleIds: string[];
  evidenceKeys: string[];
  createdAt: string;
  status?: "staged" | "promoted";
  promotedAt?: string;
  memoryKey?: string;
};

const STORAGE_KEY = "nexus.learning-candidates.v1";

function isCandidate(value: unknown): value is StoredLearningCandidate {
  if (!value || typeof value !== "object") {
    return false;
  }

  const candidate = value as Record<string, unknown>;

  return (
    (typeof candidate.id === "string" || candidate.id === undefined) &&
    typeof candidate.insightId === "string" &&
    typeof candidate.title === "string" &&
    typeof candidate.knowledge === "string" &&
    Array.isArray(candidate.sourceCycleIds) &&
    candidate.sourceCycleIds.every((item) => typeof item === "string") &&
    Array.isArray(candidate.evidenceKeys) &&
    candidate.evidenceKeys.every((item) => typeof item === "string") &&
    typeof candidate.createdAt === "string"
  );
}

export function loadLearningCandidates(): StoredLearningCandidate[] {
  try {
    const raw = window.localStorage.getItem(STORAGE_KEY);

    if (!raw) {
      return [];
    }

    const parsed: unknown = JSON.parse(raw);

    if (!Array.isArray(parsed)) {
      return [];
    }

    return parsed.filter(isCandidate).map((candidate) => ({
      ...candidate,
      id:
        typeof candidate.id === "string"
          ? candidate.id
          : `insight-${candidate.insightId}-${candidate.sourceCycleIds.slice().sort().join("--")}`,
      status: candidate.status ?? "staged",
    }));
  } catch {
    return [];
  }
}

function persist(candidates: StoredLearningCandidate[]): void {
  try {
    window.localStorage.setItem(
      STORAGE_KEY,
      JSON.stringify(candidates),
    );
  } catch {
    // Local storage is an optional UI persistence layer.
  }
}

export function saveLearningCandidate(
  candidate: StoredLearningCandidate,
): void {
  const current = loadLearningCandidates();

  const exists = current.some(
    (item) =>
      item.insightId === candidate.insightId &&
      item.sourceCycleIds.length === candidate.sourceCycleIds.length &&
      item.sourceCycleIds.every((id) => candidate.sourceCycleIds.includes(id)),
  );

  if (exists) {
    return;
  }

  persist([candidate, ...current]);
}

export function updateLearningCandidate(
  candidate: StoredLearningCandidate,
): void {
  const current = loadLearningCandidates();
  const index = current.findIndex((item) => item.id === candidate.id);

  if (index === -1) {
    return;
  }

  current[index] = candidate;
  persist(current);
}

export function removeLearningCandidate(
  candidate: StoredLearningCandidate,
): void {
  const current = loadLearningCandidates();

  persist(
    current.filter(
      (item) =>
        !(
          item.insightId === candidate.insightId &&
          item.createdAt === candidate.createdAt &&
          item.sourceCycleIds.length === candidate.sourceCycleIds.length &&
          item.sourceCycleIds.every((id) => candidate.sourceCycleIds.includes(id))
        ),
    ),
  );
}

import json
import ollama
from actions.system_actions import execute_action


SYSTEM = """
Sos Nexus Planner.

Convertí la orden del usuario en un PLAN JSON.

Formato:
{
  "steps": [
    {"action": "create_folder", "value": "PDF"},
    {"action": "move_files", "value": {"ext": ".pdf", "to": "PDF"}}
  ]
}

Acciones disponibles:
- open_app
- open_file
- create_folder
- move_files
- run_cmd

Solo devolver JSON.
"""


def run_plan(user_message: str):

    res = ollama.chat(
        model="llama3",
        messages=[
            {"role": "system", "content": SYSTEM},
            {"role": "user", "content": user_message}
        ]
    )

    content = res["message"]["content"]

    try:
        plan = json.loads(content)
    except:
        return "No pude generar un plan."

    outputs = []

    for step in plan.get("steps", []):
        out = execute_action(step["action"], step["value"])
        outputs.append(out)

    return "\n".join(outputs)

import os

from pathlib import Path

from actions.registry import register_action


def create_folder(value):

    Path(value).mkdir(parents=True, exist_ok=True)

    return f"Carpeta creada: {value}"


def list_dir(value):

    files = os.listdir(value or ".")

    return "\n".join(files)


def move_files(value):

    ext = value["ext"]
    to = value["to"]

    Path(to).mkdir(exist_ok=True)

    moved = 0

    for f in os.listdir("."):
        if f.lower().endswith(ext):
            shutil.move(f, os.path.join(to, f))
            moved += 1

    return f"{moved} archivos movidos a {to}"

ALIASES = {
    "open_browser": "open_app",
    "launch_app": "open_app",

    "delete": "delete_file",
    "remove_file": "delete_file",

def normalize(action: str):
    return ALIASES.get(action, action)

def check_role(user, action):

    role = user.get("role")

    if role == "admin":
        return True

    if role == "staff" and action in [
        "delete_file", "run_cmd", "kill_process"
    ]:
        return False

    if role == "viewer":
        return False

    return True

# ==================================================

def confirm_if_needed(action, value, auto=True):

    if auto:
        return True

    try:
        resp = input(f"Confirmar acción {action}? (y/n): ")
        return resp.lower() == "y"
    except:
        return False


def sandbox_check(value):
    if isinstance(value, str) and ("\\" in value or "/" in value):
        if not is_path_allowed(value):
            return False
    return True


# ==================================================
# 🔥 MAIN EXECUTOR (ASYNC)
# ==================================================

async def execute_action(action, value, user=None):

    if user is None:
        user = {"role": "admin"}

    action = normalize(action)

    if not check_role(user, action):
        return "⛔ Permiso denegado"

    if not sandbox_check(value):
        return "⛔ Ruta no permitida por seguridad"

    if not confirm_if_needed(action, value):
        return "Acción cancelada"

    result = None

    plugin = get_action(action)

    if plugin:
      result = plugin(value)
      return result

    # ==================================================
    # 📦 APPS
    # ==================================================

    if action == "open_app":

        shortcuts = {
            "chrome": r"C:\Program Files\Google\Chrome\Application\chrome.exe",
            "edge": r"C:\Program Files (x86)\Microsoft\Edge\Application\msedge.exe",
            "notepad": "notepad",
            "calc": "calc",
        }

        value = shortcuts.get(str(value).lower(), value)
        subprocess.Popen(value)
        result = f"Abriendo {value}"

    elif action == "open_url":

        webbrowser.open(value)
        result = f"Abriendo URL {value}"

    # ==================================================
    # 📁 FILESYSTEM
    # ==================================================

    elif action == "list_dir":
        files = os.listdir(value or ".")
        result = "\n".join(files)

    elif action == "read_file":
        result = Path(value).read_text(encoding="utf-8")

    elif action == "write_file":
        Path(value["path"]).write_text(value["content"], encoding="utf-8")
        result = f"Archivo guardado: {value['path']}"

    elif action == "delete_file":
        os.remove(value)
        result = f"Archivo eliminado: {value}"

    elif action == "copy_file":
        shutil.copy(value["src"], value["dst"])
        result = "Archivo copiado"

    elif action == "rename_file":
        os.rename(value["src"], value["dst"])
        result = "Archivo renombrado"

    elif action == "create_folder":
        Path(value).mkdir(parents=True, exist_ok=True)
        result = f"Carpeta creada: {value}"

    elif action == "move_files":

        ext = value["ext"]
        to = value["to"]

        Path(to).mkdir(exist_ok=True)

        moved = 0

        for f in os.listdir("."):
            if f.lower().endswith(ext):
                shutil.move(f, os.path.join(to, f))
                moved += 1

        result = f"{moved} archivos movidos a {to}"

    # ==================================================
    # 🌐 INTEGRACIONES EXTERNAS
    # ==================================================

    elif action == "http_request":

        result = await http_request(
            value["method"],
            value["url"],
            value.get("headers"),
            value.get("body"),
            value.get("timeout", 10)
        )

    elif action == "send_slack":

        result = await send_slack(
            value["webhook_url"],
            value["message"]
        )

    elif action == "send_email":

        result = await send_email(
            value["smtp_host"],
            value["smtp_port"],
            value["username"],
            value["password"],
            value["to"],
            value["subject"],
            value["body"]
        )

    # ==================================================
    # 🖥 SISTEMA
    # ==================================================

    elif action == "list_processes":
        names = [p.info['name'] for p in psutil.process_iter(['name'])]
        result = "\n".join(names)

    elif action == "kill_process":

        for p in psutil.process_iter(['name']):
            if p.info['name'].lower() == value.lower():
                p.kill()
                result = f"{value} finalizado"
                break
        else:
            result = "Proceso no encontrado"

    elif action == "get_time":
        from datetime import datetime
        result = datetime.now().strftime("%H:%M:%S")

    elif action == "system_info":

        result = f"""
        OS: {os.name}
        CPU: {psutil.cpu_percent()}%
        RAM: {psutil.virtual_memory().percent}%
        """

    elif action == "sleep":
        await asyncio.sleep(float(value))
        result = f"Esperado {value}s"

    # ==================================================
    # 💻 SHELL
    # ==================================================

    elif action == "run_cmd":
        result = subprocess.getoutput(value)

    # ==================================================
    # 🔌 PLUGINS
    # ==================================================

    else:

        plugin_result = run_plugin(action, value)

        if plugin_result is not None:
            result = plugin_result
        else:
            result = f"Acción no reconocida: {action}"

    # ==================================================
    # 🔥 LOG + TRACK
    # ==================================================

    log(action, result)
    track(action, value)

    return result

# =========================
# FIX JSON (🔥 CLAVE)
# =========================

def fix_json(text: str):

    # cerrar llaves faltantes
    open_braces = text.count("{")
    close_braces = text.count("}")

    if open_braces > close_braces:
        text += "}" * (open_braces - close_braces)

    # cerrar corchetes faltantes
    open_brackets = text.count("[")
    close_brackets = text.count("]")

    if open_brackets > close_brackets:
        text += "]" * (open_brackets - close_brackets)

    return text


# =========================
# EXTRACT JSON
# =========================

def extract_json(text):

    match = re.search(r"\{.*", text, re.DOTALL)

    if match:
        return match.group(0)

    return text

def validate_steps(steps):
