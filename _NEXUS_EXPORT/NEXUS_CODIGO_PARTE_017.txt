"""Origin and Host validation for the local NEXUS HTTP API.

The API can execute local actions, so browser callers are restricted to the
known local development UIs. Non-browser clients without an Origin header
remain supported; Host is still restricted to loopback names.
"""
from __future__ import annotations

from urllib.parse import urlsplit

ALLOWED_BROWSER_ORIGINS = frozenset(
    {
        "http://localhost:5173",
        "http://127.0.0.1:5173",
        "http://localhost:5174",
        "http://127.0.0.1:5174",
    }
)
ALLOWED_HOST_NAMES = frozenset({"localhost", "127.0.0.1"})
API_PORT = 8765
MAX_REQUEST_BODY_BYTES = 1_048_576  # 1 MiB por solicitud JSON


def is_allowed_browser_origin(origin: str | None) -> bool:
    """Allow non-browser clients and only the known local browser origins."""
    if origin is None or not origin.strip():
        return True
    return origin.strip() in ALLOWED_BROWSER_ORIGINS


def cors_origin_for(origin: str | None) -> str | None:
    """Return an origin to echo in CORS headers, never a wildcard."""
    if origin is None:
        return None
    normalized = origin.strip()
    return normalized if normalized in ALLOWED_BROWSER_ORIGINS else None


def is_allowed_host(host_header: str | None) -> bool:
    """Reject non-loopback Host headers (including DNS-rebinding domains)."""
    if host_header is None or not host_header.strip():
        # Preserve compatibility with HTTP/1.0 clients that omit Host.
        return True
    try:
        parsed = urlsplit("//" + host_header.strip())
        hostname = (parsed.hostname or "").lower()
        port = parsed.port
    except ValueError:
        return False
    return hostname in ALLOWED_HOST_NAMES and port in (None, API_PORT)


def is_allowed_content_length(content_length: int) -> bool:
    """Accept non-empty request bodies up to the configured safety limit."""
    return 0 < content_length <= MAX_REQUEST_BODY_BYTES

from __future__ import annotations

from datetime import datetime

from actions.base import Action
from planner.execution_context import ExecutionContext
from planner.plan import PlanStep


class GetTimeAction(Action):
    """Acción para consultar la fecha y hora local del equipo."""

    name = "system.get_time"
    description = "Consultar la fecha y hora local del sistema."
    internal = True

    produces = ["date", "time"]

    async def execute(
        self,
        context: ExecutionContext,
        step: PlanStep,
    ) -> None:
        now = datetime.now()
        current_date = now.strftime("%d/%m/%Y")
        current_time = now.strftime("%H:%M")

        context.add_result("date", current_date)
        context.add_result("time", current_time)
        context.add_result("status", "read")

        title="Consultar la hora",
        description="Consultar la hora",
    )
    step = PlanStep(
        id="step-time",
        name="Consultar hora",
        description="Obtener la hora local",
        action="system.get_time",
    )
    plan = ExecutionPlan(
        objective_id=objective.id,
        steps=[step],
    )
    context = ExecutionContext(
        objective=objective,
        plan=plan,
    )

        mocked_datetime.now.return_value.strftime.side_effect = [
            "28/10/2026",
            "14:35",

        await GetTimeAction().execute(context, step)

    assert context.results["time"] == "14:35"
    assert context.results["status"] == "read"

        for episode in episodes:
            description = ResponseFormatter._describe_episode(
                episode.get("action"),
                episode.get("parameters", {}),
            )

            if description:
                descriptions.append(description)
                continue

            original_text = episode.get("original_text")

            if original_text:
                descriptions.append(original_text)

        if not descriptions:
            return "Encontré acciones recientes, pero no pude describirlas."

        if len(descriptions) == 1:
            return (
                "La acción reciente fue "
                f"{descriptions[0]}."
            )

        return (
            "Las acciones recientes fueron: "
            + "; ".join(descriptions)
            + "."
        )

            return "activar el sonido"

        if action == "system.process_list":
            return "consultar los procesos activos"

        return None

    @staticmethod
    def _describe_episode_infinitive(
        action: str | None,
        parameters: dict[str, Any],
    ) -> str | None:
        """
        Describe un episodio usando infinitivo.
        """

        return ResponseFormatter._describe_episode(
            action,
            parameters,
        )

    @staticmethod
    def _format_history_not_found(
        mode: str | None,
    ) -> str:
        if mode == "previous":
            return "No encontré un volumen anterior registrado."

        if mode == "find":
            return "No encontré ese valor en el historial del volumen."

        return "No encontré información en el historial del volumen."
