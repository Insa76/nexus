import asyncio

from planner.execution_context import ExecutionContext
from planner.objective import Objective, ObjectiveStatus
from planner.plan import ExecutionPlan, PlanStep, StepStatus
from runtime.executor import Executor


class FakeActionManager:
    def __init__(self, failing_steps=None):
        self.failing_steps = set(failing_steps or [])
        self.execution_order = []

    async def execute(self, context):
        step = context.plan.steps[context.current_step]

        self.execution_order.append(step.id)

        if step.id in self.failing_steps:
            step.status = StepStatus.FAILED
            context.add_error(
                f"Step '{step.id}' failed."
            )
            return

        step.status = StepStatus.COMPLETED

        context.results[step.id] = {
            "status": "completed"
        }

        for artifact in step.produces:
            context.artifacts[artifact] = {
                "produced_by": step.id
            }


class FakeRuntime:
    def __init__(self, failing_steps=None):
        self.action_manager = FakeActionManager(
            failing_steps=failing_steps
        )


def make_step(
    step_id,
    name,
    depends_on=None,
    produces=None,
    consumes=None,
):
    return PlanStep(
        id=step_id,
        name=name,
        description=name,
        action=name,
        status=StepStatus.PENDING,
        depends_on=depends_on or [],
        consumes=consumes or [],
        produces=produces or [],
    )


def make_context(steps):
    objective = Objective(
        id="objective-state-test",
        title="State Consistency Test",
        status=ObjectiveStatus.READY,
        priority=1,
        metadata={},
    )

    plan = ExecutionPlan(
        objective_id=objective.id,
        steps=steps,
        current_step=0,
        completed=False,
        metadata={},
    )

    return ExecutionContext(
        objective=objective,
        plan=plan,
        variables={},
        artifacts={},
        results={},
        errors=[],
        current_step=0,
        finished=False,
        step_evidence={},
    )


def run_executor(context, runtime):
    executor = Executor(runtime)

    return asyncio.run(
        executor.execute(context)
    )


def get_step(context, step_id):
    for step in context.plan.steps:
        if step.id == step_id:
            return step

    raise AssertionError(
        f"Step '{step_id}' not found."
    )


def test_successful_plan_has_consistent_global_state():
    runtime = FakeRuntime()

    context = make_context(
        [
            make_step(
                "step_a",
                "action_a",
            ),
            make_step(
                "step_b",
                "action_b",
                depends_on=["step_a"],
            ),
        ]
    )

    result = run_executor(
        context,
        runtime,
    )

    assert result.success is True

    assert context.finished is True
    assert context.plan.completed is True
    assert (
        context.objective.status
        == ObjectiveStatus.COMPLETED
    )

    assert all(
        step.status == StepStatus.COMPLETED
        for step in context.plan.steps
    )

from __future__ import annotations

from planner.intent import Intent
from planner.objective import (
    Objective,
    ObjectiveStatus,
)
from planner.planner import Planner


class FakeTool:

    def __init__(
        self,
        *,
        produces=None,
        consumes=None,
    ):
        self.produces = list(
            produces or []
        )
        self.consumes = list(
            consumes or []
        )


class FakeToolRegistry:

    def __init__(self, tools):
        self.tools = tools

    def resolve(self, action):
        return self.tools.get(action)


def make_objective(
    objective_id="objective-1",
):
    return Objective(
        id=objective_id,
        title="test",
        description="test objective",
    )


def test_consumer_without_producer_is_detected():

    registry = FakeToolRegistry(
        {
            "system.inspect": FakeTool(
                consumes=["image"],
            ),
        }
    )

    planner = Planner(
        tool_resolver=registry.resolve
    )

    objective = make_objective()

    intent = Intent(
        action="system.inspect",
        original_text="inspeccionar imagen",
    )

    plan = planner.create_plan(
        objective,
        intent,
    )

    assert len(plan.steps) == 1

    assert plan.steps[0].consumes == [
        "image"
    ]

    assert (
        objective.status
        == ObjectiveStatus.FAILED
    )

                consumes=["inspection"],

            ),
        }
    )

    planner = Planner(
        tool_resolver=registry.resolve
    )

    objective = make_objective()

    capture = Intent(
        action="system.capture",
        original_text="capturar pantalla",

    inspect_capture = Intent(
        action="system.inspect",
        original_text="inspeccionar captura",
    )

    process = Intent(
        action="system.process",
        original_text="procesar inspección",
    )

    intent = Intent(
        action=capture.action,
        original_text=(
            "capturar, inspeccionar "
            "y procesar"

        sub_intents=[
            capture,
            inspect_capture,
            process,

from planner.objective import Objective
from planner.plan import (
    ExecutionPlan,
    PlanStep,
    StepStatus,
)
from planner.verification import Verifier
from runtime.execution_result import ExecutionResult


def make_step(
    step_id: str,
    action: str,
    result_keys=None,
    artifact_keys=None,
):
    metadata = {
        "execution": {
            "status": "completed",
            "result_keys": result_keys or [],
            "artifact_keys": artifact_keys or [],
        }
    }

    return PlanStep(
        id=step_id,
        name=action,
        description=action,
        action=action,
        status=StepStatus.COMPLETED,
        metadata=metadata,
    )


def make_plan(steps):
    return ExecutionPlan(
        objective_id="objective-test",
        steps=steps,
        current_step=len(steps),
        completed=True,
    )


def make_result(
    output=None,
    artifacts=None,
    step_evidence=None,
    success=True,
):
    metadata = {}

    if step_evidence is not None:
        metadata["step_evidence"] = step_evidence

    return ExecutionResult(
        success=success,
        message="",
        output=output or {},
        artifacts=artifacts or {},
        metadata=metadata,
    )


def check(condition, message):
    if not condition:
        raise AssertionError(message)


def test_compound_plan_all_steps_verified():
    objective = Objective(
        id="objective-test",
        title="compound",
        metadata={
            "application": "Chrome",
            "level": 60,
        },
    )

    plan = make_plan(
        [
            make_step(
                "step-1",
                "system.open_app",
                result_keys=["application", "status"],
            ),
            make_step(
                "step-2",
                "system.set_volume",
                result_keys=["volume", "status"],
            ),
            make_step(
                "step-3",
                "system.screen_capture",
                result_keys=["status"],
                artifact_keys=["image"],
            ),
        ]
    )

    result = make_result(
        output={
            "application": "Chrome",
            "status": "opened",
            "volume": 60,
        },
        artifacts={
            "image": object(),
        },
        step_evidence={
            "step-1": {
                "step_id": "step-1",
                "action": "system.open_app",
                "status": "completed",
                "results": {
                    "application": "Chrome",
                    "status": "opened",
                },
                "artifacts": {},
            },
            "step-2": {
                "step_id": "step-2",
                "action": "system.set_volume",
                "status": "completed",
                "results": {
                    "volume": 60,
                    "status": "set",
                },
                "artifacts": {},
            },
            "step-3": {
                "step_id": "step-3",
                "action": "system.screen_capture",
                "status": "completed",
                "results": {
                    "status": "captured",
                },
                "artifacts": {
                    "image": object(),
                },
            },
        },
    )

    verification = Verifier().verify(
        objective,
        plan,
        result,
    )

    check(
        verification.verified,
        "All compound steps should be verified.",
    )

    check(
        verification.metadata["steps_verified"] == 3,
        "All three steps should be verified.",
    )

    check(
        verification.metadata["steps_total"] == 3,
        "Plan should contain three steps.",
    )

from unittest.mock import MagicMock

from context import AssistantContext
from memory import WorkingMemory
from planner.context_builder import ContextBuilder


def test_context_builder_includes_context_provider_output():
    assistant_context = MagicMock(spec=AssistantContext)
    assistant_context.to_dict.return_value = {
        "user_id": "user-1",
        "organization_id": "org-1",
    }

    memory = WorkingMemory()

    captured = {}

    def provider(semantic_query, effective_assistant_context):
        captured["semantic_query"] = semantic_query
        captured["assistant_context"] = effective_assistant_context

        return {
            "documents": [
                {
                    "id": "doc-1",
                    "name": "manual.pdf",
                }
            ]
        }

    builder = ContextBuilder(
        assistant_context=assistant_context,
        context_providers=[provider],
    )

    result = builder.build(
        memory,
        semantic_query="buscar manual",
    )

    assert captured["semantic_query"] == "buscar manual"
    assert captured["assistant_context"] is assistant_context

    assert result["documents"] == [
        {
            "id": "doc-1",
            "name": "manual.pdf",
        }
    ]


def test_context_builder_ignores_provider_returning_none():
    memory = WorkingMemory()

    def provider(semantic_query, effective_assistant_context):
        return None

    builder = ContextBuilder(
        context_providers=[provider],
    )

    result = builder.build(
        memory,
        semantic_query="buscar algo",
    )

    assert "documents" not in result

def test_context_includes_semantic_knowledge() -> None:
    working_memory = WorkingMemory()

    semantic_memory = SemanticMemory()

    semantic_memory.learn(
        "chrome",
        "es una aplicación",
        True,
        confidence=1.0,
        source="episode",
    )

    retriever = SemanticRetriever(
        semantic_memory
    )

    builder = ContextBuilder(
        semantic_retriever=retriever
    )

    context = builder.build(
        working_memory,
        semantic_query="chrome",
    )

    knowledge = context[
        "semantic_knowledge"
    ]

    assert len(knowledge) == 1
    assert knowledge[0]["subject"] == "chrome"
    assert (
        knowledge[0]["predicate"]
        == "es una aplicación"
    )
    assert knowledge[0]["value"] is True
    assert knowledge[0]["confidence"] == 1.0
    assert knowledge[0]["source"] == "episode"
