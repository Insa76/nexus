import pytest

from documents import (
    DocumentChange,
    DocumentChangeType,
    DocumentSemanticChange,
)


def build_source_change() -> DocumentChange:
    return DocumentChange.modified(
        before="El contrato tiene vigencia de 12 meses.",
        after="El contrato tiene vigencia de 24 meses.",
        before_line=1,
        after_line=1,
    )


def test_semantic_change_preserves_source_change():
    source_change = build_source_change()

    result = DocumentSemanticChange(
        source_change=source_change,
        meaning="La vigencia contractual fue modificada.",
    )

    assert result.source_change == source_change
    assert result.source_change.change_type == (
        DocumentChangeType.MODIFIED
    )


def test_semantic_change_requires_meaning():
    source_change = build_source_change()

    with pytest.raises(
        ValueError,
        match="Semantic change meaning cannot be empty",
    ):
        DocumentSemanticChange(
            source_change=source_change,
            meaning="   ",
        )


def test_semantic_change_accepts_topics():
    source_change = build_source_change()

    result = DocumentSemanticChange(
        source_change=source_change,
        meaning="La vigencia contractual fue modificada.",
        topics=(" contrato ", "vigencia", " "),
    )

    assert result.topics == (
        "contrato",
        "vigencia",
    )


def test_semantic_change_accepts_impact():
    source_change = build_source_change()

    result = DocumentSemanticChange(
        source_change=source_change,
        meaning="La vigencia contractual fue modificada.",
        impact="Requiere revisión contractual.",
    )

    assert result.impact == (
        "Requiere revisión contractual."
    )


def test_semantic_change_normalizes_empty_impact():
    source_change = build_source_change()

    result = DocumentSemanticChange(
        source_change=source_change,
        meaning="La vigencia contractual fue modificada.",
        impact="   ",
    )

    assert result.impact is None


def test_semantic_change_accepts_confidence():
    source_change = build_source_change()

    result = DocumentSemanticChange(
        source_change=source_change,
        meaning="La vigencia contractual fue modificada.",
        confidence=0.92,
    )

    assert result.confidence == 0.92


@pytest.mark.parametrize(
    "confidence",
    [-0.01, 1.01],
)
def test_semantic_change_rejects_invalid_confidence(
    confidence,
):
    source_change = build_source_change()

    with pytest.raises(
        ValueError,
        match="Semantic change confidence must be between 0 and 1",
    ):
        DocumentSemanticChange(
            source_change=source_change,
            meaning="La vigencia contractual fue modificada.",
            confidence=confidence,
        )


def test_semantic_change_accepts_boundary_confidence_values():
    source_change = build_source_change()

    low = DocumentSemanticChange(
        source_change=source_change,
        meaning="Cambio detectado.",
        confidence=0.0,
    )

    high = DocumentSemanticChange(
        source_change=source_change,
        meaning="Cambio detectado.",
        confidence=1.0,
    )

    assert low.confidence == 0.0
    assert high.confidence == 1.0


def test_semantic_change_is_immutable():
    source_change = build_source_change()

    result = DocumentSemanticChange(
        source_change=source_change,
        meaning="La vigencia contractual fue modificada.",
    )

    with pytest.raises(AttributeError):
        result.meaning = "Otro significado"

from documents import (
    DocumentArtifact,
    IngestionResult,
    TextIngestor,
)


class FakeArtifactReader:
    def __init__(self, text: str) -> None:
        self.text = text
        self.received_artifact = None

    def read(self, artifact: DocumentArtifact) -> str:
        self.received_artifact = artifact
        return self.text


def test_text_ingestor_reads_artifact_and_creates_content():
    artifact = DocumentArtifact.create(
        location="local:/documents/notes.txt",
    )

    reader = FakeArtifactReader(
        "Este es el contenido del documento.",
    )

    ingestor = TextIngestor(reader)

    result = ingestor.ingest(artifact)

    assert isinstance(result, IngestionResult)
    assert result.artifact is artifact
    assert result.content.text == "Este es el contenido del documento."
    assert result.content.content_type == "text/plain"
    assert reader.received_artifact is artifact


def test_text_ingestor_preserves_empty_text():
    artifact = DocumentArtifact.create(
        location="local:/documents/empty.txt",
    )

    reader = FakeArtifactReader("")

    result = TextIngestor(reader).ingest(artifact)

    assert result.content.text == ""
    assert result.content.content_type == "text/plain"

from documents import (
    DocumentContentVersion,
    DocumentContentVersionStore,
    DocumentVersionComparator,
    DocumentVersionResolver,
)


def add_version(
    store: DocumentContentVersionStore,
    document_id: str,
    version: int,
    content_id: str,
) -> DocumentContentVersion:
    content_version = DocumentContentVersion.create(
        document_id=document_id,
        content_id=content_id,
        version=version,
        content_type="text/plain",
    )
    store.add(content_version)
    return content_version


def build_resolution(
    document_id: str = "doc-1",
    from_version: int = 1,
    from_content_id: str = "content-1",
    to_version: int = 2,
    to_content_id: str = "content-2",
):
    store = DocumentContentVersionStore()

    add_version(
        store,
        document_id=document_id,
        version=from_version,
        content_id=from_content_id,
    )

    add_version(
        store,
        document_id=document_id,
        version=to_version,
        content_id=to_content_id,
    )

    resolver = DocumentVersionResolver(store)

    return resolver.resolve_pair(
        document_id=document_id,
        from_version=from_version,
        to_version=to_version,
    )


def test_comparator_detects_changed_content():
    resolution = build_resolution(
        from_content_id="content-1",
        to_content_id="content-2",
    )

    comparator = DocumentVersionComparator()

    result = comparator.compare(resolution)

    assert result.document_id == "doc-1"
    assert result.from_version == 1
    assert result.to_version == 2
    assert result.from_content_id == "content-1"
    assert result.to_content_id == "content-2"
    assert result.changed is True
    assert result.unchanged is False

        from_content_id="content-1",

import pytest

from documents import (
    DocumentContentVersion,
    DocumentContentVersionStore,
    DocumentVersionResolver,
)


def build_store() -> DocumentContentVersionStore:
    return DocumentContentVersionStore()


def add_version(
    store: DocumentContentVersionStore,
    document_id: str,
    version: int,
    content_id: str,
) -> DocumentContentVersion:
    content_version = DocumentContentVersion.create(
        document_id=document_id,
        content_id=content_id,
        version=version,
        content_type="text/plain",
    )
    store.add(content_version)
    return content_version


def test_resolver_returns_specific_version():
    store = build_store()

    v1 = add_version(
        store,
        document_id="doc-1",
        version=1,
        content_id="content-1",
    )

    add_version(
        store,
        document_id="doc-1",
        version=2,
        content_id="content-2",
    )

    resolver = DocumentVersionResolver(store)

    result = resolver.resolve_version("doc-1", 1)

    assert result == v1


def test_resolver_raises_when_version_does_not_exist():
    store = build_store()

    add_version(
        store,
        document_id="doc-1",
        version=1,
        content_id="content-1",
    )

    resolver = DocumentVersionResolver(store)

    with pytest.raises(
        ValueError,
        match="Document content version not found: doc-1 v2",

        resolver.resolve_version("doc-1", 2)

from __future__ import annotations

from dataclasses import dataclass
from typing import Any

from documents.cognitive import (
    DocumentCognitiveAccess,
    DocumentCognitiveResult,
)
from processes.cognitive import (
    ProcessCognitiveAccess,
    ProcessCognitiveResult,
)
from processes.models import Process
from rules.cognitive import (
    RuleCognitiveAccess,
    RuleCognitiveResult,
)
from rules.models import Rule


@dataclass(frozen=True, slots=True)
class CognitiveCapabilityContext:
    """
    Contexto disponible para una operación cognitiva.

    Mantiene la información de entrada separada de las
    implementaciones concretas de las capabilities.
    """

    context: object
    parameters: dict[str, Any]


@dataclass(frozen=True, slots=True)
class CognitiveCapabilityInvocation:
    """
    Representa una capability cognitiva seleccionada,
    pero todavía no ejecutada.
    """

    capability: str
    operation: str
    parameters: dict[str, Any]


@dataclass(frozen=True, slots=True)
class CognitiveCapabilityDescriptor:
    """
    Describe una capability cognitiva disponible.

    El descriptor permite separar:

    - qué capability existe
    - qué operación expone
    - qué acción puede seleccionarla

    No contiene implementación ni ejecuta nada.
    """

    capability: str
    operation: str
    action: str


@dataclass(frozen=True, slots=True)
class CognitiveCapabilityOperation:
    """
    Describe una operación cognitiva disponible.

    Representa la capacidad concreta que el Cognitive Core
    puede utilizar.
    """

    capability: str
    operation: str
    action: str


class CognitiveCapabilityRouter:
    """
    Punto de entrada para que el Cognitive Core utilice capabilities.

    No conoce stores ni detalles internos de Documents, Processes o Rules.
    """

    def __init__(
        self,
        *,
        document_access: DocumentCognitiveAccess | None = None,
        process_access: ProcessCognitiveAccess | None = None,
        rule_access: RuleCognitiveAccess | None = None,
    ) -> None:
        self._document_access = document_access
        self._process_access = process_access
        self._rule_access = rule_access

    @property
    def document_access(self) -> DocumentCognitiveAccess | None:
        return self._document_access

    @property
    def process_access(self) -> ProcessCognitiveAccess | None:
        return self._process_access

    @property
    def rule_access(self) -> RuleCognitiveAccess | None:
        return self._rule_access

    @property
    def descriptors(self) -> tuple[CognitiveCapabilityDescriptor, ...]:
        """
        Devuelve las capabilities cognitivas actualmente disponibles.
        """

        descriptors: list[CognitiveCapabilityDescriptor] = []

        if self._document_access is not None:
            descriptors.append(
                CognitiveCapabilityDescriptor(
                    capability="documents",
                    operation="find_relevant",
                    action="documents.find_relevant",
                )
            )

        if self._process_access is not None:
            descriptors.extend(
                (
                    CognitiveCapabilityDescriptor(
                        capability="processes",
                        operation="find_relevant",
                        action="processes.find_relevant",
                    ),
                    CognitiveCapabilityDescriptor(
                        capability="processes",
                        operation="get",
                        action="processes.get",
                    ),
                    CognitiveCapabilityDescriptor(
                        capability="processes",
                        operation="list",
                        action="processes.list",
                    ),
                )
            )

        if self._rule_access is not None:
            descriptors.extend(
                (
                    CognitiveCapabilityDescriptor(
                        capability="rules",
                        operation="find_relevant",
                        action="rules.find_relevant",
                    ),
                    CognitiveCapabilityDescriptor(
                        capability="rules",
                        operation="get",
                        action="rules.get",
                    ),
                    CognitiveCapabilityDescriptor(
                        capability="rules",
                        operation="list",
                        action="rules.list",
                    ),
                )
            )

        return tuple(descriptors)

    def available_capabilities(self) -> tuple[str, ...]:
        """
        Devuelve los identificadores de las capabilities cognitivas
        actualmente disponibles para el Cognitive Core.
        """

        return tuple(
            descriptor.capability
            for descriptor in self.descriptors
        )

    def available_operations(
        self,
    ) -> tuple[CognitiveCapabilityOperation, ...]:
        """
        Devuelve las operaciones cognitivas actualmente disponibles.
        """

        return tuple(
            CognitiveCapabilityOperation(
                capability=descriptor.capability,
                operation=descriptor.operation,
                action=descriptor.action,
            )
            for descriptor in self.descriptors
        )

    def find_operation(
        self,
        *,
        action: str,
    ) -> CognitiveCapabilityOperation | None:
        """
        Busca la operación cognitiva asociada a una acción.
        """

        for operation in self.available_operations():
            if operation.action == action:
                return operation

        return None

    @staticmethod
    def summarize_operation_result(
        result: (
            DocumentCognitiveResult
            | ProcessCognitiveResult
            | RuleCognitiveResult
            | Process
            | tuple[Process, ...]
            | Rule
            | tuple[Rule, ...]
            | None
        ),
        *,
        operation: str,
    ) -> dict[str, Any]:
        if isinstance(result, DocumentCognitiveResult):
            count = len(result.documents)

            return {
                "operation": "documents.find_relevant",
                "result_count": count,
                "document_count": count,
                "query_count": len(result.query.documents),
                "has_results": count > 0,
                "status": "found" if count > 0 else "empty",
            }

        if isinstance(result, ProcessCognitiveResult):
            count = len(result.processes)

            return {
                "operation": "processes.find_relevant",
                "result_count": count,
                "has_results": count > 0,
                "status": "found" if count > 0 else "empty",
            }

        if isinstance(result, RuleCognitiveResult):
            count = len(result.rules)

            return {
                "operation": "rules.find_relevant",
                "result_count": count,
                "has_results": count > 0,
                "status": "found" if count > 0 else "empty",
            }

        if operation == "processes.get":
            count = 1 if isinstance(result, Process) else 0

            return {
                "operation": operation,
                "result_count": count,
                "has_results": count > 0,
                "status": "found" if count > 0 else "empty",
            }

        if operation == "processes.list":
            if not isinstance(result, tuple):
                raise TypeError(
                    f"Expected tuple result for {operation}, "
                    f"got {type(result)!r}"
                )

            if result and not all(
                isinstance(item, Process)
                for item in result
            ):
                raise TypeError(
                    f"Expected tuple[Process, ...] result for {operation}, "
                    f"got {type(result)!r}"
                )

            count = len(result)

            return {
                "operation": operation,
                "result_count": count,
                "has_results": count > 0,
                "status": "found" if count > 0 else "empty",
            }

        if operation == "rules.get":
            count = 1 if isinstance(result, Rule) else 0

            return {
                "operation": operation,
                "result_count": count,
                "has_results": count > 0,
                "status": "found" if count > 0 else "empty",
            }

        if operation == "rules.list":
            if not isinstance(result, tuple):
                raise TypeError(
                    f"Expected tuple result for {operation}, "
                    f"got {type(result)!r}"
                )

            if result and not all(
                isinstance(item, Rule)
                for item in result
            ):
                raise TypeError(
                    f"Expected tuple[Rule, ...] result for {operation}, "
                    f"got {type(result)!r}"
                )

            count = len(result)

            return {
                "operation": operation,
                "result_count": count,
                "has_results": count > 0,
                "status": "found" if count > 0 else "empty",
            }

        raise TypeError(
            f"Unsupported cognitive capability result type: {type(result)!r}"
        )

    def find_relevant_documents(
        self,
        *,
        context: object,
        name: str | None = None,
        mime_type: str | None = None,
        offset: int = 0,
        limit: int | None = None,
    ) -> DocumentCognitiveResult:
        if self._document_access is None:
            raise RuntimeError(
                "Document cognitive capability is not available."
            )

        return self._document_access.find_relevant(
            context=context,
            name=name,
            mime_type=mime_type,
            offset=offset,
            limit=limit,
        )

    def find_relevant_processes(
        self,
        *,
        context: object,
    ) -> ProcessCognitiveResult:
        if self._process_access is None:
            raise RuntimeError(
                "Process cognitive capability is not available."
            )

        return self._process_access.find_relevant(
            context=context,
        )

    def get_process(
        self,
        *,
        process_id: str,
    ) -> Process | None:
        if self._process_access is None:
            raise RuntimeError(
                "Process cognitive capability is not available."
            )

        return self._process_access.get(process_id)

    def list_processes(self) -> tuple[Process, ...]:
        if self._process_access is None:
            raise RuntimeError(
                "Process cognitive capability is not available."
            )

        return self._process_access.list()

    def find_relevant_rules(
        self,
        *,
        context: object,
    ) -> RuleCognitiveResult:
        if self._rule_access is None:
            raise RuntimeError(
                "Rule cognitive capability is not available."
            )

        return self._rule_access.find_relevant(
            context=context,
        )

    def get_rule(
        self,
        *,
        rule_id: str,
    ) -> Rule | None:
        if self._rule_access is None:
            raise RuntimeError(
                "Rule cognitive capability is not available."
            )

        return self._rule_access.get(rule_id)

    def list_rules(self) -> tuple[Rule, ...]:
        if self._rule_access is None:
            raise RuntimeError(
                "Rule cognitive capability is not available."
            )

        return self._rule_access.list()

    def select(
        self,
        *,
        action: str,
        parameters: dict[str, Any] | None = None,
    ) -> CognitiveCapabilityInvocation | None:
        """
        Determina si una intención corresponde a una
        capability cognitiva disponible.

        La selección no ejecuta ninguna operación.
        """

        parameters = parameters or {}

        for descriptor in self.descriptors:
            if descriptor.action != action:
                continue

            return CognitiveCapabilityInvocation(
                capability=descriptor.capability,
                operation=descriptor.operation,
                parameters=dict(parameters),
            )

        return None

    def invoke(
        self,
        invocation: CognitiveCapabilityInvocation,
        *,
        context: object,
        assistant_context: object | None = None,
    ) -> (
        DocumentCognitiveResult
        | ProcessCognitiveResult
        | RuleCognitiveResult
        | Process
        | tuple[Process, ...]
        | Rule
        | tuple[Rule, ...]
        | None
    ):
        """
        Ejecuta la operación cognitiva previamente seleccionada.

        La invocación ocurre sobre el acceso cognitivo de la capability.
        No ejecuta acciones del Runtime.
        """

        if invocation.capability == "documents":
            if invocation.operation != "find_relevant":
                raise ValueError(
                    f"Unsupported documents operation: "
                    f"{invocation.operation}"
                )

            if self._document_access is None:
                raise RuntimeError(
                    "Document cognitive capability is not available."
                )


            return self._document_access.find_relevant(
    context=(
        assistant_context
        if assistant_context is not None
        else context
    ),
    name=invocation.parameters.get("name"),
    mime_type=invocation.parameters.get("mime_type"),
    offset=invocation.parameters.get("offset", 0),
    limit=invocation.parameters.get("limit"),
)

        if invocation.capability == "processes":
            if self._process_access is None:
                raise RuntimeError(
                    "Process cognitive capability is not available."
                )

            if invocation.operation == "find_relevant":
                if assistant_context is None:
                    raise ValueError(
                        "processes.find_relevant requires 'assistant_context'."
                    )

                return self._process_access.find_relevant(
                    context=assistant_context,
                )

            if invocation.operation == "get":
                process_id = invocation.parameters.get("process_id")

                if not process_id:
                    raise ValueError(
                        "processes.get requires 'process_id'."
                    )

                if assistant_context is None:
                    raise ValueError(
                        "processes.get requires 'assistant_context'."
                    )

                return self._process_access.get(
                    process_id,
                    context=assistant_context,
                )

            if invocation.operation == "list":
                if assistant_context is None:
                    raise ValueError(
                        "processes.list requires 'assistant_context'."
                    )

                return self._process_access.list(
                    context=assistant_context,
                )

            raise ValueError(
                f"Unsupported processes operation: "
                f"{invocation.operation}"
            )


        if invocation.capability == "rules":
            if self._rule_access is None:
                raise RuntimeError(
                    "Rule cognitive capability is not available."
                )

            if invocation.operation == "find_relevant":
                if assistant_context is None:
                    raise ValueError(
                        "rules.find_relevant requires 'assistant_context'."
                    )

                return self._rule_access.find_relevant(
                    context=assistant_context,
                )

            if invocation.operation == "get":
                rule_id = invocation.parameters.get("rule_id")

                if not rule_id:
                    raise ValueError(
                        "rules.get requires 'rule_id'."
                    )

                if assistant_context is None:
                    raise ValueError(
                        "rules.get requires 'assistant_context'."
                    )

                return self._rule_access.get(
                    rule_id,
                    context=assistant_context,
                )

            if invocation.operation == "list":
                if assistant_context is None:
                    raise ValueError(
                        "rules.list requires 'assistant_context'."
                    )

                return self._rule_access.list(
                    context=assistant_context,
                )

        raise ValueError(
            f"Unsupported cognitive capability: "
            f"{invocation.capability}"
        )
