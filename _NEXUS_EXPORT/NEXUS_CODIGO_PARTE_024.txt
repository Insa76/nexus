// ✅ AXIOS CONFIG CON TOKEN
const api = axios.create({
  baseURL: API
})

api.interceptors.request.use(config => {
  const token = localStorage.getItem("token")


  if (token) {
    config.headers.Authorization = `Bearer ${token}`
  }

  return config

export default function App() {

  const [installed, setInstalled] = useState(true)
  const [kiosk, setKiosk] = useState(false)

  const [stats, setStats] = useState({})
  const [jobs, setJobs] = useState({})
  const [logs, setLogs] = useState([])
  const [plugins, setPlugins] = useState([])
  const [suggestions, setSuggestions] = useState([])
  const [market, setMarket] = useState([])

  // ======================
  // KIOSK MODE
  // ======================
  useEffect(() => {
    const params = new URLSearchParams(window.location.search)

    if (params.get("kiosk")) {
      setKiosk(true)
      document.body.style.margin = "0"
      document.body.style.overflow = "hidden"
      document.body.style.cursor = "none"
    }
  }, [])

  // ======================
  // SETUP CHECK (SAFE)
  // ======================
  useEffect(() => {
    api.get("/setup")
      .then(r => setInstalled(r.data?.installed ?? true))
      .catch(() => setInstalled(true)) // fallback
  }, [])

  // ======================
  // DASHBOARD LOADER (SAFE MODE)
  // ======================
  const load = async () => {

    const token = localStorage.getItem("token")
     if (!token) return

    try {
      const safe = async (fn, fallback) => {
        try {
          const r = await fn()
          return r.data
        } catch {
          return fallback
        }
      }

      const [
        s,
        l,
        p,
        sug,
        j
      ] = await Promise.all([
        safe(() => api.get("/dashboard/stats"), {}),
        safe(() => api.get("/dashboard/logs"), { logs: [] }),
        safe(() => api.get("/dashboard/plugins"), { plugins: [] }),
        safe(() => api.get("/dashboard/suggestions"), { suggestions: [] }),
        safe(() => api.get("/jobs/"), {})
      ])

      setStats(s)
      setLogs(l.logs || [])
      setPlugins(p.plugins || [])
      setSuggestions(sug.suggestions || [])
      setJobs(j || {})

    } catch (e) {
      console.log("dashboard load fail", e)
    }
  }

  useEffect(() => {

  const start = () => {
    const token = localStorage.getItem("token")
    if (!token) return

    load()
    const i = setInterval(load, 5000)
    return () => clearInterval(i)
  }

  // pequeña espera para asegurar login
  setTimeout(start, 300)

}, [])

  // ======================
  // AUTH
  // ======================
  const token = localStorage.getItem("token")

  if (window.location.pathname === "/login") return <Login />
  if (window.location.pathname === "/register") return <Register />

  if (!token) {
    window.location.href = "/login"
    return null
  }

  if (window.location.pathname === "/assistant") return <AssistantMode />
  if (window.location.pathname === "/wallboard") return <Wallboard />
  if (window.location.pathname === "/users") return <Users />
  if (window.location.pathname === "/builder") return <Builder />

  if (!installed) return <Setup />

  return <Dashboard />
}

import { useEffect, useState } from "react"
import Presence from "../presence/Presence"

export default function AssistantMode(){

  const [expanded,setExpanded]=useState(false)
  const [mood,setMood]=useState("idle")

  /* ===============================
     HOTKEY
  =============================== */

  useEffect(()=>{

    const handler=(e)=>{
      if(e.ctrlKey && e.code==="Space"){
        setExpanded(v=>!v)
      }
    }

    window.addEventListener("keydown",handler)
    return ()=>window.removeEventListener("keydown",handler)

  },[])


  /* ===============================
     WEBSOCKET SISTEMA
  =============================== */

  useEffect(()=>{

    const ws = new WebSocket("ws://127.0.0.1:8000/ws")

    ws.onmessage = (e)=>{

      const msg = JSON.parse(e.data)

      if(msg.type==="job_failed"){
        setMood("error")
      }

      if(msg.type==="job_done"){
        setMood("happy")
      }

      if(msg.type==="job_start"){
        setMood("thinking")
      }

      // volver a idle después de 2s
      setTimeout(()=>setMood("idle"),2000)
    }

    ws.onerror = ()=>console.warn("WS Assistant desconectado")

    return ()=>ws.close()

  },[])


  return(
    <div style={styles.wrapper}>

      <Presence mood={mood}/>

      {expanded && (
        <div style={styles.panel}>
          <h3>Nexus</h3>
          <p>Modo asistente activo</p>
        </div>
      )}

    </div>
  )
}

import ReactFlow, { Background, Controls } from "reactflow"
import "reactflow/dist/style.css"

import { useEffect, useState } from "react"
import axios from "axios"

const api = axios.create({
  baseURL: "http://127.0.0.1:8010",
  headers: { "x-api-key": "nexus-secret" }
})

export default function BrainMap(){

  const [nodes,setNodes] = useState([])
  const [edges,setEdges] = useState([])

  useEffect(()=>{

    const load = async ()=>{

      try{

        const goals = await api.get("/goals")
        const jobs = await api.get("/jobs")

        buildGraph(goals.data, jobs.data)

      }catch(e){
        console.error(e)
      }

    }

    load()

    const i = setInterval(load,2000)

    return ()=>clearInterval(i)

  },[])


  function buildGraph(goals,jobs){

    const n=[]
    const e=[]

    let id=1

    goals.forEach((g,i)=>{

      const goalId = "goal_"+id++

      n.push({
        id:goalId,
        position:{x:100,y:i*150},
        data:{label:"🎯 "+g.goal},
        style:{background:"#8e44ad",color:"white"}
      })

      Object.entries(jobs).forEach(([jid,j])=>{

        const jobNode="job_"+jid

        n.push({
          id:jobNode,
          position:{x:400,y:i*150+50},
          data:{label:"⚙ "+j.workflow},
          style:{background:"#2980b9",color:"white"}
        })

        e.push({
          id:"e_"+goalId+"_"+jid,
          source:goalId,
          target:jobNode
        })

      })

    })

    setNodes(n)
    setEdges(e)

  }


  return(

    <div style={{height:"600px",background:"#111"}}>

      <ReactFlow
        nodes={nodes}
        edges={edges}
        fitView
      >

        <Background/>
        <Controls/>

      </ReactFlow>

    </div>

  )

}

import axios from "axios"
import JobTimeline from "./JobTimeline"

const api = axios.create({
  baseURL: "http://127.0.0.1:8010"
})

api.interceptors.request.use((config) => {
  const token = localStorage.getItem("token")

  if (token) {
    config.headers.Authorization = `Bearer ${token}`
  }

  return config
})

export default function JobsPanel(){

  const [jobs,setJobs]=useState({})
  const [metrics,setMetrics]=useState({})
  const [expanded,setExpanded]=useState(null)
  const [selected,setSelected]=useState(null)

  /* =========================
     polling
  ========================= */

  useEffect(() => {

  const loadInitial = async () => {
    const [jobsRes, metricsRes] = await Promise.all([
      api.get("/jobs"),
      api.get("/metrics")
    ])
    setJobs(jobsRes.data)
    setMetrics(metricsRes.data)
  }

  loadInitial()

  const ws = new WebSocket("ws://127.0.0.1:8010/ws")

  ws.onmessage = (e) => {

    const msg = JSON.parse(e.data)

    if (msg.type === "job_update") {

      setJobs(prev => ({
        ...prev,
        [msg.job_id]: {
          ...prev[msg.job_id],
          status: msg.status,
          result: msg.result
        }
      }))
    }

    if (msg.type === "metrics_update") {
      setMetrics(msg.data)
    }

    if (msg.type === "circuit_update") {
      setMetrics(prev => ({
        ...prev,
        circuit_state: msg.state
      }))
    }
  }

  return () => ws.close()

}, [])

  /* =========================
     ordenar por fecha
  ========================= */

  const orderedEntries = useMemo(()=>{
    return Object.entries(jobs)
      .sort((a,b)=>{
        const da = new Date(a[1].updated_at || 0)
        const db = new Date(b[1].updated_at || 0)
        return db - da
      })
  },[jobs])

  const list = orderedEntries.map(e=>e[1])

  /* =========================
     métricas
  ========================= */

  const stats = useMemo(()=>( {
    total: list.length,
    running: list.filter(j=>j.status==="running").length,
    queued: list.filter(j=>j.status==="queued").length,
    done: list.filter(j=>j.status==="done").length,
    error: list.filter(j=>j.status==="error").length
  }),[jobs])

  /* ========================= */

  const color=(s)=>{
    if(s==="running") return "#f6c344"
    if(s==="done") return "#2ecc71"
    if(s==="queued") return "#3498db"
    if(s==="error") return "#e74c3c"
    return "#999"
  }

  const toggle = (id)=>{
    setExpanded(expanded===id ? null : id)
  }

  const formatResult = (r)=>{
    if(!r) return "Sin resultado"
    if(typeof r === "string") return r
    return JSON.stringify(r,null,2)
  }

  /* ========================= */

  return(
  <div style={styles.wrap}>

    <h3 style={{marginBottom:14,fontSize:18}}>Jobs</h3>

    <div style={styles.stats}>
      <Stat label="Total" value={stats.total}/>
      <Stat label="Running" value={stats.running}/>
      <Stat label="Queued" value={stats.queued}/>
      <Stat label="Done" value={stats.done}/>
      <Stat label="Error" value={stats.error}/>
    </div>

    <div style={styles.list}>

      {orderedEntries.map(([id,j])=>{

        const isExpanded = expanded===id

        return(
          <div key={id} style={styles.rowWrap}>

            <div 
              style={styles.row}
              onClick={()=>toggle(id)}
            >

              <div style={{display:"flex",alignItems:"center",gap:8}}>

                <span style={styles.arrow}>
                  {isExpanded ? "▾" : "▸"}
                </span>

                <span style={styles.version}>

                </span>

                  <span style={styles.time}>

                  </span>
                )}

                  background: color(j.status),
                  animation: j.status==="running"
                    ? "pulse 1.2s infinite"
                    : "none"
                }}>
                  {j.status}
                </span>

              </div>

            </div>

            {isExpanded && (
              <div style={styles.detail}>

                {/* RESULTADO DEL JOB */}

                <pre style={{
                  ...styles.pre,

import { useEffect, useRef } from "react"
import face from "../../assets/face-mask.png"
import { NexusState } from "../../nexusState"

export default function Presence() {

  const ref = useRef()
  const particlesRef = useRef([])
  const animationRef = useRef()
  const audioLevelRef = useRef(0)

  useEffect(() => {

    const canvas = ref.current
    const ctx = canvas.getContext("2d")

    /* =========================
       CANVAS SCALE SAFE
    ========================= */

    function resize() {
      const DPR = window.devicePixelRatio || 1

      canvas.width = window.innerWidth * DPR
      canvas.height = window.innerHeight * DPR

      canvas.style.width = "100vw"
      canvas.style.height = "100vh"

      ctx.setTransform(1,0,0,1,0,0)
      ctx.scale(DPR, DPR)
    }

    resize()
    window.addEventListener("resize", resize)

    /* =========================
       VARIABLES
    ========================= */

    const particles = particlesRef.current
    let mouse = { x: window.innerWidth/2, y: window.innerHeight/2 }
    let t = 0

    /* =========================
       AUDIO REACTIVE
    ========================= */

    navigator.mediaDevices?.getUserMedia({ audio: true })
      .then(stream => {

        const audioCtx = new AudioContext()
        const analyser = audioCtx.createAnalyser()
        const source = audioCtx.createMediaStreamSource(stream)

        analyser.fftSize = 256
        source.connect(analyser)

        const data = new Uint8Array(analyser.frequencyBinCount)

        function readAudio() {
          analyser.getByteFrequencyData(data)
          const avg = data.reduce((a,b)=>a+b,0) / data.length
          audioLevelRef.current = avg / 255
          requestAnimationFrame(readAudio)
        }

        readAudio()

      })
      .catch(() => {
        audioLevelRef.current = 0
      })

    /* =========================
       MOUSE TRACK
    ========================= */

    window.addEventListener("mousemove", e => {
      mouse.x = e.clientX
      mouse.y = e.clientY
    })

    /* =========================
       CARGAR MÁSCARA
    ========================= */

    const img = new Image()
    img.src = face

    img.onload = () => {

      particles.length = 0

      const off = document.createElement("canvas")
      const octx = off.getContext("2d")

      const size = 500
      off.width = size
      off.height = size

      octx.drawImage(img, 0, 0, size, size)
      const data = octx.getImageData(0, 0, size, size).data

      const gap = 4

      for (let y = 0; y < size; y += gap) {
        for (let x = 0; x < size; x += gap) {

          const i = (y * size + x) * 4

          if (data[i + 3] > 120) {

            particles.push({
              x: Math.random() * window.innerWidth,
              y: Math.random() * window.innerHeight,
              tx: window.innerWidth/2 - size/2 + x,
              ty: window.innerHeight/2 - size/2 + y,
              vx: 0,
              vy: 0
            })
          }
        }
      }

      animate()
    }

    /* =========================
       COLOR DINÁMICO
    ========================= */

    function color() {
      switch (NexusState.mode) {
        case "listening": return "rgba(0,200,255,0.9)"
        case "thinking": return "rgba(255,170,0,0.95)"
        case "speaking": return "rgba(124,92,255,1)"
        case "error": return "rgba(255,50,80,0.9)"
        default: return "rgba(124,92,255,0.55)"
      }
    }

    /* =========================
       LOOP PRINCIPAL
    ========================= */

    function animate() {

      ctx.clearRect(0, 0, canvas.width, canvas.height)

      t += 0.02
      const breathe = Math.sin(t) * 2

      ctx.shadowBlur = 24
      ctx.shadowColor = "#7c5cff"

      const mx = (mouse.x - window.innerWidth/2) * 0.03
      const my = (mouse.y - window.innerHeight/2) * 0.03

      for (const p of particles) {

        const dx = p.tx + mx - p.x
        const dy = p.ty + my - p.y

        p.vx += dx * 0.02
        p.vy += dy * 0.02

        const audioBoost = audioLevelRef.current * 8

        p.vx += (Math.random() - 0.5) * audioBoost
        p.vy += (Math.random() - 0.5) * audioBoost

        p.vx *= 0.86
        p.vy *= 0.86

        p.x += p.vx
        p.y += p.vy + breathe

        ctx.fillStyle = color()
        ctx.fillRect(p.x, p.y, 2, 2)
      }

      animationRef.current = requestAnimationFrame(animate)
    }

    /* =========================
       CLEANUP
    ========================= */

    return () => {
      cancelAnimationFrame(animationRef.current)
      window.removeEventListener("resize", resize)
    }

  }, [])

  return (
    <canvas
      ref={ref}
      style={{
        position: "fixed",
        inset: 0,
        pointerEvents: "none",
        zIndex: 0,
        opacity: 0.45,
        transition: "opacity .4s ease"
      }}
    />
  )
}
