import uuid
import json

from core.database import get_connection

from core.embeddings import (
    fake_embedding,
    cosine_similarity
)


def save_vector_memory(

    agent,
    content,
    metadata=None
):

    embedding = fake_embedding(
        content
    )

    conn = get_connection()
    cur = conn.cursor()

    cur.execute("""
        INSERT INTO vector_memory
        (
            id,
            agent,
            content,
            embedding,
            metadata
        )
        VALUES (%s,%s,%s,%s,%s)
    """, (
        str(uuid.uuid4()),
        agent,
        content,
        json.dumps(embedding),
        json.dumps(metadata)
        if metadata
        else None
    ))

    conn.commit()
    conn.close()


def search_memory(

    agent,
    query,
    limit=5
):

    query_embedding = fake_embedding(
        query
    )

    conn = get_connection()
    cur = conn.cursor()

    cur.execute("""
        SELECT *
        FROM vector_memory
        WHERE agent = %s
    """, (agent,))

    rows = cur.fetchall()

    conn.close()

    scored = []

    for row in rows:

        embedding = row["embedding"]

        if isinstance(embedding, str):
            embedding = json.loads(
                embedding
            )

        score = cosine_similarity(
            query_embedding,
            embedding
        )

        scored.append({
            "score": score,
            "memory": dict(row)
        })

    scored.sort(
        key=lambda x: x["score"],
        reverse=True
    )

    return scored[:limit]

def detect_ui_elements(
    image_path
):

    image = Image.open(image_path)

    data = pytesseract.image_to_data(

        image,

        output_type=Output.DICT
    )

    elements = []

    total = len(data["text"])

    for i in range(total):

        text = data["text"][i].strip()

        if not text:
            continue

        elements.append({

            "text": text,

            "x": data["left"][i],

            "y": data["top"][i],

            "w": data["width"][i],

            "h": data["height"][i]
        })

    return elements

import asyncio
import inspect
from collections import defaultdict


class EventBus:

    def __init__(self):
        self.listeners = defaultdict(list)

    def subscribe(self, event_type: str, handler):
        self.listeners[event_type].append(handler)

    async def publish(self, event_type: str, payload: dict):

        if event_type not in self.listeners:
            return

        for handler in self.listeners[event_type]:

            asyncio.create_task(
                self._safe_execute(handler, event_type, payload)
            )

    async def _safe_execute(self, handler, event_type, payload):

        try:
            sig = inspect.signature(handler)

            # Handler(event_type, payload)
            if len(sig.parameters) == 2:
                await handler(event_type, payload)

            # Handler(payload)
            else:
                await handler(payload)

        except Exception as e:
            print(f"⚠ Event handler error [{event_type}]:", e)


bus = EventBus()

import asyncio
from collections import defaultdict
from events.bus import bus
from jobs.manager import manager
from workflows.repository import get_workflow_by_name


class RetryManager:

    def __init__(self):
        self.retry_count = defaultdict(int)

    async def handle_step_error(self, payload):

        workflow = payload["workflow"]
        job_id = payload.get("job_id")

        wf = get_workflow_by_name(workflow)

        if not wf:
            return

        retry_conf = wf.get("retry")
        if not retry_conf:
            return

        max_r = retry_conf.get("max_retries", 0)
        delay = retry_conf.get("delay_seconds", 0)
        backoff = retry_conf.get("backoff", "fixed")

        self.retry_count[job_id] += 1
        attempt = self.retry_count[job_id]

        if attempt > max_r:

            # 🔥 AQUÍ VA EL PASO 4
            await bus.publish("job.failed.final", {
                "workflow": workflow,
                "job_id": job_id,
                "tenant_id": wf.get("tenant_id"),
                "workspace_id": wf.get("workspace_id"),
                "error": "Max retries exceeded"
            })

            return

        # delay inteligente
        if backoff == "exponential":
            wait = delay * (2 ** (attempt - 1))
        else:
            wait = delay

        await asyncio.sleep(wait)

        await manager.create_job(
            workflow,
            wf.get("tenant_id"),
            wf.get("workspace_id")
        )


retry_manager = RetryManager()


def register_retry():
    bus.subscribe("step.error", retry_manager.handle_step_error)

import asyncio
from watchdog.observers import Observer
from watchdog.events import FileSystemEventHandler

from workflows.engine import run_workflow
from jobs.manager import manager
from events.bus import bus


class FolderHandler(FileSystemEventHandler):

    def __init__(self, path, workflow_name):
        self.path = path
        self.workflow_name = workflow_name

    def on_created(self, event):

     if event.is_directory:
        return

     asyncio.create_task(
        bus.publish("filesystem.created", {
            "path": event.src_path
        })
    )


class Watcher:

    def __init__(self):
        self.observer = Observer()
        self.active = []

    def watch(self, path, workflow_name):

        handler = FolderHandler(path, workflow_name)

        self.observer.schedule(handler, path, recursive=False)
        self.active.append((path, workflow_name))

    def start(self):
        self.observer.start()


watcher = Watcher()

    to,

):

    )

import httpx


async def http_request(method, url, headers=None, body=None, timeout=10):

    async with httpx.AsyncClient(timeout=timeout) as client:

        response = await client.request(
            method=method,
            url=url,
            headers=headers,
            json=body
        )

        response.raise_for_status()

        return {
            "status": response.status_code,
            "data": response.json() if "application/json" in response.headers.get("content-type","") else response.text
        }

                    r"\{.*\}",

                    re.DOTALL
                )

                    )

                    continue

                parsed = json.loads(
                    match.group(0)
                )

            except Exception as e:

                history.append(
                    f"JSON parse error: {str(e)}"
                )

                save_memory(
                    "main_agent",
                    "error",
                    str(e)
                )

                continue

            # ==================================================
            # DONE
            # ==================================================
            if parsed.get("done"):

                print("✅ AGENT COMPLETE")

                save_memory(
                    "main_agent",
                    "goal_completed",
                    goal
                )

                save_vector_memory(
                    "main_agent",
                    f"Completed goal: {goal}"
                )

                return {
                    "success": True,
                    "history": history
                }

            # ==================================================
            # TOOL
            # ==================================================
            tool = parsed.get("tool")

            params = parsed.get(
                "params",
                {}
            )

            print(f"🛠 TOOL: {tool}")
            print(f"📦 PARAMS: {params}")

            # ==================================================
            # TOOL EXECUTION
            # ==================================================
            try:

                tool_result = await execute_action(
                    tool,
                    params
                )

                print(f"✅ TOOL RESULT: {tool_result}")

                # ==============================================
                # SUPERVISOR
                # ==============================================
                review = supervisor.review(
                    history
                )

                print(f"🛡 SUPERVISOR: {review}")

                if "stop" in review.lower():

                    print("🛑 SUPERVISOR STOPPED AGENT")

                    save_memory(
                        "main_agent",
                        "supervisor_stop",
                        review
                    )

                    return {
                        "success": False,
                        "error": "Supervisor stopped execution"
                    }

            except Exception as tool_error:

                print(f"❌ TOOL ERROR: {tool_error}")

                save_memory(
                    "main_agent",
                    "tool_error",
                    str(tool_error)
                )

                save_vector_memory(
                    "main_agent",
                    f"Tool error: {tool_error}"
                )

                history.append(f"""
Tool:
{tool}

Params:
{params}

Error:
{str(tool_error)}
""")

                continue

            # ==================================================
            # SAVE MEMORY
            # ==================================================
            save_memory(

                "main_agent",

                "tool_execution",

                f"""
Tool: {tool}

Params:
{params}

Result:
{tool_result}
""",

                {
                    "tool": tool
                }
            )

            # ==================================================
            # SAVE VECTOR MEMORY
            # ==================================================
            save_vector_memory(

                "main_agent",

                f"""
Tool: {tool}

Params:
{params}

Result:
{tool_result}
"""
            )

            # ==================================================
            # OBSERVATION HISTORY
            # ==================================================
            history.append(f"""
Tool:
{tool}

Params:
{params}

Result:
{tool_result}
""")

        # ======================================================
        # MAX STEPS REACHED
        # ======================================================
        print("❌ MAX STEPS REACHED")

        save_memory(
            "main_agent",
            "max_steps",
            goal
        )

        save_vector_memory(
            "main_agent",
            f"Max steps reached for goal: {goal}"
        )

            "success": False,
            "error": "Max steps reached",
            "history": history
        }
