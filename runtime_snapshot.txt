from __future__ import annotations

from packages.core.src.core.kernel import Kernel

from packages.runtime.src.runtime.runtime import Runtime

from packages.actions.src.actions.registry import CapabilityRegistry
from packages.actions.src.actions.manager import CapabilityManager

from packages.actions.src.actions.echo import EchoCapability
from packages.actions.src.actions.open_app import OpenAppCapability
from packages.runtime.src.runtime.executor import Executor


class Bootstrap:
    """
    Inicializa todos los componentes principales
    del Runtime.
    """

    def build(self) -> Runtime:

        kernel = Kernel()

        runtime = Runtime(
            kernel=kernel,
        )

        #
        # Capability Registry
        #

        registry = CapabilityRegistry()

        registry.register(
            EchoCapability()
        )

        registry.register(
            OpenAppCapability()
        )

        manager = CapabilityManager(
            registry
        )

        executor = Executor(
          manager
        )

        runtime.configure_executor(
          executor
        )

        runtime.capability_registry = registry

        runtime.capability_manager = manager

        return runtime

from __future__ import annotations

from collections.abc import Awaitable, Callable
from typing import Any


EventHandler = Callable[[Any], Awaitable[None]]


class Dispatcher:
    """
    Despachador de eventos del Runtime.

    Su única responsabilidad es encontrar el
    handler registrado para un evento y ejecutarlo.

    No conoce IA.
    No conoce Planner.
    No conoce Memory.
    No conoce Providers.
    """

    def __init__(self) -> None:

        self._handlers: dict[str, EventHandler] = {}

    def register(
        self,
        event_type: str,
        handler: EventHandler,
    ) -> None:
        """
        Registra un handler para un tipo de evento.
        """

        self._handlers[event_type] = handler

    def unregister(
        self,
        event_type: str,
    ) -> None:

        self._handlers.pop(
            event_type,
            None,
        )

    def has(
        self,
        event_type: str,
    ) -> bool:

        return event_type in self._handlers

    async def dispatch(
        self,
        event: Any,
    ) -> None:
        """
        Envía el evento al handler correspondiente.
        """

        handler = self._handlers.get(
            event.type
        )

        if handler is None:
            return

        await handler(event)

    def clear(self) -> None:

        self._handlers.clear()

    def handlers(self) -> list[str]:

        return sorted(
            self._handlers.keys()
        )

from __future__ import annotations

import time

from planner.execution_context import ExecutionContext

from runtime.execution_result import ExecutionResult


class Executor:
    """
    Ejecuta un ExecutionContext completo.
    """

    def __init__(
        self,
        runtime,
    ) -> None:

        self.runtime = runtime

    async def execute(
        self,
        context: ExecutionContext,
    ) -> ExecutionResult:

        start = time.perf_counter()

        result = ExecutionResult()

        if self.runtime.action_manager is None:

            result.success = False

            result.message = (
                "ActionManager has not been configured."
            )

            return result

        while True:

            step = context.current_plan_step()

            if step is None:
                break

            await self.runtime.action_manager.execute(
                context
            )

        result.success = not context.has_errors()

        result.output = context.results

        result.artifacts = context.artifacts

        result.metadata = context.variables

        if context.errors:

            result.message = "\n".join(
                context.errors
            )

        result.execution_time = round(

            time.perf_counter() - start,

            3,

        )

        return result

from __future__ import annotations

from planner.execution_context import ExecutionContext

from runtime.execution_result import ExecutionResult


class Executor:
    """
    Ejecuta un ExecutionContext completo.

    El Executor no conoce las Actions.
    Solamente coordina el Runtime.
    """

    def __init__(
        self,
        runtime,
    ) -> None:

        self.runtime = runtime

    async def execute(
        self,
        context: ExecutionContext,
    ) -> ExecutionResult:

        result = ExecutionResult()

        if self.runtime.action_manager is None:

            result.success = False
            result.errors.append(
                "ActionManager has not been configured."
            )

            return result

        while True:

            step = context.current_plan_step()

            if step is None:
                break

            await self.runtime.action_manager.execute(
                context
            )

        result.success = len(
            context.errors
        ) == 0

        result.data = context.results.copy()

        result.errors.extend(
            context.errors
        )

        return result

from __future__ import annotations

from packages.runtime.src.runtime.runtime import Runtime


class Lifecycle:
    """
    Controla el ciclo de vida del Runtime.

    Este componente únicamente administra el
    inicio y la detención del sistema.
    """

    def __init__(self, runtime: Runtime) -> None:

        self.runtime = runtime

        self.running = False

    async def start(self) -> None:
        """
        Inicia el Runtime.
        """

        if self.running:
            return

        self.running = True

        self.runtime.kernel.start()

    async def stop(self) -> None:
        """
        Detiene el Runtime.
        """

        if not self.running:
            return

        self.running = False

        self.runtime.kernel.stop()

    async def restart(self) -> None:
        """
        Reinicia el Runtime.
        """

        await self.stop()

        await self.start()

from __future__ import annotations

from typing import Any


class Registry:
    """
    Registro de módulos del Runtime.

    Mantiene un catálogo de todos los módulos
    cargados en el sistema.

    No crea módulos.
    No ejecuta módulos.
    No conoce implementaciones.
    """

    def __init__(self) -> None:

        self._modules: dict[str, Any] = {}

    def register(
        self,
        name: str,
        module: Any,
    ) -> None:
        """
        Registra un módulo.
        """

        self._modules[name] = module

    def get(
        self,
        name: str,
    ) -> Any:
        """
        Devuelve un módulo registrado.
        """

        if name not in self._modules:

            raise KeyError(
                f"Module '{name}' not registered."
            )

        return self._modules[name]

    def has(
        self,
        name: str,
    ) -> bool:

        return name in self._modules

    def unregister(
        self,
        name: str,
    ) -> None:

        self._modules.pop(
            name,
            None,
        )

    def clear(self) -> None:

        self._modules.clear()

    def modules(self) -> list[str]:

        return sorted(
            self._modules.keys()
        )

from __future__ import annotations

from core.kernel import Kernel

from planner.execution_context import ExecutionContext

from runtime.dispatcher import Dispatcher
from runtime.executor import Executor
from runtime.registry import Registry
from runtime.scheduler import Scheduler
from runtime.service_container import ServiceContainer
from runtime.execution_result import ExecutionResult

from actions.registry import ActionRegistry
from actions.manager import ActionManager


class Runtime:
    """
    Runtime principal de NEXUS.

    Es el orquestador del sistema.

    Su responsabilidad es coordinar los distintos
    componentes sin conocer sus implementaciones.

    Nunca contiene lógica de negocio.
    """

    def __init__(
        self,
        kernel: Kernel,
    ) -> None:

        self.kernel = kernel

        self.registry = Registry()

        self.container = ServiceContainer()

        self.scheduler = Scheduler()

        self.dispatcher = Dispatcher()

        #
        # Se inicializan en Bootstrap
        #

        self.action_registry: ActionRegistry | None = None

        self.action_manager: ActionManager | None = None

        self.executor: Executor | None = None

    def configure_executor(
        self,
        executor: Executor,
    ) -> None:

        self.executor = executor

    async def execute(
        self,
        context: ExecutionContext,
    ) -> ExecutionResult:
        """
        Ejecuta un plan completo.
        """

        if self.executor is None:
            raise RuntimeError(
                "Runtime executor has not been configured."
            )

        return await self.executor.execute(
            context
        )

    def register_module(
        self,
        name: str,
        module,
    ) -> None:

        self.registry.register(
            name,
            module,
        )

    def register_service(
        self,
        name: str,
        service,
    ) -> None:

        self.container.register(
            name,
            service,
        )

    def configure_actions(
        self,
        registry: ActionRegistry,
        manager: ActionManager,
    ) -> None:
        """
        Registra el sistema de Actions del Runtime.
        """

        self.action_registry = registry
        self.action_manager = manager

from __future__ import annotations

import asyncio
from dataclasses import dataclass
from typing import Awaitable, Callable


TaskCallback = Callable[[], Awaitable[None]]


@dataclass(slots=True)
class ScheduledTask:
    """
    Representa una tarea periódica.
    """

    name: str
    interval: float
    callback: TaskCallback


class Scheduler:
    """
    Scheduler principal de NEXUS.

    Administra tareas periódicas.

    No conoce IA.
    No conoce Runtime.
    No conoce Providers.
    No conoce Planner.

    Solo ejecuta tareas registradas.
    """

    def __init__(self) -> None:

        self._tasks: list[ScheduledTask] = []

        self._running = False

    def register(
        self,
        name: str,
        interval: float,
        callback: TaskCallback,
    ) -> None:

        self._tasks.append(

            ScheduledTask(

                name=name,

                interval=interval,

                callback=callback,
            )
        )

    async def start(self) -> None:

        if self._running:
            return

        self._running = True

        await asyncio.gather(

            *[

                self._run_task(task)

                for task in self._tasks

            ]
        )

    async def stop(self) -> None:

        self._running = False

    async def _run_task(
        self,
        task: ScheduledTask,
    ) -> None:

        while self._running:

            try:

                await task.callback()

            except Exception as ex:

                print(

                    f"[Scheduler] {task.name}: {ex}"

                )

            await asyncio.sleep(

                task.interval

            )

    def clear(self) -> None:

        self._tasks.clear()

    def tasks(self) -> list[str]:

        return [

            task.name

            for task in self._tasks

        ]

from __future__ import annotations

from typing import Any


class ServiceContainer:
    """
    Contenedor de dependencias del Runtime.

    Es el único responsable de almacenar y entregar
    servicios compartidos entre los distintos módulos.

    No crea objetos.
    No conoce implementaciones.
    No contiene lógica de negocio.
    """

    def __init__(self) -> None:

        self._services: dict[str, Any] = {}

    def register(
        self,
        name: str,
        service: Any,
    ) -> None:
        """
        Registra un servicio.
        """

        self._services[name] = service

    def get(
        self,
        name: str,
    ) -> Any:
        """
        Obtiene un servicio previamente registrado.
        """

        if name not in self._services:

            raise KeyError(
                f"Service '{name}' not registered."
            )

        return self._services[name]

    def has(
        self,
        name: str,
    ) -> bool:

        return name in self._services

    def remove(
        self,
        name: str,
    ) -> None:

        self._services.pop(
            name,
            None,
        )

    def clear(self) -> None:

        self._services.clear()

    def services(self) -> list[str]:

        return sorted(
            self._services.keys()
        )
