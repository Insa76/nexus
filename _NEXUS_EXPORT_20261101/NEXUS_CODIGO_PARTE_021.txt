class OllamaIntentProvider(IntentProvider):
    """Interpreta lenguaje natural usando Ollama y valida la intención resultante."""

    MIN_CONFIDENCE = 0.50
    REQUEST_TIMEOUT_SECONDS = 10.0

    def __init__(
        self,
        model: str = "llama3:latest",
        host: str = "http://localhost:11434",
    ) -> None:
        self._model = model
        self._host = host

    async def recognize(
        self,
        text: str,
        capabilities: list[dict[str, Any]] | None = None,
        context: dict[str, Any] | None = None,
    ) -> Intent:
        capabilities = capabilities or []
        context = context or {}

        # Índices para resolver la respuesta del modelo contra el catálogo real.
        by_name = {
            capability["name"]: capability
            for capability in capabilities
            if isinstance(capability, dict)
            and isinstance(capability.get("name"), str)
        }
        by_action = {
            capability["action"]: capability
            for capability in capabilities
            if isinstance(capability, dict)
            and isinstance(capability.get("action"), str)
        }

        normalized = text.lower().strip()

        # Consultas deterministas de volumen actual.
        volume_read_patterns = (
            r"\ba cuánto está\b",
            r"\ba cuanto está\b",
            r"\ba cuánto quedó\b",
            r"\ba cuanto quedó\b",
            r"\bcuánto está\b",
            r"\bcuanto está\b",
            r"\bcuánto quedó\b",
            r"\bcuanto quedó\b",
            r"\bqué volumen\b",
            r"\bque volumen\b",
            r"\bcuál es el volumen\b",
            r"\bcual es el volumen\b",
            r"\bvolumen actual\b",
            r"\bqué nivel de volumen\b",
            r"\bque nivel de volumen\b",
        )

        if any(re.search(pattern, normalized) for pattern in volume_read_patterns):
            if by_action.get("system.get_volume") is not None:
                return Intent(
                    action="system.get_volume",
                    confidence=1.0,
                    parameters={},
                    original_text=text,
                )

        capabilities_json = json.dumps(
            capabilities,
            ensure_ascii=False,
            indent=2,
        )
        context_json = json.dumps(
            context,
            ensure_ascii=False,
            indent=2,
            default=str,

        system_prompt = f"""
Sos el módulo de interpretación de NEXUS.

Tu tarea es convertir la solicitud del usuario
 en UNA intención estructurada.

CAPACIDADES DISPONIBLES:

{capabilities_json}

CONTEXTO RECIENTE DE NEXUS:

{context_json}

IMPORTANTE SOBRE EL CONTEXTO:

El contexto sirve únicamente para ayudar a
comprender solicitudes que dependan de una
interacción anterior.

El contexto NO crea capacidades nuevas.

Nunca ejecutes una acción solamente porque
aparece en el contexto.

La acción seleccionada debe existir actualmente
en CAPACIDADES DISPONIBLES.

El estado real del sistema tiene prioridad sobre
cualquier dato almacenado en memoria.

El contexto contiene diferentes tipos de información
y cada uno tiene un propósito distinto:

- last_request:

- last_intent:
  última intención interpretada.

- last_result:

recent_episodes representa MEMORIA EPISÓDICA.

Si el usuario pregunta qué hizo, cuál fue su última
acción, qué hizo recién o qué acción realizó antes,

16. Nunca uses una capacidad solamente porque

17. El campo action debe utilizar exactamente

18. Si no existe suficiente información para

19. Respondé únicamente JSON válido.

EJEMPLOS:

            return self._unknown(text)

            confidence=confidence,
            parameters=parameters,
            original_text=text,

    def _validate_semantics(
        self,
        text: str,
        action: str,
        parameters: dict[str, Any],
        context: dict[str, Any] | None = None,
    ) -> bool:

                return False
            application = parameters.get("application")
            return isinstance(application, str) and bool(application.strip())

        if action == "system.get_volume":
            read_patterns = (
                r"\ba cuánto está\b",
                r"\ba cuanto está\b",
                r"\bcuánto está\b",
                r"\bcuanto está\b",

                r"\bque nivel de volumen\b",
                r"\bdecime el volumen actual\b",

            read_patterns = (

                r"\bcuánto está\b",
                r"\bcuanto está\b",

            if any(re.search(pattern, normalized) for pattern in read_patterns):
                return False

            level = parameters.get("level")
            if not isinstance(level, (int, float)) or isinstance(level, bool):
                return False
            if not 0 <= float(level) <= 100:
                return False

            audio_cues = (
                "volumen", "música", "musica", "sonido", "audio", "parlante",
                "altavoz", "auricular", "auriculares", "micrófono", "microfono",
            )
            volume_cues = (
                "fuerte", "bajo", "baja", "bajá", "subí", "subi", "subir",
                "subirlo", "bajarlo", "bajala", "bajalo", "subilo", "subila",
                "más fuerte", "mas fuerte", "más bajo", "mas bajo", "dejala",
                "dejalo", "dejarlo",
            )
            has_audio_context = any(cue in normalized for cue in audio_cues)
            has_volume_context = any(cue in normalized for cue in volume_cues)

            previous_intent = context.get("last_intent")
            previous_result = context.get("last_result")
            previous_action = None
            if isinstance(previous_intent, dict):

                "system.set_volume",
                "system.get_volume",

                has_previous_volume_context = True

            return has_audio_context or has_volume_context or has_previous_volume_context

        if action == "system.mute":
            return bool(re.search(r"\b(silenci[aá]r?|mute|sin sonido)\b", normalized))

        if action == "system.unmute":
            return bool(
                re.search(
                    r"\b(quit[aá]r?.*silencio|desmut|activar.*sonido)\b",
                    normalized,
                )
            )

        if action == "system.close_app":
            close_pattern = r"\b(cerr[aá]r?|termin[aá]r?|finaliz[aá]r?)\b"
            if not re.search(close_pattern, normalized):

            episode_patterns = (

                r"\bultima accion\b",
                r"\búltimo episodio\b",
                r"\bultimo episodio\b",

                r"\bque hice\b",
                r"\bqué hiciste\b",
                r"\bque hiciste\b",
                r"\bqué pasó\b",
                r"\bque paso\b",
                r"\bqué acción hice\b",
                r"\bque accion hice\b",
                r"\búltima vez\b",

                r"\brecién\b",
                r"\brecien\b",

                r"\bqué cambié antes\b",

                r"\bqué hizo nexus antes\b",
                r"\bque hizo nexus antes\b",
            )
            return any(re.search(pattern, normalized) for pattern in episode_patterns)

        # Las demás acciones conservan la política de validación existente.
        return True

    def _validate_parameters(
        self,
        capability: dict[str, Any],
        parameters: dict[str, Any],
    ) -> bool:
        """Valida parámetros contra el esquema de la capacidad seleccionada."""
        schema = capability.get("parameters", {})
        if not isinstance(schema, dict):
            return True

        # Parámetros obligatorios.
        for name, definition in schema.items():
            if not isinstance(definition, dict):
                continue
            if definition.get("required", False) and name not in parameters:
                return False

        # No aceptar parámetros que no estén definidos en el esquema.
        for name in parameters:
            if name not in schema:
                return False

        # Comprobar los tipos de los parámetros recibidos.
        for name, value in parameters.items():
            definition = schema.get(name, {})
            parameter_type = definition.get("type")

            if parameter_type == "string":
                if not isinstance(value, str):
                    return False
            elif parameter_type == "number":
                if isinstance(value, bool) or not isinstance(value, (int, float)):
                    return False
            elif parameter_type == "integer":
                if isinstance(value, bool) or not isinstance(value, int):
                    return False

        return True

    def _unknown(
        self,
        text: str,
        confidence: float = 0.0,
    ) -> Intent:
        return Intent(
            action="system.unknown",
            confidence=confidence,
            parameters={},
            original_text=text,
        )

from typing import Any

class LocalProvider(ToolProvider):
    """
    Proveedor local de NEXUS.

    Utiliza exclusivamente adapters del sistema operativo
    para ejecutar herramientas locales.

    El LocalProvider no conoce Actions y nunca debe
    invocar Action.execute() ni Action.execute_local().

    Flujo:

        Action
            ↓
        ProviderManager
            ↓
        LocalProvider
            ↓
        Windows Adapter
    """

    name = "local"

    def __init__(self) -> None:

        self._launcher = WindowsAppLauncher()

        self._process_manager = (
            WindowsProcessManager()
        )

        self._screen_capture = (
            WindowsScreenCapture()
        )

        self._volume_control = (
            WindowsVolumeControl()
        )

    def tools(self) -> list[Tool]:
        """
        Devuelve las herramientas disponibles
        en el proveedor local.
        """

        return [
            Tool(
                name="windows.open_app",
                action="system.open_app",
                description="Abre una aplicación de Windows.",
                provider=self.name,
                parameters={
                    "application": {
                        "type": "string",
                        "required": True,
                        "description": (
                            "Nombre de la aplicación."
                        ),
                    },
                },
            ),

            Tool(
                name="windows.list_processes",
                action="system.process_list",
                description=(
                    "Obtiene los procesos activos de Windows."
                ),
                provider=self.name,
                parameters={},
            ),

            Tool(
                name="windows.close_app",
                action="system.close_app",
                description=(
                    "Cierra una aplicación en ejecución."
                ),
                provider=self.name,
                parameters={
                    "application": {
                        "type": "string",
                        "required": True,
                        "description": (
                            "Nombre de la aplicación."
                        ),
                    },
                },
            ),

            Tool(
                name="windows.capture_screen",
                action="system.screen_capture",
                description="Captura la pantalla actual.",
                provider=self.name,
                parameters={},
                produces=[
                    "image",
                ],
            ),

            Tool(
                name="windows.set_volume",
                action="system.set_volume",
                description=(
                    "Establece el volumen maestro de Windows."
                ),
                provider=self.name,
                parameters={
                    "level": {
                        "type": "number",
                        "required": True,
                        "description": (
                            "Volumen entre 0 y 100."
                        ),
                    },
                },
            ),

            Tool(
                name="windows.get_volume",
                action="system.get_volume",
                description=(
                    "Obtiene el volumen actual de Windows."
                ),
                provider=self.name,
                parameters={},
            ),

            Tool(
                name="windows.mute",
                action="system.mute",
                description=(
                    "Silencia el volumen de Windows."
                ),
                provider=self.name,
                parameters={},
            ),

            Tool(
                name="windows.unmute",
                action="system.unmute",
                description=(
                    "Quita el silencio de Windows."
                ),
                provider=self.name,
                parameters={},
            ),

            Tool(
                name="windows.get_mute",
                action="system.get_mute",
                description=(
                    "Obtiene el estado actual de silencio "
                    "de Windows."
                ),
                provider=self.name,
                parameters={},
            ),
        ]

    async def execute(
        self,
        tool: str,
        parameters: dict[str, Any],
    ) -> Any:

        if tool == "windows.open_app":
            return await self._open_app(
                parameters
            )

        if tool == "windows.list_processes":
            return await self._list_processes()

        if tool == "windows.close_app":
            return await self._close_app(
                parameters
            )

        if tool == "windows.capture_screen":
            return await self._capture_screen()

        if tool == "windows.set_volume":
            return await self._set_volume(
                parameters
            )

        if tool == "windows.get_volume":
            return await self._get_volume()

        if tool == "windows.mute":
            return await self._mute()

        if tool == "windows.unmute":
            return await self._unmute()

        if tool == "windows.get_mute":
            return await self._get_mute()

        raise KeyError(
            f"Tool '{tool}' not available "
            f"in provider '{self.name}'."
        )

    async def _open_app(
        self,
        parameters: dict[str, Any],
    ) -> Any:

        application = parameters.get(
            "application"
        )

        if not application:
            return False

        return self._launcher.open(
            application
        )

    async def _list_processes(self) -> Any:

        return self._process_manager.list()

    async def _close_app(
        self,
        parameters: dict[str, Any],
    ) -> Any:

        application = parameters.get(
            "application"
        )

        if not application:
            return False

        return self._process_manager.close(
            application
        )

    async def _capture_screen(self) -> Any:

        return self._screen_capture.capture()

    async def _set_volume(
        self,
        parameters: dict[str, Any],
    ) -> Any:

        level = parameters.get("level")

        if level is None:
            return False

        return self._volume_control.set(
            float(level)
        )

    async def _get_volume(self) -> Any:

        return self._volume_control.get()

    async def _mute(self) -> Any:

        return self._volume_control.mute()

    async def _unmute(self) -> Any:

        return self._volume_control.unmute()

    async def _get_mute(self) -> Any:

        return self._volume_control.is_muted()

def build_engine(
    assistant_context: AssistantContext | None = None,
    assistant_session: AssistantSession | None = None,
) -> Engine:
    """
    Construye y configura completamente NEXUS.

    El AssistantContext es opcional para mantener compatibilidad
    con el arranque actual de NEXUS.
    """

    security_context = (
        assistant_session.context
        if assistant_session is not None
        else assistant_context
    )

    security_user = getattr(
        security_context,
        "user",
        None,
    )

    security_actor_id = getattr(
        security_user,
        "id",
        None,
    )

    local_console_actor = Actor.system(
        "local-console",
        name="NEXUS Local Console",
    )

    if (
        isinstance(security_actor_id, str)
        and security_actor_id.strip()
    ):
        security_actor = Actor.user(
            security_actor_id.strip()
        )
    else:
        security_actor = local_console_actor

    #
    # Working Memory
    #

    # Una única instancia compartida por todos
    # los componentes que necesiten memoria.

    memory = WorkingMemory()

    episode_memory = EpisodeMemory(
        memory
    )

    #
    # State History
    #

    state_history = StateHistory(
        memory
    )

    #
    # Kernel
    #

    kernel = Kernel()

    #
    # Runtime
    #

    runtime = Runtime(
        kernel=kernel
    )

    #
    # Provider Manager
    #

    provider_manager = ProviderManager()

    provider_manager.register(
        LocalProvider()
    )

    #
    # Action Registry
    #

    registry = ActionRegistry()

    #
    # Applications
    #

    registry.register(
        OpenAppAction(
            provider_manager
        )
    )

    registry.register(
        CloseAppAction()
    )

    registry.register(
        ProcessListAction(
            provider_manager
        )
    )

    registry.register(
        ScreenCaptureAction(
            provider_manager
        )
    )

    registry.register(
        InspectScreenAction()
    )

    registry.register(
        InspectCaptureAction()
    )

    registry.register(GetTimeAction())

    #
    # Episodic Memory
    #

    registry.register(
        QueryEpisodeAction(
            episode_memory
        )
    )

    #
    # Volume
    #

    registry.register(
        SetVolumeAction(
            provider_manager
        )
    )

    registry.register(
        GetVolumeAction(
            provider_manager
        )
    )

    registry.register(
        GetVolumeHistoryAction(
            state_history
        )
    )

    registry.register(
        GetRecentEpisodesAction(
            episode_memory
        )
    )

    registry.register(
        GetPreviousEpisodeAction(
            episode_memory
        )
    )

    registry.register(
        GetLastEpisodeAction(
            episode_memory
        )
    )

    registry.register(
        RestorePreviousVolumeAction(
            provider_manager,
            state_history,
        )
    )

    #
    # Mute / Unmute
    #

    registry.register(
        MuteAction(
            provider_manager
        )
    )

    registry.register(
        UnmuteAction(
            provider_manager
        )
    )

    registry.register(
        GetMuteAction(
            provider_manager
        )
    )

    registry.register(
        RestorePreviousMuteAction(
            provider_manager,
            state_history,
        )
    )

    #
    # Action Manager
    #

        #
    # Security: permisos de ejecución.
    #

    action_names = registry.list()

    permissions = frozenset(
        Permission(
            resource=action_name,
            action=PermissionAction.EXECUTE,
        )
        for action_name in action_names
    )


    local_role = Role(
        id="local_operator",
        name="Local Operator",
        permissions=permissions,
    )

    security_db_path = os.environ.get(
        "NEXUS_SECURITY_DB_PATH",
        str(Path.home() / ".nexus" / "security.db"),
    )

    security_database = SQLiteSecurityStore(security_db_path)
    role_store = SQLiteRoleStore(security_database)
    assignment_store = SQLiteAssignmentStore(security_database)

    stored_role = role_store.get(local_role.id)

    if stored_role is None:
        role_store.add(local_role)
    elif stored_role != local_role:
        role_store.update(local_role)

    persisted_local_role = role_store.get(local_role.id)

    if persisted_local_role is None:
        raise RuntimeError(
            "No se pudo recuperar el rol local desde SQLite."
        )

    assignment_store.add(
        ActorRoleAssignment(
            actor=local_console_actor,
            role=persisted_local_role,
        )
    )


    policy_store = ActionPolicyStore()

    for action_name in action_names:
        policy_store.register(
            ActionPolicy(
                action_name=action_name,
                permission=Permission(
                    resource=action_name,
                    action=PermissionAction.EXECUTE,
                ),
            )
        )

    authorization_service = AuthorizationService(
        assignment_store
    )

    action_authorization = ActionAuthorization(
        policy_store=policy_store,
        authorization=authorization_service,
    )

    manager = ActionManager(
        registry,
        action_authorization=action_authorization,
    )

    runtime.configure_actions(
        registry,
        manager
    )

    #
    # Tool Registry
    #

    tool_registry = ToolRegistry()

    tool_registry.register_many(
        provider_manager.tools()
    )

    tool_registry.register_many(
        registry.describe()
    )

    #
    # Exponer Tool Registry al Runtime
    #

    runtime.tool_registry = tool_registry

    #
    # Intent Providers
    #

    rule_provider = RuleIntentProvider(
        IntentRecognizer()
    )

    ollama_provider = OllamaIntentProvider(
        model="llama3:latest"
    )

    intent_provider = HybridIntentProvider(
        rule_provider=rule_provider,
        ollama_provider=ollama_provider,
    )

    #
    # Executor
    #

    executor = Executor(
        runtime
    )

    runtime.configure_executor(
        executor
    )

    session_manager = AssistantSessionManager()

    if assistant_session is not None:
        active_session = assistant_session

    elif assistant_context is not None:
        active_session = session_manager.start(
            assistant_context
        )

    else:
        active_session = None

    if active_session is not None:
        assistant_context = active_session.context

    #
    # Documents Capability
    #

    document_services = build_document_services()

    document_capability = DocumentCapability(
        context_access=document_services.context_access,
    )

    document_cognitive_access = DocumentCognitiveAccess(
        document_capability
    )

    #
    # Processes Capability
    #

    process_store = ProcessStore()

    process_resolver = ProcessResolver(
        process_store,
    )

    process_capability = ProcessCapability(
        store=process_store,
        resolver=process_resolver,
    )

    process_cognitive_access = ProcessCognitiveAccess(
        process_capability,
    )

    #
    # Rules Capability
    #

    rule_store = RuleStore()

    rule_resolver = RuleResolver(
        rule_store
    )

    rule_capability = RuleCapability(
        store=rule_store,
        resolver=rule_resolver,
    )

    rule_cognitive_access = RuleCognitiveAccess(
        rule_capability
    )

    #
    # Cognitive Capability Router
    #

    cognitive_capability_router = CognitiveCapabilityRouter(
        document_access=document_cognitive_access,
        process_access=process_cognitive_access,
        rule_access=rule_cognitive_access,
    )

    #
    # Engine
    #

    return Engine(
        runtime,
        intent_provider=intent_provider,
        memory=memory,
        assistant_context=assistant_context,
        document_capability=document_capability,
        document_cognitive_access=document_cognitive_access,
        cognitive_capability_router=cognitive_capability_router,
        process_capability=process_capability,
        process_cognitive_access=process_cognitive_access,
        rule_capability=rule_capability,
        rule_cognitive_access=rule_cognitive_access,
        default_actor_id=security_actor.id,
        default_actor_type=security_actor.actor_type.value,
    )


def build_assistant_runtime(
    assistant_context: AssistantContext | None = None,
    assistant_session: AssistantSession | None = None,
) -> tuple[
    Engine,
    AssistantServices,
    AssistantServicesLifecycle,
]:
    """
    Construye el Engine cognitivo, los servicios del Assistant Layer
    y su lifecycle.

    El Engine, los servicios y su ciclo de vida mantienen
    responsabilidades separadas.
    """

    engine = build_engine(
        assistant_context=assistant_context,
        assistant_session=assistant_session,
    )

    services = build_assistant_services()

    lifecycle = AssistantServicesLifecycle(
        services
    )

    return engine, services, lifecycle


def _print_result(result) -> None:
    """
    Presenta el resultado de una ejecución
    de NEXUS de forma legible.
    """


    if result.success:

        print("NEXUS >", end=" ")

        response = (
            result.metadata.get("response")
            if isinstance(result.metadata, dict)
            else None
        )

        if response:
            print(response)
        elif result.message and result.message != "Execution completed successfully.":
            print(result.message)
        elif result.output:
            print(result.output)
        else:
            print("Listo.")

        return

    print(
        f"NEXUS > {result.message}"
    )



async def _interactive_loop(
    engine: Engine,
) -> None:
    """
    Loop interactivo persistente de NEXUS.

    El Engine y el event loop permanecen vivos
    durante toda la sesión.
    """

    print()

    print("Escribí una instrucción.")

    print("Escribí 'salir' para cerrar NEXUS.")

    print()

    while True:

        try:

            text = input(
                "Usuario > "
            ).strip()

        except (
            EOFError,
            KeyboardInterrupt,
        ):

            print()

            break

        if not text:
            continue

        if text.lower() in {
            "salir",
            "exit",
            "quit",
        }:

            break

        try:

            result = await engine.execute(
                text
            )

            _print_result(
                result
            )

        except Exception as exc:

            print(
                f"NEXUS > Error interno: {exc}"
            )

        print()


async def _main_async() -> None:
    """
    Inicializa NEXUS y mantiene una sesión
    interactiva persistente.
    """

    print()

    print("=" * 50)

    print("          NEXUS Cognitive OS")

    print("=" * 50)

    print()

    engine = build_engine()

    print("✓ Kernel")

    print("✓ Runtime")

    print("✓ Providers")

    print("✓ Actions")

    print("✓ Intent Providers")

    print("✓ Executor")

    print("✓ Engine")

    print()

    print("=" * 50)

    print("NEXUS READY")

    print("=" * 50)

    await _interactive_loop(
        engine
    )

    print()

    print("NEXUS detenido.")


def main() -> None:
    """
    Punto de entrada de NEXUS.
    """

    import asyncio

    asyncio.run(
        _main_async()
    )

from __future__ import annotations

from dataclasses import dataclass

from documents.context_access import DocumentContextAccess
from documents.query import DocumentQueryResult
from documents.context_relevance import DocumentContextMatch
from documents.models import Document


@dataclass(slots=True)
class DocumentCapability:
    """
    Interfaz cognitiva de acceso a documentos.

    Expone operaciones orientadas al Assistant/Cognitive Core
    sin exponer stores ni detalles internos del subsistema Documents.
    """

    context_access: DocumentContextAccess

    def query(
        self,
        *,
        context: object,
        name: str | None = None,
        mime_type: str | None = None,
        offset: int = 0,
        limit: int | None = None,
    ) -> DocumentQueryResult:
        return self.context_access.query_from_context(
            context,
            name=name,
            mime_type=mime_type,
            offset=offset,
            limit=limit,
        )

    def relevant(
        self,
        *,
        context: object,
    ) -> tuple[DocumentContextMatch, ...]:
        return self.context_access.resolve_ordered_matches_from_context(
            context,
        )

    def relevant_documents(
        self,
        *,
        context: object,
    ) -> tuple[Document, ...]:
        return self.context_access.resolve_ordered_documents_from_context(
            context,
        )

from __future__ import annotations

from dataclasses import dataclass
from typing import Any

from documents.capability import DocumentCapability
from documents.models import Document
from documents.query import DocumentQueryResult


@dataclass(frozen=True, slots=True)
class DocumentCognitiveResult:
    """
    Resultado de una operación cognitiva sobre Documents.

    Representa información que el Cognitive Core puede utilizar
    para razonar sobre documentos sin conocer stores ni detalles
    internos del subsistema Documents.
    """

    query: DocumentQueryResult
    documents: tuple[Document, ...]
    metadata: dict[str, Any]


class DocumentCognitiveAccess:
    """
    Punto de acceso cognitivo a Documents.

    Esta capa traduce una necesidad cognitiva del Assistant
    en una operación sobre DocumentCapability.

    No conoce stores.
    No ejecuta acciones del Runtime.
    No interpreta lenguaje natural.
    """

    def __init__(
        self,
        capability: DocumentCapability,
    ) -> None:
        self._capability = capability

    def find_relevant(
        self,
        *,
        context: object,
        name: str | None = None,
        mime_type: str | None = None,
        offset: int = 0,
        limit: int | None = None,
    ) -> DocumentCognitiveResult:
        query = self._capability.query(
            context=context,
            name=name,
            mime_type=mime_type,
            offset=offset,
            limit=limit,
        )

        # La consulta ya contiene los filtros, el orden y la
        # paginación aplicados. No recuperar otra colección
        # independiente que pueda contradecir ese resultado.
        documents = query.documents

        return DocumentCognitiveResult(
            query=query,
            documents=documents,
            metadata={
                "operation": "find_relevant",
                "name": name,
                "mime_type": mime_type,
                "offset": offset,
                "limit": limit,
                "count": len(documents),
                "total": query.total,
                "has_more": query.has_more,
            },
        )
