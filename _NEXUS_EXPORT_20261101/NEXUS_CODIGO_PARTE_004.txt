function getCycleRequestText(
  cycle: { request?: { message?: string; data?: unknown } } | null,
): string {
  const requestData = cycle?.request?.data;

  if (requestData && typeof requestData === "object") {
    const record = requestData as Record<string, unknown>;

    const candidates = [
      record.text,
      record.original_text,
      record.request,
    ];

    const requestValue = candidates.find(
      (value): value is string =>
        typeof value === "string" && value.trim().length > 0,
    );

    if (requestValue) {
      return requestValue;
    }
  }

  if (
    typeof cycle?.request?.message === "string" &&
    cycle.request.message.trim().length > 0
  ) {
    return cycle.request.message;
  }

  return "Cognitive cycle";
}

function buildInsightLearningCandidate(
  insight: CycleInsight | null,
  leftCycle: { id: string } | null,
  rightCycle: { id: string } | null,
): InsightLearningCandidate | null {
  if (!insight || !leftCycle || !rightCycle) {
    return null;
  }

  const knowledgeByInsight: Record<string, string> = {
    "same-intent-different-execution":
      "Cuando NEXUS identifica el mismo intento con objetivos de ejecución diferentes, debe conservar la relación entre intención y aplicación como evidencia contextual del ciclo.",
    "same-decision-different-result":
      "Una misma decisión puede producir resultados diferentes; la decisión no debe considerarse suficiente para explicar el resultado sin consultar la evidencia de ejecución.",
    "plan-changed":
      "Cambios en la secuencia de pasos deben conservarse como una diferencia de estrategia entre ciclos comparables.",
    "objective-changed":
      "Un cambio de objetivo indica que ciclos aparentemente similares pueden estar resolviendo metas distintas y deben mantenerse separados en el aprendizaje.",
    "verification-differs":
      "La verificación debe conservarse como evidencia independiente del resultado de ejecución para explicar por qué dos ciclos no son equivalentes.",
    "reasoning-stable":
      "Cuando intención y decisión permanecen estables entre ciclos, esa estabilidad puede utilizarse como referencia al analizar diferencias posteriores en ejecución o verificación.",
  };

  const sourceCycleIds = [leftCycle.id, rightCycle.id].sort();

  return {
    id: `insight-${insight.id}-${sourceCycleIds.join("--")}`,
    insightId: insight.id,
    title: insight.title,
    knowledge:
      knowledgeByInsight[insight.id] ??
      `El patrón “${insight.title}” fue observado entre dos ciclos y queda registrado como candidato para conocimiento reutilizable.`,
    sourceCycleIds,
    evidenceKeys: insight.evidenceKeys,
    createdAt: new Date().toISOString(),
    status: "staged",
  };
}

function formatTime(timestamp?: string): string {
  if (!timestamp) {
    return "--:--:--";
  }

  const date = new Date(timestamp);

  if (Number.isNaN(date.getTime())) {
    return "--:--:--";
  }

  return date.toLocaleTimeString("es-AR", {
    hour: "2-digit",
    minute: "2-digit",
    second: "2-digit",
  });
}

function eventLabel(type: string, message?: string): string {
  if (message) {
    return message;
  }

  const labels: Record<string, string> = {
    request: "Request processed",
    context: "Context recovered",
    intent: "Intent identified",

    objective: "Objective defined",
    plan: "Plan generated",
    execution: "Action executed",
    verification: "Execution verified",
    memory: "Episode stored",
    learning: "Experience learned",
  };

  return labels[type] ?? type;
}

function getEventDetail(
  eventType: string,
  data?: unknown,
): string | undefined {
  if (!data || typeof data !== "object") {
    return undefined;
  }

  const record = data as Record<string, unknown>;

  /*
   * ============================================================
   * 9.13 — COGNITIVE FOCUS
   * ============================================================
   */

  if (eventType === "user_request") {
    const text =
      typeof record.text === "string"
        ? record.text
        : undefined;

    return text;
  }

  if (eventType === "cognitive_context") {
    const keys = Object.keys(record);

    if (keys.length > 0) {
      return `${keys.length} context fields`;
    }

    return undefined;
  }

  if (eventType === "intent") {
    const action =
      typeof record.action === "string"
        ? record.action
        : undefined;

    const confidence =
      typeof record.confidence === "number"
        ? record.confidence
        : undefined;

    if (action && confidence !== undefined) {
      return `${action} · confidence ${confidence.toFixed(2)}`;
    }

    if (action) {
      return action;
    }

    if (confidence !== undefined) {
      return `confidence ${confidence.toFixed(2)}`;
    }

    return undefined;
  }

  if (eventType === "decision") {
    const action =
      typeof record.selected_action === "string"
        ? record.selected_action
        : undefined;

    const confidence =
      typeof record.confidence === "number"
        ? record.confidence
        : undefined;

    const reason =
      typeof record.reason === "string"
        ? record.reason
        : undefined;

    const parts: string[] = [];

    if (action) {
      parts.push(action);
    }

    if (confidence !== undefined) {
      parts.push(`confidence ${confidence.toFixed(2)}`);
    }

    if (reason) {
      parts.push(reason);
    }

    return parts.length > 0
      ? parts.join(" · ")
      : undefined;
  }

  if (eventType === "objective") {
    const title =
      typeof record.title === "string"
        ? record.title
        : undefined;

    const status =
      typeof record.status === "string"
        ? record.status
        : undefined;

    let application: string | undefined;

    if (
      record.metadata &&
      typeof record.metadata === "object"
    ) {
      const metadata =
        record.metadata as Record<string, unknown>;

      if (
        typeof metadata.application === "string"
      ) {
        application = metadata.application;
      }
    }

    const parts: string[] = [];

    if (title) {
      parts.push(title);
    }

    if (status) {
      parts.push(`status ${status}`);
    }

    if (application) {
      parts.push(`application ${application}`);
    }

    return parts.length > 0
      ? parts.join(" · ")
      : undefined;
  }

  if (
    eventType === "plan" ||
    eventType === "plan_final"
  ) {
    const steps = Array.isArray(record.steps)
      ? record.steps
      : [];

    if (steps.length > 0) {
      const firstStep = steps[0];

      if (
        firstStep &&
        typeof firstStep === "object"
      ) {
        const step =
          firstStep as Record<string, unknown>;

        const action =
          typeof step.action === "string"
            ? step.action
            : undefined;

        const status =
          typeof step.status === "string"
            ? step.status
            : undefined;

        const parts: string[] = [];

        if (action) {
          parts.push(action);
        }

        if (status) {
          parts.push(`status ${status}`);
        }

        if (steps.length > 1) {
          parts.push(`${steps.length} steps`);
        }

        if (parts.length > 0) {
          return parts.join(" · ");
        }
      }
    }

    const completed =
      typeof record.completed === "boolean"
        ? record.completed
        : undefined;

    if (completed !== undefined) {
      return completed
        ? "plan completed"
        : "plan pending";
    }

    return undefined;
  }

  if (
    eventType === "verification" ||
    eventType === "verification_failure"
  ) {
    const verified =
      typeof record.verified === "boolean"
        ? record.verified
        : undefined;

    const confidence =
      typeof record.confidence === "number"
        ? record.confidence
        : undefined;

    const reason =
      typeof record.reason === "string"
        ? record.reason
        : undefined;

    const parts: string[] = [];

    if (verified !== undefined) {
      parts.push(
        verified
          ? "verified"
          : "not verified",
      );
    }

    if (confidence !== undefined) {
      parts.push(
        `confidence ${confidence.toFixed(2)}`,
      );
    }

    if (reason) {
      parts.push(reason);
    }

    return parts.length > 0
      ? parts.join(" · ")
      : undefined;
  }

  if (eventType === "execution_result") {
    const success =
      typeof record.success === "boolean"
        ? record.success
        : undefined;

    const message =
      typeof record.message === "string"
        ? record.message
        : undefined;

    let output:
      | Record<string, unknown>
      | undefined;

    if (
      record.output &&
      typeof record.output === "object"
    ) {
      output =
        record.output as Record<string, unknown>;
    }

    const parts: string[] = [];

    if (success !== undefined) {
      parts.push(
        success
          ? "success"
          : "failed",
      );
    }

    if (output) {
      const application =
        typeof output.application === "string"
          ? output.application
          : undefined;

      const status =
        typeof output.status === "string"
          ? output.status
          : undefined;

      if (application) {
        parts.push(application);
      }

      if (status) {
        parts.push(status);
      }
    }

    if (message) {
      parts.push(message);
    }

    return parts.length > 0
      ? parts.join(" · ")
      : undefined;
  }

  if (eventType === "episode") {
    return "Experience stored for future reuse.";
  }

  return undefined;
}

function getFocusFields(
  eventType: string,
  data?: unknown,
): FocusField[] {
  if (!data || typeof data !== "object") {
    return [];
  }

  const record = data as Record<string, unknown>;

  if (eventType === "user_request") {
    const text =
      typeof record.text === "string"
        ? record.text
        : undefined;

    return text
      ? [{ label: "REQUEST", value: text }]
      : [];
  }

  if (eventType === "cognitive_context") {
    return [
      {
        label: "FIELDS",
        value: `${Object.keys(record).length}`,
      },
    ];
  }

  if (eventType === "intent") {
    const fields: FocusField[] = [];

    if (typeof record.action === "string") {
      fields.push({
        label: "ACTION",
        value: record.action,
      });
    }

    if (typeof record.confidence === "number") {
      fields.push({
        label: "CONFIDENCE",
        value: record.confidence.toFixed(2),
      });
    }

    if (
      record.parameters &&
      typeof record.parameters === "object"
    ) {
      fields.push({
        label: "PARAMETERS",
        value: JSON.stringify(record.parameters),
      });
    }

    if (typeof record.original_text === "string") {
      fields.push({
        label: "ORIGINAL",
        value: record.original_text,
      });
    }

    return fields;
  }

  if (eventType === "decision") {
    const fields: FocusField[] = [];

    if (typeof record.selected_action === "string") {
      fields.push({
        label: "ACTION",
        value: record.selected_action,
      });
    }

    if (typeof record.confidence === "number") {
      fields.push({
        label: "CONFIDENCE",
        value: record.confidence.toFixed(2),
      });
    }

    if (typeof record.requires_execution === "boolean") {
      fields.push({
        label: "EXECUTION",
        value: record.requires_execution
          ? "required"
          : "not required",
      });
    }

    if (typeof record.reason === "string") {
      fields.push({
        label: "REASON",
        value: record.reason,
      });
    }

    return fields;
  }

  if (eventType === "objective") {
    const fields: FocusField[] = [];

    if (typeof record.id === "string") {
      fields.push({
        label: "ID",
        value: record.id,
      });
    }

    if (typeof record.title === "string") {
      fields.push({
        label: "TITLE",
        value: record.title,
      });
    }

    if (typeof record.status === "string") {
      fields.push({
        label: "STATUS",
        value: record.status,
      });
    }

    if (
      record.metadata &&
      typeof record.metadata === "object"
    ) {
      const metadata =
        record.metadata as Record<string, unknown>;

      if (typeof metadata.application === "string") {
        fields.push({
          label: "APPLICATION",
          value: metadata.application,
        });
      }
    }

    return fields;
  }

  if (
    eventType === "plan" ||
    eventType === "plan_final"
  ) {
    const fields: FocusField[] = [];

    if (typeof record.objective_id === "string") {
      fields.push({
        label: "OBJECTIVE",
        value: record.objective_id,
      });
    }

    const steps = Array.isArray(record.steps)
      ? record.steps
      : [];

    fields.push({
      label: "STEPS",
      value: `${steps.length}`,
    });

    if (steps.length > 0) {
      const firstStep = steps[0];

      if (
        firstStep &&
        typeof firstStep === "object"
      ) {
        const step =
          firstStep as Record<string, unknown>;

        if (typeof step.action === "string") {
          fields.push({
            label: "ACTION",
            value: step.action,
          });
        }

        if (typeof step.status === "string") {
          fields.push({
            label: "STATUS",
            value: step.status,
          });
        }

        if (typeof step.description === "string") {
          fields.push({
            label: "DESCRIPTION",
            value: step.description,
          });
        }
      }
    }

    if (typeof record.current_step === "number") {
      fields.push({
        label: "CURRENT STEP",
        value: String(record.current_step),
      });
    }

    if (typeof record.completed === "boolean") {
      fields.push({
        label: "COMPLETED",
        value: record.completed
          ? "yes"
          : "no",
      });
    }

    return fields;
  }

  if (
    eventType === "verification" ||
    eventType === "verification_failure"
  ) {
    const fields: FocusField[] = [];

    if (typeof record.verified === "boolean") {
      fields.push({
        label: "STATUS",
        value: record.verified
          ? "verified"
          : "not verified",
      });
    }

    if (typeof record.confidence === "number") {
      fields.push({
        label: "CONFIDENCE",
        value: record.confidence.toFixed(2),
      });
    }

    if (typeof record.reason === "string") {
      fields.push({
        label: "REASON",
        value: record.reason,
      });
    }

    if (Array.isArray(record.evidence)) {
      fields.push({
        label: "EVIDENCE",
        value: `${record.evidence.length} item${
          record.evidence.length === 1
            ? ""
            : "s"
        }`,
      });
    }

    return fields;
  }

  if (eventType === "execution_result") {
    const fields: FocusField[] = [];

    if (typeof record.success === "boolean") {
      fields.push({
        label: "STATUS",
        value: record.success
          ? "success"
          : "failed",
      });
    }

    if (typeof record.message === "string") {
      fields.push({
        label: "MESSAGE",
        value: record.message,
      });
    }

    if (
      record.output &&
      typeof record.output === "object"
    ) {
      const output =
        record.output as Record<string, unknown>;

      if (typeof output.application === "string") {
        fields.push({
          label: "APPLICATION",
          value: output.application,
        });
      }

      if (typeof output.status === "string") {
        fields.push({
          label: "RESULT",
          value: output.status,
        });
      }
    }

    return fields;
  }

  if (eventType === "episode") {
    return [
      {
        label: "MEMORY",
        value: "Experience stored for future reuse.",
      },
    ];
  }

  return [];
}

function getTimelineStatus(
  eventIndex: number,
  totalEvents: number,
  state: NexusPresenceState,
): TimelineStatus {
  if (
    state === "failed" &&
    eventIndex === totalEvents - 1
  ) {
    return "failed";
  }

  if (
    (state === "recovering" ||
      state === "needs_confirmation") &&
    eventIndex === totalEvents - 1
  ) {
    return "active";
  }

  if (
    state !== "completed" &&
    state !== "idle" &&
    eventIndex === totalEvents - 1
  ) {
    return "active";
  }

  return "done";
}

function buildTimeline(
  presence: NexusPresenceStateModel,
): TimelineEvent[] {
  const events = presence.events;

  if (events.length === 0) {
    return [];
  }

  /*
   * ============================================================
   * 9.12C — TIMELINE CYCLE ISOLATION
   * ============================================================
   */

  let cycleEvents = events;

  for (let index = events.length - 1; index >= 0; index -= 1) {
    if (events[index].type === "user_request") {
      cycleEvents = events.slice(index);
      break;
    }
  }

  return cycleEvents.map((event, index) => {
    const state = event.type as NexusPresenceState;

    return {
      id: event.id,
      time: formatTime(event.timestamp),
      label: eventLabel(event.type, event.message),
      status: getTimelineStatus(
        index,
        cycleEvents.length,
        presence.state,
      ),
      state,
      eventType: event.type,
      data: event.data,
      detail: getEventDetail(
        event.type,
        event.data,
      ),
    };
  });
}

function getCycleEventData(
  cycle: {
    events: Array<{
      type: string;
      data?: unknown;
    }>;
  } | null,
  type: string,
): unknown {
  if (!cycle) {
    return undefined;
  }

  return cycle.events.find((event) => event.type === type)?.data;
}

function asRecord(value: unknown): Record<string, unknown> | null {
  return value && typeof value === "object"
    ? (value as Record<string, unknown>)
    : null;
}

function textValue(value: unknown): string {
  if (typeof value === "string" && value.trim()) {
    return value.trim();
  }

  if (typeof value === "number" || typeof value === "boolean") {
    return String(value);
  }

  return "—";
}

function formatComparisonData(
  type: string,
  data: unknown,
): string {
  const record = asRecord(data);

  if (!record) {
    return "—";
  }

  switch (type) {
    case "user_request":
      return textValue(
        record.text ??
          record.original_text ??
          record.request,
      );

    case "intent":
      return [
        `action=${textValue(record.name ?? record.intent)}`,
        `confidence=${textValue(record.confidence)}`,
      ].join(" · ");

    case "decision":
      return [
        `action=${textValue(record.selected_action ?? record.type)}`,
        `confidence=${textValue(record.confidence)}`,
        record.requires_execution !== undefined
          ? `execute=${textValue(record.requires_execution)}`
          : "",
      ]
        .filter(Boolean)
        .join(" · ");

    case "objective":
      return [
        textValue(record.title ?? record.description),
        record.status
          ? `status=${textValue(record.status)}`
          : "",
      ]
        .filter(Boolean)
        .join(" · ");

    case "plan": {
      const steps = Array.isArray(record.steps)
        ? record.steps
            .map((step) => {
              const stepRecord = asRecord(step);
              return textValue(
                stepRecord?.name ??
                  stepRecord?.description ??
                  stepRecord?.id,
              );
            })
            .filter((step) => step !== "—")
        : [];

      return [
        steps.length > 0
          ? `${steps.length} steps`
          : "",
        record.current_step !== undefined
          ? `current=${textValue(record.current_step)}`
          : "",
        record.completed !== undefined
          ? `completed=${textValue(record.completed)}`
          : "",
      ]
        .filter(Boolean)
        .join(" · ") || "—";
    }

    case "plan_final":
      return [
        record.completed !== undefined
          ? `completed=${textValue(record.completed)}`
          : "",
        Array.isArray(record.steps)
          ? `${record.steps.length} steps`
          : "",
      ]
        .filter(Boolean)
        .join(" · ") || "—";

    case "execution_result": {
      const output = asRecord(record.output);

      return [
        `success=${textValue(record.success)}`,
        record.message
          ? `message=${textValue(record.message)}`
          : "",
        output?.application
          ? `app=${textValue(output.application)}`
          : "",
        output?.status
          ? `status=${textValue(output.status)}`
          : "",
      ]
        .filter(Boolean)
        .join(" · ");
    }

    case "verification":
    case "verification_failure":
      return [
        `verified=${textValue(record.verified)}`,
        `confidence=${textValue(record.confidence)}`,
        record.reason
          ? `reason=${textValue(record.reason)}`
          : "",
      ]
        .filter(Boolean)
        .join(" · ");

    case "episode": {
      const lesson =
        record.lesson ??
        record.learning ??
        record.summary;

      return lesson
        ? textValue(lesson)
        : Object.keys(record).length > 0
          ? `${Object.keys(record).length} fields`
          : "—";
    }

    default:
      return Object.keys(record).length > 0
        ? `${Object.keys(record).length} fields`
        : "—";
  }
}

function buildCycleComparisonRows(
  leftCycle: {
    events: Array<{
      type: string;
      data?: unknown;
    }>;
  } | null,
  rightCycle: {
    events: Array<{
      type: string;
      data?: unknown;
    }>;
  } | null,
): CycleComparisonRow[] {
  const definitions = [
    ["request", "REQUEST", "user_request"],
    ["intent", "INTENT", "intent"],
    ["decision", "DECISION", "decision"],
    ["objective", "OBJECTIVE", "objective"],
    ["plan", "PLAN", "plan"],
    ["plan_final", "PLAN FINAL", "plan_final"],
    ["execution", "EXECUTION", "execution_result"],
    ["verification", "VERIFICATION", "verification"],
    ["episode", "MEMORY / EPISODE", "episode"],
  ] as const;

  return definitions.map(([key, label, eventType]) => {
    const left = formatComparisonData(
      eventType,
      getCycleEventData(leftCycle, eventType),
    );

    const right = formatComparisonData(
      eventType,
      getCycleEventData(rightCycle, eventType),
    );

    return {
      key,
      label,
      left,
      right,
      changed: left !== right,
    };
  });
}

function getCycleStringField(
  cycle: {
    events: Array<{
      type: string;
      data?: unknown;
    }>;
  } | null,
  eventType: string,
  fields: string[],
): string | null {
  const record = asRecord(getCycleEventData(cycle, eventType));

  if (!record) {
    return null;
  }

  for (const field of fields) {
    const value = record[field];

    if (typeof value === "string" && value.trim()) {
      return value.trim();
    }
  }

  return null;
}

function getCycleExecutionApplication(
  cycle: {
    events: Array<{
      type: string;
      data?: unknown;
    }>;
  } | null,
): string | null {
  const record = asRecord(
    getCycleEventData(cycle, "execution_result"),
  );

  const output = asRecord(record?.output);

  return typeof output?.application === "string"
    ? output.application
    : null;
}

function getCycleExecutionSuccess(
  cycle: {
    events: Array<{
      type: string;
      data?: unknown;
    }>;
  } | null,
): boolean | null {
  const record = asRecord(
    getCycleEventData(cycle, "execution_result"),
  );

  return typeof record?.success === "boolean"
    ? record.success
    : null;
}

function getCyclePlanSignature(
  cycle: {
    events: Array<{
      type: string;
      data?: unknown;
    }>;
  } | null,
): string {
  const record = asRecord(
    getCycleEventData(cycle, "plan"),
  );

  const steps = Array.isArray(record?.steps)
    ? record.steps
        .map((step) => {
          const stepRecord = asRecord(step);

          return (
            stepRecord?.action ??
            stepRecord?.name ??
            stepRecord?.description ??
            stepRecord?.id
          );
        })
        .filter(
          (value): value is string =>
            typeof value === "string" && value.trim().length > 0,
        )
        .map((value) => value.trim())
    : [];

  return steps.join(" → ");
}

function getCycleVerificationSignature(
  cycle: {
    events: Array<{
      type: string;
      data?: unknown;
    }>;
  } | null,
): string {
  const record = asRecord(
    getCycleEventData(cycle, "verification"),
  );

  if (!record) {
    return "";
  }

  return [
    typeof record.verified === "boolean"
      ? String(record.verified)
      : "",
    typeof record.confidence === "number"
      ? record.confidence.toFixed(2)
      : "",
    typeof record.reason === "string"
      ? record.reason.trim()
      : "",
  ].join("|");
}

function buildCycleInsights(
  leftCycle: {
    id: string;
    events: Array<{
      type: string;
      data?: unknown;
    }>;
  } | null,
  rightCycle: {
    id: string;
    events: Array<{
      type: string;
      data?: unknown;
    }>;
  } | null,
): CycleInsight[] {
  if (!leftCycle || !rightCycle || leftCycle.id === rightCycle.id) {
    return [];
  }

  const insights: CycleInsight[] = [];

  const leftIntent = getCycleStringField(
    leftCycle,
    "intent",
    ["name", "intent", "action"],
  );
  const rightIntent = getCycleStringField(
    rightCycle,
    "intent",
    ["name", "intent", "action"],
  );

  const leftDecision = getCycleStringField(
    leftCycle,
    "decision",
    ["selected_action", "type", "action"],
  );
  const rightDecision = getCycleStringField(
    rightCycle,
    "decision",
    ["selected_action", "type", "action"],
  );

  const leftApp = getCycleExecutionApplication(leftCycle);
  const rightApp = getCycleExecutionApplication(rightCycle);

  const leftSuccess = getCycleExecutionSuccess(leftCycle);
  const rightSuccess = getCycleExecutionSuccess(rightCycle);

  const leftPlan = getCyclePlanSignature(leftCycle);
  const rightPlan = getCyclePlanSignature(rightCycle);

  const leftVerification = getCycleVerificationSignature(leftCycle);
  const rightVerification = getCycleVerificationSignature(rightCycle);

  const leftObjective = getCycleStringField(
    leftCycle,
    "objective",
    ["title", "description"],
  );
  const rightObjective = getCycleStringField(
    rightCycle,
    "objective",
    ["title", "description"],
  );

  if (
    leftIntent &&
    rightIntent &&
    leftIntent === rightIntent &&
    leftApp &&
    rightApp &&
    leftApp !== rightApp
  ) {
    insights.push({
      id: "same-intent-different-execution",
      label: "PATTERN",
      title: "Mismo intento · distinta ejecución",
      detail: `Ambos ciclos interpretaron el intento como “${leftIntent}”, pero la ejecución terminó sobre ${leftApp} y ${rightApp}.`,
      tone: "difference",
      evidenceKeys: ["intent", "execution"],
    });
  }

  if (
    leftDecision &&
    rightDecision &&
    leftDecision === rightDecision &&
    leftSuccess !== null &&
    rightSuccess !== null &&
    leftSuccess !== rightSuccess
  ) {
    insights.push({
      id: "same-decision-different-result",
      label: "PATTERN",
      title: "Misma decisión · distinto resultado",
      detail: `La decisión seleccionada fue “${leftDecision}” en ambos ciclos, mientras el resultado de ejecución fue ${leftSuccess ? "exitoso" : "no exitoso"} y ${rightSuccess ? "exitoso" : "no exitoso"}.`,
      tone: "difference",
      evidenceKeys: ["decision", "execution"],
    });
  }

  if (leftPlan && rightPlan && leftPlan !== rightPlan) {
    insights.push({
      id: "plan-changed",
      label: "CHANGE",
      title: "El plan cambió",
      detail: `La secuencia de pasos difiere entre los ciclos: “${leftPlan}” frente a “${rightPlan}”.`,
      tone: "change",
      evidenceKeys: ["plan"],
    });
  }

  if (
    leftObjective &&
    rightObjective &&
    leftObjective !== rightObjective
  ) {
    insights.push({
      id: "objective-changed",
      label: "CHANGE",
      title: "El objetivo cambió",
      detail: `El objetivo registrado fue “${leftObjective}” en un ciclo y “${rightObjective}” en el otro.`,
      tone: "change",
      evidenceKeys: ["objective"],
    });
  }

  if (
    leftVerification &&
    rightVerification &&
    leftVerification !== rightVerification
  ) {
    insights.push({
      id: "verification-changed",
      label: "EVIDENCE",
      title: "La verificación difiere",
      detail: "Los datos de verificación no son iguales entre los dos ciclos comparados.",
      tone: "difference",
      evidenceKeys: ["verification"],
    });
  }

  if (
    leftIntent &&
    rightIntent &&
    leftIntent === rightIntent &&
    leftDecision &&
    rightDecision &&
    leftDecision === rightDecision
  ) {
    insights.push({
      id: "reasoning-stable",
      label: "STABLE",
      title: "Razonamiento estable",
      detail: `El intento “${leftIntent}” condujo a la misma decisión “${leftDecision}” en ambos ciclos.`,
      tone: "stable",
      evidenceKeys: ["intent", "decision"],
    });
  }

  if (insights.length === 0) {
    insights.push({
      id: "no-pattern",
      label: "OBSERVATION",
      title: "No se detectaron patrones definidos",
      detail: "Los datos disponibles no muestran una coincidencia o diferencia suficientemente clara para generar un insight específico.",
      tone: "stable",
      evidenceKeys: [],
    });
  }

  return insights;
}

function buildHistoricalReplayEvents(
  cycle: {
    events: Array<{
      id: string;
      type: string;
      timestamp: string;
      message?: string;
      data?: unknown;
    }>;
  } | null,
): HistoricalReplayEvent[] {
  if (!cycle) {
    return [];
  }

  return cycle.events.map((event) => ({
    id: event.id,
    time: formatTime(event.timestamp),
    label: eventLabel(event.type, event.message),
    eventType: event.type,
    data: event.data,
    detail: getEventDetail(event.type, event.data),
  }));
}

   * usando timeouts encadenados.

   * Una interacción manual cancela inmediatamente el proceso.

    stopAutoFollow();

    let currentIndex = 0;
    let cancelled = false;

    setSelectedTimelineEventId(timeline[0].id);

    function advance(): void {
      if (cancelled) {
        return;
      }

      if (!focusAutoFollow || focusManualSelection) {
        return;
      }

      currentIndex += 1;

      if (currentIndex >= timeline.length) {
        autoFollowTimerRef.current = null;
        return;
      }

      setSelectedTimelineEventId(
        timeline[currentIndex].id,
      );

      autoFollowTimerRef.current =
        window.setTimeout(advance, 450);
    }

    autoFollowTimerRef.current =
      window.setTimeout(advance, 450);

    return () => {
      cancelled = true;
      stopAutoFollow();
    };
  }, [
    timeline.length,
    timeline[0]?.id,
    focusAutoFollow,
    focusManualSelection,
  ]);

  useEffect(() => {
    if (!historyReplayActive || !historyReplayPlaying) {

  async function executeCommand(
    event?: React.FormEvent,
  ): Promise<void> {
    event?.preventDefault();

    const text = command.trim();

    if (!text || isSubmitting) {
      return;
    }

    setIsSubmitting(true);
    setCommandError(null);

    /*
     * Una nueva solicitud significa un nuevo ciclo.
     * Permitimos que Auto-Follow vuelva a tomar control.
     */

    stopAutoFollow();
    resetReplay();
    resetHistoryReplay();
    autoFollowCycleRef.current = null;
    setFocusManualSelection(false);
    setFocusAutoFollow(true);
    setSelectedTimelineEventId(null);
    setSelectedHistoryCycleId(null);

    try {
      const response = await fetch(
        "http://127.0.0.1:8765/api/execute",
        {
          method: "POST",
          headers: {
            "Content-Type": "application/json",
            Accept: "application/json",
          },
          body: JSON.stringify({
            text,
          }),
        },
      );

      let payload: unknown = null;

      try {
        payload = await response.json();
      } catch {
        payload = null;
      }

      if (!response.ok) {
        const message =
          typeof payload === "object" &&
          payload !== null &&
          "error" in payload &&
          typeof payload.error === "string"
            ? payload.error
            : `NEXUS returned HTTP ${response.status}`;

        throw new Error(message);
      }

      setCommand("");

      await new Promise((resolve) =>
        setTimeout(resolve, 100),
      );
    } catch (error) {
      const message =
        error instanceof Error
          ? error.message
          : "Unable to execute request.";

      setCommandError(message);
    } finally {
      setIsSubmitting(false);
    }
  }

                          key={insight.id}
                          type="button"
                          className={`cycle-insight-card cycle-insight-${insight.tone}${
                            isSelected
                              ? " cycle-insight-selected"
                              : ""
                          }`}
                          aria-pressed={isSelected}
                          onClick={() =>
                            setSelectedInsightId(
                              isSelected ? null : insight.id,
                            )
                          }
                        >
                          <div className="cycle-insight-meta">
                            <span>{insight.label}</span>
                            <span>
                              {insight.evidenceKeys.length > 0
                                ? `${insight.evidenceKeys.length} EVIDENCE`
                                : "OBSERVATION"}
                            </span>
                          </div>
                          <strong>{insight.title}</strong>
                          <p>{insight.detail}</p>
                          <span className="cycle-insight-action">
                            {isSelected
                              ? "HIDE EVIDENCE"
                              : "SHOW EVIDENCE"}
                          </span>
                        </button>
                      );
                    })}
                  </div>

                  {selectedInsight && (
                    <div className="cycle-insight-trace">
                      <div className="cycle-insight-trace-header">
                        <span>INSIGHT TRACE</span>
                        <span>
                          {selectedInsight.evidenceKeys.length > 0
                            ? `${selectedInsight.evidenceKeys.length} DIMENSIONS`
                            : "NO DIRECT EVIDENCE"}
                        </span>
                      </div>

                      <strong>{selectedInsight.title}</strong>

                      {selectedInsight.evidenceKeys.length > 0 ? (
                        <div className="cycle-insight-evidence-list">
                          {selectedInsight.evidenceKeys.map(
                            (key) => {
                              const row =
                                cycleComparisonRows.find(
                                  (item) => item.key === key,
                                );

                              if (!row) {
                                return null;
                              }

                              return (
                                <div
                                  key={key}
                                  className="cycle-insight-evidence-row"
                                >
                                  <span>{row.label}</span>
                                  <span>{row.left}</span>
                                  <span>→</span>
                                  <span>{row.right}</span>
                                </div>
                              );
                            },
                          )}
                        </div>
                      ) : (
                        <p>
                          Este insight es una observación general y
                          no tiene una dimensión específica asociada.
                        </p>
                      )}
                    </div>
                  )}
                </div>
              )}
            </section>

            <section className="panel cycle-learning-panel">
              <div className="panel-header">
                <span>LEARNING CANDIDATE</span>
                <span className="panel-indicator">
                  {learningCandidate ? "REGISTERED" : "DRAFT"}
                </span>
              </div>

              {!selectedInsight || !comparisonLeftCycle || !comparisonRightCycle ? (
                <div className="cycle-learning-empty">
                  Select an insight from the comparison to prepare a learning candidate.
                </div>
              ) : (
                <div className="cycle-learning-content">
                  <div className="cycle-learning-source">
                    <span>SOURCE INSIGHT</span>
                    <strong>{selectedInsight.title}</strong>
                    <small>
                      {getCycleRequestText(comparisonLeftCycle)} ↔ {getCycleRequestText(comparisonRightCycle)}
                    </small>
                  </div>

                  {learningCandidate?.insightId === selectedInsight.id ? (
                    <>
                      <div className="cycle-learning-status">
                        <span>LEARNING STATUS</span>
                        <strong>{learningCandidate.status === "promoted" ? "PROMOTED TO SEMANTIC MEMORY" : "STAGED FOR LEARNING"}</strong>
                      </div>

                      <div className="cycle-learning-knowledge">
                        <span>REUSABLE KNOWLEDGE</span>
                        <p>{learningCandidate.knowledge}</p>
                      </div>

                      <div className="cycle-learning-evidence">
                        <span>EVIDENCE</span>
                        <strong>{learningCandidate.evidenceKeys.length} DIMENSIONS · {learningCandidate.status === "promoted" ? "SEMANTIC MEMORY" : "LOCAL STORE"}</strong>
                      </div>

                      {learningCandidate.status === "promoted" ? (
                        <div className="cycle-learning-promoted">
                          <strong>LEARNING PROMOTED</strong>
                          <span>Stored in NEXUS semantic memory.</span>
                          {learningCandidate.memoryKey ? (
                            <small>{learningCandidate.memoryKey}</small>
                          ) : null}
                        </div>
                      ) : (
                        <button
                          type="button"
                          className="cycle-learning-button"
                          onClick={promoteLearningCandidate}
                        >
                          PROMOTE TO SEMANTIC MEMORY
                        </button>
                      )}

                      <button
                        type="button"
                        className="cycle-learning-button secondary"
                        onClick={clearLearningCandidate}
                      >
                        CLEAR CANDIDATE
                      </button>
                    </>
                  ) : (
                    <>
                      <div className="cycle-learning-note">
                        Registering this candidate stores it in the local Learning Candidate Store. It does not write episodic memory or alter NEXUS reasoning.
                      </div>

                      <button
                        type="button"
                        className="cycle-learning-button"
                        onClick={registerLearningCandidate}
                      >
                        REGISTER LEARNING CANDIDATE
                      </button>
                    </>
                  )}
                </div>
              )}
            </section>

            <section className="panel cycle-learning-store-panel">
              <div className="panel-header">
                <span>LEARNING CANDIDATE STORE</span>
                <span className="panel-indicator">
                  {learningCandidates.length} STORED
                </span>
              </div>

              {learningCandidates.length === 0 ? (
                <div className="cycle-learning-empty">
                  No learning candidates have been staged yet.
                </div>
              ) : (
                <div className="cycle-learning-store-list">
                  {learningCandidates.map((candidate) => (
                    <button
                      key={`${candidate.insightId}-${candidate.createdAt}-${candidate.sourceCycleIds.join("-")}`}
                      type="button"
                      className={`cycle-learning-store-item ${
                        learningCandidate === candidate
                          ? "cycle-learning-store-item-selected"
                          : ""
                      }`}
                      onClick={() => setLearningCandidate(candidate)}
                    >
                      <span className="cycle-learning-store-marker">✦</span>
                      <span className="cycle-learning-store-summary">
                        <strong>{candidate.title}</strong>
                        <small>
                          {candidate.sourceCycleIds.length} CYCLES · {candidate.evidenceKeys.length} EVIDENCE · {candidate.status === "promoted" ? "PROMOTED" : "STAGED"} · {new Date(candidate.createdAt).toLocaleTimeString()}
                        </small>
                      </span>
                    </button>
                  ))}
                </div>
              )}

              <div className="cycle-learning-store-note">
                STAGED KNOWLEDGE · PROMOTION TO SEMANTIC MEMORY REMAINS EXPLICIT
              </div>
            </section>

            <section className="panel result-panel">
              <div className="panel-header">
                <span>RESULT</span>
              </div>

              <div className="result-content">
                <div className="result-icon">
                  {presence.state === "failed"
                    ? "!"
                    : "✓"}
                </div>

                <div>
                  <div className="result-title">
                    {resultTitle}
                  </div>

                  <p>
                    {presence.execution?.result
                      ? "Execution completed."
                      : executionStatus ===
                          "running"
                        ? "NEXUS is executing the requested action."
                        : verificationMessage}
                  </p>
                </div>
              </div>

              <div className="verification-badge">
                <span>
                  {verificationStatus ===
                  "verified"
                    ? "✓"
                    : "•"}
                </span>

                {verificationStatus
                  .replaceAll("_", " ")
                  .toUpperCase()}
              </div>
            </section>
          </div>

          <form
            className="command-bar"
            onSubmit={executeCommand}
          >
            <div className="command-prefix">
              NEXUS
            </div>

            <input
              type="text"
              placeholder={
                isSubmitting
                  ? "NEXUS está ejecutando..."
                  : "Escribí una solicitud..."
              }
              aria-label="NEXUS request"
              value={command}
              onChange={(event) =>
                setCommand(event.target.value)
              }
              disabled={isSubmitting}
            />

            <button
              type="submit"
              className="command-button"
              aria-label="Send request"
              disabled={
                isSubmitting ||
                command.trim().length === 0
              }
            >
              {isSubmitting ? "…" : "↑"}
            </button>
          </form>

          {commandError && (
            <div
              role="alert"
              className="command-error"
            >
              {commandError}
            </div>
          )}
        </section>
      </div>
    </main>
  );
}

export default App;
