from workflows.repository import save_workflow_version
from workflows.validator import validate_workflow
from copilot.debugger import debugger
from copilot.fixer import fixer

router = APIRouter(prefix="/copilot", tags=["copilot"])


@router.post("/generate")
def generate(data: dict):

    prompt = data.get("prompt")

    workflow = copilot.interpret(prompt)
    workflow = validate_workflow(workflow)

    if not workflow:
        return {
            "ok": False,
            "message": "No pude interpretar la instrucción."
        }

    save_workflow_version(
        workflow["name"],
        workflow["trigger"],
        workflow["steps"],
        "default",
        "main"
    )

    return {
        "ok": True,
        "workflow": workflow
    }

@router.post("/run")
async def run(data: dict, user=Depends(get_current_user)):

    tenant_id = user["tenant_id"]

    try:

        # =========================
        # 1. RUN DESDE BUILDER
        # =========================
        if "nodes" in data:

          workflow = {
           "nodes": data["nodes"],
           "edges": data.get("edges", [])
         }

          job_id = await manager_module.manager.create_job(
           workflow,
           tenant_id
         )

          return {"job_id": job_id}

        # =========================
        # 2. RUN POR NOMBRE (DB)
        # =========================
        elif "name" in data:

            job_id = await manager_module.manager.create_job(
                data["name"],
                tenant_id
            )

            return {"job_id": job_id}

        else:
            raise Exception("Payload inválido")

    except Exception as e:
        return {"error": str(e)}

@router.post("/")
def create_schedule(data: dict, user=Depends(get_current_user)):

    tenant_id = user["tenant_id"]

    required = ["name", "workflow", "type", "expression"]

    for r in required:
        if r not in data:
            raise HTTPException(400, f"Falta campo: {r}")

    conn = get_connection()
    cur = conn.cursor()

    cur.execute("""
        INSERT INTO schedules
        (name, workflow, type, expression, enabled, last_run, tenant_id)
        VALUES (%s, %s, %s, %s, %s, %s, %s)
        RETURNING id
    """, (
        data["name"],
        data["workflow"],
        data["type"],       # interval / cron
        data["expression"],
        True,
        None,
        tenant_id
    ))

    new_id = cur.fetchone()[0]

    conn.commit()
    conn.close()

    return {"ok": True, "id": new_id}

async def system_errors():

    conn = get_connection()
    cur = conn.cursor()

    cur.execute("""
        SELECT error, COUNT(*) as total
        FROM (
            SELECT json_extract(result,'$.error') as error
            FROM jobs
            WHERE status='error'
        )
        WHERE error IS NOT NULL
        GROUP BY error
        ORDER BY total DESC
        LIMIT 10
    """)

    rows = cur.fetchall()
    conn.close()

    errors = []

    for r in rows:
        errors.append({
            "error": r["error"],
            "count": r["total"]
        })

    return errors

from fastapi import Header, HTTPException, Depends
from jose import JWTError

from auth.security import decode_token


def get_current_user(authorization: str = Header(None)):

    if not authorization:
        raise HTTPException(status_code=401, detail="Token requerido")

    if not authorization.startswith("Bearer "):
        raise HTTPException(status_code=401, detail="Formato de token inválido")

    token = authorization.split(" ")[1]

    try:
        payload = decode_token(token)
        return payload

    except JWTError:
        raise HTTPException(status_code=401, detail="Token inválido")
    
def require_admin(user=Depends(get_current_user)):

    if user["role"] != "admin":
        raise HTTPException(status_code=403, detail="Admin only")

    return user    
    
def get_current_tenant(user=Depends(get_current_user)):
    return user["tenant_id"]

from fastapi import APIRouter, HTTPException, Depends
from datetime import datetime
import uuid

from auth.security import create_token
from auth.service import authenticate, create_user
from auth.deps import require_admin, get_current_user
from core.database import get_connection
from audit.service import log_event

router = APIRouter(prefix="/auth", tags=["auth"])


# =========================
# REGISTER (MULTI-TENANT REAL)
# =========================
@router.post("/register")
def register(data: dict):

    username = data.get("username")
    password = data.get("password")

    if not username or not password:
        raise HTTPException(400, "Username y password requeridos")

    tenant_id = str(uuid.uuid4())

    conn = get_connection()
    cur = conn.cursor()

    try:
        # 1️⃣ Crear tenant
        cur.execute("""
            INSERT INTO tenants (id, name, created_at)
            VALUES (%s, %s, %s)
        """, (
            tenant_id,
            username,
            datetime.utcnow()
        ))

        # 2️⃣ Crear usuario admin del tenant
        create_user(
            username=username,
            password=password,
            role="admin",
            tenant_id=tenant_id,
            workspace_id="main"
        )

        conn.commit()

    except Exception as e:
        conn.rollback()
        raise HTTPException(400, f"Error creando usuario: {str(e)}")

    finally:
        conn.close()

    return {
        "ok": True,
        "tenant_id": tenant_id
    }

@router.post("/login")
def login(data: dict):

    username = data.get("username")
    password = data.get("password")

    if not username or not password:
        raise HTTPException(400, "Username y password requeridos")

    user = authenticate(username, password)

    if not user:
        raise HTTPException(401, "Credenciales inválidas")

    token = create_token({
        "sub": user["id"],
        "username": user["username"],
        "role": user["role"],
        "tenant_id": user["tenant_id"],
        "workspace_id": user["workspace_id"]
    })

    # audit seguro
    try:
        log_event(
            user,
            action="login",
            entity="user",
            entity_id=user["id"]
        )
    except Exception:
        pass

    return {
        "access_token": token,
        "token_type": "bearer",
        "user": {
            "username": user["username"],
            "role": user["role"],
            "tenant_id": user["tenant_id"],
            "workspace_id": user["workspace_id"]

SECRET = "nexus-super-secret"  # 🔥 MISMO QUE ANTES

TOKEN_EXPIRE_HOURS = 8

pwd = CryptContext(
    schemes=["argon2"],
    deprecated="auto"
)

# ======================================================
# PASSWORDS
# ======================================================

def hash_password(password: str) -> str:
    return pwd.hash(password)


def verify_password(password: str, hashed: str) -> bool:
    return pwd.verify(password, hashed)


# ======================================================
# JWT
# ======================================================

def create_token(data: dict) -> str:

    payload = data.copy()
    payload["exp"] = datetime.utcnow() + timedelta(hours=TOKEN_EXPIRE_HOURS)

    return jwt.encode(payload, SECRET, algorithm=ALGORITHM)


def decode_token(token: str) -> dict:

    try:
        return jwt.decode(token, SECRET, algorithms=[ALGORITHM])
    except JWTError:
        return None

def create_user(
    username,
    password,
    role="user",
    tenant_id="default",
    workspace_id="main"
):

    conn = get_connection()
    cur = conn.cursor()

    user_id = str(uuid.uuid4())

    try:
        cur.execute("""
        INSERT INTO users
        (id, username, password_hash, role, tenant_id, workspace_id, created_at)
        VALUES (%s, %s, %s, %s, %s, %s, %s)
        """, (
            user_id,
            username,
            hash_password(password),
            role,
            tenant_id,
            workspace_id,
            datetime.utcnow()
        ))

        conn.commit()

    except Exception as e:
        conn.rollback()
        raise e

    finally:
        conn.close()


def get_user(username):

    conn = get_connection()
    cur = conn.cursor()

    cur.execute("""
        SELECT *
        FROM users
        WHERE username = %s
    """, (username,))

    row = cur.fetchone()
    conn.close()

    if not row:
        return None

    return dict(row)


def authenticate(username, password):

    user = get_user(username)

    if not user:
        return None

    if not verify_password(password, user["password_hash"]):
        return None

    return user

                print("🧠 Nexus suggestions:", suggestions)

            learning = learning_engine.analyze_recent_failures()

from core.metrics import metrics
from core.self_model import self_model
import jobs.manager as manager_module


async def adjust_workers():

    snapshot = self_model.snapshot()
    health = self_model.health_score(snapshot)

    queue_size = snapshot["queue_size"]

    manager = manager_module.manager

    if not manager:
        return

    current = manager.max_workers

    # ======================
    # Escalar hacia arriba
    # ======================

    if queue_size > 5 and health > 70:

        new_workers = min(current + 1, 8)

        if new_workers != current:
            await manager.update_workers(new_workers)
            print(f"🚀 Workers aumentados a {new_workers}")

    # ======================
    # Escalar hacia abajo
    # ======================

    if queue_size == 0 and current > 2:

        new_workers = current - 1

        await manager.update_workers(new_workers)
        print(f"📉 Workers reducidos a {new_workers}")
