class NexusHandler(BaseHTTPRequestHandler):

    # =========================================================
    # JSON SAFE
    # =========================================================

    def _json_safe(
        self,
        value: Any,
    ) -> Any:
        """
        Convierte estructuras internas de NEXUS
        en valores compatibles con JSON.

        Esta conversión ocurre únicamente
        en la frontera HTTP.
        """

        if value is None:
            return None

        if isinstance(
            value,
            (
                str,
                int,
                float,
                bool,
            ),
        ):
            return value

        if isinstance(
            value,
            (datetime, date),
        ):
            return value.isoformat()

        if isinstance(value, bytes):
            return {
                "type": "bytes",
                "size": len(value),
            }

        # PIL.Image.Image
        if (
            hasattr(value, "size")
            and hasattr(value, "mode")
            and value.__class__.__module__.startswith(
                "PIL"
            )
        ):
            width, height = value.size

            return {
                "type": "image",
                "width": width,
                "height": height,
                "mode": value.mode,
            }

        if isinstance(value, dict):
            return {
                str(key): self._json_safe(item)
                for key, item in value.items()
            }

        if isinstance(
            value,
            (list, tuple, set),
        ):
            return [
                self._json_safe(item)
                for item in value
            ]

        # Objetos Pydantic u otros objetos
        # con model_dump().
        model_dump = getattr(
            value,
            "model_dump",
            None,
        )

        if callable(model_dump):
            try:
                return self._json_safe(
                    model_dump()
                )
            except Exception:
                pass

        # Último recurso.
        return {
            "type": "object",
            "class": value.__class__.__name__,
            "repr": repr(value),
        }

    # =========================================================
    # HTTP HELPERS
    # =========================================================

    def _send_json(
        self,
        data: dict,
        status: int = 200,
    ) -> None:

        safe_data = self._json_safe(data)

        payload = json.dumps(
            safe_data,
            ensure_ascii=False,
        ).encode("utf-8")

        self.send_response(status)

        self.send_header(
            "Content-Type",
            "application/json; charset=utf-8",
        )

        self.send_header(
            "Content-Length",
            str(len(payload)),
        )

        # Permitimos que la Console Vite
        # consuma el Runtime local.
        self.send_header(
            "Access-Control-Allow-Origin",
            "*",
        )

        self.send_header(
            "Access-Control-Allow-Headers",
            "Content-Type",
        )

        self.send_header(
            "Access-Control-Allow-Methods",
            "GET, POST, OPTIONS",
        )

        self.end_headers()

        self.wfile.write(payload)

    def _send_html(
        self,
        content: bytes,
    ) -> None:

        self.send_response(200)

        self.send_header(
            "Content-Type",
            "text/html; charset=utf-8",
        )

        self.send_header(
            "Content-Length",
            str(len(content)),
        )

        self.end_headers()

        self.wfile.write(content)

    # =========================================================
    # MEMORY / TRACE HELPERS
    # =========================================================

    def _entry_to_dict(
        self,
        entry,
    ) -> dict:

        return {
            "id": entry.id,
            "timestamp": entry.timestamp.isoformat(),
            "type": entry.type,
            "data": entry.data,
        }

    def _build_trace(
        self,
        entries: list,
    ) -> dict:
        """
        Construye un Execution Trace a partir
        de las entradas generadas por una ejecución.
        """

        trace = {
            "request": None,
            "intent": None,
            "objective": None,
            "plan": None,
            "result": None,
            "error": None,
        }

        for entry in entries:

            if entry.type == "user_request":
                trace["request"] = entry.data

            elif entry.type == "intent":
                trace["intent"] = entry.data

            elif entry.type == "objective":
                trace["objective"] = entry.data

            elif entry.type == "plan":
                trace["plan"] = entry.data

            elif entry.type == "execution_result":
                trace["result"] = entry.data

            elif entry.type == "error":
                trace["error"] = entry.data

        return trace

    # =========================================================
    # PRESENCE EVENT STREAM (SSE)
    # =========================================================

    def _send_sse_headers(self) -> None:
        self.send_response(200)

        self.send_header(
            "Content-Type",
            "text/event-stream; charset=utf-8",
        )

        self.send_header(
            "Cache-Control",
            "no-cache",
        )

        self.send_header(
            "Connection",
            "keep-alive",
        )

        self.send_header(
            "Access-Control-Allow-Origin",
            "*",
        )

        self.send_header(
            "Access-Control-Allow-Headers",
            "Content-Type",
        )

        self.send_header(
            "Access-Control-Allow-Methods",
            "GET, POST, OPTIONS",
        )

        self.end_headers()

    def _write_sse(
        self,
        event: str,
        data: Any,
    ) -> None:
        payload = json.dumps(
            self._json_safe(data),
            ensure_ascii=False,
        )

        message = (
            f"event: {event}\\n"
            f"data: {payload}\\n\\n"
        ).encode("utf-8")

        self.wfile.write(message)
        self.wfile.flush()

    def _handle_presence_events(self) -> None:
        event_queue: queue.Queue[PresenceEvent] = queue.Queue()
        closed = threading.Event()

        def listener(event: PresenceEvent) -> None:
            if not closed.is_set():
                event_queue.put(event)

        unsubscribe = engine.presence_events.subscribe(listener)

        try:
            self._send_sse_headers()

            # Confirma que el canal está conectado.
            self._write_sse(
                "connected",
                {
                    "state": "connected",
                    "message": "Presence Event Stream conectado.",
                },
            )

            while True:
                try:
                    event = event_queue.get(
                        timeout=15.0,
                    )

                    self._write_sse(
                        "presence",
                        event.to_dict(),
                    )

                except queue.Empty:
                    # Keep-alive para mantener abierta
                    # la conexión SSE.
                    self.wfile.write(
                        b": keep-alive\\n\\n"
                    )
                    self.wfile.flush()

        except (BrokenPipeError, ConnectionResetError):
            pass
        finally:
            closed.set()
            unsubscribe()

    # =========================================================
    # OPTIONS
    # =========================================================

    def do_OPTIONS(self) -> None:

        self.send_response(204)

        self.send_header(
            "Access-Control-Allow-Origin",
            "*",
        )

        self.send_header(
            "Access-Control-Allow-Headers",
            "Content-Type",
        )

        self.send_header(
            "Access-Control-Allow-Methods",
            "GET, POST, OPTIONS",
        )

        self.end_headers()

    # =========================================================
    # GET
    # =========================================================

    def do_GET(self) -> None:

        # -----------------------------------------------------
        # PRESENCE EVENT STREAM
        # -----------------------------------------------------

        if self.path == "/api/presence/events":
            self._handle_presence_events()
            return

        # -----------------------------------------------------
        # Console HTML legacy
        # -----------------------------------------------------

        if self.path in (
            "/",
            "/index.html",
        ):

            index_file = (
                BASE_DIR / "index.html"
            )

            content = index_file.read_bytes()

            self._send_html(content)

            return

        # -----------------------------------------------------
        # STATUS
        # -----------------------------------------------------

        if self.path == "/api/status":

            self._send_json(
                {
                    "status": "online",
                    "memory": engine.memory.count(),
                }
            )

            return

        # -----------------------------------------------------
        # PRESENCE
        # -----------------------------------------------------

        if self.path == "/api/presence":

            snapshot = presence.snapshot()

            self._send_json(
                snapshot
            )

            return

        # -----------------------------------------------------
        # MEMORY
        # -----------------------------------------------------

        if self.path == "/api/memory":

            entries = [
                self._entry_to_dict(entry)
                for entry in engine.memory.all()
            ]

            self._send_json(
                {
                    "entries": entries,
                    "count": len(entries),
                }
            )

            return

        self.send_error(404)

    # =========================================================
    # LEARNING CANDIDATE PROMOTION
    # =========================================================

    def _handle_learning_candidate_promote(self) -> None:
        try:
            content_length = int(
                self.headers.get("Content-Length", "0")
            )

            body = self.rfile.read(content_length)
            candidate = json.loads(body.decode("utf-8"))

            if not isinstance(candidate, dict):
                self._send_json(
                    {"success": False, "message": "Learning candidate must be an object."},
                    status=400,
                )
                return

            candidate_id = candidate.get("id")
            knowledge = candidate.get("knowledge")
            insight_id = candidate.get("insightId")
            source_cycle_ids = candidate.get("sourceCycleIds")
            evidence_keys = candidate.get("evidenceKeys")

            if not isinstance(candidate_id, str) or not candidate_id.strip():
                self._send_json(
                    {"success": False, "message": "Candidate id is required."},
                    status=400,
                )
                return

            if not isinstance(knowledge, str) or not knowledge.strip():
                self._send_json(
                    {"success": False, "message": "Candidate knowledge is required."},
                    status=400,
                )
                return

            if not isinstance(insight_id, str) or not insight_id.strip():
                self._send_json(
                    {"success": False, "message": "Insight id is required."},
                    status=400,
                )
                return

            if not isinstance(source_cycle_ids, list):
                source_cycle_ids = []

            if not isinstance(evidence_keys, list):
                evidence_keys = []

            memory_key = f"insight:{candidate_id}"

            with engine_lock:
                fact = engine.semantic_memory.learn(
                    memory_key,
                    "cognitive_pattern",
                    knowledge.strip(),
                    confidence=0.85,
                    source="cognitive_insight",
                    metadata={
                        "candidate_id": candidate_id,
                        "insight_id": insight_id,
                        "source_cycle_ids": [str(item) for item in source_cycle_ids],
                        "evidence_keys": [str(item) for item in evidence_keys],
                        "promoted_from": "console_learning_candidate",
                    },
                )

            self._send_json(
                {
                    "success": True,
                    "memory_key": memory_key,
                    "promoted_at": datetime.now().astimezone().isoformat(),
                    "fact": fact,
                }
            )

        except json.JSONDecodeError:
            self._send_json(
                {"success": False, "message": "Invalid JSON."},
                status=400,
            )
        except Exception as exc:
            self._send_json(
                {"success": False, "message": str(exc)},
                status=500,
            )

    # =========================================================
    # POST
    # =========================================================

    def do_POST(self) -> None:

        if self.path == "/api/learning/candidates/promote":
            self._handle_learning_candidate_promote()
            return

        if self.path != "/api/execute":

            self.send_error(404)

            return

        try:

            # -------------------------------------------------
            # BODY
            # -------------------------------------------------

            content_length = int(
                self.headers.get(
                    "Content-Length",
                    "0",
                )
            )

            body = self.rfile.read(
                content_length
            )

            payload = json.loads(
                body.decode("utf-8")
            )

            text = payload.get("text")

            # -------------------------------------------------
            # VALIDATION
            # -------------------------------------------------

            if not isinstance(
                text,
                str,
            ):

                self._send_json(
                    {
                        "success": False,
                        "message": (
                            "El campo 'text' "
                            "es obligatorio."
                        ),
                    },
                    400,
                )

                return

            text = text.strip()

            if not text:

                self._send_json(
                    {
                        "success": False,
                        "message": (
                            "La instrucción "
                            "no puede estar vacía."
                        ),
                    },
                    400,
                )

                return

            # -------------------------------------------------
            # EXECUTION
            # -------------------------------------------------

            with engine_lock:

                # Memoria antes
                memory_before = (
                    engine.memory.count()
                )

                # Ejecutar NEXUS
                result = asyncio.run(
                    engine.execute(text)
                )

                # Memoria después
                memory_after = (
                    engine.memory.count()
                )

                # Entradas generadas
                all_entries = (
                    engine.memory.all()
                )

                execution_entries = (
                    all_entries[
                        memory_before:
                        memory_after
                    ]
                )

                # Trace
                trace = self._build_trace(
                    execution_entries
                )

            # -------------------------------------------------
            # PUBLIC RESPONSE
            # -------------------------------------------------

            response_text = (
                result.metadata.get(
                    "response"
                )
                if isinstance(
                    result.metadata,
                    dict,
                )
                else None
            )

            if not isinstance(
                response_text,
                str,
            ) or not response_text.strip():

                response_text = result.message

            self._send_json(
                {
                    "success": result.success,
                    "message": response_text,
                    "output": result.output,
                    "artifacts": result.artifacts,
                    "metadata": result.metadata,
                    "execution_time": result.execution_time,
                    "memory_count": (
                        engine.memory.count()
                    ),
                    "trace": trace,
                }
            )

        except Exception as exc:

            self._send_json(
                {
                    "success": False,
                    "message": (
                        f"Error interno: {exc}"
                    ),
                },
                500,
            )

    # =========================================================
    # LOG
    # =========================================================

    def log_message(
        self,
        format,
        *args,
    ) -> None:

        print(
            f"[NEXUS] {format % args}"
        )

def main() -> None:
    server = ThreadingHTTPServer(
        (HOST, PORT),
        NexusHandler,
    )

    async def start_assistant_services() -> None:
        await assistant_lifecycle.start()

    async def stop_assistant_services() -> None:
        await assistant_lifecycle.stop()

    try:
        asyncio.run(start_assistant_services())
        print("✓ Reminder Store")
        print("✓ Reminder Scheduler")

        server.serve_forever()

    except KeyboardInterrupt:
        print()
        print("NEXUS detenido.")

    finally:
        server.server_close()

        try:
            asyncio.run(stop_assistant_services())
        finally:
            engine.semantic_memory.close()


if __name__ == "__main__":
    main()

function Test-HttpEndpoint([string]$Url) {
    try {
        $response = Invoke-WebRequest -Uri $Url -UseBasicParsing -TimeoutSec 2
        return ($response.StatusCode -ge 200 -and $response.StatusCode -lt 500)
    } catch { return $false }
}
function Wait-HttpEndpoint([string]$Url, [int]$Seconds, [string]$Name) {
    $deadline = (Get-Date).AddSeconds($Seconds)
    while ((Get-Date) -lt $deadline) {
        if (Test-HttpEndpoint $Url) { return }
        Start-Sleep -Milliseconds 750
    }
    throw "No se pudo validar $Name en $Url dentro de $Seconds segundos. Logs: $logsDir"
}

from context import (
    ContextStore,
    OrganizationProfile,
    ProjectContext,
    UserProfile,
)


def test_user_organization_and_project_context() -> None:
    store = ContextStore()

    organization = OrganizationProfile.create(
        "Estudio Jurídico Demo",
        industry="legal",
        areas=("Civil", "Comercial"),
        rules=("Los contratos requieren revisión antes de firma.",),
        priorities=("Vencimientos", "Clientes"),
    )

    user = UserProfile.create(
        "Diego",
        role="Abogado",
        organization_id=organization.id,
        responsibilities=("Revisión contractual",),
        priorities=("Vencimientos",),
    )

    project = ProjectContext.create(
        "Cliente Demo",
        organization.id,
        description="Proyecto de prueba.",
        responsible_user_ids=(user.id,),
    )

    store.add_organization(organization)
    store.add_user(user)
    store.add_project(project)

    context = store.build_context(user_id=user.id)

    assert context.user is user
    assert context.organization is organization
    assert len(context.projects) == 1
    assert context.projects[0] is project
    assert context.projects[0].name == "Cliente Demo"

def test_start_runs_scheduler_in_background() -> None:
    services = build_assistant_services()
    lifecycle = AssistantServicesLifecycle(services)

    async def run() -> None:
        await lifecycle.start()

        assert lifecycle.running is True
        assert lifecycle._scheduler_task is not None
        assert not lifecycle._scheduler_task.done()

        await lifecycle.stop()

    asyncio.run(run())


def test_stop_stops_background_scheduler() -> None:
    services = build_assistant_services()
    lifecycle = AssistantServicesLifecycle(services)

    async def run() -> None:
        await lifecycle.start()
        await lifecycle.stop()

        assert lifecycle.running is False
        assert lifecycle._scheduler_task is None

    asyncio.run(run())


def test_scheduler_executes_pending_reminder_in_background() -> None:
    services = build_assistant_services()

    reminder = Reminder.create(
        message="Reminder de prueba",

            seconds=0.05

    services.reminder_scheduler.schedule(reminder)
