class FakeToolRegistry:

    def describe(self):
        return [
            {
                "name": "windows.open_app",
                "action": "system.open_app",
                "description": "Open application",
                "provider": "windows",
                "parameters": {
                    "application": {
                        "type": "string",
                        "required": True,
                    }
                },
            },
            {
                "name": "windows.set_volume",
                "action": "system.set_volume",
                "description": "Set volume",
                "provider": "windows",
                "parameters": {
                    "level": {
                        "type": "number",
                        "required": True,
                    }
                },
            },
        ]

    def resolve(self, action):
        return object()


class FakeAction:

    def __init__(self, name):
        self.name = name
        self.internal = False


class FakeActionRegistry:

    def get(self, action):
        return FakeAction(action)


class FakeRuntime:

    def __init__(self, result):
        self.tool_registry = FakeToolRegistry()
        self.action_registry = FakeActionRegistry()
        self.result = result

    async def execute(self, context):
        """
        Simula el resultado final del runtime real.

        Como este FakeRuntime no atraviesa ActionManager,
        reproduce explícitamente la evidencia estructural
        que ActionManager registra durante una ejecución real.
        """

        result_output = (
            self.result.output
            if isinstance(self.result.output, dict)
            else {}
        )

        result_artifacts = (
            self.result.artifacts
            if isinstance(self.result.artifacts, dict)
            else {}
        )

        result_keys = list(result_output.keys())
        artifact_keys = list(result_artifacts.keys())

        for step in context.plan.steps:

            step.status = StepStatus.COMPLETED

            step.metadata["execution"] = {
                "status": "completed",
                "result_keys": result_keys,
                "artifact_keys": artifact_keys,
            }

        context.plan.completed = True

        return self.result


class FakeIntentProvider(IntentProvider):

    def __init__(self, intent):
        self.intent = intent

    async def recognize(
        self,
        text,
        capabilities=None,
        context=None,
    ):
        return self.intent


def run(coro):
    return asyncio.run(coro)


def check(condition, message):

        raise AssertionError(message)

from __future__ import annotations

from memory.semantic_memory import SemanticMemory
from memory.semantic_knowledge_extractor import (
    SemanticKnowledgeExtractor,
)
from memory.semantic_retriever import SemanticRetriever


def test_semantic_components_work_together() -> None:
    semantic_memory = SemanticMemory()

    extractor = SemanticKnowledgeExtractor(
        semantic_memory
    )

    retriever = SemanticRetriever(
        semantic_memory
    )

    from memory.episode import Episode

    episode = Episode(
        action="system.open_app",
        parameters={
            "application": "Chrome",
        },
        original_text="abrí chrome",
        success=True,
        output={
            "status": "opened",
            "application": "Chrome",
        },
    )

    facts = extractor.extract(
        episode
    )

    assert len(facts) == 1

    results = retriever.search(
        "chrome"
    )

    assert len(results) == 1

    match = results[0]

    assert match.fact.subject == "chrome"
    assert (
        match.fact.predicate
        == "es una aplicación"
    )
    assert match.fact.value is True


def test_failed_episode_does_not_create_knowledge() -> None:
    semantic_memory = SemanticMemory()

    extractor = SemanticKnowledgeExtractor(
        semantic_memory
    )

    from memory.episode import Episode

    episode = Episode(
        action="system.open_app",
        parameters={
            "application": "Chrome",
        },
        original_text="abrí chrome",
        success=False,
        output={},
    )

    facts = extractor.extract(
        episode
    )

    assert facts == []
    assert semantic_memory.count() == 0


def test_semantic_memory_is_reusable() -> None:
    semantic_memory = SemanticMemory()

    extractor = SemanticKnowledgeExtractor(
        semantic_memory
    )

    retriever = SemanticRetriever(
        semantic_memory
    )

    from memory.episode import Episode

    extractor.extract(
        Episode(
            action="system.open_app",
            parameters={
                "application": "Chrome",
            },
            original_text="abrí chrome",
            success=True,
            output={
                "status": "opened",
                "application": "Chrome",
            },
        )
    )

    first = retriever.search(
        "chrome"
    )

    second = retriever.search_by_subject(
        "chrome"
    )

    assert len(first) == 1
    assert len(second) == 1

    assert (
        first[0].fact
        is second[0].fact
    )


if __name__ == "__main__":

from __future__ import annotations

import asyncio

from planner.decision import DecisionEngine
from planner.intent import Intent
from planner.plan import ExecutionPlan, PlanStep, StepStatus
from planner.verification import Verifier
from runtime.execution_result import ExecutionResult


def check(condition: bool, message: str) -> None:
    if not condition:
        raise AssertionError(message)


def test_verifier_success() -> None:

    verifier = Verifier()

    class Objective:
        id = "objective-1"

    plan = ExecutionPlan(
        objective_id="objective-1",
        steps=[
            PlanStep(
                id="step-1",
                name="test",
                description="test",
                action="system.test",
                status=StepStatus.COMPLETED,
            )
        ],
        completed=True,
    )

    result = ExecutionResult(
        success=True,
        message="",
        output={
            "status": "ok",
        },
    )

    verification = verifier.verify(
        objective=Objective(),
        plan=plan,
        result=result,
    )

    check(
        verification.verified,
        "Successful completed execution must verify.",
    )

    check(
        verification.confidence == 1.0,
        "Expected full verification confidence.",
    )


def test_verifier_failed_execution() -> None:

    verifier = Verifier()

    class Objective:
        id = "objective-2"

    plan = ExecutionPlan(
        objective_id="objective-2",
        completed=True,
    )

    result = ExecutionResult(
        success=False,
        message="execution failed",
        output={},
    )

    verification = verifier.verify(
        objective=Objective(),
        plan=plan,
        result=result,
    )

    check(
        not verification.verified,
        "Failed execution must not verify.",
    )


def test_verifier_incomplete_plan() -> None:

    verifier = Verifier()

    class Objective:
        id = "objective-3"

    plan = ExecutionPlan(

        completed=False,
    )

    result = ExecutionResult(
        success=True,
        message="",
        output={
            "status": "ok",
        },
    )

    verification = verifier.verify(
        objective=Objective(),
        plan=plan,
        result=result,
    )

    check(

        "Incomplete plan must not verify.",

def test_decision_known_action() -> None:

    engine = DecisionEngine()

    intent = Intent(
        action="system.open_app",
        confidence=1.0,
        parameters={
            "application": "chrome",
        },
        original_text="Abrí Chrome",
    )

        capabilities=[],

    )

from planner.episode_query_resolver import EpisodeQueryResolver
from memory.episode import Episode
from memory.episode_memory import EpisodeMemory
from memory.working_memory import WorkingMemory


def check(
    condition: bool,
    message: str,
) -> None:

    if not condition:
        raise AssertionError(message)


def main() -> None:

    memory = WorkingMemory()
    episodes = EpisodeMemory(memory)
    resolver = EpisodeQueryResolver()

    #
    # Episodios de prueba
    #

    episodes.add(
        Episode(
            action="system.open_app",
            parameters={
                "application": "chrome",
            },
            original_text="abrí Chrome",
            success=True,
            output={
                "status": "opened",
                "application": "chrome",
            },
        )
    )

    episodes.add(
        Episode(
            action="system.set_volume",
            parameters={
                "level": 30,
            },
            original_text="poné el volumen al 30%",
            success=True,
            output={
                "status": "set",
                "volume": 30,
            },
        )
    )

    episodes.add(
        Episode(
            action="system.close_app",
            parameters={
                "application": "chrome",
            },
            original_text="cerrá Chrome",
            success=True,
            output={
                "status": "closed",
                "application": "chrome",
            },
        )
    )

    #
    # Consulta semántica
    #

    query = resolver.resolve(
        "qué aplicación cerré"
    )

    check(
        query is not None,
        "La consulta debería resolverse.",
    )

    episode = episodes.find_latest_by_action(
        query.action
    )

    check(
        episode is not None,
        "No se encontró el episodio de cierre.",
    )

    application = episodes.find_parameter(
        query.action,
        query.field,
    )

    check(
        application == "chrome",
        "La aplicación recuperada debería ser chrome.",
    )

    #
    # Apertura
    #

    query = resolver.resolve(
        "qué aplicación abrí"
    )

    check(
        query is not None,
        "La consulta de apertura debería resolverse.",
    )

    application = episodes.find_parameter(
        query.action,
        query.field,
    )

    check(
        application == "chrome",
        "La aplicación abierta debería ser chrome.",
    )

    #
    # Volumen
    #

    query = resolver.resolve(
        "qué volumen puse"
    )

    check(
        query is not None,
        "La consulta de volumen debería resolverse.",
    )

    volume = episodes.find_parameter(
        query.action,
        query.field,
    )

    check(
        volume == 30,
        "El volumen recuperado debería ser 30.",
    )

    #
    # Acción inexistente
    #

    result = episodes.find_latest_by_action(
        "system.mute"
    )

    check(
        result is None,
        "No debería encontrar una acción inexistente.",
    )

    #
    # Solo episodios exitosos
    #

    episodes.add(
        Episode(
            action="system.close_app",
            parameters={
                "application": "notepad",
            },
            original_text="cerrá Notepad",
            success=False,
            output={},
        )
    )

    application = episodes.find_parameter(
        "system.close_app",
        "application",
    )

    check(
        application == "chrome",
        "Los episodios fallidos no deben reemplazar "
        "al último episodio exitoso.",
    )

    print("=" * 72)
    print("EPISODE MEMORY")
    print("=" * 72)
    print("PASSED : 6")
    print("FAILED : 0")
    print("=" * 72)


if __name__ == "__main__":
    main()

        "http://localhost:5174",
        "http://127.0.0.1:5174",

        return {}

def check(
    condition: bool,
    message: str,
) -> None:

    if not condition:
        raise AssertionError(message)


def create_context(
    plan: ExecutionPlan,
) -> ExecutionContext:

    objective = Objective(
        id="test-objective",
        title="Consultar memoria episódica",
    )

    return ExecutionContext(
        objective=objective,
        plan=plan,
    )


async def run() -> None:

    memory = WorkingMemory()
    episode_memory = EpisodeMemory(memory)

    episode_memory.add(
        Episode(
            action="system.close_app",
            parameters={
                "application": "chrome",
            },
            original_text="cerrá Chrome",
            success=True,
            output={
                "status": "closed",
                "application": "chrome",
            },
        )
    )

    action = QueryEpisodeAction(
        episode_memory
    )

    #
    # Consulta válida
    #

    plan = ExecutionPlan(
        objective_id="test-objective",
        steps=[
            PlanStep(
                id="query",
                name="Query episode",
                description="Consulta un episodio anterior.",
                action="system.query_episode",
                metadata={
                    "action": "system.close_app",
                    "field": "application",
                },
            )
        ],
    )

    context = create_context(plan)

    await action.execute(
        context,
        plan.steps[0],
    )

    result = context.results.get(
        "episode_query_result"
    )

    check(
        isinstance(result, dict),
        "La acción debería producir un resultado estructurado.",
    )

    check(
        result.get("status") == "found",
        "El estado debería ser found.",
    )

    check(
        result.get("action") == "system.close_app",
        "La acción consultada debería ser system.close_app.",
    )

    check(
        result.get("field") == "application",
        "El campo consultado debería ser application.",
    )

    check(
        result.get("value") == "chrome",
        "El valor recuperado debería ser chrome.",
    )

    check(
        result.get("original_text") == "cerrá Chrome",
        "Debería conservarse el texto original del episodio.",
    )

    #
    # Consulta sin episodio
    #

    missing_plan = ExecutionPlan(
        objective_id="missing-objective",
        steps=[
            PlanStep(
                id="query",
                name="Query missing episode",
                description="Consulta un episodio inexistente.",
                action="system.query_episode",
                metadata={
                    "action": "system.mute",
                    "field": "status",
                },
            )
        ],
    )

    missing_context = create_context(
        missing_plan
    )

    await action.execute(
        missing_context,
        missing_plan.steps[0],
    )

    check(
        bool(missing_context.errors),
        "Una consulta sin episodio debería producir un error.",
    )

    #
    # Campo inexistente
    #

    invalid_field_plan = ExecutionPlan(
        objective_id="invalid-field-objective",
        steps=[
            PlanStep(
                id="query",
                name="Query invalid field",
                description="Consulta un campo inexistente.",
                action="system.query_episode",
                metadata={
                    "action": "system.close_app",
                    "field": "unknown_field",
                },
            )
        ],
    )

    invalid_field_context = create_context(
        invalid_field_plan
    )

    await action.execute(
        invalid_field_context,
        invalid_field_plan.steps[0],
    )

    check(
        bool(invalid_field_context.errors),
        "Un campo inexistente debería producir un error.",
    )

    #
    # Parámetros inválidos
    #

    invalid_parameters_plan = ExecutionPlan(
        objective_id="invalid-parameters-objective",
        steps=[
            PlanStep(
                id="query",
                name="Query invalid parameters",
