from __future__ import annotations
import asyncio
import json
import re
from typing import Any

import httpx
from ollama import AsyncClient, RequestError

from planner.intent import Intent
from planner.intent_provider import IntentProvider


class OllamaIntentProvider(IntentProvider):
    """Interpreta lenguaje natural usando Ollama y valida la intención resultante."""

    MIN_CONFIDENCE = 0.50
    REQUEST_TIMEOUT_SECONDS = 10.0

    def __init__(
        self,
        model: str = "llama3:latest",
        host: str = "http://localhost:11434",
    ) -> None:
        self._model = model
        self._host = host

    async def recognize(
        self,
        text: str,
        capabilities: list[dict[str, Any]] | None = None,
        context: dict[str, Any] | None = None,
    ) -> Intent:
        capabilities = capabilities or []
        context = context or {}

        # Índices para resolver la respuesta del modelo contra el catálogo real.
        by_name = {
            capability["name"]: capability
            for capability in capabilities
            if isinstance(capability, dict)
            and isinstance(capability.get("name"), str)
        }
        by_action = {
            capability["action"]: capability
            for capability in capabilities
            if isinstance(capability, dict)
            and isinstance(capability.get("action"), str)
        }

        normalized = text.lower().strip()

        # Consultas deterministas de volumen actual.
        volume_read_patterns = (
            r"\ba cuánto está\b",
            r"\ba cuanto está\b",
            r"\ba cuánto quedó\b",
            r"\ba cuanto quedó\b",
            r"\bcuánto está\b",
            r"\bcuanto está\b",
            r"\bcuánto quedó\b",
            r"\bcuanto quedó\b",
            r"\bqué volumen\b",
            r"\bque volumen\b",
            r"\bcuál es el volumen\b",
            r"\bcual es el volumen\b",
            r"\bvolumen actual\b",
            r"\bqué nivel de volumen\b",
            r"\bque nivel de volumen\b",
        )

        if any(re.search(pattern, normalized) for pattern in volume_read_patterns):
            if by_action.get("system.get_volume") is not None:
                return Intent(
                    action="system.get_volume",
                    confidence=1.0,
                    parameters={},
                    original_text=text,
                )

        capabilities_json = json.dumps(
            capabilities,
            ensure_ascii=False,
            indent=2,
        )
        context_json = json.dumps(
            context,
            ensure_ascii=False,
            indent=2,
            default=str,
        )

        system_prompt = f"""
Sos el módulo de interpretación de NEXUS.

Tu tarea es convertir la solicitud del usuario
 en UNA intención estructurada.

CAPACIDADES DISPONIBLES:

{capabilities_json}

CONTEXTO RECIENTE DE NEXUS:

{context_json}

IMPORTANTE SOBRE EL CONTEXTO:

El contexto sirve únicamente para ayudar a
comprender solicitudes que dependan de una
interacción anterior.

El contexto NO crea capacidades nuevas.

Nunca ejecutes una acción solamente porque
aparece en el contexto.

La acción seleccionada debe existir actualmente
en CAPACIDADES DISPONIBLES.

El estado real del sistema tiene prioridad sobre
cualquier dato almacenado en memoria.

El contexto contiene diferentes tipos de información
y cada uno tiene un propósito distinto:

- last_request:
  última solicitud textual del usuario.

- last_intent:
  última intención interpretada.

- last_result:
  resultado de la última ejecución.

- recent_episodes:
  registro de acciones que NEXUS ejecutó realmente,
  incluyendo la acción, los parámetros, el texto
  original y el resultado.

IMPORTANTE SOBRE recent_episodes:

recent_episodes representa MEMORIA EPISÓDICA.

Si el usuario pregunta qué hizo, cuál fue su última
acción, qué hizo recién o qué acción realizó antes,
está preguntando por un EPISODIO de interacción,
no por el estado actual del sistema.

Ejemplos:

- "¿Qué hice recién?"
- "¿Cuál fue mi última acción?"
- "¿Qué hice antes?"
- "¿Qué cambié antes?"
- "¿Qué hice antes de subir el volumen?"

Estas preguntas deben resolverse utilizando
recent_episodes cuando exista una capacidad disponible
para consultar memoria episódica.

No deben interpretarse automáticamente como una
consulta sobre el estado actual del sistema.

Diferencias importantes:

- "¿A cuánto está el volumen?"
  → consulta el estado actual.

- "¿Qué volumen tenía antes?"
  → consulta el historial del estado del volumen.

- "¿Qué hice recién?"
  → consulta memoria episódica.

- "¿Cuál fue mi última acción?"
  → consulta memoria episódica.

El estado actual del sistema tiene prioridad cuando
el usuario pregunta por el estado actual.

La memoria episódica tiene prioridad cuando el usuario
pregunta explícitamente por acciones realizadas
anteriormente.

REGLAS ESTRICTAS:

1. Solo podés seleccionar una acción que exista
   en las capacidades disponibles.

2. Si la solicitud no corresponde claramente
   a una capacidad, devolvé system.unknown.

3. NO intentes adivinar la intención del usuario.

4. NO confundas verbos de conversación con
   acciones del sistema. Un saludo como "hola"
   debe devolver system.unknown.

5. Podés utilizar el contexto reciente para
   interpretar referencias como:

   - "dejalo como estaba"
   - "dejala en 30%"
   - "eso"
   - "lo mismo"
   - "el volumen"
   - "esa aplicación"

   siempre que exista suficiente información
   en el contexto.

6. Para system.open_app, el usuario debe estar
   pidiendo explícitamente abrir, ejecutar o
   iniciar una aplicación. No lo uses para documentos.
   Para documentos usa system.open_file solo si hay
   nombre de archivo y carpeta autorizada.

7. Para system.set_volume, el usuario puede
   referirse al volumen, música, sonido, audio
   o utilizar expresiones como subir, bajar,
   fuerte o bajo cuando el contexto indique
   claramente que habla del sonido.

8. Para system.get_volume, el usuario está
   consultando el volumen actual.

   Ejemplos:
   - "¿A cuánto está el volumen?"
   - "¿Cuál es el volumen?"
   - "¿Qué volumen tengo?"
   - "¿A cuánto está?"
   - "Decime el volumen actual"

   Si la solicitud es una pregunta sobre el
   estado actual del volumen, utilizá
   system.get_volume.

9. Una consulta sobre el volumen NUNCA debe
   convertirse en system.set_volume.

10. Si la solicitud contiene un valor numérico
    acompañado de una instrucción para cambiar
    el volumen, utilizá system.set_volume.

11. Si la solicitud solamente pregunta por el
    volumen actual y no solicita modificarlo,
    utilizá system.get_volume.

12. Para system.mute, el usuario debe pedir
    silenciar.

13. Para system.unmute, el usuario debe pedir
    quitar el silencio.

14. Para system.close_app, el usuario debe pedir
    explícitamente cerrar una aplicación.

15. Si la solicitud es enviar mensajes, hablar
    con una persona, realizar una tarea que no
    aparezca en las capacidades, o cualquier
    otra tarea no disponible, devolvé:

    system.unknown

16. Nunca uses una capacidad solamente porque
    una palabra del usuario parece coincidir
    con su nombre.

17. El campo action debe utilizar exactamente
    el campo action de la capacidad.

18. Si no existe suficiente información para
    completar los parámetros obligatorios,
    devolvé system.unknown.

19. Respondé únicamente JSON válido.

EJEMPLOS:

Usuario:
"Poné el volumen al 40%"

Respuesta:
{{
    "action": "system.set_volume",
    "confidence": 1.0,
    "parameters": {{
        "level": 40
    }}
}}

        "level": 30

}}

"Está muy fuerte el audio, bajalo al 25%"

{{

    "confidence": 1.0,
    "parameters": {{
        "level": 25
    }}
}}

Usuario:
"Mandale un mensaje a Juan"

Respuesta:
{{
    "action": "system.unknown",
    "confidence": 0.0,
    "parameters": {{}}
}}

Usuario:
"Abrí Spotify"

Respuesta:
{{
    "action": "system.open_app",
    "confidence": 1.0,
    "parameters": {{
        "application": "Spotify"
    }}
}}

Usuario:
"hola"

Respuesta:
{{
    "action": "system.unknown",
    "confidence": 0.0,
    "parameters": {{}}
}}

FORMATO:

{{
    "action": "system.unknown",
    "confidence": 0.0,
    "parameters": {{}}
}}

SOLICITUD DEL USUARIO:

{text}

        try:
            async with asyncio.timeout(self.REQUEST_TIMEOUT_SECONDS):
                async with AsyncClient(
                    host=self._host,
                    timeout=self.REQUEST_TIMEOUT_SECONDS,
                ) as client:
                    response = await client.chat(
                        model=self._model,
                        messages=[{"role": "system", "content": system_prompt}],
                        format="json",
                        options={"temperature": 0},
                    )
        except (
            TimeoutError,
            httpx.TimeoutException,
            httpx.RequestError,
            RequestError,
        ):
            return self._unknown(text)

        try:
            data = json.loads(response.message.content)
        except (json.JSONDecodeError, TypeError):
            return self._unknown(text)

        if not isinstance(data, dict):
            return self._unknown(text)

        requested_action = str(data.get("action", "system.unknown"))
        try:
            confidence = float(data.get("confidence", 0.0))
        except (TypeError, ValueError):
            confidence = 0.0

        parameters = data.get("parameters", {})
        if not isinstance(parameters, dict):
            parameters = {}

        if confidence < self.MIN_CONFIDENCE:
            return self._unknown(text, confidence)

        capability = by_action.get(requested_action)
        if capability is None:
            capability = by_name.get(requested_action)
        if capability is None:
            return self._unknown(text, confidence)

        if not self._validate_parameters(capability, parameters):
            return self._unknown(text, confidence)

        if not self._validate_semantics(
            text,
            capability["action"],
            parameters,
            context,
        ):
            return self._unknown(text, confidence)

        return Intent(
            action=capability["action"],
            confidence=confidence,
            parameters=parameters,
            original_text=text,
        )

    def _validate_semantics(
        self,
        text: str,
        action: str,
        parameters: dict[str, Any],
        context: dict[str, Any] | None = None,
    ) -> bool:
        """Comprueba que la acción propuesta sea compatible con la solicitud."""
        context = context or {}
        normalized = text.lower().strip()

        if action == "system.open_file":
            open_pattern = r"\b(abr[ií]r?|abrime|abrirme)\b"
            if not re.search(open_pattern, normalized):
                return False
            file_name = parameters.get("file_name")
            location = parameters.get("location")
            return (
                isinstance(file_name, str) and bool(file_name.strip())
                and location in {"desktop", "documents", "downloads"}
            )

        if action == "system.open_app":
            open_pattern = r"\b(abr[ií]r?|ejecut[aá]r?|inici[aá]r?|lanzar?)\b"
            if not re.search(open_pattern, normalized):
                return False
            application = parameters.get("application")
            return isinstance(application, str) and bool(application.strip())

        if action == "system.get_volume":
            read_patterns = (
                r"\ba cuánto está\b",
                r"\ba cuanto está\b",
                r"\bcuánto está\b",
                r"\bcuanto está\b",
                r"\bqué volumen\b",
                r"\bque volumen\b",
                r"\bcuál es el volumen\b",
                r"\bcual es el volumen\b",
                r"\bvolumen actual\b",
                r"\bqué nivel de volumen\b",
                r"\bque nivel de volumen\b",
                r"\bdecime el volumen actual\b",
                r"\bdecime el volumen\b",
            )
            return any(re.search(pattern, normalized) for pattern in read_patterns)

        if action == "system.set_volume":
            read_patterns = (
                r"\ba cuánto está\b",
                r"\ba cuanto está\b",
                r"\bcuánto está\b",
                r"\bcuanto está\b",
                r"\bqué volumen\b",
                r"\bque volumen\b",
                r"\bcuál es el volumen\b",
                r"\bcual es el volumen\b",
                r"\bvolumen actual\b",
                r"\bqué nivel de volumen\b",
                r"\bque nivel de volumen\b",
            )
            if any(re.search(pattern, normalized) for pattern in read_patterns):
                return False

            level = parameters.get("level")
            if not isinstance(level, (int, float)) or isinstance(level, bool):
                return False
            if not 0 <= float(level) <= 100:
                return False

            audio_cues = (
                "volumen", "música", "musica", "sonido", "audio", "parlante",
                "altavoz", "auricular", "auriculares", "micrófono", "microfono",
            )
            volume_cues = (
                "fuerte", "bajo", "baja", "bajá", "subí", "subi", "subir",
                "subirlo", "bajarlo", "bajala", "bajalo", "subilo", "subila",
                "más fuerte", "mas fuerte", "más bajo", "mas bajo", "dejala",
                "dejalo", "dejarlo",
            )
            has_audio_context = any(cue in normalized for cue in audio_cues)
            has_volume_context = any(cue in normalized for cue in volume_cues)

            previous_intent = context.get("last_intent")
            previous_result = context.get("last_result")
            previous_action = None
            if isinstance(previous_intent, dict):
                previous_action = previous_intent.get("action")

            previous_volume = None
            if isinstance(previous_result, dict):
                output = previous_result.get("output")
                if isinstance(output, dict):
                    previous_volume = output.get("volume")

            has_previous_volume_context = previous_action in {
                "system.set_volume",
                "system.get_volume",
            }
            if isinstance(previous_volume, (int, float)) and not isinstance(previous_volume, bool):
                has_previous_volume_context = True

            return has_audio_context or has_volume_context or has_previous_volume_context

        if action == "system.mute":
            return bool(re.search(r"\b(silenci[aá]r?|mute|sin sonido)\b", normalized))

        if action == "system.unmute":
            return bool(
                re.search(
                    r"\b(quit[aá]r?.*silencio|desmut|activar.*sonido)\b",
                    normalized,
                )
            )

        if action == "system.close_app":
            close_pattern = r"\b(cerr[aá]r?|termin[aá]r?|finaliz[aá]r?)\b"
            if not re.search(close_pattern, normalized):
                return False
            application = parameters.get("application")
            return isinstance(application, str) and bool(application.strip())

        if action == "system.get_last_episode":
            episode_patterns = (
                r"\búltima acción\b",
                r"\bultima accion\b",
                r"\búltimo episodio\b",
                r"\bultimo episodio\b",
                r"\bqué hice\b",
                r"\bque hice\b",
                r"\bqué hiciste\b",
                r"\bque hiciste\b",
                r"\bqué pasó\b",
                r"\bque paso\b",
                r"\bqué acción hice\b",
                r"\bque accion hice\b",
                r"\búltima vez\b",
                r"\bultima vez\b",
                r"\brecién\b",
                r"\brecien\b",
                r"\bqué hice antes\b",
                r"\bque hice antes\b",
                r"\bqué cambié antes\b",
                r"\bque cambie antes\b",
                r"\bqué hizo nexus antes\b",
                r"\bque hizo nexus antes\b",
            )
            return any(re.search(pattern, normalized) for pattern in episode_patterns)

        # Las demás acciones conservan la política de validación existente.
        return True

    def _validate_parameters(
        self,
        capability: dict[str, Any],
        parameters: dict[str, Any],
    ) -> bool:
        """Valida parámetros contra el esquema de la capacidad seleccionada."""
        schema = capability.get("parameters", {})
        if not isinstance(schema, dict):
            return True

        # Parámetros obligatorios.
        for name, definition in schema.items():
            if not isinstance(definition, dict):
                continue
            if definition.get("required", False) and name not in parameters:
                return False

        # No aceptar parámetros que no estén definidos en el esquema.
        for name in parameters:
            if name not in schema:
                return False

        # Comprobar los tipos de los parámetros recibidos.
        for name, value in parameters.items():
            definition = schema.get(name, {})
            parameter_type = definition.get("type")

            if parameter_type == "string":
                if not isinstance(value, str):
                    return False
            elif parameter_type == "number":
                if isinstance(value, bool) or not isinstance(value, (int, float)):
                    return False
            elif parameter_type == "integer":
                if isinstance(value, bool) or not isinstance(value, int):
                    return False

        return True

    def _unknown(
        self,
        text: str,
        confidence: float = 0.0,
    ) -> Intent:
        return Intent(
            action="system.unknown",
            confidence=confidence,
            parameters={},
            original_text=text,
        )

    def _validate_semantics(

                return False

            if not isinstance(level, (int, float)) or isinstance(level, bool):

            if not 0 <= float(level) <= 100:
                return False

                "volumen", "música", "musica", "sonido", "audio", "parlante",

        return True

    def _validate_parameters(
        self,
        capability: dict[str, Any],
        parameters: dict[str, Any],
    ) -> bool:
        """Valida parámetros contra el esquema de la capacidad seleccionada."""
        schema = capability.get("parameters", {})
        if not isinstance(schema, dict):
            return True

        # Parámetros obligatorios.
        for name, definition in schema.items():
            if not isinstance(definition, dict):
                continue
            if definition.get("required", False) and name not in parameters:
                return False

        # No aceptar parámetros que no estén definidos en el esquema.
        for name in parameters:
            if name not in schema:
                return False

        # Comprobar los tipos de los parámetros recibidos.
        for name, value in parameters.items():
            definition = schema.get(name, {})
            parameter_type = definition.get("type")

            if parameter_type == "string":
                if not isinstance(value, str):
                    return False
            elif parameter_type == "number":
                if isinstance(value, bool) or not isinstance(value, (int, float)):
                    return False
            elif parameter_type == "integer":
                if isinstance(value, bool) or not isinstance(value, int):
                    return False

        return True

    def _unknown(
        self,
        text: str,
        confidence: float = 0.0,
    ) -> Intent:
        return Intent(
            action="system.unknown",
            confidence=confidence,
            parameters={},
            original_text=text,
        )

from __future__ import annotations

from enum import Enum

from pydantic import BaseModel, Field


class StepStatus(str, Enum):
    """
    Estado de un paso del plan.
    """

    PENDING = "pending"
    READY = "ready"
    RUNNING = "running"
    COMPLETED = "completed"
    FAILED = "failed"
    SKIPPED = "skipped"


class PlanStep(BaseModel):
    """
    Representa un paso del plan.

    Un paso puede declarar:

    - depends_on: pasos que deben completarse antes
    - consumes: artifacts que necesita como entrada
    - produces: artifacts que genera como salida

    Estas relaciones forman el contrato de ejecución
    del paso.
    """

    id: str

    name: str

    description: str

    action: str

    status: StepStatus = StepStatus.PENDING

    metadata: dict = Field(
        default_factory=dict
    )

    depends_on: list[str] = Field(
        default_factory=list
    )

    consumes: list[str] = Field(
        default_factory=list
    )

    produces: list[str] = Field(
        default_factory=list
    )


class ExecutionPlan(BaseModel):
    """
    Plan generado para cumplir un Objective.

    Los pasos forman una estructura de ejecución
    basada en dependencias.
    """

    objective_id: str

    steps: list[PlanStep] = Field(
        default_factory=list
    )

    current_step: int = 0

    completed: bool = False

    metadata: dict = Field(
        default_factory=dict
    )

from __future__ import annotations

from typing import Any, Callable
from uuid import uuid4

from planner.intent import Intent
from planner.objective import (
    Objective,
    ObjectiveStatus,
)
from planner.plan import (
    ExecutionPlan,
    PlanStep,
)


class Planner:
    """
    Convierte un Intent en un ExecutionPlan.

    El Planner consulta el catálogo de herramientas
    para resolver las capacidades disponibles.

    No conoce las implementaciones concretas de
    las Actions.

    Las relaciones entre pasos se determinan a partir
    del contrato de artifacts:

        produces → salida de un paso
        consumes → entrada de otro paso

    El Planner no asume que un paso depende
    necesariamente del paso anterior.
    """

    def __init__(
        self,
        tool_resolver: Callable[[str], Any] | None = None,
    ) -> None:

        self._tool_resolver = tool_resolver

    def create_plan(
        self,
        objective: Objective,
        intent: Intent,
    ) -> ExecutionPlan:

        objective.status = (
            ObjectiveStatus.PLANNING
        )

        plan = ExecutionPlan(
            objective_id=objective.id,
        )

        #
        # Determinar las intenciones que forman
        # el plan.
        #

        if intent.is_compound:

            intents = intent.sub_intents

        else:

            intents = [intent]

        #
        # Resolver las herramientas.
        #

        resolved_tools: list[
            tuple[Intent, Any]
        ] = []

        for sub_intent in intents:

            tool = None

            if self._tool_resolver is not None:

                tool = self._tool_resolver(
                    sub_intent.action
                )

                if tool is None:

                    objective.status = (
                        ObjectiveStatus.FAILED
                    )

                    plan.metadata = {
                        "error": (
                            "Tool not available: "
                            f"{sub_intent.action}"
                        ),
                    }

                    return plan

            resolved_tools.append(
                (
                    sub_intent,
                    tool,
                )
            )

        #
        # Crear todos los pasos.
        #

        for sub_intent, tool in resolved_tools:

            produces: list[str] = []

            consumes: list[str] = []

            if tool is not None:

                produces = list(
                    getattr(
                        tool,
                        "produces",
                        [],
                    )
                )

                consumes = list(
                    getattr(
                        tool,
                        "consumes",
                        [],
                    )
                )

            step = PlanStep(
                id=str(uuid4()),
                name=sub_intent.action,
                description=sub_intent.original_text,
                action=sub_intent.action,
                metadata=sub_intent.parameters.copy(),
                depends_on=[],
                produces=produces,
                consumes=consumes,
            )

            plan.steps.append(
                step
            )

        #
        # Resolver las dependencias mediante
        # produces / consumes.
        #

        self._resolve_dependencies(
            plan
        )

        #
        # Validar la estructura del plan.
        #

        validation_error = (
            self._validate_plan_structure(
                plan
            )
        )

        if validation_error is not None:

            objective.status = (
                ObjectiveStatus.FAILED
            )

            plan.metadata = {
                "error": validation_error,
            }

            return plan

        #
        # Validar el grafo de dependencias.
        #

        graph_error = (
            self._validate_dependency_graph(
                plan
            )
        )

        if graph_error is not None:

            objective.status = (
                ObjectiveStatus.FAILED
            )

            plan.metadata = {
                "error": graph_error,
            }

            return plan

        #
        # Un plan sin pasos no puede ejecutarse.
        #

        if not plan.steps:

            objective.status = (
                ObjectiveStatus.FAILED
            )

            plan.metadata = {
                "error": (
                    "No executable steps generated."
                ),
            }

            return plan

        objective.status = (
            ObjectiveStatus.READY
        )

        return plan

    def _resolve_dependencies(
        self,
        plan: ExecutionPlan,
    ) -> None:
        """
        Construye las dependencias del plan
        mediante los artifacts.

        Un paso depende de otro cuando consume
        un artifact producido por ese otro paso.

        La dependencia se determina únicamente
        mediante produces / consumes y no depende
        de la posición de los pasos dentro del plan.
        """

        for step in plan.steps:

            if not step.consumes:

                continue

            for candidate in plan.steps:

                #
                # El propio paso no puede ser
                # su propia dependencia.
                #

                if candidate.id == step.id:

                    continue

                #
                # Comprobar relación artifact.
                #

                if not self._produces_required_artifact(
                    candidate,
                    step,
                ):

                    continue

                #
                # Agregar dependencia.
                #

                if candidate.id not in step.depends_on:

                    step.depends_on.append(
                        candidate.id
                    )

            #
            # Orden estable para facilitar
            # debugging y serialización.
            #

            step.depends_on.sort()

    @staticmethod
    def _validate_plan_structure(
        plan: ExecutionPlan,
    ) -> str | None:
        """
        Valida la estructura de artifacts del plan.

        Cada artifact consumido debe tener exactamente
        un productor dentro del mismo plan.

        No se permite:

        - un artifact requerido sin productor;
        - un artifact requerido con múltiples
          productores posibles.

        Los pasos que no consumen artifacts no requieren
        ninguna validación adicional en esta etapa.
        """

        for step in plan.steps:

            if not step.consumes:

                continue

            for artifact in step.consumes:

                producers = [
                    candidate
                    for candidate in plan.steps
                    if candidate.id != step.id
                    and artifact in candidate.produces
                ]

                if not producers:

                    return (
                        "Missing producer for artifact: "
                        f"{artifact}"
                    )

                if len(producers) > 1:

                    return (
                        "Ambiguous producer for artifact: "
                        f"{artifact}"
                    )

        return None

    @staticmethod
    def _validate_dependency_graph(
        plan: ExecutionPlan,
    ) -> str | None:
        """
        Valida que el grafo de dependencias
        sea acíclico.

        Cada dependencia apunta desde un paso
        consumidor hacia el paso que debe ejecutarse
        previamente.

        Un ciclo significa que no existe un orden
        válido de ejecución para completar el plan.
        """

        step_ids = {
            step.id
            for step in plan.steps
        }

        visiting: set[str] = set()
        visited: set[str] = set()

        def visit(step_id: str) -> bool:

            if step_id in visiting:

                return True

            if step_id in visited:

                return False

            visiting.add(
                step_id
            )

            step = next(
                (
                    candidate
                    for candidate in plan.steps
                    if candidate.id == step_id
                ),
                None,
            )

            if step is not None:

                for dependency_id in step.depends_on:

                    #
                    # Las dependencias ya fueron
                    # construidas a partir de pasos
                    # existentes. Este control evita
                    # que una referencia inválida
                    # sea tratada como un ciclo.
                    #

                    if dependency_id not in step_ids:

                        continue

                    if visit(
                        dependency_id
                    ):

                        return True

            visiting.remove(
                step_id
            )

            visited.add(
                step_id
            )

            return False

        for step in plan.steps:

            if visit(step.id):

                return (
                    "Cyclic dependency detected."
                )

        return None

    @staticmethod
    def _produces_required_artifact(
        producer: PlanStep,
        consumer: PlanStep,
    ) -> bool:
        """
        Determina si el productor genera al menos
        uno de los artifacts que necesita el consumidor.
        """

        return any(
            artifact in consumer.consumes
            for artifact in producer.produces
        )
