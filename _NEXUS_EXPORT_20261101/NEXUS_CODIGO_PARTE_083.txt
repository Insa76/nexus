def make_episode(
    *,
    action: str,
    parameters: dict | None = None,
    output: dict | None = None,
    success: bool = True,
) -> Episode:
    return Episode(
        action=action,
        parameters=parameters or {},
        original_text="test",
        success=success,
        output=output or {},
        timestamp=datetime.now(timezone.utc),
    )


def test_extract_open_app() -> None:
    memory = SemanticMemory()
    extractor = SemanticKnowledgeExtractor(memory)

    episode = make_episode(
        action="system.open_app",
        parameters={
            "application": "Chrome",
        },
    )

    facts = extractor.extract(episode)

    assert len(facts) == 1
    assert facts[0].subject == "chrome"
    assert facts[0].predicate == "es una aplicación"
    assert facts[0].value is True

    assert (
        memory.get_value(
            "chrome",
            "es una aplicación",
        )
        is True
    )

import pytest

from tasks import (
    InvalidTaskTransition,
    Task,
    TaskStatus,
    can_transition,
    transition_task,
)


def test_pending_can_start() -> None:
    assert can_transition(
        TaskStatus.PENDING,
        TaskStatus.IN_PROGRESS,
    )


def test_pending_can_be_cancelled() -> None:
    assert can_transition(
        TaskStatus.PENDING,
        TaskStatus.CANCELLED,
    )


def test_in_progress_can_complete() -> None:
    assert can_transition(
        TaskStatus.IN_PROGRESS,
        TaskStatus.COMPLETED,
    )


def test_in_progress_can_be_cancelled() -> None:
    assert can_transition(
        TaskStatus.IN_PROGRESS,
        TaskStatus.CANCELLED,
    )


def test_completed_has_no_outgoing_transitions() -> None:
    assert not can_transition(
        TaskStatus.COMPLETED,
        TaskStatus.IN_PROGRESS,
    )

    assert not can_transition(
        TaskStatus.COMPLETED,
        TaskStatus.CANCELLED,
    )


def test_cancelled_has_no_outgoing_transitions() -> None:
    assert not can_transition(
        TaskStatus.CANCELLED,
        TaskStatus.PENDING,
    )

    assert not can_transition(
        TaskStatus.CANCELLED,
        TaskStatus.IN_PROGRESS,
    )


def test_transition_task_returns_new_task() -> None:
    task = Task.create(title="Preparar reunión")

    started = transition_task(
        task,
        TaskStatus.IN_PROGRESS,
    )

    assert started.id == task.id
    assert started.title == task.title
    assert started.status == TaskStatus.IN_PROGRESS
    assert task.status == TaskStatus.PENDING


def test_transition_task_can_complete() -> None:
    task = Task.create(title="Preparar reunión")

    started = transition_task(
        task,
        TaskStatus.IN_PROGRESS,
    )

    completed = transition_task(
        started,
        TaskStatus.COMPLETED,
    )

    assert completed.status == TaskStatus.COMPLETED


def test_invalid_transition_raises() -> None:
    task = Task.create(title="Tarea")

    with pytest.raises(InvalidTaskTransition):
        transition_task(
            task,
            TaskStatus.COMPLETED,
        )


def test_completed_task_cannot_be_reopened() -> None:
    task = Task.create(title="Tarea")

    started = transition_task(
        task,
        TaskStatus.IN_PROGRESS,
    )

    completed = transition_task(
        started,
        TaskStatus.COMPLETED,
    )

    with pytest.raises(InvalidTaskTransition):
        transition_task(
            completed,
            TaskStatus.IN_PROGRESS,
        )

    adapter = TemporalSchedulerAdapter(scheduler)

    executed: list[str] = []

    schedule = TemporalSchedule.create(
        execute_at=(
            datetime.now(timezone.utc)
            - timedelta(seconds=0.01)
        ),
        target_id="reminder-123",
        target_type="reminder",

    async def callback() -> None:

def make_objective(
    action: str,
    metadata: dict,
) -> Objective:
    return Objective(
        id="objective-test",
        title=action,
        metadata=metadata,
    )


def make_plan(
    action: str,
    completed: bool = True,
) -> ExecutionPlan:
    return ExecutionPlan(
        objective_id="objective-test",
        steps=[
            PlanStep(
                id="step-test",
                name=action,
                description=action,
                action=action,
                status=(
                    StepStatus.COMPLETED
                    if completed
                    else StepStatus.PENDING
                ),
            )
        ],
        current_step=1 if completed else 0,
        completed=completed,
    )


def make_result(
    output: dict,

) -> ExecutionResult:

        success=success,
        output=output,

    )
