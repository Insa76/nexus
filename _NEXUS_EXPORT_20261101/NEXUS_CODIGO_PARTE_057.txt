def test_access_can_list_persisted_knowledge_history(tmp_path):
    services = build_document_services()

    source = tmp_path / "document.txt"
    source.write_text(
        "Contenido original.",
        encoding="utf-8",
    )

    artifact = DocumentArtifact.create(
        location=f"local:{source}",
    )

    services.access.add_artifact(artifact)

    document = services.access.create_document(
        name=source.name,
        mime_type="text/plain",
        source=f"local:{source}",
        artifact_id=artifact.id,
    )

    services.document_service.ingest(document)

    source.write_text(
        "Contenido actualizado.",
        encoding="utf-8",
    )

    services.document_service.ingest(document)

    extraction = services.document_service.extract_version_knowledge(
        document_id=document.id,
        from_version=1,
        to_version=2,
    )

    history = services.access.list_knowledge_history(
        document.id,
    )

    assert len(history) == len(extraction.knowledge)

    assert all(
        record.document_id == document.id
        for record in history
    )

    assert [
        record.knowledge
        for record in history
    ] == list(extraction.knowledge)

from documents import (
    DocumentArtifact,
    DocumentType,
    FormatIngestionService,
    FormatReaderRegistry,

class FakeFormatReader:
    def __init__(self, text: str = "document content") -> None:
        self.text = text

    def supports(self, document_type):
        return True

    def read(self, artifact, document_type):
        return self.text


def test_format_ingestion_reads_registered_format():
    registry = FormatReaderRegistry()

    reader = FakeFormatReader(
        text="hello nexus"
    )

    registry.register(
        "text/plain",
        reader,
    )

    service = FormatIngestionService(
        reader_registry=registry,
    )

    artifact = DocumentArtifact.create(
        location="local:C:/documents/test.txt",
    )

    result = service.ingest(
        artifact=artifact,
        document_type="text/plain",
    )

    assert result.artifact is artifact
    assert result.content.text == "hello nexus"
    assert result.content.content_type == "text/plain"
