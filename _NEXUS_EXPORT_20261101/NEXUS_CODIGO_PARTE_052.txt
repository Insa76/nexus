    def ingest_with_validation(
        self,
        document: Document,
    ) -> DocumentIngestionResult:
        if document.artifact_id is None:
            raise ValueError(
                f"Document has no artifact: {document.id}"
            )

        artifact = self.artifact_store.get(
            document.artifact_id
        )

        if artifact is None:
            raise ValueError(
                f"Artifact not found: {document.artifact_id}"
            )

        type_validation: DocumentTypeValidation | None = None
        detection_evidence = None
        ingestion_evidence = None

        if self.format_ingestion_service is None:
            result = self.ingestion_service.ingest(
                artifact=artifact,
                content_type=document.mime_type,
            )
        else:
            format_result = (
                self.format_ingestion_service.ingest_with_validation(
                    artifact=artifact,
                    declared_type=document.mime_type,
                )
            )

            result = format_result.ingestion

            type_validation = format_result.type_validation
            detection_evidence = format_result.detection_evidence

            if (
                type_validation is not None
                and detection_evidence is not None
            ):
                ingestion_evidence = DocumentIngestionEvidence(
                    declared_type=type_validation.declared_type,
                    detected_type=detection_evidence.detected_type,
                    detection_strategy=detection_evidence.strategy,
                )

        self.content_store.add(result.content)

        if self.content_version_store is not None:
            version_number = (
                self.content_version_store.get_next_version(
                    document.id
                )
            )

            content_version = DocumentContentVersion.create(
                document_id=document.id,
                content_id=result.content.id,
                version=version_number,
                content_type=result.content.content_type,
            )

            self.content_version_store.add(
                content_version
            )

        updated_document = Document(
            id=document.id,
            name=document.name,
            mime_type=document.mime_type,
            source=document.source,
            description=document.description,
            user_id=document.user_id,
            organization_id=document.organization_id,
            project_id=document.project_id,
            artifact_id=document.artifact_id,
            content_id=result.content.id,
            metadata=dict(document.metadata),
            ingestion_evidence=(
                ingestion_evidence
                if ingestion_evidence is not None
                else document.ingestion_evidence
            ),
            created_at=document.created_at,
            updated_at=_utc_now(),
        )

        self.document_store.update(
            updated_document
        )

        if self.ingestion_record_store is not None:
            record = DocumentIngestionRecord.create(
                document_id=updated_document.id,
                artifact_id=artifact.id,
                content_id=result.content.id,
                declared_type=document.mime_type,
                detected_type=(
                    detection_evidence.detected_type
                    if detection_evidence is not None
                    else None
                ),
                detection_strategy=(
                    detection_evidence.strategy
                    if detection_evidence is not None
                    else None
                ),
                type_matches=(
                    type_validation.matches
                    if type_validation is not None
                    else None
                ),
            )

            self.ingestion_record_store.add(record)

        resolution = self.resolve(
            updated_document.id
        )

        return DocumentIngestionResult(
            resolution=resolution,
            type_validation=type_validation,
            detection_evidence=detection_evidence,
        )

    def validate_latest_knowledge(
        self,
        document_id: str,
    ) -> bool:
        if (
            self.knowledge_record_store is None
            or self.content_version_store is None
        ):
            return False

        latest_version = (
            self.content_version_store.get_latest_for_document(
            document_id
            )
        )

        if latest_version is None:
            return False

        records = (
            self.knowledge_record_store.list_for_document(
                document_id
            )
        )

        return all(
            record.content_version_id
            == latest_version.id
            for record in records
        )

    def has_orphaned_knowledge(
        self,
            document_id: str,
    ) -> bool:
        if (
            self.knowledge_record_store is None
            or self.content_version_store is None
        ):
            return False

        records = self.knowledge_record_store.list_for_document(
            document_id
        )

        for record in records:
            try:
                self.content_version_store.get(
                    record.content_version_id
                )
            except AttributeError:
                return False

            if (
                self.content_version_store.get(
                    record.content_version_id
                )
                is None
            ):
                return True

        return False

    def list_orphaned_knowledge(
        self,
        document_id: str,
    ) -> list[DocumentKnowledgeRecord]:
        if (
            self.knowledge_record_store is None
            or self.content_version_store is None
        ):
            return []

        records = self.knowledge_record_store.list_for_document(
            document_id
        )

        orphaned: list[DocumentKnowledgeRecord] = []

        for record in records:
            content_version = self.content_version_store.get(
                record.content_version_id
            )

            if content_version is None:
                orphaned.append(record)

        return orphaned

        content_version_id = self._resolve_content_version_id(
            document_id=document_id,

        return self.knowledge_record_store.remove_for_content_version(
            content_version_id
        )

    def remove_orphaned_knowledge(
        self,
        document_id: str,
    ) -> int:
        if (
            self.knowledge_record_store is None
            or self.content_version_store is None
        ):
            return 0

        orphaned = self.list_orphaned_knowledge(
            document_id
        )

        removed = 0

        for record in orphaned:
            self.knowledge_record_store.remove(
                record.id
            )
            removed += 1

        return removed

    def _resolve_content_version_id(
        self,
        *,
        document_id: str,
        version: int,
    ) -> str:
        if self.content_version_store is None:
            raise ValueError(
                "Document content version store is not configured"
            )

        resolver = DocumentVersionResolver(
            store=self.content_version_store,
        )

        content_version = resolver.resolve_version(
            document_id=document_id,
            version=version,
        )

        return content_version.id

    def compare_versions(
        self,
        *,
        document_id: str,
        from_version: int,
        to_version: int,
    ) -> DocumentVersionChangeResult:
        if self.content_version_store is None:
            raise ValueError(
                "Document content version store is not configured"
            )

        resolver = DocumentVersionResolver(
            store=self.content_version_store,
        )

        resolution = resolver.resolve_pair(
            document_id=document_id,
            from_version=from_version,
            to_version=to_version,
        )

        from_content = self.content_store.get(
            resolution.from_version.content_id
        )

        if from_content is None:
            raise ValueError(
                "Document content not found: "
                f"{resolution.from_version.content_id}"
            )

        to_content = self.content_store.get(
            resolution.to_version.content_id
        )

        if to_content is None:
            raise ValueError(
                "Document content not found: "
                f"{resolution.to_version.content_id}"
            )

        comparator = DocumentVersionComparator()
        comparison = comparator.compare(resolution)

        detector = DocumentChangeDetector()
        detection = detector.detect(
            comparison=comparison,
            before_text=from_content.text,
            after_text=to_content.text,
        )

        return DocumentVersionChangeResult(
            comparison=comparison,
            changes=detection.changes,
        )

    def compare_latest_versions(
        self,
        *,
        document_id: str,
    ) -> DocumentVersionChangeResult:
        if self.content_version_store is None:
            raise ValueError(
                "Document content version store is not configured"
            )

        resolver = DocumentVersionResolver(
            store=self.content_version_store,
        )

        resolution = resolver.resolve_latest_pair(
            document_id=document_id,
        )

        from_content = self.content_store.get(
            resolution.from_version.content_id
        )

        if from_content is None:
            raise ValueError(
                "Document content not found: "
                f"{resolution.from_version.content_id}"
            )

        to_content = self.content_store.get(
            resolution.to_version.content_id
        )

        if to_content is None:
            raise ValueError(
                "Document content not found: "
                f"{resolution.to_version.content_id}"
            )

        comparator = DocumentVersionComparator()
        comparison = comparator.compare(resolution)

        detector = DocumentChangeDetector()
        detection = detector.detect(
            comparison=comparison,
            before_text=from_content.text,
            after_text=to_content.text,
        )

        return DocumentVersionChangeResult(
            comparison=comparison,
            changes=detection.changes,
        )

    def analyze_version_changes(
        self,

        document_id: str,
        from_version: int,
        to_version: int,

        if self.semantic_analysis_service is None:

                "Document semantic analysis service is not configured"

        result = self.compare_versions(
            document_id=document_id,
            from_version=from_version,

        if self.semantic_analysis_service is None:

    def _store_knowledge_records(
        self,
        *,
        document_id: str,
        content_version_id: str,
        extraction: DocumentKnowledgeExtraction,
    ) -> None:
        if self.knowledge_record_store is None:
            return

        for knowledge in extraction.knowledge:
            existing = self.knowledge_record_store.find(
                document_id=document_id,
                content_version_id=content_version_id,
                knowledge=knowledge,
            )

            if existing is not None:
                continue

            record = DocumentKnowledgeRecord.create(
                document_id=document_id,
                content_version_id=content_version_id,
                knowledge=knowledge,
            )

            self.knowledge_record_store.add(record)

    def extract_version_knowledge(
        self,
        *,
        document_id: str,
        from_version: int,
        to_version: int,
    ) -> DocumentKnowledgeExtraction:
        if self.semantic_analysis_service is None:
            raise ValueError(
                "Document semantic analysis service is not configured"
            )

        if self.knowledge_extraction_service is None:
            raise ValueError(
                "Document knowledge extraction service is not configured"
            )

        analysis = self.analyze_version_changes(
            document_id=document_id,
            from_version=from_version,
            to_version=to_version,
        )

        extraction = self.knowledge_extraction_service.extract(
            analysis
        )

        content_version_id = self._resolve_content_version_id(
            document_id=document_id,
            version=to_version,
        )

        self._store_knowledge_records(
            document_id=document_id,
            content_version_id=content_version_id,
            extraction=extraction,
        )

        return extraction

    def refresh_version_knowledge(
        self,
        *,
        document_id: str,
        from_version: int,
        to_version: int,
    ) -> DocumentKnowledgeExtraction:
        if self.semantic_analysis_service is None:
            raise ValueError(

            )

        if self.knowledge_extraction_service is None:
            raise ValueError(
                "Document knowledge extraction service is not configured"
            )

        analysis = self.analyze_version_changes(
            document_id=document_id,
            from_version=from_version,
            to_version=to_version,
        )

        extraction = self.knowledge_extraction_service.extract(
            analysis
        )

        content_version_id = self._resolve_content_version_id(
            document_id=document_id,
            version=to_version,
        )

        if self.knowledge_record_store is not None:
            self.knowledge_record_store.remove_for_content_version(
                content_version_id
            )

        self._store_knowledge_records(
            document_id=document_id,
            content_version_id=content_version_id,
            extraction=extraction,
        )

        return extraction

    def extract_latest_version_knowledge(
        self,
        *,
        document_id: str,
    ) -> DocumentKnowledgeExtraction:
        if self.semantic_analysis_service is None:
            raise ValueError(
                "Document semantic analysis service is not configured"

        if self.knowledge_extraction_service is None:
            raise ValueError(

        analysis = self.analyze_latest_version_changes(

        )

        extraction = self.knowledge_extraction_service.extract(

        )

        content_version_id = self._resolve_content_version_id(

            version=analysis.source.comparison.to_version,
        )

        self._store_knowledge_records(
            document_id=document_id,
            content_version_id=content_version_id,
            extraction=extraction,
        )

        return extraction

    def list_documents_for_user(
        self,

    knowledge_record_store = DocumentKnowledgeRecordStore()

    semantic_analysis_service = DocumentSemanticAnalysisService(
        analyzer=RuleBasedSemanticAnalyzer(),
    )

    knowledge_extraction_service = DocumentKnowledgeExtractionService(
    extractor=RuleBasedDocumentKnowledgeExtractor(),
    )

    text_ingestor = TextIngestor(
        reader=artifact_reader,
    )

    ingestion_registry.register(
        "text/plain",
        text_ingestor,
    )

    ingestion_service = DocumentIngestionService(
        registry=ingestion_registry,
        type_resolver=type_resolver,
    )

    format_reader_registry = FormatReaderRegistry()

    text_format_reader = TextFormatReader(
        artifact_reader=artifact_reader,
    )

    format_reader_registry.register(
        "text/plain",
        text_format_reader,
    )

    format_ingestion_service = FormatIngestionService(
        reader_registry=format_reader_registry,
        type_resolver=artifact_type_resolver,

from __future__ import annotations

from dataclasses import dataclass


@dataclass(frozen=True)
class DocumentSource:
    scheme: str
    location: str

    @classmethod
    def from_location(cls, location: str) -> "DocumentSource":
        value = location.strip()

        if not value:
            raise ValueError(
                "Document source location cannot be empty"
            )

        # Windows drive path:
        # C:/archivo.txt
        # C:\archivo.txt
        if (
            len(value) >= 2
            and value[0].isalpha()
            and value[1] == ":"
        ):
            return cls(
                scheme="local",
                location=value,
            )

        # Plain local path.
        if ":" not in value:
            return cls(
                scheme="local",
                location=value,
            )

        scheme, source_location = value.split(
            ":",
            1,
        )

        scheme = scheme.strip().lower()
        source_location = source_location.strip()

        if not scheme:
            raise ValueError(
                "Document source scheme cannot be empty"
            )

        if not source_location:
            raise ValueError(
                "Document source location cannot be empty"
            )

        return cls(
            scheme=scheme,
            location=source_location,
        )

from __future__ import annotations

from documents.models import Document


class DocumentStore:
    """
    Almacenamiento en memoria de documentos.
    """

    def __init__(self) -> None:
        self._documents: dict[str, Document] = {}

    def add(self, document: Document) -> Document:
        self._documents[document.id] = document
        return document

    def get(self, document_id: str) -> Document | None:
        return self._documents.get(document_id)

    def list(self) -> list[Document]:
        return list(self._documents.values())

    def list_for_user(
        self,
        user_id: str,
    ) -> list[Document]:
        return [
            document
            for document in self._documents.values()
            if document.user_id == user_id
        ]

    def list_for_organization(
        self,
        organization_id: str,
    ) -> list[Document]:
        return [
            document
            for document in self._documents.values()
            if document.organization_id == organization_id
        ]

    def list_for_project(
        self,
        project_id: str,
    ) -> list[Document]:
        return [
            document
            for document in self._documents.values()
            if document.project_id == project_id
        ]

    def list_for_context(
        self,
        *,
        user_id: str | None = None,
        organization_id: str | None = None,
        project_id: str | None = None,
    ) -> list[Document]:
        if (
            user_id is None
            and organization_id is None
            and project_id is None
        ):
            raise ValueError(
                "At least one document context filter is required"
            )

        return [
            document
            for document in self._documents.values()
            if (
                user_id is None
                or document.user_id == user_id
            )
            and (
                organization_id is None
                or document.organization_id == organization_id
            )
            and (
                project_id is None
                or document.project_id == project_id
            )
        ]

    def update(self, document: Document) -> Document:
        if document.id not in self._documents:
            raise KeyError(
                f"Document '{document.id}' not found"
            )

        self._documents[document.id] = document
        return document

    def remove(self, document_id: str) -> Document | None:
        return self._documents.pop(
            document_id,
            None,
        )

    def clear(self) -> None:
        self._documents.clear()

    def count(self) -> int:
        return len(self._documents)

from documents.artifacts import DocumentArtifact
from documents.document_type import DocumentType

class TextFormatReader:
    """
    Reader especializado para documentos de texto plano.
    """

    def __init__(
        self,
        artifact_reader: ArtifactReader,
    ) -> None:
        self._artifact_reader = artifact_reader

    def supports(
        self,
        document_type: DocumentType,
    ) -> bool:
        return document_type.mime_type == "text/plain"

    def read(
        self,
        artifact: DocumentArtifact,
        document_type: DocumentType,
    ) -> str:
        if not self.supports(document_type):
            raise ValueError(
                "TextFormatReader does not support "
                f"{document_type.mime_type}"
            )

        return self._artifact_reader.read(
            artifact
        )

from __future__ import annotations

from dataclasses import dataclass

from documents.version_resolver import DocumentVersionResolution


@dataclass(frozen=True)
class DocumentVersionComparison:
    document_id: str
    from_version: int
    to_version: int
    from_content_id: str
    to_content_id: str
    changed: bool

    @property
    def unchanged(self) -> bool:
        return not self.changed


class DocumentVersionComparator:
    def compare(
        self,
        resolution: DocumentVersionResolution,
    ) -> DocumentVersionComparison:
        from_version = resolution.from_version
        to_version = resolution.to_version

        changed = from_version.content_id != to_version.content_id

        return DocumentVersionComparison(
            document_id=resolution.document_id,
            from_version=from_version.version,
            to_version=to_version.version,
            from_content_id=from_version.content_id,
            to_content_id=to_version.content_id,
            changed=changed,
        )
