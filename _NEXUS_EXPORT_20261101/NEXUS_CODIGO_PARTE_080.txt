class RegressionFailure(Exception):
    pass


def check(condition: bool, message: str) -> None:
    if not condition:
        raise RegressionFailure(message)


def output_of(result):
    output = result.output
    if output is None:
        return {}
    return output


async def execute(engine, text: str):
    return await engine.execute(text)

async def test_open_app(engine):
    print("[1/13] Acción simple: abrir aplicación")

    result = await execute(engine, "abrí notepad")

        result.success,

        output.get("application") == "notepad",

async def test_screen_capture(engine):
    print("[6/13] Acción simple: captura de pantalla")

    result = await execute(engine, "capturá la pantalla")

    check(
        result.success,
        f"Falló captura de pantalla: {result.message}",
    )

    artifacts = result.artifacts or {}

    check(
        "image" in artifacts,
        "La captura no produjo el artifact 'image'.",
    )

    image = artifacts["image"]

    check(
        image is not None,
        "El artifact 'image' es None.",
    )

    check(
        hasattr(image, "size"),
        "El artifact 'image' no tiene dimensiones.",
    )

    width, height = image.size

    check(
        width > 0,
        "El ancho de la captura es inválido.",
    )

    check(
        height > 0,
        "El alto de la captura es inválido.",
    )

    print(f"      OK ({width} × {height})")


async def test_compound_capture_inspection(engine):
    print("[7/13] Acción compuesta: captura → inspección")

    result = await execute(
        engine,
        "capturá la pantalla e inspeccionala",
    )

    check(
        result.success,
        f"Falló acción compuesta: {result.message}",
    )

    output = output_of(result)

    check(
        output.get("status") == "inspected",
        "La acción compuesta no terminó con status='inspected'.",
    )

    inspection = output.get("capture_inspection")

    check(
        isinstance(inspection, dict),
        "No existe 'capture_inspection' como diccionario.",
    )

    width = inspection.get("width")
    height = inspection.get("height")

    check(
        isinstance(width, int) and width > 0,
        "Ancho de inspección inválido.",
    )

    check(
        isinstance(height, int) and height > 0,
        "Alto de inspección inválido.",
    )

    check(
        "image" in (result.artifacts or {}),
        "La acción compuesta perdió el artifact 'image'.",
    )

    metadata = result.metadata or {}

    response = metadata.get("response")

    check(
        isinstance(response, str),
        "La respuesta humana no está presente en metadata.",
    )

    check(
        "Inspeccioné la pantalla" in response,
        f"ResponseFormatter no produjo la respuesta esperada: {response}",
    )

    print(f"      OK ({width} × {height})")

async def test_contextual_volume_reference(engine):
    print("[12/13] Memoria contextual: referencia al volumen")

    """
    Verifica que NEXUS pueda interpretar una referencia
    contextual al volumen a partir de una interacción anterior.
    """

    first = await execute(
        engine,
        "poné el volumen al 30%",
    )

    check(
        first.success,
        "La primera modificación de volumen debería ser exitosa.",
    )

    second = await execute(
        engine,
        "ahora ponelo al 50%",
    )

    check(
        second.success,
        "La referencia contextual al volumen debería resolverse.",
    )

    intent_entries = engine.memory.find("intent")

    check(
        bool(intent_entries),
        "La intención contextual no quedó registrada en memoria.",
    )

    latest_intent = intent_entries[-1].data

    check(
        latest_intent.get("action") == "system.set_volume",
        "La segunda intención debería ser system.set_volume.",
    )

    parameters = latest_intent.get(
        "parameters",
        {},
    )

    check(
        parameters.get("level") == 50,
        "La referencia contextual debería conservar el nivel 50.",
    )

    print("      OK")


async def test_contextual_close_app(engine):
    print("[13/13] Memoria contextual: referencia a aplicación")

    """
    Verifica el flujo completo:

        abrir aplicación
            ↓
        memoria contextual
            ↓
        "cerralo"
            ↓
        system.close_app

    La aplicación se abre dentro de esta misma prueba para que
    la referencia contextual dependa de una interacción real
    inmediatamente anterior.
    """

    opened = await execute(
        engine,
        "abrí notepad",
    )

    check(
        opened.success,
        f"Falló la apertura previa de notepad: {opened.message}",
    )

    opened_output = output_of(opened)

    check(
        opened_output.get("status") == "opened",
        f"Notepad no quedó marcado como abierto: {opened_output}",
    )

    closed = await execute(
        engine,
        "cerralo",
    )

    check(
        closed.success,
        f"Falló el cierre contextual de notepad: {closed.message}",
    )

    closed_output = output_of(closed)

    check(
        isinstance(closed_output, dict),
        "El output del cierre contextual no es un diccionario.",
    )

    intent_entries = engine.memory.find("intent")

    check(
        bool(intent_entries),
        "El cierre contextual no dejó una intención registrada.",
    )

    latest_intent = intent_entries[-1].data

    check(
        latest_intent.get("action") == "system.close_app",
        (
            "La referencia 'cerralo' no se resolvió como "
            "system.close_app."
        ),
    )

    parameters = latest_intent.get(
        "parameters",
        {},
    )

    check(
        parameters.get("application") == "notepad",
        (
            "La referencia contextual no recuperó "
            f"notepad: {parameters}"
        ),
    )

    print("      OK")

    ]

    passed = 0
    failed = 0

    for test in tests:
        try:
            await test(engine)
            passed += 1

        except Exception as exc:
            failed += 1

            print("      FAILED")
            print(f"      {type(exc).__name__}: {exc}")
            print()

            traceback.print_exc()

    print()
    print("=" * 72)
    print("REGRESSION RESULT")
    print("=" * 72)
    print(f"PASSED : {passed}")
    print(f"FAILED : {failed}")
    print("=" * 72)
    print()

    if failed:
        raise SystemExit(1)

    print("NEXUS CORE REGRESSION: PASS")
    print()

from planner.decision import DecisionEngine
from planner.intent import Intent


def test_known_action_without_capability_list():
    engine = DecisionEngine()

    intent = Intent(
        action="system.open_app",
        confidence=0.95,
        parameters={"application": "Chrome"},
        original_text="Abrí Chrome",

    assert decision.selected_action == "system.open_app"
    assert decision.requires_execution is True
    assert decision.is_executable is True

def test_compound_intent_selects_first_available_action():
    engine = DecisionEngine()

    first = Intent(
        action="system.open_app",
        confidence=0.9,
        parameters={"application": "Chrome"},
        original_text="Abrí Chrome",
    )

    second = Intent(
        action="system.set_volume",
        confidence=0.9,
        parameters={"level": 50},
        original_text="poné el volumen al 50",
    )

    compound = Intent(
        action="compound",
        confidence=0.9,
        original_text="Abrí Chrome y poné el volumen al 50",
        sub_intents=[first, second],
    )

    capabilities = [
        {
            "name": "windows.set_volume",
            "action": "system.set_volume",
        }
    ]

    decision = engine.decide(
        compound,
        capabilities=capabilities,
    )

    assert decision.selected_action == "system.set_volume"
    assert decision.requires_execution is True
    assert decision.alternatives == []
    assert decision.metadata["compound"] is True

    document = Document.create(
        name="Documento.txt",
        mime_type="text/plain",
        source="memory:test",
    )

    try:
        document.name = "Otro.txt"
    except AttributeError:
        pass
    else:

import pytest

from documents import Document, DocumentStore


def create_document(
    name: str = "Documento.pdf",
) -> Document:
    return Document.create(
        name=name,
        mime_type="application/pdf",
        source=f"local:/documents/{name}",
    )


def test_add_and_get_document() -> None:
    store = DocumentStore()
    document = create_document()

    result = store.add(document)

    assert result == document
    assert store.get(document.id) == document


def test_get_unknown_document_returns_none() -> None:
    store = DocumentStore()

    assert store.get("unknown") is None


def test_list_documents() -> None:
    store = DocumentStore()

    first = create_document("uno.pdf")
    second = create_document("dos.pdf")

    store.add(first)
    store.add(second)

    assert store.list() == [first, second]


def test_update_document() -> None:
    store = DocumentStore()

    document = create_document()

    store.add(document)

    updated = Document(
        **{
            **document.__dict__,
            "name": "Documento actualizado.pdf",
        }
    )

    result = store.update(updated)

    assert result == updated
    assert store.get(document.id) == updated


def test_update_unknown_document_raises() -> None:
    store = DocumentStore()
    document = create_document()

    with pytest.raises(KeyError):
        store.update(document)


def test_remove_document() -> None:
    store = DocumentStore()
    document = create_document()

    store.add(document)

    result = store.remove(document.id)

    assert result == document
    assert store.get(document.id) is None


def test_remove_unknown_document_returns_none() -> None:
    store = DocumentStore()

    assert store.remove("unknown") is None


def test_clear_documents() -> None:
    store = DocumentStore()

    store.add(create_document("uno.pdf"))
    store.add(create_document("dos.pdf"))

    store.clear()

    assert store.list() == []
    assert store.count() == 0


def test_count_documents() -> None:
    store = DocumentStore()

    assert store.count() == 0

    store.add(create_document("uno.pdf"))
    store.add(create_document("dos.pdf"))

    assert store.count() == 2

from context import (
    AssistantContext,
    OrganizationProfile,
    ProjectContext,
    UserProfile,
)

from memory import WorkingMemory

from planner.context_builder import ContextBuilder


def test_engine_context_builder_receives_assistant_context():
    organization = OrganizationProfile.create(
        name="NEXUS Labs",
        industry="Software",
        description="Organización de desarrollo.",
        areas=["Technology"],
        rules=["Security first"],
        priorities=["Reliability"],
    )

    user = UserProfile.create(
        name="Carlos",
        role="Founder",
        organization_id=organization.id,
        preferences={
            "language": "es",
        },
        responsibilities=[
            "Product",
            "Architecture",
        ],
        priorities=[
            "NEXUS",
        ],
    )

    project = ProjectContext.create(
        name="NEXUS",
        organization_id=organization.id,
        description="Cognitive runtime.",
        status="active",
        responsible_user_ids=[
            user.id,
        ],
    )

    assistant_context = AssistantContext(
        user=user,
        organization=organization,
        projects=[project],
    )

    builder = ContextBuilder(
        assistant_context=assistant_context
    )

    memory = WorkingMemory()

    memory.add(
        "user_request",
        {
            "text": "Abrí Chrome",
        },
    )

    context = builder.build(
        memory,
        semantic_query="Abrí Chrome",
    )

    assistant = context["assistant_context"]

    assert assistant is not None

    assert assistant["user"]["id"] == user.id
    assert assistant["user"]["name"] == "Carlos"

    assert (
        assistant["organization"]["id"]
        == organization.id
    )

    assert (
        assistant["organization"]["name"]
        == "NEXUS Labs"
    )

    assert (
        assistant["projects"][0]["id"]
        == project.id
    )

    assert (
        assistant["projects"][0]["name"]
        == "NEXUS"
    )

from __future__ import annotations

import asyncio

from planner.execution_context import ExecutionContext
from planner.intent import Intent
from planner.objective import Objective
from planner.plan import ExecutionPlan, PlanStep, StepStatus
from runtime.execution_result import ExecutionResult


def check(condition: bool, message: str) -> None:
    if not condition:
        raise AssertionError(message)


class FakeAction:
    def __init__(
        self,
        action: str,
        output: dict,
        artifact: tuple[str, object] | None = None,
    ):
        self.action = action
        self.output = output
        self.artifact = artifact
        self.internal = False

    async def execute(
        self,
        context: ExecutionContext,
        step: PlanStep,
    ) -> None:

        for key, value in self.output.items():
            context.add_result(
                key,
                value,
            )

        if self.artifact is not None:
            key, value = self.artifact

            context.add_artifact(
                key,
                value,
            )


class FakeActionManager:
    def __init__(self, actions: dict[str, FakeAction]):
        self.actions = actions

    async def execute(
        self,
        context: ExecutionContext,
    ) -> None:

        step = context.current_plan_step()

        if step is None:
            return

        action = self.actions.get(step.action)

        if action is None:
            context.add_error(
                f"Unknown action: {step.action}"
            )
            return

        step.status = StepStatus.RUNNING

        results_before = dict(
            context.results
        )

        artifacts_before = dict(
            context.artifacts
        )

        try:
            await action.execute(
                context,
                step,
            )

            step.status = (
                StepStatus.COMPLETED
            )

            context.record_step_evidence(
                step,
                status="completed",
                results_before=(
                    results_before
                ),
                artifacts_before=(
                    artifacts_before
                ),
            )

        except Exception as exc:
            step.status = StepStatus.FAILED

            context.add_error(
                str(exc)
            )

            context.record_step_evidence(
                step,
                status="failed",
                results_before=(
                    results_before
                ),
                artifacts_before=(
                    artifacts_before
                ),
            )


class FakeRuntime:
    def __init__(
        self,
        actions: dict[str, FakeAction],
    ):
        self.action_manager = (
            FakeActionManager(actions)
        )


def make_step(
    step_id: str,
    action: str,
    depends_on: list[str] | None = None,
) -> PlanStep:

    return PlanStep(
        id=step_id,
        name=step_id,
        description=step_id,
        action=action,
        status=StepStatus.PENDING,
        depends_on=(
            depends_on
            if depends_on is not None
            else []
        ),
        consumes=[],
        produces=[],
        metadata={},
    )


def make_compound_context() -> ExecutionContext:

    objective = Objective(
        id="objective-compound",
        title="compound",
        metadata={
            "application": "Chrome",
            "level": 60,
        },
    )

    plan = ExecutionPlan(
        objective_id=objective.id,
        steps=[
            make_step(
                "step-1",
                "system.open_app",
            ),
            make_step(
                "step-2",
                "system.set_volume",
                depends_on=["step-1"],
            ),
        ],
        current_step=0,
        completed=False,
        metadata={},
    )

    return ExecutionContext(
        objective=objective,
        plan=plan,
        variables={},
        artifacts={},
        results={},
        errors=[],
        current_step=0,
        finished=False,
        step_evidence={},
    )


async def execute_compound_plan(
    context: ExecutionContext,
) -> ExecutionResult:

    runtime = FakeRuntime(
        {
            "system.open_app": FakeAction(
                "system.open_app",
                {
                    "application": "Chrome",
                    "status": "opened",
                },
            ),
            "system.set_volume": FakeAction(
                "system.set_volume",
                {
                    "volume": 60,
                    "status": "set",
                },
            ),
        }
    )

    while not context.finished:

        step = context.current_plan_step()

        if step is None:
            break

        await runtime.action_manager.execute(
            context
        )

        if context.has_errors():
            context.finished = True
            break

        completed = all(
            current.status
            == StepStatus.COMPLETED
            for current
            in context.plan.steps
        )

        if completed:
            context.plan.completed = True
            context.finished = True
            break

        context.current_step += 1

    return ExecutionResult(
        success=(
            context.plan.completed
            and not context.has_errors()
        ),
        message=(
            "Execution completed successfully."
            if context.plan.completed
            else "Execution failed."
        ),
        output=dict(
            context.results
        ),
        artifacts=dict(
            context.artifacts
        ),
        metadata={
            "step_evidence": (
                context.all_step_evidence()
            ),
        },
    )


def test_engine_compound_execution_produces_step_evidence():
    context = (
        make_compound_context()
    )

    result = asyncio.run(
        execute_compound_plan(
            context
        )
    )

    check(
        result.success,
        "Compound execution must succeed.",
    )

    step_evidence = (
        result.metadata.get(
            "step_evidence"
        )
    )

    check(
        isinstance(
            step_evidence,
            dict,
        ),
        "ExecutionResult must contain step_evidence.",
    )

    check(
        set(step_evidence.keys())
        == {
            "step-1",
            "step-2",
        },
        "Every compound step must have individual evidence.",
    )


def test_engine_compound_execution_preserves_step_outputs():
    context = (
        make_compound_context()
    )

    result = asyncio.run(
        execute_compound_plan(

    )

    step_evidence = (

    check(
        step_1["results"][
            "status"
        ]
        == "opened",
        "Step 1 evidence must contain its own execution status.",
    )

    check(
        step_2["results"][
            "volume"
        ]
        == 60,
        "Step 2 evidence must contain its own volume result.",
    )

    check(
        step_2["results"][
            "status"
        ]
        == "set",
        "Step 2 evidence must contain its own execution status.",
    )


def test_engine_compound_execution_does_not_collapse_evidence_into_global_output():

from planner.decision import DecisionEngine
from planner.intent import Intent


def test_engine_decision_layer_is_available():
    engine = DecisionEngine()

    intent = Intent(
        action="system.open_app",
        confidence=0.95,
        parameters={
            "application": "Chrome",
        },
        original_text="Abrí Chrome",
    )

    decision = engine.decide(intent)

    assert decision.is_executable is True
    assert decision.selected_action == "system.open_app"


def test_unknown_intent_is_blocked():
    engine = DecisionEngine()

    intent = Intent(
        action="system.unknown",
        original_text="Hacé algo que no conozco",
    )

    decision = engine.decide(intent)

    assert decision.is_executable is False
    assert decision.selected_action is None


def test_decision_preserves_traceability():
    engine = DecisionEngine()

    intent = Intent(
        action="system.set_volume",
        confidence=0.91,
        parameters={
            "level": 60,
        },
        original_text="Poné el volumen al 60",
    )

    decision = engine.decide(
        intent,
        context={
            "last_request": "subí el volumen",
        },
    )

    assert decision.decision_id
    assert decision.reason
    assert decision.confidence == 0.91
    assert decision.metadata["status"] == "selected"


def test_decision_never_executes_an_action():
    engine = DecisionEngine()

    execution_happened = False

    intent = Intent(
        action="system.open_app",
        confidence=1.0,
        parameters={
            "application": "Chrome",
        },
        original_text="Abrí Chrome",
    )

    decision = engine.decide(intent)

    assert decision.is_executable is True
    assert execution_happened is False


def test_decision_respects_capabilities():
    engine = DecisionEngine()

    intent = Intent(
        action="system.open_app",
        confidence=0.9,
        parameters={
            "application": "Chrome",
        },
        original_text="Abrí Chrome",
    )

    capabilities = [
        {
            "name": "windows.set_volume",
            "action": "system.set_volume",
        }
    ]

    decision = engine.decide(
        intent,
        capabilities=capabilities,
    )

    assert decision.is_executable is False
    assert decision.selected_action is None
    assert decision.metadata["status"] == "unavailable"


def test_compound_decision_preserves_available_actions():
    engine = DecisionEngine()

    first = Intent(
        action="system.open_app",
        confidence=0.9,
        parameters={
            "application": "Chrome",
        },
        original_text="Abrí Chrome",
    )

    second = Intent(
        action="system.set_volume",
        confidence=0.9,
        parameters={
            "level": 50,
        },
        original_text="poné el volumen al 50",
    )

    compound = Intent(
        action="compound",
        confidence=0.9,
        original_text=(
            "Abrí Chrome y poné el volumen al 50"
        ),
        sub_intents=[
            first,
            second,
        ],
    )

    capabilities = [
        {
            "name": "windows.open_app",
            "action": "system.open_app",
        },
        {
            "name": "windows.set_volume",
            "action": "system.set_volume",
        },
    ]

    decision = engine.decide(
        compound,
        capabilities=capabilities,
    )

    assert decision.is_executable is True
    assert decision.selected_action == "system.open_app"
    assert decision.alternatives == [
        "system.set_volume"
    ]
