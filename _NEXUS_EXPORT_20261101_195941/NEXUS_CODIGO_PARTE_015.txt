# ==========================================================
# RUN WORKFLOW (ENTRYPOINT OFICIAL)
# ==========================================================

async def run_workflow(
    name,
    tenant_id=None,
    workspace_id=None,
    job_id=None,
    user=None,
    shared_context=None
):

    wf = get_workflow_by_name(
        name,
        tenant_id=tenant_id,
        workspace_id=workspace_id
    )

    if not wf:
        raise Exception("Workflow no encontrado")

    context = shared_context if shared_context is not None else {}
    history = []

    for step in wf["steps"]:
        result = await execute_step(
            step,
            user,
            context,
            workflow_name=name,
            tenant_id=tenant_id,
            workspace_id=workspace_id,
            job_id=job_id
        )
        history.extend(result)

    return history


# ==========================================================
# EXECUTE STEP (MOTOR REAL)
# ==========================================================

async def execute_step(
    step,
    user,
    context,
    workflow_name=None,
    tenant_id=None,
    workspace_id=None,
    job_id=None
):

    history = []
    step_type = step.get("type", "action")

    try:

        # ==================================================
        # 1️⃣ ACCIÓN NORMAL
        # ==================================================

        if step_type == "action":

            value = resolve_variables(step.get("value"), context)

            if step.get("input_type"):
                if not validate_type(value, step.get("input_type")):
                    raise Exception(
                        f"Input inválido para {step['action']}"
                    )

            await bus.publish("step.start", {
                "workflow": workflow_name,
                "step": step["action"],
                "job_id": job_id
            })

            out = execute_action(
                step["action"],
                value,
                user
            )

            # 🔥 FIX IMPORTANTE (si la acción es async)
            if asyncio.iscoroutine(out):
                out = await out

            if step.get("output_type"):
                if not validate_type(out, step.get("output_type")):
                    raise Exception(
                        f"Output inválido en {step['action']}"
                    )

            if step.get("assign"):
                context[step["assign"]] = out

            history.append({
                "type": "action",
                "action": step["action"],
                "result": out
            })

            # 🔥 LOG DEL STEP
            log_step(job_id, step["action"], "success", out)

            await bus.publish("step.success", {
                "workflow": workflow_name,
                "step": step["action"],
                "job_id": job_id
            })

            await manager.broadcast({
                "type": "workflow_step",
                "action": step["action"]
            })

            await manager.broadcast({
                "type": "workflow_result",
                "action": step["action"],
                "result": out
            })

            for child in step.get("next", []):
                child_result = await execute_step(
                    child,
                    user,
                    context,
                    workflow_name,
                    tenant_id,
                    workspace_id,
                    job_id
                )
                history.extend(child_result)

            return history


        # ==================================================
        # 2️⃣ CONDICIÓN
        # ==================================================

        if step_type == "condition":

            test_value = resolve_variables(
                step.get("value"),
                context
            )

            test_result = execute_action(
                step["if_action"],
                test_value,
                user
            )

            if asyncio.iscoroutine(test_result):
                test_result = await test_result

            condition_met = evaluate_condition(
                test_result,
                step.get("operator"),
                step.get("compare")
            )

            history.append({
                "type": "condition",
                "result": condition_met
            })

            branch = step.get("then") if condition_met else step.get("else_")

            if branch:
                for s in branch:
                    branch_result = await execute_step(
                        s,
                        user,
                        context,
                        workflow_name,
                        tenant_id,
                        workspace_id,
                        job_id
                    )
                    history.extend(branch_result)

            return history


        # ==================================================
        # 3️⃣ SUBWORKFLOW
        # ==================================================

        if step_type == "subworkflow":

            sub_name = step.get("name")

            await manager.broadcast({
                "type": "workflow_sub_start",
                "workflow": sub_name
            })

            sub_history = await run_workflow(
                sub_name,
                tenant_id=tenant_id,
                workspace_id=workspace_id,
                job_id=job_id,
                user=user,
                shared_context=context
            )

            history.append({
                "type": "subworkflow",
                "workflow": sub_name,
                "result": sub_history
            })

            await manager.broadcast({
                "type": "workflow_sub_end",
                "workflow": sub_name
            })

            return history

        return history


    except Exception as e:

        error = str(e)

        print("🔥 ERROR REAL:", repr(e))

        await bus.publish("step.error", {
            "workflow": workflow_name,
            "step": step.get("action"),
            "error": error,
            "job_id": job_id,
            "tenant_id": tenant_id,
            "workspace_id": workspace_id
        })

        await manager.broadcast({
            "type": "workflow_error",
            "error": error
        })

        history.append({
            "type": "error",
            "step": step.get("action"),
            "error": error
        })

        log_step(job_id, step.get("action"), "error", str(e))

        raise

def resolve_variables(value, context):

    if value is None:
        return value

    if isinstance(value, str):
        for key, val in context.items():
            placeholder = "{{" + key + "}}"
            if placeholder in value:
                value = value.replace(
                    placeholder,
                    str(val)
                )
        return value

    if isinstance(value, dict):
        return {
            k: resolve_variables(v, context)
            for k, v in value.items()
        }

    return value


# ==========================================================
# CONDITION EVALUATOR
# ==========================================================

def evaluate_condition(result, operator, compare):

    if operator == "contains":
        return str(compare) in str(result)

    if operator == "equals":
        return str(result) == str(compare)

    if operator == "not_equals":
        return str(result) != str(compare)

    if operator == "gt":
        try:
            return float(result) > float(compare)
        except:
            return False

    if operator == "lt":
        try:
            return float(result) < float(compare)
        except:
            return False

    return False

def log_step(job_id, step, status, result):

    conn = get_connection()
    cur = conn.cursor()

    cur.execute("""
        INSERT INTO job_steps
        (job_id, step, status, result, timestamp)
        VALUES (%s, %s, %s, %s, %s)
    """, (
        job_id,
        step,
        status,
        json.dumps(result) if result else "",
        datetime.utcnow()
    ))

    conn.commit()
    conn.close()

def check_workflow_health(workflow_name):

    conn = get_connection()
    cur = conn.cursor()

    cur.execute("""
        SELECT COUNT(*) as fails
        FROM jobs
        WHERE workflow=%s
        AND status='error'
        AND updated_at > datetime('now','-10 minutes')
    """,(workflow_name,))

    row = cur.fetchone()

import json
from datetime import datetime, timezone

from core.database import get_connection


# ==========================================================
# SAVE WORKFLOW VERSION
# ==========================================================

def save_workflow_version(
    name,
    trigger,
    steps,
    tenant_id="default",
    workspace_id="main"
):

    conn = get_connection()
    cur = conn.cursor()

    # 1️⃣ Obtener versión actual
    cur.execute("""
        SELECT MAX(version)
        FROM workflows
        WHERE name = %s
        AND tenant_id = %s
        AND workspace_id = %s
    """, (name, tenant_id, workspace_id))

    row = cur.fetchone()
    current_version = row[0] if row and row[0] else 0
    new_version = current_version + 1

    # 2️⃣ Desactivar versiones anteriores
    cur.execute("""
        UPDATE workflows
        SET active = FALSE
        WHERE name = %s
        AND tenant_id = %s
        AND workspace_id = %s
    """, (name, tenant_id, workspace_id))

    # 3️⃣ Crear definición
    definition = {
        "name": name,
        "trigger": trigger,
        "steps": steps
    }

    now = datetime.now(timezone.utc)

    # 4️⃣ Insertar nueva versión activa
    cur.execute("""
        INSERT INTO workflows
        (name, version, definition, created_at, tenant_id, workspace_id, active)
        VALUES (%s, %s, %s, %s, %s, %s, TRUE)
    """, (
        name,
        new_version,
        json.dumps(definition),
        now,
        tenant_id,
        workspace_id
    ))

    conn.commit()
    conn.close()

    return new_version

def get_workflow_by_name(name, tenant_id="default", workspace_id="main"):

    conn = get_connection()
    cur = conn.cursor()

    cur.execute("""
        SELECT definition
        FROM workflows
        WHERE name = %s
        AND tenant_id = %s
        AND workspace_id = %s
        AND active = 1
        ORDER BY version DESC
        LIMIT 1
    """, (name, tenant_id, workspace_id))

    row = cur.fetchone()
    conn.close()

    if not row:
        return None

    definition = row[0]

    if not definition:
        raise Exception("Workflow definition vacía en DB")

    return json.loads(definition)

def load_workflows(tenant_id="default", workspace_id="main"):

    conn = get_connection()
    cur = conn.cursor()

    cur.execute("""
        SELECT name, version
        FROM workflows
        WHERE tenant_id = %s
        AND workspace_id = %s
        AND active = 1
    """, (tenant_id, workspace_id))

    rows = cur.fetchall()
    conn.close()

    return [
        {
            "name": r[0],
            "version": r[1]
        }
        for r in rows
    ]


# ==========================================================
# GET WORKFLOWS BY TRIGGER
# ==========================================================

def get_workflows_by_trigger(trigger, tenant_id="default", workspace_id="main"):

    conn = get_connection()
    cur = conn.cursor()

    cur.execute("""
        SELECT definition
        FROM workflows
        WHERE tenant_id = %s
        AND workspace_id = %s
        AND active = 1
    """, (tenant_id, workspace_id))

    rows = cur.fetchall()
    conn.close()

    matched = []

    for row in rows:
        wf = json.loads(row[0])
        if wf.get("trigger") == trigger:
            matched.append(wf)

    return matched

from pydantic import BaseModel, Field, validator
from typing import List, Optional, Any


ALLOWED_ACTIONS = {
    "open_app",
    "open_url",
    "list_dir",
    "read_file",
    "write_file",
    "delete_file",
    "copy_file",
    "rename_file",
    "create_folder",
    "move_files",
    "list_processes",
    "kill_process",
    "get_time",
    "run_cmd",
    "sleep"
}


class StepModel(BaseModel):
    action: str
    value: Optional[Any] = None

    @validator("action")
    def validate_action(cls, v):
        if v not in ALLOWED_ACTIONS:
            raise ValueError(f"Acción no permitida: {v}")
        return v


class WorkflowModel(BaseModel):
    name: str
    trigger: str = Field(default="manual")
    fail_fast: bool = Field(default=False)  # 🔥 NUEVO
    steps: List[StepModel]

    @validator("steps")
    def validate_steps_not_empty(cls, v):
        if not v:
            raise ValueError("Workflow sin pasos")
        return v

import json
from pathlib import Path
from workflows.engine import load_workflows, save_workflows

TEMPLATES_DIR = Path("workflows/templates")


def list_templates():

    return [f.stem for f in TEMPLATES_DIR.glob("*.json")]


def load_template(name):

    file = TEMPLATES_DIR / f"{name}.json"

    if not file.exists():
        return None

    return json.loads(file.read_text())


def import_template(name):

    tpl = load_template(name)

    if not tpl:
        return False

    current = load_workflows()

    merged = current + tpl

    save_workflows(merged)

    return True

from typing import Any


TYPE_MAP = {
    "string": str,
    "int": int,
    "float": float,
    "bool": bool,
    "list": list,
    "dict": dict
}


def validate_type(value: Any, declared_type: str):

    if not declared_type:
        return True

    py_type = TYPE_MAP.get(declared_type)

    if not py_type:
        return True  # tipo desconocido → no bloquea

    return isinstance(value, py_type)

VALID_ACTIONS = {
    "create_folder",
    "move_files",
    "list_dir",
    "read_file",
    "write_file",
    "delete_file",
    "run_cmd",
    "open_url"
}

# acciones que requieren confirmación humana
DANGEROUS_ACTIONS = {
    "delete_file",
    "run_cmd"
}

MAX_STEPS = 25


def validate_workflow(workflow):

    if not workflow:
        raise Exception("Workflow vacío")

    if "steps" not in workflow:
        raise Exception("Workflow sin steps")

    steps = workflow["steps"]

    if not isinstance(steps, list):
        raise Exception("Steps debe ser una lista")

    if len(steps) > MAX_STEPS:
        raise Exception("Workflow demasiado grande")

    for step in steps:
        validate_step(step)

    return workflow


def validate_step(step):

    if not isinstance(step, dict):
        raise Exception("Step inválido")

    if "action" not in step and step.get("type") != "condition":
        raise Exception("Step sin acción")

    # =========================
    # ACTION STEPS
    # =========================

    if "action" in step:

        action = step["action"]

        if action not in VALID_ACTIONS:
            raise Exception(f"Acción no válida: {action}")

        # advertencia acciones peligrosas
        if action in DANGEROUS_ACTIONS:
            print(f"⚠ Acción peligrosa detectada: {action}")

        value = step.get("value")

        # ---------------------
        # autocorrecciones
        # ---------------------

        if action == "move_files":

            if isinstance(value, str):

                step["value"] = {
                    "ext": ".pdf",
                    "to": "PDF"
                }

        if action == "list_dir":

            if not value:
                step["value"] = "."

    # =========================
    # SUBSTEPS (next)
    # =========================

    if "next" in step:

        if not isinstance(step["next"], list):
            raise Exception("next debe ser lista")

        for s in step["next"]:
            validate_step(s)

    # =========================
    # CONDITIONS
    # =========================

    if step.get("type") == "condition":

        for branch in ["then", "else_"]:

            if branch in step:

                if not isinstance(step[branch], list):
                    raise Exception(f"{branch} debe ser lista")

                for s in step[branch]:
                    validate_step(s)

"""Origin and Host validation for the local NEXUS HTTP API.

The API can execute local actions, so browser callers are restricted to the
known local development UIs. Non-browser clients without an Origin header
remain supported; Host is still restricted to loopback names.
"""
from __future__ import annotations

from urllib.parse import urlsplit

ALLOWED_BROWSER_ORIGINS = frozenset(
    {
        "http://localhost:5173",
        "http://127.0.0.1:5173",
        "http://localhost:5174",
        "http://127.0.0.1:5174",
    }
)
ALLOWED_HOST_NAMES = frozenset({"localhost", "127.0.0.1"})
API_PORT = 8765
MAX_REQUEST_BODY_BYTES = 1_048_576  # 1 MiB por solicitud JSON


def is_allowed_browser_origin(origin: str | None) -> bool:
    """Allow non-browser clients and only the known local browser origins."""
    if origin is None or not origin.strip():
        return True
    return origin.strip() in ALLOWED_BROWSER_ORIGINS


def cors_origin_for(origin: str | None) -> str | None:
    """Return an origin to echo in CORS headers, never a wildcard."""
    if origin is None:
        return None
    normalized = origin.strip()
    return normalized if normalized in ALLOWED_BROWSER_ORIGINS else None


def is_allowed_host(host_header: str | None) -> bool:
    """Reject non-loopback Host headers (including DNS-rebinding domains)."""
    if host_header is None or not host_header.strip():
        # Preserve compatibility with HTTP/1.0 clients that omit Host.
        return True
    try:
        parsed = urlsplit("//" + host_header.strip())
        hostname = (parsed.hostname or "").lower()
        port = parsed.port
    except ValueError:
        return False
    return hostname in ALLOWED_HOST_NAMES and port in (None, API_PORT)


def is_allowed_content_length(content_length: int) -> bool:
    """Accept non-empty request bodies up to the configured safety limit."""
    return 0 < content_length <= MAX_REQUEST_BODY_BYTES

from __future__ import annotations

from datetime import datetime

from actions.base import Action
from planner.execution_context import ExecutionContext
from planner.plan import PlanStep


class GetTimeAction(Action):
    """Acción para consultar la fecha y hora local del equipo."""

    name = "system.get_time"
    description = "Consultar la fecha y hora local del sistema."
    internal = True

    produces = ["date", "time"]

    async def execute(
        self,
        context: ExecutionContext,
        step: PlanStep,
    ) -> None:
        now = datetime.now()
        current_date = now.strftime("%d/%m/%Y")
        current_time = now.strftime("%H:%M")

        context.add_result("date", current_date)
        context.add_result("time", current_time)
        context.add_result("status", "read")
