from __future__ import annotations

from dataclasses import dataclass

from documents.content_version import DocumentContentVersion
from documents.content_version_store import DocumentContentVersionStore


@dataclass(frozen=True)
class DocumentVersionResolution:
    document_id: str
    from_version: DocumentContentVersion
    to_version: DocumentContentVersion


class DocumentVersionResolver:
    def __init__(self, store: DocumentContentVersionStore):
        self.store = store

    def resolve_version(
        self,
        document_id: str,
        version: int,
    ) -> DocumentContentVersion:
        versions = self.store.list_for_document(document_id)

        for content_version in versions:
            if content_version.version == version:
                return content_version

        raise ValueError(
            f"Document content version not found: "
            f"{document_id} v{version}"
        )

    def resolve_latest(
        self,
        document_id: str,
    ) -> DocumentContentVersion:
        latest = self.store.get_latest_for_document(document_id)

        if latest is None:
            raise ValueError(
                f"No content versions found for document: {document_id}"
            )

        return latest

    def resolve_previous(
        self,
        document_id: str,
        version: int | None = None,
    ) -> DocumentContentVersion:
        versions = self.store.list_for_document(document_id)

        if not versions:
            raise ValueError(
                f"No content versions found for document: {document_id}"
            )

        target_version = (
            self.resolve_latest(document_id).version
            if version is None
            else version
        )

        previous = [
            content_version
            for content_version in versions
            if content_version.version < target_version
        ]

        if not previous:
            raise ValueError(
                f"No previous content version found for document: "
                f"{document_id} v{target_version}"
            )

        return max(previous, key=lambda item: item.version)

    def resolve_pair(
        self,
        document_id: str,
        from_version: int,
        to_version: int,
    ) -> DocumentVersionResolution:
        if from_version == to_version:
            raise ValueError(
                "Cannot resolve a version comparison pair with "
                "the same version"
            )

        from_content_version = self.resolve_version(
            document_id,
            from_version,
        )
        to_content_version = self.resolve_version(
            document_id,
            to_version,
        )

        return DocumentVersionResolution(
            document_id=document_id,
            from_version=from_content_version,
            to_version=to_content_version,
        )

    def resolve_latest_pair(
        self,
        document_id: str,
    ) -> DocumentVersionResolution:
        latest = self.resolve_latest(document_id)
        previous = self.resolve_previous(
            document_id,
            version=latest.version,
        )

        return DocumentVersionResolution(
            document_id=document_id,
            from_version=previous,
            to_version=latest,
        )

from documents import (
    DocumentChangeDetector,
    DocumentChangeType,
    DocumentContentVersion,
    DocumentContentVersionStore,
    DocumentVersionComparator,
    DocumentVersionResolver,
)


def add_version(
    store: DocumentContentVersionStore,
    document_id: str,
    version: int,
    content_id: str,
) -> DocumentContentVersion:
    content_version = DocumentContentVersion.create(
        document_id=document_id,
        content_id=content_id,
        version=version,
        content_type="text/plain",
    )
    store.add(content_version)
    return content_version


def build_comparison(
    document_id: str = "doc-1",
    from_content_id: str = "content-1",
    to_content_id: str = "content-2",
):
    store = DocumentContentVersionStore()

    add_version(
        store,
        document_id=document_id,
        version=1,
        content_id=from_content_id,
    )

    add_version(
        store,
        document_id=document_id,
        version=2,
        content_id=to_content_id,
    )

    resolver = DocumentVersionResolver(store)
    resolution = resolver.resolve_pair(
        document_id=document_id,
        from_version=1,
        to_version=2,
    )

    return DocumentVersionComparator().compare(resolution)


def test_detector_detects_modified_line():
    comparison = build_comparison()

    detector = DocumentChangeDetector()

    result = detector.detect(
        comparison,
        before_text="El contrato tiene vigencia de 12 meses.",
        after_text="El contrato tiene vigencia de 24 meses.",
    )

    assert result.changed is True
    assert result.unchanged is False
    assert len(result.changes) == 1

    change = result.changes[0]

    assert change.change_type == DocumentChangeType.MODIFIED
    assert change.before == "El contrato tiene vigencia de 12 meses."
    assert change.after == "El contrato tiene vigencia de 24 meses."
    assert change.before_line == 1
    assert change.after_line == 1


def test_detector_detects_added_line():
    comparison = build_comparison()

    detector = DocumentChangeDetector()

    result = detector.detect(
        comparison,
        before_text="Primera línea.",
        after_text="Primera línea.\nSegunda línea.",
    )

    assert result.changed is True
    assert len(result.changes) == 1

    change = result.changes[0]

    assert change.change_type == DocumentChangeType.ADDED
    assert change.before is None
    assert change.after == "Segunda línea."
    assert change.after_line == 2


def test_detector_detects_removed_line():
    comparison = build_comparison()

def test_detector_detects_multiple_changes():
    comparison = build_comparison()

    detector = DocumentChangeDetector()

    result = detector.detect(
        comparison,
        before_text="A\nB\nC",
        after_text="A\nX\nC\nD",
    )

    assert len(result.changes) == 2

    modified = result.changes[0]
    added = result.changes[1]

    assert modified.change_type == DocumentChangeType.MODIFIED
    assert modified.before == "B"
    assert modified.after == "X"
    assert modified.before_line == 2
    assert modified.after_line == 2

    assert added.change_type == DocumentChangeType.ADDED
    assert added.after == "D"
    assert added.after_line == 4
