class AnomalyDetector:

    def detect(self, history):

        if len(history) < 5:
            return None

        last = history[-1]
        prev = history[-5]

        # error spike
        if last["error_rate"] > prev["error_rate"] * 2 and last["error_rate"] > 0.2:
            return {
                "type": "error_spike",
                "message": "Estoy observando un aumento anormal de errores."
            }

        # queue growth
        if last["queue_size"] > prev["queue_size"] + 10:
            return {
                "type": "queue_growth",
                "message": "La cola de trabajos está creciendo rápidamente."
            }

        if len(last["open_circuits"]) >= 2:
            return {
                "type": "circuit_alert",
                "message": "Varios circuit breakers están abiertos."
            }

        return None


anomaly_detector = AnomalyDetector()

class DecisionEngine:

    def decide(self, snapshot, risk, anomaly):

        # riesgo alto
        if risk == "HIGH":
            return {
                "action": "force_stable_mode",
                "reason": "Riesgo alto detectado"
            }

        # anomalía detectada
        if anomaly:
            return {
                "action": "investigate",
                "reason": anomaly["message"]
            }

        # cola creciendo
        if snapshot["queue_size"] > 20:
            return {
                "action": "scale_workers",
                "reason": "Cola de trabajos elevada"
            }

        return None

import math


def fake_embedding(text):

    text = text.lower()

    vector = []

    for c in text[:128]:

        vector.append(
            ord(c) / 255
        )

    while len(vector) < 128:
        vector.append(0)

    return vector


def cosine_similarity(a, b):

    dot = sum(x*y for x,y in zip(a,b))

    norm_a = math.sqrt(
        sum(x*x for x in a)
    )

    norm_b = math.sqrt(
        sum(x*x for x in b)
    )

    if norm_a == 0 or norm_b == 0:
        return 0

    return dot / (
        norm_a * norm_b
    )

def explain_decision(decision, snapshot):

    action = decision.get("action")

    if action == "reduce_workers":

        return (
            f"He reducido los workers porque la tasa de error "
            f"alcanzó {round(snapshot['error_rate']*100,2)}%."
        )

    if action == "scale_workers":

        return (
            f"He aumentado los workers porque la cola de trabajos "
            f"llegó a {snapshot['queue_size']}."
        )

    if action == "force_stable_mode":

        return (
            "Activé modo estable porque hay circuit breakers abiertos."
        )

    if action == "investigate":

        return decision.get("reason")

    return "He tomado una acción automática para proteger el sistema."

def extract_json(text):

    match = re.search(r"\{.*\}", text, re.DOTALL)

    if match:
        return match.group(0)

    return None


def generate_steps(prompt: str):

    full_prompt = f"{SYSTEM_PROMPT}\n\nUsuario: {prompt}"

    try:

        response = requests.post(
            OLLAMA_URL,

import logging
import json
from datetime import datetime

class JsonFormatter(logging.Formatter):
    def format(self, record):

        log_record = {
            "timestamp": datetime.utcnow().isoformat(),
            "level": record.levelname,
            "message": record.getMessage(),
            "module": record.module,
        }

        if hasattr(record, "extra_data"):
            log_record["data"] = record.extra_data

        return json.dumps(log_record)

import time
from datetime import datetime


class Metrics:

    def __init__(self):
        self.start_time = time.time()

        # contadores
        self.counters = {}

        # histogramas
        self.histograms = {}

        # gauges (estado actual)
        self.gauges = {}

        # métricas globales
        self.total_workflows = 0
        self.total_steps = 0
        self.total_jobs = 0
        self.total_errors = 0
        self.active_jobs = 0

    # =========================
    # COUNTERS
    # =========================

    def inc(self, name):
        self.counters[name] = self.counters.get(name, 0) + 1

    def inc_job(self):
        self.inc("jobs_created")
        self.total_jobs += 1

    def inc_error(self):
        self.inc("jobs_failed")
        self.total_errors += 1

    # =========================
    # HISTOGRAMS
    # =========================

    def observe(self, name, value):
        if name not in self.histograms:
            self.histograms[name] = []

        self.histograms[name].append(value)

    # =========================
    # GAUGES
    # =========================

    def set_active_jobs(self, value):
        self.active_jobs = value
        self.gauges["active_jobs"] = value

    def set_gauge(self, name, value):
        self.gauges[name] = value

    # =========================
    # WORKFLOW METRICS
    # =========================

    def inc_workflow(self):
        self.total_workflows += 1

    def inc_step(self):
        self.total_steps += 1

    # =========================
    # UPTIME
    # =========================

    def uptime(self):
        return round(time.time() - self.start_time)

    # =========================
    # SNAPSHOT
    # =========================

    def snapshot(self):

        return {
            "uptime_seconds": self.uptime(),
            "total_workflows_executed": self.total_workflows,
            "total_steps_executed": self.total_steps,
            "total_jobs_created": self.total_jobs,
            "total_errors": self.total_errors,
            "active_jobs": self.active_jobs,
            "timestamp": datetime.utcnow().isoformat()
        }

    # =========================
    # CONCURRENCY INFO
    # =========================

    def concurrency_info(self, queue_size, workers):

        return {
            "queue_size": queue_size,
            "max_workers": workers
        }


metrics = Metrics()

async def create_mission(

    goal,
    model="llama3"
):

    mission_id = str(uuid.uuid4())

    conn = get_connection()
    cur = conn.cursor()

    cur.execute("""
        INSERT INTO missions
        (
            id,
            goal,
            status
        )
        VALUES (%s,%s,%s)
    """, (
        mission_id,
        goal,
        "planning"
    ))

    conn.commit()

    # =========================
    # GENERATE DAG
    # =========================
    workflow = generate_workflow(
        goal,
        model
    )

    # =========================
    # EXECUTE
    # =========================
    job_id = await manager.create_job(
        workflow,
        "system"
    )

    # =========================
    # UPDATE
    # =========================
    cur.execute("""
        UPDATE missions
        SET status = %s,
            result = %s
        WHERE id = %s
    """, (
        "running",
        str({
            "job_id": job_id
        }),
        mission_id
    ))

    conn.commit()
    conn.close()

    return {
        "mission_id": mission_id,
        "job_id": job_id
    }

from ultralytics import YOLO

import cv2


model = YOLO("yolov8n.pt")


def detect_objects(
    image_path
):

    results = model(image_path)

    detections = []

    for r in results:

        boxes = r.boxes

        for box in boxes:

            cls = int(box.cls[0])

            conf = float(box.conf[0])

            coords = box.xyxy[0].tolist()

            label = model.names[cls]

            detections.append({

                "label": label,

                "confidence": conf,

                "x1": coords[0],

                "y1": coords[1],

                "x2": coords[2],

                "y2": coords[3]
            })

    return detections

BLOCKED_PATHS = [

    "C:/Windows",
    "C:/Program Files",
    "C:/Users"
]

BLOCKED_DOMAINS = [

    "malware",
    "phishing",
    "tor"
]

MAX_AUTONOMOUS_STEPS = 20


def validate_action(
    action,
    params
):

    # =========================
    # BLOCK DELETE
    # =========================
    if action == "delete_files":

        return False, "delete_files blocked"

    # =========================
    # VALIDATE PATHS
    # =========================
    path = (
        params.get("path")
        or params.get("to")
        or ""
    )

    for blocked in BLOCKED_PATHS:

        if blocked.lower() in path.lower():

            return False, (
                f"Blocked path: {blocked}"
            )

    # =========================
    # VALIDATE URLS
    # =========================
    url = params.get(
        "url",
        ""
    )

    for domain in BLOCKED_DOMAINS:

        if domain in url.lower():

            return False, (
                f"Blocked domain: {domain}"
            )

    return True, "safe"

from core.metrics import metrics
from resilience.circuit import circuit
from core.database import get_connection
from datetime import datetime
import json


class SelfModel:

    def __init__(self):
        self.last_evaluation = None
        self.history = []

    # =========================
    # SYSTEM SNAPSHOT
    # =========================

    def snapshot(self):

        errors = metrics.counters.get("jobs_failed", 0)
        done = metrics.counters.get("jobs_done", 0)

        total = errors + done

        error_rate = errors / total if total > 0 else 0

        open_circuits = [
            k for k, v in circuit.state.items()
            if v == "OPEN"
        ]

        queue_size = metrics.gauges.get("active_jobs", 0)

        snapshot = {
            "timestamp": datetime.utcnow().isoformat(),
            "error_rate": error_rate,
            "open_circuits": open_circuits,
            "queue_size": queue_size,
            "total_jobs": total
        }

        # guardar historial
        self.history.append(snapshot)

        # mantener historial corto (~5 minutos si loop es 10s)
        if len(self.history) > 30:
            self.history.pop(0)

        return snapshot

    # =========================
    # RISK EVALUATION
    # =========================

    def evaluate_risk(self, snapshot):

        if snapshot["open_circuits"]:
            return "HIGH"

        if snapshot["error_rate"] > 0.3:
            return "MEDIUM"

        return "LOW"

    # =========================
    # HEALTH SCORE
    # =========================

    def health_score(self, snapshot):

        score = 100

        # penalizar error rate
        score -= snapshot["error_rate"] * 50

        # penalizar circuit breakers abiertos
        score -= len(snapshot["open_circuits"]) * 20

        # penalizar cola grande
        if snapshot["queue_size"] > 10:
            score -= 10

        if score < 0:
            score = 0

        return round(score, 2)

    # =========================
    # ANOMALY DETECTION
    # =========================

    def detect_anomaly(self):

        if len(self.history) < 5:
            return None

        last = self.history[-1]
        prev = self.history[-5]

        # spike de errores
        if last["error_rate"] > prev["error_rate"] * 2 and last["error_rate"] > 0.2:
            return {
                "type": "error_spike",
                "message": "He detectado un aumento abrupto en la tasa de errores."
            }

        # crecimiento de cola
        if last["queue_size"] > prev["queue_size"] + 10:
            return {
                "type": "queue_growth",
                "message": "La cola de trabajos está creciendo rápidamente."
            }

        # múltiples circuitos abiertos
        if len(last["open_circuits"]) >= 2:
            return {
                "type": "multiple_circuits",
                "message": "Varios circuit breakers están abiertos."
            }

        return None

    # =========================
    # DECISION ENGINE
    # =========================

    def propose_action(self, snapshot, risk):

        if risk == "HIGH":
            return {
                "action": "force_stable_mode",
                "reason": "Circuitos abiertos detectados"
            }

        if risk == "MEDIUM":
            return {
                "action": "reduce_workers",
                "reason": "Tasa de error elevada"
            }

        return None

    # =========================
    # DECISION LOG
    # =========================

    def log_decision(self, decision):

        conn = get_connection()
        cur = conn.cursor()

        cur.execute("""
            INSERT INTO decisions_log
            (timestamp, action, reason)
            VALUES (%s, %s, %s)
        """, (
            datetime.utcnow(),
            decision["action"],
            decision["reason"]
        ))

        conn.commit()
        conn.close()

import psutil


def get_running_processes():

    result = []

    for p in psutil.process_iter([

        "pid",
        "name"
    ]):

        try:

            result.append({

                "pid": p.info["pid"],

                "name": p.info["name"]
            })

        except:
            pass

    return result
