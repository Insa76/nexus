from __future__ import annotations

import os
from dataclasses import dataclass
from pathlib import Path

from runtime.scheduler import Scheduler
from tasks import ReminderStore

from bootstrap.reminder_scheduler import ReminderScheduler
from bootstrap.temporal_scheduler import TemporalSchedulerAdapter


@dataclass(slots=True)
class AssistantServices:
    scheduler: Scheduler
    reminder_store: ReminderStore
    temporal_scheduler: TemporalSchedulerAdapter
    reminder_scheduler: ReminderScheduler


def build_assistant_services(
    reminder_db_path: str | Path | None = None,
) -> AssistantServices:
    scheduler = Scheduler()

    if reminder_db_path is None:
        reminder_db_path = os.environ.get(
            "NEXUS_REMINDERS_DB_PATH",
            str(Path.home() / ".nexus" / "reminders.db"),
        )

    reminder_store = ReminderStore(db_path=reminder_db_path)
    temporal_scheduler = TemporalSchedulerAdapter(scheduler)

    reminder_scheduler = ReminderScheduler(
        scheduler_adapter=temporal_scheduler,
        reminder_store=reminder_store,
    )

    return AssistantServices(
        scheduler=scheduler,
        reminder_store=reminder_store,
        temporal_scheduler=temporal_scheduler,
        reminder_scheduler=reminder_scheduler,
    )

from __future__ import annotations

import asyncio

from bootstrap.assistant_services import AssistantServices


class AssistantServicesLifecycle:
    """
    Controls the lifecycle of Assistant Layer services.
    """

    def __init__(self, services: AssistantServices) -> None:
        self.services = services
        self.running = False
        self._scheduler_task: asyncio.Task[None] | None = None

    async def start(self) -> None:
        """Restore pending reminder schedules and start the scheduler."""

        if self.running:
            return

        self.services.reminder_scheduler.reconcile()

        self.running = True

        try:
            self._scheduler_task = asyncio.create_task(
                self.services.scheduler.start()
            )
        except Exception:
            self.running = False
            raise

    async def stop(self) -> None:
        """Stop the background scheduler."""

        if not self.running:
            return

        self.running = False

        await self.services.scheduler.stop()

        if self._scheduler_task is not None:
            try:
                await self._scheduler_task
            except asyncio.CancelledError:
                pass
            finally:
                self._scheduler_task = None

    async def restart(self) -> None:
        """Restart the services."""

        await self.stop()
        await self.start()

def build_engine(
    assistant_context: AssistantContext | None = None,
    assistant_session: AssistantSession | None = None,
) -> Engine:
    """
    Construye y configura completamente NEXUS.

    El AssistantContext es opcional para mantener compatibilidad
    con el arranque actual de NEXUS.
    """

    security_context = (
        assistant_session.context
        if assistant_session is not None
        else assistant_context
    )

    security_user = getattr(
        security_context,
        "user",
        None,
    )

    security_actor_id = getattr(
        security_user,
        "id",
        None,
    )

    local_console_actor = Actor.system(
        "local-console",
        name="NEXUS Local Console",
    )

    if (
        isinstance(security_actor_id, str)
        and security_actor_id.strip()
    ):
        security_actor = Actor.user(
            security_actor_id.strip()
        )
    else:
        security_actor = local_console_actor

    #
    # Working Memory
    #

    # Una única instancia compartida por todos
    # los componentes que necesiten memoria.

    memory = WorkingMemory()

    episode_memory = EpisodeMemory(
        memory
    )

    #
    # State History
    #

    state_history = StateHistory(
        memory
    )

    #
    # Kernel
    #

    kernel = Kernel()

    #
    # Runtime
    #

    runtime = Runtime(
        kernel=kernel
    )

    #
    # Provider Manager
    #

    provider_manager = ProviderManager()

    provider_manager.register(
        LocalProvider()
    )

    #
    # Action Registry
    #

    registry = ActionRegistry()

    #
    # Applications
    #

    registry.register(
        OpenAppAction(
            provider_manager
        )
    )

    registry.register(
        OpenFileAction(
            provider_manager
        )
    )

    registry.register(
        CloseAppAction()
    )

    registry.register(
        ProcessListAction(
            provider_manager
        )
    )

    registry.register(
        ScreenCaptureAction(
            provider_manager
        )
    )

    registry.register(
        InspectScreenAction()
    )

    registry.register(
        InspectCaptureAction()
    )

    registry.register(GetTimeAction())

    #
    # Episodic Memory
    #

    registry.register(
        QueryEpisodeAction(
            episode_memory
        )
    )

    #
    # Volume
    #

    registry.register(
        SetVolumeAction(
            provider_manager
        )
    )

    registry.register(
        GetVolumeAction(
            provider_manager
        )
    )

    registry.register(
        GetVolumeHistoryAction(
            state_history
        )
    )

    registry.register(
        GetRecentEpisodesAction(
            episode_memory
        )
    )

    registry.register(
        GetPreviousEpisodeAction(
            episode_memory
        )
    )

    registry.register(
        GetLastEpisodeAction(
            episode_memory
        )
    )

    registry.register(
        RestorePreviousVolumeAction(
            provider_manager,
            state_history,
        )
    )

    #
    # Mute / Unmute
    #

    registry.register(
        MuteAction(
            provider_manager
        )
    )

    registry.register(
        UnmuteAction(
            provider_manager
        )
    )

    registry.register(
        GetMuteAction(
            provider_manager
        )
    )

    registry.register(
        RestorePreviousMuteAction(
            provider_manager,
            state_history,
        )
    )

    #
    # Action Manager
    #

        #
    # Security: permisos de ejecución.
    #

    action_names = registry.list()

    permissions = frozenset(
        Permission(
            resource=action_name,
            action=PermissionAction.EXECUTE,
        )
        for action_name in action_names
    )


    local_role = Role(
        id="local_operator",
        name="Local Operator",
        permissions=permissions,
    )

    security_db_path = os.environ.get(
        "NEXUS_SECURITY_DB_PATH",
        str(Path.home() / ".nexus" / "security.db"),
    )

    security_database = SQLiteSecurityStore(security_db_path)
    role_store = SQLiteRoleStore(security_database)
    assignment_store = SQLiteAssignmentStore(security_database)

    stored_role = role_store.get(local_role.id)

    if stored_role is None:
        role_store.add(local_role)
    elif stored_role != local_role:
        role_store.update(local_role)

    persisted_local_role = role_store.get(local_role.id)

    if persisted_local_role is None:
        raise RuntimeError(
            "No se pudo recuperar el rol local desde SQLite."
        )

    assignment_store.add(
        ActorRoleAssignment(
            actor=local_console_actor,
            role=persisted_local_role,
        )
    )


    policy_store = ActionPolicyStore()

    for action_name in action_names:
        policy_store.register(
            ActionPolicy(
                action_name=action_name,
                permission=Permission(
                    resource=action_name,
                    action=PermissionAction.EXECUTE,
                ),
            )
        )

    authorization_service = AuthorizationService(
        assignment_store
    )

    action_authorization = ActionAuthorization(
        policy_store=policy_store,
        authorization=authorization_service,
    )

    manager = ActionManager(
        registry,
        action_authorization=action_authorization,
    )

    runtime.configure_actions(
        registry,
        manager
    )

    #
    # Tool Registry
    #

    tool_registry = ToolRegistry()

    tool_registry.register_many(
        provider_manager.tools()
    )

    tool_registry.register_many(
        registry.describe()
    )

    #
    # Exponer Tool Registry al Runtime
    #

    runtime.tool_registry = tool_registry

    #
    # Intent Providers
    #

    rule_provider = RuleIntentProvider(
        IntentRecognizer()
    )

    ollama_provider = OllamaIntentProvider(
        model="llama3:latest"
    )

    intent_provider = HybridIntentProvider(
        rule_provider=rule_provider,
        ollama_provider=ollama_provider,
    )

    #
    # Executor
    #

    executor = Executor(
        runtime
    )

    runtime.configure_executor(
        executor
    )

    session_manager = AssistantSessionManager()

    if assistant_session is not None:
        active_session = assistant_session

    elif assistant_context is not None:
        active_session = session_manager.start(
            assistant_context
        )

    else:
        active_session = None

    if active_session is not None:
        assistant_context = active_session.context

    #
    # Documents Capability
    #

    document_services = build_document_services()

    document_capability = DocumentCapability(
        context_access=document_services.context_access,
    )

    document_cognitive_access = DocumentCognitiveAccess(
        document_capability
    )

    #
    # Processes Capability
    #

    process_store = ProcessStore()

    process_resolver = ProcessResolver(
        process_store,
    )

    process_capability = ProcessCapability(
        store=process_store,
        resolver=process_resolver,
    )

    process_cognitive_access = ProcessCognitiveAccess(
        process_capability,
    )

    #
    # Rules Capability
    #

    rule_store = RuleStore()

    rule_resolver = RuleResolver(
        rule_store
    )

    rule_capability = RuleCapability(
        store=rule_store,
        resolver=rule_resolver,
    )

    rule_cognitive_access = RuleCognitiveAccess(
        rule_capability
    )

    #
    # Cognitive Capability Router
    #

    cognitive_capability_router = CognitiveCapabilityRouter(
        document_access=document_cognitive_access,
        process_access=process_cognitive_access,
        rule_access=rule_cognitive_access,
    )

    #
    # Engine
    #

    semantic_memory_db_path = os.environ.get(
        "NEXUS_SEMANTIC_MEMORY_DB_PATH",
        str(Path.home() / ".nexus" / "semantic_memory.db"),
    )
    semantic_memory = SemanticMemory(db_path=semantic_memory_db_path)

    return Engine(
        runtime,
        intent_provider=intent_provider,
        semantic_memory=semantic_memory,
        memory=memory,
        assistant_context=assistant_context,
        document_capability=document_capability,
        document_cognitive_access=document_cognitive_access,
        cognitive_capability_router=cognitive_capability_router,
        process_capability=process_capability,
        process_cognitive_access=process_cognitive_access,
        rule_capability=rule_capability,
        rule_cognitive_access=rule_cognitive_access,
        default_actor_id=security_actor.id,
        default_actor_type=security_actor.actor_type.value,
    )


def build_assistant_runtime(
    assistant_context: AssistantContext | None = None,
    assistant_session: AssistantSession | None = None,
) -> tuple[
    Engine,
    AssistantServices,
    AssistantServicesLifecycle,
]:
    """
    Construye el Engine cognitivo, los servicios del Assistant Layer
    y su lifecycle.

    El Engine, los servicios y su ciclo de vida mantienen
    responsabilidades separadas.
    """

    engine = build_engine(
        assistant_context=assistant_context,
        assistant_session=assistant_session,
    )

    services = build_assistant_services()

    lifecycle = AssistantServicesLifecycle(
        services
    )

    return engine, services, lifecycle


def _print_result(result) -> None:
    """
    Presenta el resultado de una ejecución
    de NEXUS de forma legible.
    """


    if result.success:

        print("NEXUS >", end=" ")

        response = (
            result.metadata.get("response")
            if isinstance(result.metadata, dict)
            else None
        )

        if response:
            print(response)
        elif result.message and result.message != "Execution completed successfully.":
            print(result.message)
        elif result.output:
            print(result.output)
        else:
            print("Listo.")

        return

    print(
        f"NEXUS > {result.message}"
    )



async def _interactive_loop(
    engine: Engine,
) -> None:
    """
    Loop interactivo persistente de NEXUS.

    El Engine y el event loop permanecen vivos
    durante toda la sesión.
    """

    print()

    print("Escribí una instrucción.")

    print("Escribí 'salir' para cerrar NEXUS.")

    print()

    while True:

        try:

            text = input(
                "Usuario > "
            ).strip()

        except (
            EOFError,
            KeyboardInterrupt,
        ):

            print()

            break

        if not text:
            continue

        if text.lower() in {
            "salir",
            "exit",
            "quit",
        }:

            break

        try:

            result = await engine.execute(
                text
            )

            _print_result(
                result
            )

        except Exception as exc:

            print(
                f"NEXUS > Error interno: {exc}"
            )

        print()



async def _main_async() -> None:
    """
    Inicializa NEXUS, mantiene una sesión interactiva
    persistente y gestiona el ciclo de vida de sus servicios.
    """

    print()
    print("=" * 50)
    print("          NEXUS Cognitive OS")
    print("=" * 50)
    print()

    engine, services, lifecycle = build_assistant_runtime()

    print("✓ Kernel")
    print("✓ Runtime")
    print("✓ Providers")
    print("✓ Actions")
    print("✓ Intent Providers")
    print("✓ Executor")
    print("✓ Engine")
    print("✓ Reminder Store")
    print("✓ Reminder Scheduler")
    print()

    try:
        await lifecycle.start()

        print("=" * 50)
        print("NEXUS READY")
        print("=" * 50)

        await _interactive_loop(engine)

    finally:
        await lifecycle.stop()

        close_semantic_memory = getattr(
            engine.semantic_memory,
            "close",
            None,
        )

        if callable(close_semantic_memory):
            close_semantic_memory()

        print()
        print("NEXUS detenido.")



def main() -> None:
    """
    Punto de entrada de NEXUS.
    """

    import asyncio

    asyncio.run(
        _main_async()
    )

def _print_result(result) -> None:
    """
    Presenta el resultado de una ejecución
    de NEXUS de forma legible.
    """


    if result.success:

        print("NEXUS >", end=" ")

        response = (
            result.metadata.get("response")
            if isinstance(result.metadata, dict)
            else None
        )

        if response:
            print(response)
        elif result.message and result.message != "Execution completed successfully.":
            print(result.message)
        elif result.output:
            print(result.output)
        else:
            print("Listo.")

        return

    print(
        f"NEXUS > {result.message}"
    )



async def _interactive_loop(
    engine: Engine,
) -> None:
    """
    Loop interactivo persistente de NEXUS.

    El Engine y el event loop permanecen vivos
    durante toda la sesión.
    """

    print()

    print("Escribí una instrucción.")

    print("Escribí 'salir' para cerrar NEXUS.")

    print()

    while True:

        try:

            text = input(
                "Usuario > "
            ).strip()

        except (
            EOFError,
            KeyboardInterrupt,
        ):

            print()

            break

        if not text:
            continue

        if text.lower() in {
            "salir",
            "exit",
            "quit",
        }:

            break

        try:

            result = await engine.execute(
                text
            )

            _print_result(
                result
            )

        except Exception as exc:

            print(
                f"NEXUS > Error interno: {exc}"
            )

        print()

from __future__ import annotations

from collections.abc import Awaitable, Callable
from dataclasses import replace
from datetime import datetime

from tasks import (
    Reminder,
    ReminderStatus,
    ReminderStore,
    ReminderTriggerHandler,
    ReminderTriggerResult,
    reminder_to_schedule,
    transition_reminder,
)

from bootstrap.temporal_scheduler import TemporalSchedulerAdapter


TriggerCallback = Callable[[ReminderTriggerResult], Awaitable[None]]


class ReminderScheduler:
    """
    Composition layer for end-to-end reminder scheduling.

    Responsibilities:
    - Convert a Reminder into a TemporalSchedule.
    - Register the schedule through TemporalSchedulerAdapter.
    - Trigger the reminder when its scheduled time arrives.
    - Persist the resulting triggered Reminder.
    - Optionally notify an external async callback.

    The Reminder domain remains independent from the Scheduler.
    """

    def __init__(
        self,
        scheduler_adapter: TemporalSchedulerAdapter,
        reminder_store: ReminderStore,
        trigger_handler: ReminderTriggerHandler | None = None,
    ) -> None:
        self._scheduler_adapter = scheduler_adapter
        self._reminder_store = reminder_store
        self._trigger_handler = trigger_handler or ReminderTriggerHandler()

    def schedule(
        self,
        reminder: Reminder,
        on_trigger: TriggerCallback | None = None,
    ) -> None:
        """
        Schedule a pending reminder for future execution.
        """

        temporal_schedule = reminder_to_schedule(reminder)

        async def handle_trigger() -> None:
            result = self._trigger_handler.handle(reminder)

            self._reminder_store.update(result.reminder)

            if on_trigger is not None:
                await on_trigger(result)

        self._scheduler_adapter.schedule(
            temporal_schedule=temporal_schedule,
            callback=handle_trigger,
        )

    def cancel(
        self,
        reminder_id: str,
    ) -> bool:
        """
        Cancela un reminder pendiente y elimina su ejecución temporal.

        Devuelve True si el reminder fue cancelado.
        Devuelve False si no existe.
        """

        reminder = self._reminder_store.get(reminder_id)

        if reminder is None:
            return False

        cancelled = transition_reminder(
            reminder,
            ReminderStatus.CANCELLED,
        )

        temporal_schedule = reminder_to_schedule(
            reminder,
        )

        scheduler_cancelled = self._scheduler_adapter.cancel(
            temporal_schedule,
        )

        self._reminder_store.update(cancelled)

        return scheduler_cancelled

    def reschedule(
        self,
        reminder_id: str,
        trigger_at: datetime,
    ) -> bool:
        """
        Cambia la fecha de ejecución de un reminder pendiente.

        Elimina la ejecución temporal anterior, actualiza el reminder
        y registra una nueva ejecución temporal.

        Devuelve True si el reminder fue reprogramado.
        Devuelve False si no existe.
        """

        reminder = self._reminder_store.get(reminder_id)

        if reminder is None:
            return False

        if reminder.status is not ReminderStatus.PENDING:
            return False

        current_schedule = reminder_to_schedule(
            reminder,
        )

        self._scheduler_adapter.cancel(
            current_schedule,
        )

        updated_reminder = replace(
            reminder,
            trigger_at=trigger_at,
        )

        self._reminder_store.update(
            updated_reminder,
        )

        self.schedule(
            updated_reminder,
        )

        return True

    def reconcile(self) -> int:
        """
        Reconstruye las ejecuciones temporales de todos los reminders
        pendientes actualmente almacenados.

        Devuelve la cantidad de reminders programados.
        """

        scheduled_count = 0

        for reminder in self._reminder_store.list():
            if reminder.status is not ReminderStatus.PENDING:
                continue

            schedule = reminder_to_schedule(reminder)

            schedule_name = (
                f"{schedule.target_type}:"
                f"{schedule.target_id}"
            )

            if schedule_name in self._scheduler_adapter.scheduled():
                continue

            self.schedule(reminder)
            scheduled_count += 1

        return scheduled_count

    def get(self, reminder_id: str) -> Reminder | None:
        """
        Return the current persisted state of a reminder.
        """

        return self._reminder_store.get(reminder_id)

from __future__ import annotations

from collections.abc import Awaitable, Callable

from runtime.scheduler import Scheduler
from tasks import TemporalSchedule


TemporalCallback = Callable[[], Awaitable[None]]


class TemporalSchedulerAdapter:
    """
    Connects domain temporal schedules with the Runtime Scheduler.

    The adapter knows both sides of the boundary:
    - Assistant domain: TemporalSchedule
    - Runtime infrastructure: Scheduler

    Domain objects themselves remain unaware of Runtime.
    """

    def __init__(self, scheduler: Scheduler) -> None:
        self._scheduler = scheduler

    def schedule(
        self,
        temporal_schedule: TemporalSchedule,
        callback: TemporalCallback,
    ) -> None:
        self._scheduler.register_at(
            name=(
                f"{temporal_schedule.target_type}:"
                f"{temporal_schedule.target_id}"
            ),
            execute_at=temporal_schedule.execute_at,
            callback=callback,
        )

    def cancel(
        self,
        temporal_schedule: TemporalSchedule,
    ) -> bool:
        """
        Cancela una ejecución temporal previamente registrada.

        Devuelve True si la ejecución fue encontrada y cancelada.
        Devuelve False si no existía.
        """

        name = (
            f"{temporal_schedule.target_type}:"
            f"{temporal_schedule.target_id}"
        )

        return self._scheduler.cancel_one_shot(name)

    def scheduled(self) -> list[str]:
        return self._scheduler.one_shots()
