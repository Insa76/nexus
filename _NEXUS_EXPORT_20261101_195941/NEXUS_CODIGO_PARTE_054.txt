def build_service() -> DocumentService:
    document_store = DocumentStore()
    artifact_store = ArtifactStore()
    content_store = ContentStore()

    registry = IngestionRegistry()
    registry.register(
        "text/plain",
        TextIngestor(LocalArtifactReader()),
    )

    ingestion_service = DocumentIngestionService(
        registry=registry,
    )

    content_version_store = DocumentContentVersionStore()

    semantic_analysis_service = DocumentSemanticAnalysisService(
        analyzer=RuleBasedSemanticAnalyzer(),
    )
    knowledge_extraction_service = DocumentKnowledgeExtractionService(
        extractor=RuleBasedDocumentKnowledgeExtractor(),
    )
    knowledge_record_store = DocumentKnowledgeRecordStore()

    return DocumentService(
        document_store=document_store,
        artifact_store=artifact_store,
        content_store=content_store,
        ingestion_service=ingestion_service,
        content_version_store=content_version_store,
        semantic_analysis_service=semantic_analysis_service,
        knowledge_extraction_service=knowledge_extraction_service,
        knowledge_record_store=knowledge_record_store,
    )


def create_document(
    service: DocumentService,
    source: Path,
):
    artifact = DocumentArtifact.create(
        location=f"local:{source}",
    )

    service.add_artifact(artifact)

    return service.create_document(
        name=source.name,
        mime_type="text/plain",
        source=f"local:{source}",
        artifact_id=artifact.id,
    )


def test_compare_versions_detects_modified_content(tmp_path):
    source = tmp_path / "contract.txt"
    source.write_text(
        "El contrato tiene vigencia de 12 meses.",
        encoding="utf-8",
    )

    service = build_service()
    document = create_document(service, source)

    service.ingest(document)

    source.write_text(
        "El contrato tiene vigencia de 24 meses.",
        encoding="utf-8",
    )

    service.ingest(document)

    result = service.compare_versions(
        document_id=document.id,
        from_version=1,
        to_version=2,
    )

    assert result.changed is True
    assert result.unchanged is False

    assert result.comparison.document_id == document.id
    assert result.comparison.from_version == 1
    assert result.comparison.to_version == 2

    assert len(result.changes) == 1

    change = result.changes[0]

    assert change.before == (
        "El contrato tiene vigencia de 12 meses."
    )
    assert change.after == (
        "El contrato tiene vigencia de 24 meses."
    )

def test_list_knowledge_history_returns_persisted_records() -> None:
    service = build_service()

    document = Document.create(
        name="knowledge-history.txt",
        mime_type="text/plain",
        source="memory://knowledge-history.txt",
    )

    service.document_store.add(document)

    artifact_v1 = DocumentArtifact.create(
        location="memory://knowledge-history-v1.txt",
    )
    service.artifact_store.add(artifact_v1)

    content_v1 = DocumentContent(
        id="content-v1",
        text="Contenido versión 1",
        content_type="text/plain",
    )
    service.content_store.add(content_v1)

    version_v1 = DocumentContentVersion.create(
        document_id=document.id,
        content_id=content_v1.id,
        version=1,
        content_type="text/plain",
    )
    service.content_version_store.add(version_v1)

    artifact_v2 = DocumentArtifact.create(
        location="memory://knowledge-history-v2.txt",
    )
    service.artifact_store.add(artifact_v2)

    content_v2 = DocumentContent(
        id="content-v2",
        text="Contenido versión 2",
        content_type="text/plain",
    )
    service.content_store.add(content_v2)

    version_v2 = DocumentContentVersion.create(
        document_id=document.id,
        content_id=content_v2.id,
        version=2,
        content_type="text/plain",
    )
    service.content_version_store.add(version_v2)

    extraction = service.extract_version_knowledge(
    document_id=document.id,
    from_version=1,
    to_version=2,
    )

    history = service.list_knowledge_history(document.id)

    assert len(history) == len(extraction.knowledge)

    assert all(
        record.document_id == document.id
        for record in history
    )

    assert all(
        record.content_version_id == version_v2.id
        for record in history
    )

    assert [
        record.knowledge
        for record in history
    ] == list(extraction.knowledge)

def test_refresh_version_knowledge_removes_stale_records(
    tmp_path,
):
    source = tmp_path / "refresh-stale.txt"
    source.write_text(
        "Contenido inicial.",
        encoding="utf-8",
    )

    service = build_service()
    document = create_document(service, source)

    service.ingest(document)

    source.write_text(
        "Vigencia: 24 meses.",
        encoding="utf-8",
    )

    service.ingest(document)

    extraction = service.extract_version_knowledge(
        document_id=document.id,
        from_version=1,
        to_version=2,
    )

    assert extraction.knowledge

    records_before = service.list_knowledge_for_version(
        document.id,
        2,
    )

    assert len(records_before) == len(
        extraction.knowledge
    )

    version = service.content_version_store.get_latest_for_document(
        document.id
    )

    assert version is not None

    stale_record = DocumentKnowledgeRecord.create(
        document_id=document.id,
        content_version_id=version.id,
        knowledge=extraction.knowledge[0],
    )

    service.knowledge_record_store.add(
        stale_record
    )

    assert len(
        service.list_knowledge_for_version(
            document.id,
            2,
        )
    ) == len(records_before) + 1

    refreshed = service.refresh_version_knowledge(
        document_id=document.id,
        from_version=1,
        to_version=2,
    )

    records_after = service.list_knowledge_for_version(
        document.id,
        2,
    )

    assert len(records_after) == len(
        refreshed.knowledge
    )

    assert records_after == [
        record
        for record in records_after
        if record.knowledge in refreshed.knowledge
    ]
