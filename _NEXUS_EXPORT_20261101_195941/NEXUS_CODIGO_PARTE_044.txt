import { useState, useEffect } from "react"
import axios from "axios"

const API = "http://192.168.0.15:8000" // ← poné tu IP real

export default function App() {

  const [msg, setMsg] = useState("")
  const [chat, setChat] = useState([])
  const [stats, setStats] = useState({})

  const send = async (text) => {

    const t = text || msg
    if (!t) return

    setChat(c => [...c, { role: "user", text: t }])
    setMsg("")

    const res = await axios.post(API + "/ask", { message: t }, {
      headers: { "x-api-key": "nexus-secret" }
    })

    setChat(c => [...c, { role: "assistant", text: res.data.result }])
  }

  const loadStats = async () => {
    const s = await axios.get(API + "/dashboard/stats", {
      headers: { "x-api-key": "nexus-secret" }
    })
    setStats(s.data)
  }

  useEffect(() => {
    loadStats()
    const i = setInterval(loadStats, 3000)
    return () => clearInterval(i)
  }, [])

  return (
    <div style={styles.page}>

      <h2>Nexus Mobile</h2>

      {/* Stats */}
      <div style={styles.stats}>
        <Card label="CPU" value={stats.cpu + "%"} />
        <Card label="RAM" value={stats.ram + "%"} />
      </div>

      {/* Chat */}
      <div style={styles.chat}>
        {chat.map((m,i)=>(
          <div key={i} style={{
            ...styles.bubble,
            alignSelf: m.role === "user" ? "flex-end" : "flex-start",
            background: m.role === "user" ? "#6c5ce7" : "#2d3436"
          }}>
            {m.text}
          </div>
        ))}
      </div>

      {/* Quick actions */}
      <div style={styles.quick}>
        <button onClick={()=>send("ordená descargas")}>Ordenar</button>
        <button onClick={()=>send("abrí chrome")}>Chrome</button>
        <button onClick={()=>send("cerrá esta app")}>Cerrar app</button>
      </div>

      {/* Input */}
      <div style={styles.row}>
        <input
          value={msg}
          onChange={e=>setMsg(e.target.value)}
          placeholder="Decile algo a Nexus..."
          style={styles.input}
        />
        <button onClick={()=>send()}>Enviar</button>
      </div>

    </div>
  )
}

function Card({label,value}){
  return (
    <div style={styles.card}>
      <div>{label}</div>
      <strong>{value}</strong>
    </div>
  )
}

from __future__ import annotations

from abc import ABC, abstractmethod
from typing import Any

from planner.execution_context import ExecutionContext
from planner.plan import PlanStep


class Action(ABC):
    """
    Clase base para todas las Actions de NEXUS.
    """

    name: str

    description: str = ""

    parameters: dict[str, Any] = {}

    internal: bool = False

    produces: list[str] = []

    consumes: list[str] = []

    @abstractmethod
    async def execute(
        self,
        context: ExecutionContext,
        step: PlanStep,
    ) -> None:
        ...

from __future__ import annotations

from actions.base import Action

from planner.execution_context import (
    ExecutionContext,
)

from planner.plan import PlanStep

from adapters.windows.process_manager import (
    WindowsProcessManager,
)


class CloseAppAction(Action):

    name = "system.close_app"

    def __init__(self):

        self._manager = WindowsProcessManager()

    async def execute(

        self,

        context: ExecutionContext,

        step: PlanStep,

    ) -> None:

        application = step.metadata.get(
            "application"
        )

        if not application:

            context.add_error(
                "Application not specified."
            )

            return

        success = self._manager.close(
            application
        )

        if not success:

            context.add_error(

                f"{application} is not running."

            )

            return

        context.add_result(

            "application",

            application,

        )

        context.add_result(

            "status",

            "closed",

        )

from __future__ import annotations

from planner.plan import PlanStep
from planner.execution_context import ExecutionContext

from actions.base import Action

from memory.episode_memory import EpisodeMemory


class GetRecentEpisodesAction(Action):
    """
    Action para consultar la memoria episódica reciente
    de NEXUS.

    No interpreta intenciones.
    No ejecuta acciones del sistema.
    No genera lenguaje natural.

    Devuelve únicamente episodios estructurados.
    """

    name = "system.get_recent_episodes"

    description = (
        "Consulta las acciones realizadas "
        "recientemente por NEXUS."
    )

    parameters = {
        "limit": {
            "type": "integer",
            "required": False,
            "description": (
                "Cantidad de episodios recientes "
                "a devolver."
            ),
        },
    }

    internal = True

    def __init__(
        self,
        episode_memory: EpisodeMemory,
    ) -> None:

        self._episode_memory = episode_memory

    async def execute(
        self,
        context: ExecutionContext,
        step: PlanStep,
    ) -> None:

        limit = step.metadata.get(
            "limit",
            5,
        )

        #
        # Validar limit
        #

        if isinstance(
            limit,
            bool,
        ):

            context.add_error(
                "Episode limit must be an integer."
            )

            return

        try:

            limit = int(limit)

        except (
            TypeError,
            ValueError,
        ):

            context.add_error(
                "Episode limit must be an integer."
            )

            return

        #
        # Rango razonable
        #

        if limit <= 0:

            context.add_error(
                "Episode limit must be greater than zero."
            )

            return

        #
        # Evitamos consultas excesivamente grandes.
        #

        limit = min(
            limit,
            20,
        )

        #
        # Obtener episodios
        #

        episodes = (
            self._episode_memory.recent(
                limit=limit,
            )
        )

        #
        # Convertir a estructuras serializables
        #

        items = [
            episode.model_dump()
            for episode in episodes
        ]

        #
        # Resultado
        #

        context.add_result(
            "episodes",
            items,
        )

        context.add_result(
            "count",
            len(items),
        )

        context.add_result(
            "limit",
            limit,
        )

        context.add_result(
            "status",
            "read",
        )

from __future__ import annotations

from planner.plan import PlanStep
from planner.execution_context import ExecutionContext

from actions.base import Action

from memory.state_history import StateHistory


class GetVolumeHistoryAction(Action):
    """
    Action para consultar el historial del volumen.

    No genera lenguaje natural.
    Devuelve únicamente datos estructurados
    para que las capas superiores puedan decidir
    cómo presentar el resultado.
    """

    name = "system.get_volume_history"

    internal = True

    description = (
        "Consulta el historial del volumen."
    )

    parameters = {
        "mode": {
            "type": "string",
            "required": True,
            "description": (
                "Modo de consulta: "
                "previous o find."
            ),
        },
        "value": {
            "type": "number",
            "required": False,
            "description": (
                "Valor de volumen a buscar "
                "cuando el modo es find."
            ),
        },
    }

    def __init__(
        self,
        state_history: StateHistory,
    ) -> None:

        self._state_history = state_history

    async def execute(
        self,
        context: ExecutionContext,
        step: PlanStep,
    ) -> None:

        mode = step.metadata.get(
            "mode"
        )

        #
        # Volumen anterior
        #

        if mode == "previous":

            value = self._state_history.previous(
                "volume"
            )

            context.add_result(
                "found",
                value is not None,
            )

            context.add_result(
                "value",
                value,
            )

            context.add_result(
                "mode",
                "previous",
            )

            return

        #
        # Buscar un valor concreto
        #
        # Ejemplo:
        #
        # ¿Qué dejé en 50%?
        #

        if mode == "find":

            requested_value = step.metadata.get(
                "value"
            )

            events = self._state_history.find_events(
                "volume",
                requested_value,
                status="set",
            )

            context.add_result(
                "found",
                bool(events),
            )

            context.add_result(
                "value",
                requested_value
                if events
                else None,
            )

            context.add_result(
                "mode",
                "find",
            )

            return

        #
        # Modo desconocido
        #

        context.add_error(
            "Unknown volume history mode."
        )

from __future__ import annotations

from planner.execution_context import ExecutionContext
from planner.objective import ObjectiveStatus
from planner.plan import StepStatus

from actions.registry import ActionRegistry
from security import ActionAuthorization, Actor, ActorType


class ActionManager:
    """
    Ejecuta Actions a partir del ExecutionContext.

    El ActionManager coordina la ejecución del paso
    actual y mantiene sincronizado el estado del
    Objective con el resultado real de la Action.

    También registra la evidencia producida
    específicamente por cada PlanStep.

    La autorización, cuando está configurada,
    se comprueba antes de ejecutar cada Action.

    No decide cuándo finaliza globalmente el Plan.
    Esa responsabilidad pertenece al Executor.
    """

    def __init__(
        self,
        registry: ActionRegistry,
        action_authorization: ActionAuthorization | None = None,
    ) -> None:
        self._registry = registry
        self._action_authorization = action_authorization

    async def execute(
        self,
        context: ExecutionContext,
    ) -> None:
        """
        Ejecuta el paso actual del plan.

        Si Security está configurado, la acción debe
        estar autorizada antes de llegar a execute().
        """

        step = context.current_plan_step()

        if step is None:
            return

        #
        # Paso en ejecución.
        #

        step.status = StepStatus.RUNNING
        context.objective.status = ObjectiveStatus.RUNNING

        #
        # Capturamos el estado previo para registrar
        # únicamente la evidencia producida por este paso.
        #

        results_before = context.results.copy()
        artifacts_before = context.artifacts.copy()

        #
        # Resolver Action.
        #

        try:
            action = self._registry.get(step.action)

        except Exception as exc:
            error = (
                f"Failed to resolve action "
                f"'{step.action}': {exc}"
            )

            step.status = StepStatus.FAILED
            context.add_error(error)

            context.record_step_evidence(
                step,
                status="failed",
                results_before=results_before,
                artifacts_before=artifacts_before,
            )

            context.objective.status = ObjectiveStatus.FAILED
            context.finished = True
            context.plan.completed = False
            return

        #
        # Autorizar antes de ejecutar la Action.
        #

        if self._action_authorization is not None:
            actor = None

            if context.actor_id and context.actor_type:
                try:
                    actor = Actor(
                        id=context.actor_id,
                        actor_type=ActorType(context.actor_type),
                    )
                except (ValueError, TypeError):
                    actor = None

            allowed = (
                actor is not None
                and self._action_authorization.is_allowed(
                    actor,
                    step.action,
                )
            )

            if not allowed:
                actor_description = (
                    context.actor_id
                    if context.actor_id
                    else "unknown"
                )

                error = (
                    "Authorization denied: actor "
                    f"'{actor_description}' cannot execute "
                    f"action '{step.action}'."
                )

                step.status = StepStatus.FAILED
                context.add_error(error)

                context.record_step_evidence(
                    step,
                    status="failed",
                    results_before=results_before,
                    artifacts_before=artifacts_before,
                )

                context.step_evidence[step.id]["authorization"] = {
                    "allowed": False,
                    "actor_id": context.actor_id,
                    "action": step.action,
                }

                context.objective.status = ObjectiveStatus.FAILED
                context.finished = True
                context.plan.completed = False
                return

        #
        # Ejecutar Action.
        #

        try:
            await action.execute(
                context,
                step,
            )

        except Exception as exc:
            error = (
                f"Action '{step.action}' "
                f"failed: {exc}"
            )

            step.status = StepStatus.FAILED
            context.add_error(error)

            context.record_step_evidence(
                step,
                status="failed",
                results_before=results_before,
                artifacts_before=artifacts_before,
            )

            context.objective.status = ObjectiveStatus.FAILED
            context.finished = True
            context.plan.completed = False
            return

        #
        # La Action puede haber capturado una excepción
        # internamente y registrado un error en el contexto.
        #

        if context.has_errors():
            step.status = StepStatus.FAILED

            context.record_step_evidence(
                step,
                status="failed",
                results_before=results_before,
                artifacts_before=artifacts_before,
            )

            context.objective.status = ObjectiveStatus.FAILED
            context.finished = True
            context.plan.completed = False
            return

        #
        # Registrar evidencia real producida por el Step.
        #

        context.record_step_evidence(
            step,
            status="completed",
            results_before=results_before,
            artifacts_before=artifacts_before,
        )

        #
        # La Action terminó correctamente.
        # El Executor determina la finalización global.
        #

        step.status = StepStatus.COMPLETED

from __future__ import annotations

from planner.plan import PlanStep
from planner.execution_context import ExecutionContext

from actions.base import Action

from providers.provider_manager import ProviderManager


class OpenAppAction(Action):
    """
    Action para abrir aplicaciones
    del sistema operativo.

    La Action no conoce directamente
    los adapters de Windows.

    Delega la ejecución al ProviderManager.
    """

    name = "system.open_app"

    def __init__(
        self,
        provider_manager: ProviderManager,
    ) -> None:

        self._provider_manager = provider_manager

    async def execute(
        self,
        context: ExecutionContext,
        step: PlanStep,
    ) -> None:

        application = step.metadata.get(
            "application"
        )

        if not application:

            context.add_error(
                "No application specified."
            )

            return

        try:

            success = await self._provider_manager.execute(
                provider="local",
                tool="windows.open_app",
                parameters={
                    "application": application,
                },
            )

        except Exception as exc:

            context.add_error(
                f"Failed to open application "
                f"'{application}': {exc}"
            )

            return

        if not success:

            context.add_error(
                f"Application '{application}' "
                f"is not supported."
            )

            return

        context.add_result(
            "application",
            application,
        )

        context.add_result(
            "status",
            "opened",
        )

from __future__ import annotations
from actions.base import Action
from planner.execution_context import ExecutionContext
from planner.plan import PlanStep
from providers.provider_manager import ProviderManager

class OpenFileAction(Action):
    """Abre un archivo existente con la aplicación predeterminada."""
    name = "system.open_file"
    description = "Busca y abre un archivo en una carpeta personal autorizada."
    parameters = {
        "file_name": {"type": "string", "required": True, "description": "Nombre del archivo, sin ruta."},
        "location": {"type": "string", "required": True, "description": "desktop, documents o downloads."},
        "extension": {"type": "string", "required": False, "description": "Extensión opcional."},
    }

    def __init__(self, provider_manager: ProviderManager) -> None:
        self._provider_manager = provider_manager

    async def execute(self, context: ExecutionContext, step: PlanStep) -> None:
        file_name = step.metadata.get("file_name")
        location = step.metadata.get("location", "desktop")
        extension = step.metadata.get("extension")
        if not isinstance(file_name, str) or not file_name.strip():
            context.add_error("No se especificó el nombre del archivo.")
            return
        try:
            success = await self._provider_manager.execute(
                provider="local", tool="windows.open_file",
                parameters={"file_name": file_name, "location": location, "extension": extension},
            )
        except Exception as exc:
            context.add_error(f"No se pudo abrir el archivo '{file_name}': {exc}")
            return
        if not success:
            context.add_error(
                f"No se encontró una coincidencia única para '{file_name}' "
                f"en '{location}', o Windows no pudo abrirla."
            )
            return
        context.add_result("file_name", file_name)
        context.add_result("location", location)
        context.add_result("status", "opened")
