from documents import (
    DocumentArtifact,
    DocumentContent,
    IngestionRegistry,
    IngestionResult,
)

from documents import DocumentType


class FakeIngestor:
    def __init__(self, label: str) -> None:
        self.label = label

    def ingest(self, artifact: DocumentArtifact) -> IngestionResult:
        return IngestionResult(
            artifact=artifact,
            content=DocumentContent.create(
                text=f"Content from {self.label}",
            ),
        )


def test_register_and_get_ingestor():
    registry = IngestionRegistry()
    ingestor = FakeIngestor("text")

    registry.register("text/plain", ingestor)

    assert registry.get("text/plain") is ingestor


def test_get_unknown_ingestor_returns_none():
    registry = IngestionRegistry()

    assert registry.get("application/pdf") is None


def test_supports_content_type():
    registry = IngestionRegistry()
    ingestor = FakeIngestor("text")

    registry.register("text/plain", ingestor)

    assert registry.supports("text/plain")
    assert not registry.supports("application/pdf")


def test_register_replaces_existing_ingestor():
    registry = IngestionRegistry()

    first = FakeIngestor("first")
    second = FakeIngestor("second")

    registry.register("text/plain", first)
    registry.register("text/plain", second)

    assert registry.get("text/plain") is second


def test_remove_ingestor():
    registry = IngestionRegistry()
    ingestor = FakeIngestor("text")

    registry.register("text/plain", ingestor)

    result = registry.remove("text/plain")

    assert result is ingestor
    assert registry.get("text/plain") is None


def test_remove_unknown_ingestor_returns_none():
    registry = IngestionRegistry()

    assert registry.remove("text/plain") is None


def test_content_types():
    registry = IngestionRegistry()

    registry.register("text/plain", FakeIngestor("text"))
    registry.register("text/markdown", FakeIngestor("markdown"))

    assert registry.content_types() == [
        "text/plain",
        "text/markdown",
    ]


def test_clear_registry():
    registry = IngestionRegistry()

    registry.register("text/plain", FakeIngestor("text"))
    registry.register("text/markdown", FakeIngestor("markdown"))

    registry.clear()

    assert registry.content_types() == []
    assert not registry.supports("text/plain")

import pytest

from documents import (
    DocumentArtifact,
    DocumentIngestionService,
    IngestionRegistry,
    LocalArtifactReader,
    TextIngestor,
)


def build_service() -> DocumentIngestionService:
    registry = IngestionRegistry()

    registry.register(
        "text/plain",
        TextIngestor(LocalArtifactReader()),
    )

    return DocumentIngestionService(registry)


def test_service_ingests_registered_content_type(tmp_path):
    file_path = tmp_path / "document.txt"
    file_path.write_text(
        "Contenido del documento.",
        encoding="utf-8",
    )

    artifact = DocumentArtifact.create(
        location=f"local:{file_path}",
    )

    service = build_service()

    result = service.ingest(
        artifact=artifact,
        content_type="text/plain",
    )

    assert result.artifact is artifact
    assert result.content.text == "Contenido del documento."
    assert result.content.content_type == "text/plain"


def test_service_rejects_unsupported_content_type(tmp_path):
    file_path = tmp_path / "document.txt"
    file_path.write_text(
        "Contenido.",
        encoding="utf-8",
    )

    artifact = DocumentArtifact.create(
        location=f"local:{file_path}",
    )

    service = build_service()

    with pytest.raises(
        ValueError,
        match="No ingestor registered for content type: application/pdf",
    ):
        service.ingest(
            artifact=artifact,
            content_type="application/pdf",
        )


def test_service_uses_registry_selected_ingestor(tmp_path):
    file_path = tmp_path / "notes.txt"
    file_path.write_text(
        "Notas de NEXUS.",
        encoding="utf-8",
    )

    artifact = DocumentArtifact.create(
        location=f"local:{file_path}",
    )

    service = build_service()

    result = service.ingest(
        artifact=artifact,
        content_type="text/plain",
    )

    assert result.artifact.id == artifact.id
    assert result.content.text == "Notas de NEXUS."

from documents import (
    DocumentChange,
    DocumentKnowledge,
    DocumentKnowledgeExtraction,
    DocumentSemanticAnalysis,
    DocumentSemanticChange,
    DocumentVersionChangeResult,
    DocumentVersionComparison,
)


def build_analysis() -> DocumentSemanticAnalysis:
    comparison = DocumentVersionComparison(
        document_id="doc-1",
        from_version=1,
        to_version=2,
        from_content_id="content-1",
        to_content_id="content-2",
        changed=True,
    )

    source_result = DocumentVersionChangeResult(
        comparison=comparison,
        changes=(
            DocumentChange.modified(
                "Vigencia: 12 meses.",
                "Vigencia: 24 meses.",
                before_line=1,
                after_line=1,
            ),
            DocumentChange.added(
                "Nueva cláusula.",
                line=3,
            ),
        ),
    )

    semantic_changes = (
        DocumentSemanticChange(
            source_change=source_result.changes[0],
            meaning="Se modificó la vigencia.",
            topics=("vigencia",),
            confidence=0.9,
        ),
        DocumentSemanticChange(
            source_change=source_result.changes[1],
            meaning="Se agregó una nueva cláusula.",
            topics=("contrato",),
            confidence=1.0,
        ),
    )

    return DocumentSemanticAnalysis(
        source=source_result,
        changes=semantic_changes,
    )


class FakeKnowledgeExtractor:
    def extract(
        self,
        change: DocumentSemanticChange,
    ) -> tuple[DocumentKnowledge, ...]:
        return (
            DocumentKnowledge(
                source_change=change,
                statement=change.meaning,
                topic=change.topics[0] if change.topics else None,
                confidence=change.confidence,
            ),
        )


def test_knowledge_extraction_preserves_source_analysis():
    analysis = build_analysis()

    extraction = DocumentKnowledgeExtraction(
        source=analysis,
        knowledge=(),
    )

    assert extraction.source == analysis


def test_knowledge_extraction_contains_knowledge():

        ),

    extraction = DocumentKnowledgeExtraction(

    )

    assert len(extraction.knowledge) == 1

def build_record(
    document_id: str = "doc-1",
    content_version_id: str = "version-1",
) -> DocumentKnowledgeRecord:
    change = DocumentChange.modified(
        "Vigencia: 12 meses.",
        "Vigencia: 24 meses.",
        before_line=1,
        after_line=1,
    )

    semantic_change = DocumentSemanticChange(
        source_change=change,
        meaning="Se modificó la vigencia.",
        topics=("vigencia",),
        confidence=0.9,
    )

    knowledge = DocumentKnowledge(
        source_change=semantic_change,
        statement="La vigencia cambió.",
        topic="vigencia",
        confidence=0.9,
    )

    return DocumentKnowledgeRecord.create(
        document_id=document_id,
        content_version_id=content_version_id,
        knowledge=knowledge,
    )

    store = DocumentKnowledgeRecordStore()
