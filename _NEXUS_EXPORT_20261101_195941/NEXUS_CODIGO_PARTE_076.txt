class FakeToolRegistry:
    def describe(self):
        return [
            {
                "name": "windows.open_app",
                "action": "system.open_app",
                "description": "Open application",
                "provider": "windows",
                "parameters": {
                    "application": {
                        "type": "string",
                        "required": True,
                    }
                },
            },
            {
                "name": "windows.close_app",
                "action": "system.close_app",
                "description": "Close application",
                "provider": "windows",
                "parameters": {
                    "application": {
                        "type": "string",
                        "required": True,
                    }
                },
            },
        ]

    def resolve(self, action):
        return object()


class FakeAction:
    def __init__(self, name):
        self.name = name
        self.internal = False


class FakeActionRegistry:
    def get(self, action):
        return FakeAction(action)


class FakeRuntime:
    """
    Runtime mínimo para probar el ciclo cognitivo completo.

    Reproduce la evidencia estructural que el Runtime real
    entrega al Verifier después de una ejecución.
    """

    def __init__(self):
        self.tool_registry = FakeToolRegistry()
        self.action_registry = FakeActionRegistry()
        self.executions = []

    async def execute(self, context):
        self.executions.append(
            context.objective.metadata.copy()
        )

        application = (
            context.objective.metadata.get(
                "application"
            )
        )

        action = context.objective.title

        if action == "system.open_app":
            output = {
                "status": "opened",
                "application": application,
            }
        elif action == "system.close_app":
            output = {
                "status": "closed",
                "application": application,
            }
        else:
            output = {
                "status": "completed",
                "action": action,
            }

        for step in context.plan.steps:
            step.status = StepStatus.COMPLETED

            step.metadata["execution"] = {
                "status": "completed",
                "result_keys": list(
                    output.keys()
                ),
                "artifact_keys": [],
            }

        context.plan.completed = True

        return ExecutionResult(
            success=True,
            message="",
            output=output,
            artifacts={},
            metadata={},
        )


class RecordingIntentProvider(IntentProvider):
    """
    Devuelve intents predefinidos y conserva todos los contextos
    que recibe para comprobar qué conocimiento llega al segundo ciclo.
    """

    def __init__(self, intents):
        self.intents = list(intents)
        self.calls = []

    async def recognize(
        self,
        text,
        capabilities=None,
        context=None,
    ):
        self.calls.append(
            {
                "text": text,
                "capabilities": capabilities,
                "context": context,
            }
        )

        if not self.intents:
            raise AssertionError(
                "No more fake intents are available."
            )

        return self.intents.pop(0)


def run(coro):
    return asyncio.run(coro)


def check(condition, message):
    if not condition:
        raise AssertionError(message)


def test_closed_cognitive_cycle():
    """
    Demuestra el ciclo:

        REQUEST 1
            ↓
        EXECUTE
            ↓
        VERIFY
            ↓
        LEARN
            ↓
        SEMANTIC MEMORY
            ↓
        REQUEST 2
            ↓
        CONTEXT
            ↓
        INTENT

    La segunda solicitud debe recibir conocimiento
    derivado de la primera experiencia verificada.
    """

    first_intent = Intent(
        action="system.open_app",
        confidence=1.0,
        parameters={
            "application": "Chrome",
        },
        original_text="Abrí Chrome",
    )

    second_intent = Intent(
        action="system.close_app",
        confidence=1.0,
        parameters={
            "application": "Chrome",
        },
        original_text="Cerrá la aplicación",
    )

    runtime = FakeRuntime()

    intent_provider = RecordingIntentProvider(
        [
            first_intent,
            second_intent,
        ]
    )

    engine = Engine(
        runtime=runtime,
        intent_provider=intent_provider,
        memory=WorkingMemory(),
    )

    #
    # ---------------------------------------------------------
    # CICLO 1
    # ---------------------------------------------------------
    #

    first_result = run(
        engine.execute("Abrí Chrome")
    )

    check(
        first_result.success,
        "First execution should succeed.",
    )

    check(
        first_result.metadata["verification"]["verified"],
        "First execution must be verified.",
    )

    #
    # ---------------------------------------------------------
    # LEARNING
    # ---------------------------------------------------------
    #

    semantic_fact = engine.semantic_memory.get(
        "chrome",
        "es una aplicación",
    )

    check(
        semantic_fact is not None,
        "Verified execution must create semantic knowledge.",
    )

    check(
        semantic_fact.value is True,
        "Learned application fact must be true.",
    )

    check(
        semantic_fact.source == "episode",
        "Semantic knowledge must preserve episode provenance.",
    )

    #
    # ---------------------------------------------------------
    # EPISODIC MEMORY
    # ---------------------------------------------------------
    #

    episodes = engine.memory.find(
        "episode"
    )

    check(
        len(episodes) >= 1,
        "Verified execution must reach episodic memory.",
    )

    #
    # ---------------------------------------------------------
    # CICLO 2
    # ---------------------------------------------------------
    #

    second_result = run(
        engine.execute("Cerrá la aplicación")
    )

    check(
        second_result.success,
        "Second execution should succeed.",
    )

    check(
        second_result.metadata["verification"]["verified"],
        "Second execution must also be verified.",
    )

    #
    # ---------------------------------------------------------
    # CONTEXT RECEIVED BY INTENT PROVIDER
    # ---------------------------------------------------------
    #

    check(
        len(intent_provider.calls) == 2,
        "IntentProvider must have received two requests.",
    )

    second_call = intent_provider.calls[1]

    second_context = second_call["context"]

    check(
        isinstance(second_context, dict),
        "Second request must receive a cognitive context.",
    )

    #
    # The semantic fact learned from the first execution
    # must be available to the second cognitive cycle.
    #
    semantic_knowledge = second_context.get(
        "semantic_knowledge"
    )

    check(
        semantic_knowledge is not None,
        (
            "Second cognitive cycle must receive "
            "semantic knowledge from the first cycle."
        ),
    )

    #
    # Accept either a single fact representation or a list
    # of facts, while keeping the assertion deterministic.
    #
    if isinstance(
        semantic_knowledge,
        list,
    ):
        matching_fact = any(
            isinstance(item, dict)
            and item.get("subject") == "chrome"
            and item.get("predicate")
            == "es una aplicación"
            and item.get("value") is True
            for item in semantic_knowledge
        )
    elif isinstance(
        semantic_knowledge,
        dict,
    ):
        matching_fact = (
            semantic_knowledge.get("subject")
            == "chrome"
            and semantic_knowledge.get("predicate")
            == "es una aplicación"
            and semantic_knowledge.get("value")
            is True
        )
    else:
        matching_fact = False

    check(
        matching_fact,
        (
            "Second cognitive cycle must receive "
            "the learned Chrome semantic fact."
        ),
    )

    #
    # ---------------------------------------------------------
    # NEXT DECISION
    # ---------------------------------------------------------
    #

    decision_entries = engine.memory.find(
        "decision"
    )

    check(
        len(decision_entries) == 2,
        "Both requests must reach the DecisionEngine.",
    )

    check(
        decision_entries[0].data[
            "selected_action"
        ]
        == "system.open_app",
        "First decision must select system.open_app.",
    )

    check(
        decision_entries[1].data[
            "selected_action"
        ]
        == "system.close_app",
        "Second decision must select system.close_app.",
    )

    #
    # ---------------------------------------------------------
    # FINAL ASSERTION
    # ---------------------------------------------------------
    #

    print(
        "CLOSED COGNITIVE CYCLE"
    )
    print(
        "============================================================"
    )
    print(
        "PASSED : 1"
    )
    print(
        "FAILED : 0"
    )


if __name__ == "__main__":
    try:
        test_closed_cognitive_cycle()
    except Exception as exc:
        print(
            "CLOSED COGNITIVE CYCLE"
        )
        print(
            "============================================================"
        )
        print(
            "PASSED : 0"
        )
        print(
            "FAILED : 1"
        )
        print(
            f"ERROR  : {exc}"
        )
        raise

from __future__ import annotations

from planner.intent import Intent
from planner.objective import (
    Objective,
    ObjectiveStatus,
)
from planner.plan import (
    StepStatus,
)
from planner.planner import Planner


class FakeTool:

    def __init__(
        self,
        *,
        produces=None,
        consumes=None,
    ):
        self.produces = list(
            produces or []
        )
        self.consumes = list(
            consumes or []
        )


class FakeToolRegistry:

    def __init__(self, tools):
        self.tools = tools

    def resolve(self, action):
        return self.tools.get(action)


def make_objective(
    objective_id="objective-1",
):
    return Objective(
        id=objective_id,
        title="test",
        description="test objective",
    )


def test_simple_plan_remains_compatible():

    registry = FakeToolRegistry(
        {
            "system.open_app": FakeTool(
                produces=["application_opened"]
            )
        }
    )

    planner = Planner(
        tool_resolver=registry.resolve
    )

    objective = make_objective()

    intent = Intent(
        action="system.open_app",
        original_text="abrir chrome",
        parameters={
            "application": "chrome",
        },
    )

    plan = planner.create_plan(
        objective,
        intent,
    )

    assert len(plan.steps) == 1
    assert (
        plan.steps[0].action
        == "system.open_app"
    )
    assert (
        plan.steps[0].depends_on
        == []
    )
    assert (
        objective.status
        == ObjectiveStatus.READY
    )


def test_compound_plan_creates_dependency_when_producer_comes_first():

    registry = FakeToolRegistry(
        {
            "system.capture": FakeTool(
                produces=["image"]
            ),
            "system.inspect": FakeTool(
                consumes=["image"],
                produces=["inspection"],
            ),
        }
    )

    planner = Planner(
        tool_resolver=registry.resolve
    )

    objective = make_objective()

    capture = Intent(
        action="system.capture",
        original_text="capturar pantalla",
    )

    inspect_capture = Intent(
        action="system.inspect",
        original_text="inspeccionar captura",
    )

    intent = Intent(
        action=capture.action,
        original_text=(
            "capturar pantalla y "
            "inspeccionar captura"
        ),
        sub_intents=[
            capture,
            inspect_capture,
        ],
    )

    plan = planner.create_plan(
        objective,
        intent,
    )

    assert len(plan.steps) == 2

    producer = plan.steps[0]
    consumer = plan.steps[1]

    assert producer.id not in []
    assert (
    producer.id
    in consumer.depends_on
)

            ),
            "system.consume": FakeTool(

        }

        tool_resolver=registry.resolve
    )

    objective = make_objective()

    producer = Intent(
        action="system.produce",
        original_text="producir dato",
    )

    consumer = Intent(
        action="system.consume",
        original_text="usar dato",

    intent = Intent(
        action=producer.action,

    plan = planner.create_plan(
        objective,

import asyncio

from planner.execution_context import ExecutionContext
from planner.objective import Objective, ObjectiveStatus
from planner.plan import ExecutionPlan, PlanStep, StepStatus
from runtime.executor import Executor


class FakeActionManager:
    """
    ActionManager mínimo para probar el Executor real.

    No reemplaza la lógica del Executor:
    solamente ejecuta el step que éste selecciona.
    """

    def __init__(self):
        self.execution_order = []

    async def execute(self, context):
        step = context.plan.steps[context.current_step]

        self.execution_order.append(step.id)

        step.status = StepStatus.COMPLETED

        # Registrar el resultado producido por el step.
        for artifact in step.produces:
            context.artifacts[artifact] = {
                "produced_by": step.id
            }

        context.results[step.id] = {
            "status": "completed"
        }


class FakeRuntime:
    def __init__(self):
        self.action_manager = FakeActionManager()


def make_step(
    step_id,
    name,
    depends_on=None,
    produces=None,
    consumes=None,
):
    return PlanStep(
        id=step_id,
        name=name,
        description=name,
        action=name,
        status=StepStatus.PENDING,
        depends_on=depends_on or [],
        produces=produces or [],
        consumes=consumes or [],
    )


def make_context(steps):
    objective = Objective(
        id="objective-test",
        title="Cognitive Planning Execution Test",
        status=ObjectiveStatus.READY,
        priority=1,
        metadata={},
    )

    plan = ExecutionPlan(
        objective_id=objective.id,
        steps=steps,
        current_step=0,
        completed=False,
        metadata={},
    )

    return ExecutionContext(
        objective=objective,
        plan=plan,
        variables={},
        artifacts={},
        results={},
        errors=[],
        current_step=0,
        finished=False,
        step_evidence={},
    )


def run_executor(context, runtime):
    executor = Executor(runtime)
    return asyncio.run(
        executor.execute(context)
    )


def test_simple_plan_executes():
    runtime = FakeRuntime()

    context = make_context(
        [
            make_step(
                "step_a",
                "action_a",
            )
        ]
    )

    result = run_executor(
        context,
        runtime,
    )

    assert result.success is True
    assert context.plan.completed is True
    assert context.objective.status == ObjectiveStatus.COMPLETED

    assert (
        context.plan.steps[0].status
        == StepStatus.COMPLETED
    )

    assert (
        runtime.action_manager.execution_order
        == ["step_a"]
    )

def test_branching_dag_executes_all_branches():
    runtime = FakeRuntime()

    context = make_context(
        [
            make_step(
                "root",
                "root",
                produces=["root_artifact"],
            ),
            make_step(
                "branch_a",
                "branch_a",
                depends_on=["root"],
                consumes=["root_artifact"],
                produces=["artifact_a"],
            ),
            make_step(
                "branch_b",
                "branch_b",
                depends_on=["root"],
                consumes=["root_artifact"],
                produces=["artifact_b"],
            ),
            make_step(
                "join",
                "join",
                depends_on=[
                    "branch_a",
                    "branch_b",
                ],
                consumes=[
                    "artifact_a",
                    "artifact_b",
                ],
            ),
        ]
    )

    result = run_executor(
        context,
        runtime,
    )

    assert result.success is True
    assert context.plan.completed is True

    order = runtime.action_manager.execution_order

    assert order[0] == "root"

    assert (
        set(order[1:3])
        == {
            "branch_a",
            "branch_b",
        }
    )

    assert order[3] == "join"

import asyncio

from planner.execution_context import ExecutionContext
from planner.objective import Objective, ObjectiveStatus
from planner.plan import ExecutionPlan, PlanStep, StepStatus
from runtime.executor import Executor


class FakeActionManager:
    """
    ActionManager de prueba.

    Permite definir qué steps fallan y reproduce el contrato
    que utiliza el Executor real: execute(context).
    """

    def __init__(self, failing_steps=None):
        self.failing_steps = set(failing_steps or [])
        self.execution_order = []

    async def execute(self, context):
        step = context.plan.steps[context.current_step]

        self.execution_order.append(step.id)

        if step.id in self.failing_steps:
            step.status = StepStatus.FAILED

            context.add_error(
                f"Step '{step.id}' failed."
            )

            return

        step.status = StepStatus.COMPLETED

        for artifact in step.produces:
            context.artifacts[artifact] = {
                "produced_by": step.id
            }

        context.results[step.id] = {
            "status": "completed"
        }


class FakeRuntime:
    def __init__(self, failing_steps=None):
        self.action_manager = FakeActionManager(
            failing_steps=failing_steps
        )


def make_step(
    step_id,
    name,
    depends_on=None,
    produces=None,
    consumes=None,
):
    return PlanStep(
        id=step_id,
        name=name,
        description=name,
        action=name,
        status=StepStatus.PENDING,
        depends_on=depends_on or [],
        consumes=consumes or [],
        produces=produces or [],
    )


def make_context(steps):
    objective = Objective(
        id="objective-failure-test",
        title="Failure Propagation Test",
        status=ObjectiveStatus.READY,
        priority=1,
        metadata={},
    )

    plan = ExecutionPlan(
        objective_id=objective.id,
        steps=steps,
        current_step=0,
        completed=False,
        metadata={},
    )

    return ExecutionContext(
        objective=objective,
        plan=plan,
        variables={},
        artifacts={},
        results={},
        errors=[],
        current_step=0,
        finished=False,
        step_evidence={},
    )


def run_executor(context, runtime):
    executor = Executor(runtime)

    return asyncio.run(
        executor.execute(context)
    )


def get_step(context, step_id):
    for step in context.plan.steps:
        if step.id == step_id:
            return step

    raise AssertionError(
        f"Step '{step_id}' not found."
    )


def test_root_failure_marks_plan_as_failed():
    runtime = FakeRuntime(
        failing_steps={"root"}
    )

    context = make_context(
        [
            make_step(
                "root",
                "root",
            ),
            make_step(
                "dependent",
                "dependent",
                depends_on=["root"],
            ),
        ]
    )

    result = run_executor(
        context,
        runtime,
    )

    assert result.success is False
    assert context.plan.completed is False
    assert (
        context.objective.status
        == ObjectiveStatus.FAILED
    )

    assert (
        get_step(context, "root").status
        == StepStatus.FAILED
    )


def test_failed_intermediate_step_stops_execution():
    runtime = FakeRuntime(
        failing_steps={"middle"}
    )

    context = make_context(
        [
            make_step(
                "root",
                "root",
                produces=["artifact_a"],
            ),
            make_step(
                "middle",
                "middle",
                depends_on=["root"],
                consumes=["artifact_a"],
                produces=["artifact_b"],
            ),
            make_step(
                "final",
                "final",
                depends_on=["middle"],
                consumes=["artifact_b"],
            ),
        ]
    )

    result = run_executor(
        context,
        runtime,
    )

    assert result.success is False
    assert context.plan.completed is False

    assert (
        runtime.action_manager.execution_order
        == [
            "root",
            "middle",
        ]
    )

    assert (
        get_step(context, "root").status
        == StepStatus.COMPLETED
    )

    assert (
        get_step(context, "middle").status
        == StepStatus.FAILED
    )

    assert (
        get_step(context, "final").status
        == StepStatus.PENDING
    )


def test_failed_dependency_blocks_dependent_step():
    runtime = FakeRuntime(
        failing_steps={"producer"}
    )

    context = make_context(
        [
            make_step(
                "producer",
                "producer",
                produces=["artifact_a"],
            ),
            make_step(
                "consumer",
                "consumer",
                depends_on=["producer"],
                consumes=["artifact_a"],
            ),
        ]
    )

    result = run_executor(

    )

    assert (

    )

    assert (

    )

from __future__ import annotations

from planner.intent import Intent
from planner.objective import (
    Objective,
    ObjectiveStatus,
)
from planner.planner import Planner


class FakeTool:

    def __init__(
        self,
        *,
        produces=None,
        consumes=None,
    ):
        self.produces = list(
            produces or []
        )
        self.consumes = list(
            consumes or []
        )


class FakeToolRegistry:

    def __init__(self, tools):
        self.tools = tools

    def resolve(self, action):
        return self.tools.get(action)


def make_objective(
    objective_id="objective-1",
):
    return Objective(
        id=objective_id,
        title="test",
        description="test objective",
    )


def test_linear_dependency_chain_is_valid():

    registry = FakeToolRegistry(
        {
            "system.capture": FakeTool(
                produces=["image"],
            ),
            "system.inspect": FakeTool(
                consumes=["image"],
                produces=["inspection"],
            ),
            "system.process": FakeTool(
                consumes=["inspection"],
                produces=["result"],
            ),
        }
    )

    planner = Planner(
        tool_resolver=registry.resolve
    )

    objective = make_objective()

    capture = Intent(
        action="system.capture",
        original_text="capturar",
    )

    inspect_capture = Intent(
        action="system.inspect",
        original_text="inspeccionar",
    )

    process = Intent(
        action="system.process",
        original_text="procesar",
    )

    intent = Intent(
        action=capture.action,
        original_text="capturar, inspeccionar y procesar",
        sub_intents=[
            capture,
            inspect_capture,
            process,
        ],
    )

    plan = planner.create_plan(
        objective,
        intent,
    )

    assert (
        objective.status
        == ObjectiveStatus.READY
    )

    assert (
        plan.steps[1].depends_on
        == [plan.steps[0].id]
    )

    assert (
        plan.steps[2].depends_on
        == [plan.steps[1].id]
    )

def test_cyclic_artifact_dependencies_are_rejected():

    registry = FakeToolRegistry(
        {
            "system.a": FakeTool(
                consumes=["b"],
                produces=["a"],
            ),
            "system.b": FakeTool(
                consumes=["a"],
                produces=["b"],
            ),
        }
    )

    planner = Planner(
        tool_resolver=registry.resolve
    )

    objective = make_objective()

    first = Intent(
        action="system.a",
        original_text="ejecutar A",
    )

    second = Intent(
        action="system.b",
        original_text="ejecutar B",
    )

    intent = Intent(
        action=first.action,
        original_text="ejecutar A y B",
        sub_intents=[
            first,
            second,
        ],
    )

    plan = planner.create_plan(
        objective,
        intent,
    )

    assert (
        objective.status
        == ObjectiveStatus.FAILED
    )

    assert "error" in plan.metadata


def test_longer_dependency_graph_is_valid():

    registry = FakeToolRegistry(
        {
            "system.a": FakeTool(
                produces=["a"],
            ),
            "system.b": FakeTool(
                consumes=["a"],
                produces=["b"],
            ),
            "system.c": FakeTool(
                consumes=["b"],
                produces=["c"],
            ),
            "system.d": FakeTool(
                consumes=["c"],
                produces=["d"],
            ),
        }
    )

    planner = Planner(
        tool_resolver=registry.resolve
    )

    objective = make_objective()

        Intent(
            action="system.a",
            original_text="ejecutar A",

            action="system.b",
            original_text="ejecutar B",
        ),

            action="system.c",
