import os
import chromadb
from sentence_transformers import SentenceTransformer
from pypdf import PdfReader
import docx

model = SentenceTransformer("all-MiniLM-L6-v2")

client = chromadb.Client()
collection = client.get_or_create_collection("nexus_memory")


def read_file(path):
    ext = path.lower().split(".")[-1]

    if ext == "txt":
        return open(path, encoding="utf-8", errors="ignore").read()

    if ext == "pdf":
        reader = PdfReader(path)
        return "\n".join(p.extract_text() or "" for p in reader.pages)

    if ext == "docx":
        d = docx.Document(path)
        return "\n".join(p.text for p in d.paragraphs)

    return ""


def chunk_text(text, size=500):
    return [text[i:i+size] for i in range(0, len(text), size)]


def index_folder(folder):
    for root, _, files in os.walk(folder):
        for f in files:
            if f.endswith((".pdf", ".txt", ".docx")):
                path = os.path.join(root, f)
                print("Indexando:", path)

                content = read_file(path)
                chunks = chunk_text(content)

                for i, chunk in enumerate(chunks):
                    emb = model.encode(chunk).tolist()

                    collection.add(
                        documents=[chunk],
                        embeddings=[emb],
                        ids=[f"{path}_{i}"],
                        metadatas=[{"path": path}]
                    )

import json
from pathlib import Path

MEMORY_FILE = Path("nexus_memory.json")


def load_memory():

    if not MEMORY_FILE.exists():
        return {"workflows":[], "errors":[], "preferences":{}}

    try:
        return json.loads(MEMORY_FILE.read_text())
    except:
        return {"workflows":[], "errors":[], "preferences":{}}


def save_memory(memory):

    MEMORY_FILE.write_text(json.dumps(memory, indent=2))

def remember_success(workflow_name):

    memory = load_memory()

    memory["workflows"].append({
        "workflow":workflow_name
    })

    save_memory(memory)
    
def remember_error(workflow_name, error):

    memory = load_memory()

    memory["errors"].append({
        "workflow":workflow_name,
        "error":error
    })

    save_memory(memory)

def search_memory(query, k=4):
    emb = model.encode(query).tolist()

    results = collection.query(
        query_embeddings=[emb],
        n_results=k
    )

    docs = results.get("documents", [[]])[0]

    return "\n".join(docs)

import json
from pathlib import Path
import sqlite3

DB = "nexus.db"
PACKS_DIR = Path("packs")


def list_packs():

    packs = []

    if not PACKS_DIR.exists():
        return packs

    for p in PACKS_DIR.iterdir():

        cfg = p / "pack.json"

        if cfg.exists():
            try:
                data = json.loads(cfg.read_text())
                packs.append(data)
            except:
                continue

    return packs


def install_pack(slug):

    pack_dir = PACKS_DIR / slug

    if not pack_dir.exists():
        return False

    workflows_file = pack_dir / "workflows.json"

    if not workflows_file.exists():
        return False

    tpl = json.loads(workflows_file.read_text())

    conn = sqlite3.connect(DB)
    cur = conn.cursor()

    for wf in tpl:

        # insertar workflow
        cur.execute("""
            INSERT INTO workflows (name, version, tenant_id)
            VALUES (?, ?, ?)
        """, (
            wf["name"],
            1,
            "default"
        ))

        workflow_id = cur.lastrowid

        # insertar steps
        for i, step in enumerate(wf.get("steps", [])):

            cur.execute("""
                INSERT INTO workflow_steps
                (workflow_id, step_order, action, value, config)
                VALUES (?, ?, ?, ?, ?)
            """, (
                workflow_id,
                i,
                step["action"],
                json.dumps(step.get("value")),
                json.dumps(step.get("config", {}))
            ))

    conn.commit()
    conn.close()

    return True

    folder = Path("plugins/installed")

    for file in folder.glob("*.py"):

        spec = importlib.util.spec_from_file_location(file.stem, file)
        module = importlib.util.module_from_spec(spec)
        spec.loader.exec_module(module)

        if hasattr(module, "Plugin"):

            plugin = module.Plugin(NexusAPI())

            plugin.on_load()

            for name, fn in plugin.actions().items():
                PLUGINS[name] = fn

    print("Plugins activos:", list(PLUGINS.keys()))

from fastapi import WebSocket

class WSManager:

    def __init__(self):
        self.clients: list[WebSocket] = []

    async def connect(self, ws: WebSocket):
        await ws.accept()
        self.clients.append(ws)
        print(f"🟢 WS conectado | Total: {len(self.clients)}")

    def disconnect(self, ws: WebSocket):
        if ws in self.clients:
            self.clients.remove(ws)
            print(f"🔌 WS desconectado | Total: {len(self.clients)}")

    async def broadcast(self, data):

        print(f"📡 Broadcasting: {data}")

        disconnected = []

        for c in self.clients:
            try:
                await c.send_json(data)
            except Exception as e:
                print("❌ Error WS:", e)
                disconnected.append(c)

        # limpiar conexiones muertas
        for d in disconnected:
            self.disconnect(d)


manager = WSManager()

import time
from collections import defaultdict

from events.bus import bus
from brain.nexus_voice import speak


class CircuitBreaker:

    def __init__(self):
        self.failures = defaultdict(int)
        self.state = {}
        self.opened_at = {}

        self.threshold = 5
        self.cooldown_seconds = 30

    # --------------------------------------------------
    # INTERNAL KEY BUILDER
    # --------------------------------------------------

    def _key(self, workflow, tenant, workspace):
        return f"{workflow}:{tenant}:{workspace}"

    # --------------------------------------------------
    # CHECK STATE
    # --------------------------------------------------

    def is_open(self, workflow, tenant, workspace):

        key = self._key(workflow, tenant, workspace)
        state = self.state.get(key, "CLOSED")

        if state == "OPEN":

            opened = self.opened_at.get(key, 0)

            # cooldown para probar recuperación
            if time.time() - opened > self.cooldown_seconds:

                print(f"🟡 Circuit HALF_OPEN para {key}")

                self.state[key] = "HALF_OPEN"
                return False

            return True

        return False

    # --------------------------------------------------
    # FAILURE
    # --------------------------------------------------

    async def handle_failure(self, payload):

        workflow = payload["workflow"]
        tenant = payload.get("tenant_id", "default")
        workspace = payload.get("workspace_id", "main")

        key = self._key(workflow, tenant, workspace)

        self.failures[key] += 1

        if self.failures[key] >= self.threshold:

            self.state[key] = "OPEN"
            self.opened_at[key] = time.time()

            print(f"🔥 Circuit OPEN para {key}")

            await speak(
                f"He bloqueado temporalmente el workflow {workflow} por fallos repetidos.",
                "warning"
            )

            await bus.publish("circuit.opened", {
                "workflow": workflow,
                "tenant_id": tenant,
                "workspace_id": workspace
            })

    # --------------------------------------------------
    # SUCCESS
    # --------------------------------------------------

    async def handle_success(self, payload):

        workflow = payload["workflow"]
        tenant = payload.get("tenant_id", "default")
        workspace = payload.get("workspace_id", "main")

        key = self._key(workflow, tenant, workspace)

        if self.state.get(key) == "HALF_OPEN":

            self.state[key] = "CLOSED"
            self.failures[key] = 0

            print(f"🟢 Circuit CLOSED para {key}")

            await speak(
                f"He restaurado el funcionamiento normal del workflow {workflow}.",
                "info"
            )

            await bus.publish("circuit.closed", {
                "workflow": workflow,
                "tenant_id": tenant,
                "workspace_id": workspace
            })


circuit = CircuitBreaker()


def register_circuit():
    bus.subscribe("job.failed", circuit.handle_failure)
    bus.subscribe("job.done", circuit.handle_success)

import asyncio
from datetime import datetime
from croniter import croniter

from core.database import get_connection
from jobs.manager import manager


class Scheduler:

    def __init__(self):
        asyncio.create_task(self.loop())

    # =========================
    # LOOP PRINCIPAL
    # =========================

    async def loop(self):

        while True:
            try:
                await self.check_schedules()
            except Exception as e:
                print("🔥 Scheduler loop error:", e)

            await asyncio.sleep(5)

    # =========================
    # CHECK SCHEDULES
    # =========================

    async def check_schedules(self):

        try:
            conn = get_connection()
            cur = conn.cursor()

            cur.execute("SELECT * FROM schedules WHERE enabled = TRUE AND tenant_id = %s", ("default",))
            rows = cur.fetchall()
        except Exception as e:
              print("⏳ Scheduler DB error:", e)
              return   

        finally:
            try:
                conn.close()
            except:
                pass

        now = datetime.utcnow()

        for row in rows:

            try:

                if row["type"] == "interval":
                    await self.check_interval(row, now)

                elif row["type"] == "cron":
                    await self.check_cron(row, now)

            except Exception as e:
                print("🔥 Scheduler check error:", e)

    # =========================
    # INTERVAL
    # =========================

    async def check_interval(self, row, now):

        interval = int(row["expression"])

        if not row["last_run"]:
            await self.run_schedule(row)
            return

        last = self.parse_datetime(row["last_run"])

        if not last:
            await self.run_schedule(row)
            return

        if (now - last).total_seconds() >= interval:
            await self.run_schedule(row)

    # =========================
    # CRON
    # =========================

    async def check_cron(self, row, now):

        expr = row["expression"]

        base = self.parse_datetime(row["last_run"]) or now

        itr = croniter(expr, base)
        next_run = itr.get_next(datetime)

        if now >= next_run:
            await self.run_schedule(row)

    # =========================
    # RUN
    # =========================

    async def run_schedule(self, row):

        try:

            await manager.create_job(row["workflow"])

            conn = get_connection()
            cur = conn.cursor()

            cur.execute("""
                UPDATE schedules
                SET last_run = %s
                WHERE id = %s
            """, (
                datetime.utcnow(),
                row["id"]
            ))

            conn.commit()
            conn.close()

        except Exception as e:
            print("🔥 Scheduler run error:", e)

    # =========================
    # HELPERS
    # =========================

    def parse_datetime(self, value):

        try:
            if isinstance(value, datetime):
                return value
            return datetime.fromisoformat(str(value))
        except Exception:
            return None

        action = step.get("action")

        # riesgo común
        if action == "move_files":

            value = step.get("value",{})

            if isinstance(value,dict):

                folder = value.get("to")

                if not folder:
                    risks.append({
                        "step":"move_files",
                        "risk":"destino no definido"
                    })

        if action == "delete_file":

            risks.append({
                "step":"delete_file",
                "risk":"acción destructiva"
            })

    return {
        "workflow": name,
        "steps": len(steps),
        "risks": risks,
        "safe": len(risks) == 0
    }

import asyncio

from watchdog.observers import Observer
from watchdog.events import FileSystemEventHandler

import jobs.manager as manager_module

from core.database import get_connection


WATCH_PATH = r"C:\Users\cdi40\Desktop"


class FolderHandler(FileSystemEventHandler):

    def on_created(self, event):

        # ignorar carpetas
        if event.is_directory:
            return

        print(f"📁 Nuevo archivo detectado: {event.src_path}")

        try:

            # =========================
            # CARGAR TRIGGERS
            # =========================
            conn = get_connection()
            cur = conn.cursor()

            cur.execute("""
                SELECT *
                FROM triggers
                WHERE enabled = TRUE
            """)

            triggers = cur.fetchall()

            conn.close()

            print(f"⚡ Triggers activos: {len(triggers)}")

            # =========================
            # EVALUAR TRIGGERS
            # =========================
            for trigger in triggers:

                config = trigger["config"] or {}

                ext = config.get("ext")

                if not ext:
                    continue

                print(f"🔍 Evaluando trigger: {ext}")

                # match extensión
                if event.src_path.lower().endswith(ext.lower()):

                    print(f"🔥 Trigger match: {ext}")

                    workflow_id = trigger["workflow_id"]

                    tenant_id = trigger["tenant_id"]

                    print(
                        f"🚀 Ejecutando workflow: {workflow_id}"
                    )

                    asyncio.run(
                        manager_module.manager.create_job(
                            workflow_id,
                            tenant_id
                        )
                    )

        except Exception as e:

            print(f"❌ Trigger error: {e}")


def start_folder_trigger():

    observer = Observer()

    observer.schedule(
        FolderHandler(),
        WATCH_PATH,
        recursive=False
    )

    observer.start()

    print(f"👀 Watching folder: {WATCH_PATH}")

    return observer

import sounddevice as sd
import whisper
import numpy as np
import threading
import time

from brain.agent import ask_nexus
from actions.system_actions import execute_action
from voice.tts import speak


model = whisper.load_model("base")
samplerate = 16000


# -----------------------------
# 🔥 detectar mic dinámicamente
# -----------------------------
def get_input_device():

    try:
        devices = sd.query_devices()

        for i, d in enumerate(devices):
            if d["max_input_channels"] > 0:
                print(f"🎤 Mic detectado: {d['name']} (id {i})")
                return i

    except Exception:
        pass

    return None   # 👈 NUNCA romper backend


# -----------------------------
def record(seconds, device):

    audio = sd.rec(
        int(seconds * samplerate),
        samplerate=samplerate,
        channels=1,
        dtype="float32",
        device=device
    )

    sd.wait()
    return audio.flatten()


# -----------------------------
def process(text):

    print("🧠 Usuario:", text)

    result = ask_nexus(text)

    if result.get("action"):
        output = execute_action(result["action"], result.get("value"))
        speak(output)
    else:
        speak(result["response"])


# -----------------------------
def listen_loop():

    device = get_input_device()

    if device is None:
        print("🔇 No hay micrófono válido → voz desactivada")
        return

    print("🎤 Nexus escuchando... decí 'nexus'")

    while True:

        try:

            audio = record(3, device)

            result = model.transcribe(audio, fp16=False)
            text = result["text"].lower().strip()

            if not text:
                continue

            print("Oído:", text)

            if "nexus" in text:

                speak("Sí, te escucho")

                audio = record(5, device)
                result = model.transcribe(audio, fp16=False)

                command = result["text"].strip()

                process(command)

        except Exception as e:
            print("⚠️ Error mic:", e)
            print("🔇 Desactivando voz...")
            return   # 👈 sale del loop sin romper server


# -----------------------------
def start_listener():

    thread = threading.Thread(
        target=listen_loop,
        daemon=True
    )

    thread.start()
