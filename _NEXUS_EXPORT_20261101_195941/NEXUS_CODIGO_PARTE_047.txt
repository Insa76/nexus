class User:

    return manager._action_authorization

from __future__ import annotations

from context.models import AssistantContext
from context.store import ContextStore


class ContextResolver:
    """
    Frontera de resolución del contexto operativo de NEXUS.

    El resolver no almacena usuarios, organizaciones ni proyectos.
    Tampoco conoce el Engine.

    Su única responsabilidad es transformar una selección de contexto
    en un AssistantContext utilizando el ContextStore.
    """

    def __init__(self, store: ContextStore) -> None:
        self._store = store

    def resolve(
        self,
        *,
        user_id: str | None = None,
        organization_id: str | None = None,
    ) -> AssistantContext:
        """
        Resuelve el contexto seleccionado por la capa superior.

        La lógica de selección y aislamiento permanece en ContextStore.
        """

        return self._store.build_context(
            user_id=user_id,
            organization_id=organization_id,
        )

from __future__ import annotations

from context.models import AssistantContext
from context.session import AssistantSession


class AssistantSessionManager:
    """
    Administra el ciclo de vida de una sesión Assistant activa.

    El manager no resuelve contexto ni persiste sesiones.
    Su única responsabilidad es crear, exponer y cerrar
    la sesión activa.
    """

    def __init__(self) -> None:
        self._active_session: AssistantSession | None = None

    def start(
        self,
        context: AssistantContext,
    ) -> AssistantSession:
        """
        Crea una nueva sesión y la establece como activa.
        """

        session = AssistantSession.create(context)
        self._active_session = session

        return session

    def get_active(self) -> AssistantSession | None:
        """
        Devuelve la sesión actualmente activa.
        """

        return self._active_session

    def close(self) -> AssistantSession | None:
        """
        Cierra la sesión activa y devuelve la sesión que estaba activa.
        """

        session = self._active_session
        self._active_session = None

        return session

from __future__ import annotations

from context.models import (
    AssistantContext,
    OrganizationProfile,
    ProjectContext,
    UserProfile,
)


class ContextStore:
    """
    Almacén en memoria para el contexto operativo de NEXUS.

    Su responsabilidad es resolver el contexto de una capa superior
    sin introducir lógica de contexto dentro del Engine.

    Flujo:

        ContextStore
            ↓
        AssistantContext
            ↓
        User / Organization / Projects

    El almacenamiento es deliberadamente en memoria en esta fase.
    """

    def __init__(self) -> None:
        self._users: dict[str, UserProfile] = {}
        self._organizations: dict[str, OrganizationProfile] = {}
        self._projects: dict[str, ProjectContext] = {}

    def add_user(self, user: UserProfile) -> None:
        self._users[user.id] = user

    def add_organization(
        self,
        organization: OrganizationProfile,
    ) -> None:
        self._organizations[organization.id] = organization

    def add_project(self, project: ProjectContext) -> None:
        self._projects[project.id] = project

    def get_user(self, user_id: str) -> UserProfile | None:
        return self._users.get(user_id)

    def get_organization(
        self,
        organization_id: str,
    ) -> OrganizationProfile | None:
        return self._organizations.get(organization_id)

    def get_project(self, project_id: str) -> ProjectContext | None:
        return self._projects.get(project_id)

    def projects_for_organization(
        self,
        organization_id: str,
    ) -> tuple[ProjectContext, ...]:
        return tuple(
            project
            for project in self._projects.values()
            if project.organization_id == organization_id
        )

    def build_context(
        self,
        *,
        user_id: str | None = None,
        organization_id: str | None = None,
    ) -> AssistantContext:
        """
        Construye un AssistantContext a partir de los identificadores
        seleccionados por la capa superior.

        Reglas:

        - user_id selecciona al usuario.
        - Si no se especifica organization_id, se utiliza la organización
          asociada al usuario.
        - Si organization_id se especifica explícitamente, tiene prioridad
          sobre la organización asociada al usuario.
        - Los proyectos siempre se limitan a la organización finalmente
          seleccionada.
        - IDs inexistentes no generan errores; simplemente producen un
          contexto parcial o vacío.
        """

        user = (
            self.get_user(user_id)
            if user_id is not None
            else None
        )

        selected_organization_id = organization_id

        if selected_organization_id is None and user is not None:
            selected_organization_id = user.organization_id

        organization = (
            self.get_organization(selected_organization_id)
            if selected_organization_id is not None
            else None
        )

        projects = (
            self.projects_for_organization(organization.id)
            if organization is not None
            else ()
        )

        return AssistantContext(
            user=user,
            organization=organization,
            projects=projects,
        )

from __future__ import annotations

from fastapi import WebSocket
from fastapi import WebSocketDisconnect


class ConnectionManager:
    """
    Administra las conexiones WebSocket activas del Kernel.

    Responsabilidades:
    - aceptar conexiones
    - cerrar conexiones
    - enviar mensajes a todos los clientes

    No conoce lógica de negocio.
    No conoce IA.
    No conoce eventos.
    """

    def __init__(self) -> None:
        self._connections: list[WebSocket] = []

    @property
    def count(self) -> int:
        """Cantidad de clientes conectados."""
        return len(self._connections)

    async def connect(self, websocket: WebSocket) -> None:
        """
        Acepta una nueva conexión.
        """

        await websocket.accept()

        self._connections.append(websocket)

        print(f"[Kernel] Client connected ({self.count})")

    def disconnect(self, websocket: WebSocket) -> None:
        """
        Elimina una conexión.
        """

        if websocket in self._connections:
            self._connections.remove(websocket)

        print(f"[Kernel] Client disconnected ({self.count})")

    async def broadcast(self, message: dict) -> None:
        """
        Envía un mensaje a todos los clientes conectados.

        Las conexiones inválidas son eliminadas automáticamente.
        """

        disconnected: list[WebSocket] = []

        for websocket in self._connections:

            try:
                await websocket.send_json(message)

            except WebSocketDisconnect:
                disconnected.append(websocket)

            except Exception:
                disconnected.append(websocket)

        for websocket in disconnected:
            self.disconnect(websocket)

from __future__ import annotations

from collections import deque
from typing import Any


class EventBus:
    """
    Bus interno de eventos del Kernel.

    Su única responsabilidad es almacenar eventos
    publicados por distintos componentes y entregarlos
    posteriormente para su procesamiento.

    No conoce WebSocket.
    No conoce IA.
    No conoce Runtime.
    """

    def __init__(self) -> None:
        self._events: deque[Any] = deque()

    def emit(self, event: Any) -> None:
        """
        Publica un nuevo evento.
        """
        self._events.append(event)

    def consume_all(self) -> list[Any]:
        """
        Devuelve todos los eventos pendientes y limpia el bus.
        """

        events = list(self._events)

        self._events.clear()

        return events

    def has_events(self) -> bool:
        """
        Indica si existen eventos pendientes.
        """

        return len(self._events) > 0

    def size(self) -> int:
        """
        Cantidad de eventos pendientes.
        """

        return len(self._events)

    def clear(self) -> None:
        """
        Elimina todos los eventos pendientes.
        """

        self._events.clear()

import asyncio
from typing import Any


class EventQueue:
    """
    Cola de eventos del Kernel.

    Centraliza el ingreso de eventos que luego serán
    procesados por el Runtime.

    La cola es completamente asíncrona y mantiene
    prioridad por evento.
    """

    def __init__(self):

        self._queue: asyncio.PriorityQueue[
            tuple[int, Any]
        ] = asyncio.PriorityQueue()

    async def put(
        self,
        priority: int,
        event: Any
    ) -> None:

        """
        Agrega un evento a la cola.

        A mayor prioridad, antes será procesado.
        """

        await self._queue.put(
            (-priority, event)
        )

    async def get(self) -> Any:

        """
        Espera hasta que exista un evento disponible.
        """

        _, event = await self._queue.get()

        return event

    def empty(self) -> bool:

        return self._queue.empty()

    def size(self) -> int:

        return self._queue.qsize()

from __future__ import annotations

from core.connection_manager import ConnectionManager
from core.event_bus import EventBus
from core.event_queue import EventQueue
from core.runtime_state import RuntimeState


class Kernel:
    """
    Núcleo principal de NEXUS.

    El Kernel coordina los componentes fundamentales
    del sistema, pero no contiene lógica de negocio.

    Responsabilidades:

    - mantener el estado global
    - administrar conexiones
    - administrar eventos
    - exponer una única interfaz para el Runtime

    El Kernel NO conoce:

    - IA
    - Memoria
    - Planner
    - Knowledge
    - Providers
    - Agentes
    """

    def __init__(self) -> None:

        self.state = RuntimeState()

        self.connections = ConnectionManager()

        self.events = EventBus()

        self.queue = EventQueue()

    @property
    def online(self) -> bool:
        return self.state.online

    @property
    def connected_clients(self) -> int:
        return self.state.connected_clients

    def start(self) -> None:
        """
        Inicia el Kernel.
        """

        self.state.online = True

    def stop(self) -> None:
        """
        Detiene el Kernel.
        """

        self.state.online = False

    def info(self) -> dict:
        """
        Devuelve información básica del Kernel.
        """

        return {

            "online": self.state.online,

            "clients": self.state.connected_clients,

            "model": self.state.current_model,

            "version": self.state.kernel_version
        }

from __future__ import annotations

from documents.artifacts import DocumentArtifact


class ArtifactStore:
    def __init__(self) -> None:
        self._artifacts: dict[str, DocumentArtifact] = {}

    def add(self, artifact: DocumentArtifact) -> DocumentArtifact:
        self._artifacts[artifact.id] = artifact
        return artifact

    def get(self, artifact_id: str) -> DocumentArtifact | None:
        return self._artifacts.get(artifact_id)

    def list(self) -> list[DocumentArtifact]:
        return list(self._artifacts.values())

    def update(self, artifact: DocumentArtifact) -> DocumentArtifact:
        if artifact.id not in self._artifacts:
            raise KeyError(f"Artifact not found: {artifact.id}")

        self._artifacts[artifact.id] = artifact
        return artifact

    def remove(self, artifact_id: str) -> DocumentArtifact | None:
        return self._artifacts.pop(artifact_id, None)

    def clear(self) -> None:
        self._artifacts.clear()

    def count(self) -> int:
        return len(self._artifacts)

from __future__ import annotations

from documents.artifacts import DocumentArtifact
from documents.declared_type_detection import DeclaredTypeDetectionStrategy
from documents.extension_type_detection import ExtensionTypeDetectionStrategy
from documents.type_detection import DocumentTypeDetectionStrategy
from documents.type_detection_evidence import DocumentTypeDetectionEvidence
from documents.document_type import DocumentType


class ArtifactTypeResolver:
    def __init__(
        self,
        strategies: list[DocumentTypeDetectionStrategy] | None = None,
    ) -> None:
        self._strategies = list(
            strategies
            if strategies is not None
            else [
                DeclaredTypeDetectionStrategy(),
                ExtensionTypeDetectionStrategy(),
            ]
        )

    def resolve(
        self,
        artifact: DocumentArtifact,
    ) -> DocumentType:
        evidence = self.resolve_with_evidence(
            artifact
        )

        return evidence.detected_type

    def resolve_with_evidence(
        self,
        artifact: DocumentArtifact,
    ) -> DocumentTypeDetectionEvidence:
        for strategy in self._strategies:
            detected_type = strategy.resolve(
                artifact
            )

            if detected_type is not None:
                return DocumentTypeDetectionEvidence(
                    detected_type=detected_type,
                    strategy=type(strategy).__name__,
                )

        raise ValueError(
            "Cannot resolve document type for artifact: "
            f"{artifact.id}"
        )

    def strategies(
        self,
    ) -> list[DocumentTypeDetectionStrategy]:
        return list(self._strategies)

from __future__ import annotations

from dataclasses import dataclass, field
from datetime import datetime, timezone
from uuid import uuid4


def _utc_now() -> datetime:
    return datetime.now(timezone.utc)


@dataclass(frozen=True)
class DocumentArtifact:
    """
    Representa el artefacto físico o digital asociado a un documento.

    El modelo describe el artefacto, pero no realiza operaciones
    de almacenamiento, lectura o hashing.
    """

    id: str
    location: str
    size: int | None = None
    checksum: str | None = None
    checksum_algorithm: str | None = None
    created_at: datetime = field(default_factory=_utc_now)

    @classmethod
    def create(
        cls,
        location: str,
        size: int | None = None,
        checksum: str | None = None,
        checksum_algorithm: str | None = None,
    ) -> DocumentArtifact:
        return cls(
            id=str(uuid4()),
            location=location,
            size=size,
            checksum=checksum,
            checksum_algorithm=checksum_algorithm,
            created_at=_utc_now(),
        )

@dataclass(slots=True)

    Expone operaciones orientadas al Assistant/Cognitive Core
    sin exponer stores ni detalles internos del subsistema Documents.
    """

    context_access: DocumentContextAccess

    def query(
        self,
        *,
        context: object,
        name: str | None = None,
        mime_type: str | None = None,
        offset: int = 0,
        limit: int | None = None,
    ) -> DocumentQueryResult:
        return self.context_access.query_from_context(
            context,
            name=name,
            mime_type=mime_type,
            offset=offset,
            limit=limit,
        )

    def relevant(
        self,
        *,
        context: object,
    ) -> tuple[DocumentContextMatch, ...]:
        return self.context_access.resolve_ordered_matches_from_context(
            context,
        )

    def relevant_documents(
        self,
        *,
        context: object,
    ) -> tuple[Document, ...]:
        return self.context_access.resolve_ordered_documents_from_context(
            context,
        )
